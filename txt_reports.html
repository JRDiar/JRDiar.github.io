def detect_current_movement(structure_data, current_price):
    """
    Detecta dirección del precio AHORA (últimas 3-5 velas)
    No la estructura histórica
    
    Returns:
        'RISING', 'FALLING', or 'SIDEWAYS'
    """
    prices = structure_data.get('prices', [])
    
    if len(prices) < 3:
        return 'UNKNOWN'
    
    # Analizar últimas 3 velas para tendencia micro
    recent = prices[-3:]
    
    # Tendencia clara subiendo
    if recent[-1] > recent[-2] and recent[-2] > recent[-3]:
        return 'RISING'
    
    # Tendencia clara bajando (CORRECCIÓN)
    elif recent[-1] < recent[-2] and recent[-2] < recent[-3]:
        return 'FALLING'
    
    # Al menos 2 de 3 subiendo
    elif sum(1 for i in range(len(recent)-1) if recent[i+1] > recent[i]) >= 2:
        return 'RISING'
    
    # Al menos 2 de 3 bajando
    elif sum(1 for i in range(len(recent)-1) if recent[i+1] < recent[i]) >= 2:
        return 'FALLING'
    
    else:
        return 'SIDEWAYS'

"""

def validate_momentum_reversal(ladder, direction, timeframe='5m'):
    """
    Detecta reversiones de momentum (señal de fin de corrección)
    
    Args:
        ladder: Dict con análisis
        direction: 'BULLISH' o 'BEARISH'
        timeframe: Timeframe a analizar
    
    Returns:
        dict: {
            'reversal_detected': bool,
            'score_adjustment': int,
            'reason': str,
            'momentum_color': str,
            'consecutive': int
        }
    """
    result = {
        'reversal_detected': False,
        'score_adjustment': 0,
        'reason': '',
        'momentum_color': 'neutral',
        'consecutive': 0
    }
    
    tf_data = ladder.get(timeframe, {})
    momentum = tf_data.get('momentum', {})
    
    color = momentum.get('color_current', 'neutral')
    color_prev = momentum.get('color_prev', color)
    consecutive = momentum.get('consecutive_count', 0)
    
    result['momentum_color'] = color
    result['consecutive'] = consecutive
    
    if direction == 'BULLISH':
        # Caso 1: Reversión confirmada (rojo → verde, 2+ velas)
        if color_prev == 'red' and color in ['green', 'lime'] and consecutive >= 2:
            result['reversal_detected'] = True
            result['score_adjustment'] = 25
            result['reason'] = f"✅ Reversión confirmada (❌→✅ {consecutive}v)"
        
        # Caso 2: Reversión iniciando (1 vela verde)
        elif color_prev == 'red' and color in ['green', 'lime'] and consecutive == 1:
            result['reversal_detected'] = True
            result['score_adjustment'] = 10
            result['reason'] = f"⚠️ Reversión iniciando (1v verde)"
        
        # Caso 3: Momentum verde maduro (3+ velas)
        elif color in ['green', 'lime'] and consecutive >= 3:
            result['score_adjustment'] = 15
            result['reason'] = f"✅ Momentum desarrollado ({consecutive}v)"
        
        # Caso 4: Momentum verde temprano (2 velas)
        elif color in ['green', 'lime'] and consecutive == 2:
            result['score_adjustment'] = 10
            result['reason'] = f"✅ Momentum confirmado ({consecutive}v)"
        
        # Caso 5: Corrección en progreso (rojo)
        elif color in ['red', 'maroon']:
            result['score_adjustment'] = -25
            result['reason'] = f"❌ Corrección activa ({consecutive}v rojas)"
        
        # Caso 6: Neutral/débil

            result['score_adjustment'] = -10
            result['reason'] = f"⚠️ Momentum neutral"
