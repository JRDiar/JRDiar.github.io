def check_5m_veto(symbol, direction, ladder_data):
    """5M puede vetar si detecta condiciones CRÍTICAS."""
    
    from brain_config import (
        VETO_5M_ADX_MIN,
        VETO_5M_ADX_VELOCITY,
        ENABLE_5M_VETO
    )
    
    if not ENABLE_5M_VETO:
        return {'should_veto': False, 'reason': '', 'penalty': 0}
    
    data_5m = ladder_data.get('5m', {})
    if not data_5m:
        return {'should_veto': False, 'reason': 'No 5M data', 'penalty': 0}
    
    vetos = []
    penalties = []
    
    adx_5m = data_5m.get('adx', {})
    adx_current = adx_5m.get('adx_current', 0)
    adx_trend = adx_5m.get('adx_trend', 'estable')
    mom_5m = data_5m.get('momentum', {})
    mom_color = mom_5m.get('color_current', 'gray')
    direction_5m = data_5m.get('direction', 'UNKNOWN')
    
    # ===============================================
    # VETO 1: ADX débil + decreciendo (COMBO)
    # ===============================================
    # Solo vetar si AMBOS: ADX bajo Y decreciendo
    if adx_current < VETO_5M_ADX_MIN and adx_trend == 'decreciendo':
        vetos.append(f'ADX 5M {adx_current:.1f} débil + decreciendo')
        penalties.append(50)
    
    # ===============================================
    # VETO 2: 5M LATERAL (penalty severa, no veto absoluto)
    # ===============================================
    if direction_5m == 'LATERAL':
        # No es veto, es penalty
        penalties.append(40)
        print(f"   ⚠️ 5M LATERAL - penalty -40")
    
    # ===============================================
    # VETO 3: Momentum 5M contradice trade (CRÍTICO)
    # ===============================================
    if direction == 'BULLISH' and mom_color in ['red', 'maroon']:
        vetos.append(f'Momentum 5M ROJO contradice LONG')
        penalties.append(60)
    elif direction == 'BEARISH' and mom_color in ['green', 'lime']:
        vetos.append(f'Momentum 5M VERDE contradice SHORT')
        penalties.append(60)
    
    # ===============================================
    # VETO 4: Conflicto de dirección 5M vs trade
    # ===============================================
    if direction_5m == 'BEARISH' and direction == 'BULLISH':
        vetos.append('5M dice BEARISH, trade es BULLISH')
        penalties.append(50)
    elif direction_5m == 'BULLISH' and direction == 'BEARISH':
        vetos.append('5M dice BULLISH, trade es BEARISH')
        penalties.append(50)
    
    # ===============================================
    # DECISIÓN: Veto si penalty total >= 100
    # ===============================================
    total_penalty = sum(penalties)
    should_veto = total_penalty >= 100
    
    return {
        'should_veto': should_veto,
        'reason': ' | '.join(vetos) if vetos else '',
        'penalty': total_penalty,
        'details': vetos
    }
