def _extract_score(score_value, default=0):
    """
    Recursively extract integer score from potentially nested dict.
    
    Args:
        score_value: The value to extract (can be int, float, dict, or None)
        default: Default value if extraction fails
    
    Returns:
        int: Extracted integer score
    """
    if score_value is None:
        return default
    if isinstance(score_value, (int, float)):
        return int(score_value)
    if isinstance(score_value, dict):
        # Recursively extract from nested dict
        return _extract_score(score_value.get('score', default), default)
    # Fallback for any other type
    return default

def find_nearest_resistance(current_price, resistances):
    """
    Encuentra la resistencia más cercana por encima del precio actual.
    Args:
        current_price (float): Precio actual.
        resistances (list): Lista de resistencias (floats).
    Returns:
        tuple: (precio_resistencia, distancia_pct) o None
    """
    if not resistances:
        return None
    above = [r for r in resistances if r > current_price]
    if not above:
        return None
    nearest = min(above, key=lambda r: r - current_price)
    distance_pct = ((nearest - current_price) / current_price) * 100
    return (nearest, distance_pct)

def find_nearest_support(current_price, supports):
    """
    Encuentra el soporte más cercano por debajo del precio actual.
    Args:
        current_price (float): Precio actual.
        supports (list): Lista de soportes (floats).
    Returns:
        tuple: (precio_soporte, distancia_pct) o None
    """
    if not supports:
        return None
    below = [s for s in supports if s < current_price]
    if not below:
        return None
    nearest = max(below, key=lambda s: current_price - s)
    distance_pct = ((current_price - nearest) / current_price) * 100
    return (nearest, distance_pct)

class WatchlistManager:
    def __init__(self):
        self.watchlist = {}
        self.load_from_file()
    
    def load_from_file(self):
        """Carga watchlist desde JSON"""
        if not ENABLE_WATCHLIST_JSON:
            return
        
        try:
            with open(WATCHLIST_JSON, 'r') as f:
                data = json.load(f)
                self.watchlist = data.get('watchlist', {})
            print(f"✅ Watchlist cargada: {len(self.watchlist)} monedas")
        except FileNotFoundError:
            print("ℹ️ Watchlist vacía (archivo no existe)")
            self.watchlist = {}
        except Exception as e:
            print(f"❌ Error cargando watchlist: {e}")
            self.watchlist = {}
    
    def save_to_file(self):
        """Guarda watchlist a JSON"""
        if not ENABLE_WATCHLIST_JSON:
            return
        
        try:
            data = {
                'version': '1.0',
                'last_updated': datetime.now(ARGENTINA_TZ).isoformat(),
                'watchlist': self.watchlist
            }
            
            with open(WATCHLIST_JSON, 'w') as f:
                json.dump(data, f, indent=2)
            
            print(f"✅ Watchlist guardada: {len(self.watchlist)} monedas")
        except Exception as e:
            print(f"❌ Error guardando watchlist: {e}")
    
    def calculate_triggers(self, analysis_data):
        """
        Calcula triggers de entrada para una moneda
        
        Args:
            analysis_data: Dict con análisis completo
        
        Returns:
            dict: Triggers calculados o None
        """
        analysis_5m = analysis_data['analysis_5m']
        analysis_4h = analysis_data['analysis_4h']
        sr_zones = analysis_data['sr_zones']
        
        if not analysis_5m or not analysis_4h:
            return None
        
        direction = analysis_5m['direction']
        current_price = analysis_5m['current_price']
        velocity = analysis_5m['velocity']
        acceleration = analysis_5m['acceleration']
        
        # Target (resistencia/soporte más cercano)
        target_price = None
        target_distance_pct = None
        
        if sr_zones:
            min_dist = 0.8 # Mínimo 0.8% de recorrido para que valga la pena
            
            if direction == 'BULLISH':
                resistances = sorted([r for r in sr_zones['resistances'] if r > current_price * (1 + min_dist/100)])
                if resistances:
                    target_price = resistances[0]
                    target_distance_pct = ((target_price - current_price) / current_price) * 100
            else:  # BEARISH
                supports = sorted([s for s in sr_zones['supports'] if s < current_price * (1 - min_dist/100)], reverse=True)
                if supports:
                    target_price = supports[0]
                    target_distance_pct = ((current_price - target_price) / current_price) * 100
        
        # Si no hay target, usar proyección simple (3% para bulls, -3% para bears)
        if not target_price:
            if direction == 'BULLISH':
                target_price = current_price * 1.03
                target_distance_pct = 3.0
            else:
                target_price = current_price * 0.97
                target_distance_pct = 3.0
        
        # Tiempo proyectado a target
        tiempo_proyectado = calculate_time_to_target(
            current_price, target_price, velocity.get('velocity_current', 0) if isinstance(velocity, dict) else velocity, 
            acceleration.get('acceleration_current', 0) if isinstance(acceleration, dict) else acceleration
        )
        
        # Risk/Reward (stop loss simple 2%)
        stop_loss = current_price * 0.98 if direction == 'BULLISH' else current_price * 1.02
        reward = abs(target_price - current_price)
        risk = abs(current_price - stop_loss)
        rr_ratio = reward / risk if risk > 0 else 0
        
        # Validar viabilidad
        if rr_ratio < MIN_RR_RATIO:
            return None  # R:R insuficiente
        
        if tiempo_proyectado and (tiempo_proyectado < MIN_TIEMPO_PROYECTADO_MIN or 
                                  tiempo_proyectado > MAX_TIEMPO_PROYECTADO_MIN):
            return None  # Timing fuera de rango
        
        return {
            'trigger_adx': ADX_THRESHOLD,
            'trigger_adx_acceleration': ADX_MIN_ACCELERATION,
            'trigger_momentum_candles': MOMENTUM_CONSECUTIVE_CANDLES,
            'trigger_volume_ratio': VOLUME_RATIO_MIN,
            'target_price': target_price,
            'target_distance_pct': target_distance_pct,
            'stop_loss': stop_loss,
            'tiempo_proyectado_min': tiempo_proyectado,
            'rr_ratio': rr_ratio
        }
    
    def add_to_watchlist(self, symbol, analysis_data):
        """
        Agrega moneda a watchlist
        
        Args:
            symbol: Símbolo
            analysis_data: Dict con análisis completo
        
        Returns:
            bool: True si agregado, False si rechazado
        """
        # Verificar límite
        if len(self.watchlist) >= MAX_WATCHLIST_SIZE:
            # Remover moneda con score más bajo
            lowest_symbol = min(self.watchlist.keys(), 
                               key=lambda s: _extract_score(self.watchlist[s].get('score', 0), 0))
            lowest_score = _extract_score(self.watchlist[lowest_symbol].get('score', 0), 0)
            
            new_score = _extract_score(analysis_data['score_data'].get('score', 0), 0)
            if new_score > lowest_score:
                del self.watchlist[lowest_symbol]
                print(f"⚠️ Watchlist llena, removido {lowest_symbol} (score {lowest_score})")
            else:
                return False  # Score muy bajo
        
        # Calcular triggers
        triggers = self.calculate_triggers(analysis_data)
        
        # LÓGICA HUNTER MODE: Si no hay triggers pero hay fuerza, guardar igual
        status = 'MONITORING'
        analysis_5m = analysis_data['analysis_5m']
        adx_current = analysis_5m['adx']['adx_current']
        dist_ema = analysis_5m.get('dist_ema_pct', 0)
        
        # Extraer estructura
        structure = analysis_data.get('structure', {})
        last_structure = structure.get('last_structure', 'N/A')

        if not triggers:
            if adx_current > 25:
                # Si está muy extendido (>2%), esperar pullback
                if dist_ema > 2.0:
                    status = 'WAITING_PULLBACK'
                    print(f"🦅 HUNTER: {symbol} extendido ({dist_ema:.1f}%), esperando pullback. Estructura: {last_structure}")
                else:
                    status = 'HUNTING'
                    adx_slope = analysis_5m['adx'].get('adx_slope', 0)
                    trend_icon = "↗️" if adx_slope > 0 else "↘️"
                    print(f"🦅 HUNTER: {symbol} acechando (ADX {adx_current:.1f} {trend_icon}). Estructura: {last_structure}")
                
                # Crear triggers dummy o vacíos para permitir el guardado
                triggers = {} 
            else:
                return False  # Triggers inviables y sin fuerza suficiente
        
        # Agregar a watchlist
        final_score = _extract_score(analysis_data['score_data'].get('score', 0), 0)
        self.watchlist[symbol] = {
            'added_timestamp': datetime.now(ARGENTINA_TZ).isoformat(),
            'last_check_timestamp': datetime.now(ARGENTINA_TZ).isoformat(),
            'checks_without_trigger': 0,
            'score': final_score,
            'clasificacion': analysis_data['score_data'].get('clasificacion', 'REGULAR'),
            'direction': analysis_data['analysis_5m']['direction'],
            'triggers': triggers,
            'analysis_snapshot': {
                'adx_5m': analysis_data['analysis_5m'].get('adx', {}),
                'adx_4h': analysis_data['analysis_4h'].get('adx', {}),
                'momentum_age_5m': analysis_data['analysis_5m'].get('momentum', {}).get('consecutive_count', 0) if isinstance(analysis_data['analysis_5m'].get('momentum'), dict) else 0,
                'emas_classification_15m': analysis_data['analysis_15m'].get('emas', {}).get('classification', 'CERCANAS') if analysis_data.get('analysis_15m') else None
            },
            'evolution': analysis_data.get('evolution', {}),
            'structure': structure,
            'status': status
        }
        
        print(f"✅ Agregado a watchlist: {symbol} (Score: {final_score}) [{status}] | Estructura: {last_structure}")
        return True
    
    def remove_from_watchlist(self, symbol, reason=""):
        """Remueve moneda de watchlist"""
        if symbol in self.watchlist:
            del self.watchlist[symbol]
            print(f"❌ Removido de watchlist: {symbol} ({reason})")
    
    def get_watchlist(self):
        """Retorna watchlist ordenada por score"""
        sorted_list = sorted(
            self.watchlist.items(),
            key=lambda x: x[1]['score'],
            reverse=True
        )
        return sorted_list
    
    def update_watchlist_from_analysis(self, symbols_analyzed):
        """
        Actualiza watchlist con nuevos análisis
        
        Args:
            symbols_analyzed: List de (symbol, analysis_data) tuples
        """
        added_count = 0
        
        for symbol, analysis_data in symbols_analyzed:
            if not analysis_data or not analysis_data.get('viable'):
                continue
            
            # Si ya está en watchlist, actualizar score
            if symbol in self.watchlist:
                old_score = _extract_score(self.watchlist[symbol].get('score', 0), 0)
                new_score = _extract_score(analysis_data['score_data'].get('score', 0), 0)
                
                # Si score bajó mucho (>15 puntos), remover
                if new_score < old_score - 15:
                    self.remove_from_watchlist(symbol, "score bajó significativamente")
                else:
                    # Actualizar score
                    self.watchlist[symbol]['score'] = new_score
            
            # Si no está, agregar
            else:
                if self.add_to_watchlist(symbol, analysis_data):
                    added_count += 1
        
        print(f"📊 Watchlist actualizada: {len(self.watchlist)} monedas ({added_count} nuevas)")
        self.save_to_file()
    
    def cleanup_stale_symbols(self, max_age_minutes=30, max_checks=20):
        """
        Remueve monedas que llevan mucho tiempo sin activar trigger
        
        Args:
            max_age_minutes: Tiempo máximo en watchlist sin trigger (30min)
            max_checks: Verificaciones máximas sin activar (20 checks)
        
        Returns:
            int: Cantidad de monedas removidas
        """
        now = datetime.now(ARGENTINA_TZ)
        to_remove = []
        
        for symbol, data in self.watchlist.items():
            # Calcular tiempo en watchlist
            added = datetime.fromisoformat(data['added_timestamp'])
            age_minutes = (now - added).total_seconds() / 60
            
            # Calcular checks sin trigger
            checks = data.get('checks_without_trigger', 0)
            
            # Remover si lleva mucho tiempo O muchos checks sin trigger
            # EXCEPCIÓN HUNTER MODE: Si está acechando, dar más tiempo
            status = data.get('status', 'MONITORING')
            
            limit_age = max_age_minutes
            limit_checks = max_checks
            
            if status in ['HUNTING', 'WAITING_PULLBACK']:
                limit_age = max_age_minutes * 4  # 2 horas en lugar de 30 min
                limit_checks = max_checks * 4    # 80 checks en lugar de 20
                
            if age_minutes > limit_age or checks > limit_checks:
                to_remove.append(symbol)
                print(f"🗑️ Removiendo {symbol} de watchlist:")
                print(f"   Tiempo: {age_minutes:.1f}min | Checks sin trigger: {checks}")
        
        for symbol in to_remove:
            del self.watchlist[symbol]
        
        if to_remove:
            print(f"✅ Limpieza watchlist: {len(to_remove)} monedas removidas")
            self.save_to_file()
        
        return len(to_remove)
    
    def increment_check_counter(self, symbol):
        """Incrementa contador de verificaciones sin trigger"""
        if symbol in self.watchlist:
            self.watchlist[symbol]['checks_without_trigger'] = \
                self.watchlist[symbol].get('checks_without_trigger', 0) + 1
            self.watchlist[symbol]['last_check_timestamp'] = \
                datetime.now(ARGENTINA_TZ).isoformat()
    
    def reset_check_counter(self, symbol):
        """Resetea contador cuando trigger se activa (casi)"""
        if symbol in self.watchlist:
            self.watchlist[symbol]['checks_without_trigger'] = 0
    
    def clear_watchlist(self):
        """Limpia watchlist completa"""
        self.watchlist = {}
        self.save_to_file()
        print("🗑️ Watchlist limpiada")
