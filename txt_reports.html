"""
ZigZag Validator - Validación de alineación estructural
Verifica que la estructura ZigZag vaya en la dirección del trade
"""

def validate_zigzag_alignment(ladder, direction, current_price):
    """
    Valida alineación de ZigZag con dirección del trade
    
    Args:
        ladder: Dict con análisis multi-timeframe
        direction: 'BULLISH' o 'BEARISH'
        current_price: Precio actual
    
    Returns:
        dict: {
            'aligned': bool,
            'score_adjustment': int (-50 a +15),
            'reason': str,
            'pivot_type': str,
            'structure': str
        }
    """
    result = {
        'aligned': True,
        'score_adjustment': 0,
        'reason': '',
        'pivot_type': 'Unknown',
        'structure': 'Unknown'
    }
    
    # Extraer datos ZigZag de 5M
    structure_data = ladder.get('5m', {}).get('zigzag', {})
    pivot_types = structure_data.get('pivot_types', [])
    last_pivot = pivot_types[-1] if pivot_types else None
    last_structure = structure_data.get('last_structure', '')
    
    result['pivot_type'] = last_pivot or 'Unknown'
    result['structure'] = last_structure
    
    if direction == 'BEARISH':
        # SHORT necesita estructura bajista
        if 'BULLISH' in last_structure or last_pivot in ['L', 'HL']:
            result['aligned'] = False
            result['score_adjustment'] = -50
            result['reason'] = f"❌ ZigZag CONTRARIO ({last_pivot}, {last_structure})"
        elif last_pivot in ['H', 'LH']:
            result['aligned'] = True
            result['score_adjustment'] = 15
            result['reason'] = f"✅ ZigZag bajista ({last_pivot})"
        
    elif direction == 'BULLISH':
        # LONG necesita estructura alcista
        if 'BEARISH' in last_structure or last_pivot in ['H', 'LH']:
            result['aligned'] = False
            result['score_adjustment'] = -50
            result['reason'] = f"❌ ZigZag CONTRARIO ({last_pivot}, {last_structure})"
        elif last_pivot in ['L', 'HL']:
            result['aligned'] = True
            result['score_adjustment'] = 15
            result['reason'] = f"✅ ZigZag alcista ({last_pivot})"
    
    return result

"""
ADX Validator - Validación de fuerza y separación
Verifica ADX mínimo y separación DI+/DI-
"""

def validate_adx_separation(ladder, direction, timeframe='5m'):
    """
    Valida separación ADX para confirmar fuerza direccional
    
    Args:
        ladder: Dict con análisis
        direction: 'BULLISH' o 'BEARISH'
        timeframe: Timeframe a validar
    
    Returns:
        dict: {
            'separated': bool,
            'score_adjustment': int,
            'reason': str,
            'adx_value': float,
            'di_spread': float
        }
    """
    result = {
        'separated': False,
        'score_adjustment': 0,
        'reason': '',
        'adx_value': 0,
        'di_spread': 0
    }
    
    tf_data = ladder.get(timeframe, {})
    adx_data = tf_data.get('adx', {})
    
    adx_val = adx_data.get('adx_current', 0)
    di_plus = adx_data.get('di_plus_current', 0)
    di_minus = adx_data.get('di_minus_current', 0)
    
    result['adx_value'] = adx_val
    result['di_spread'] = abs(di_plus - di_minus)
    
    # Validar ADX mínimo
    if adx_val < 20:
        result['separated'] = False
        result['score_adjustment'] = -25
        result['reason'] = f"⚠️ ADX débil ({adx_val:.1f} < 20)"
        return result
    
    # Validar separación DI
    if direction == 'BULLISH':
        if di_plus > di_minus + 8:
            result['separated'] = True
            result['score_adjustment'] = 15
            result['reason'] = f"✅ DI+ dominante (spread {result['di_spread']:.1f})"
        elif di_plus > di_minus + 3:
            result['separated'] = True
            result['score_adjustment'] = 5
            result['reason'] = f"⚠️ DI+ débil (spread {result['di_spread']:.1f})"
        else:
            result['separated'] = False
            result['score_adjustment'] = -20
            result['reason'] = f"❌ Sin separación DI (spread {result['di_spread']:.1f})"
    
    elif direction == 'BEARISH':
        if di_minus > di_plus + 8:
            result['separated'] = True
            result['score_adjustment'] = 15
            result['reason'] = f"✅ DI- dominante (spread {result['di_spread']:.1f})"
        elif di_minus > di_plus + 3:
            result['separated'] = True
            result['score_adjustment'] = 5
            result['reason'] = f"⚠️ DI- débil (spread {result['di_spread']:.1f})"
        else:
            result['separated'] = False
            result['score_adjustment'] = -20
            result['reason'] = f"❌ Sin separación DI"
    
    return result
