"""
brain_trend_birth.py - Detector de Tendencia Naciente vs Moribunda
==================================================================
El bot SOLO debe entrar en tendencias NACIENDO, no cuando están MURIENDO.
"""
def calculate_adx_lifecycle(adx_current, adx_history, lookback=5):
    """
    Clasifica en qué fase del ciclo de vida está el ADX
    
    Args:
        adx_current: Valor actual de ADX
        adx_history: Lista de valores históricos
        lookback: Velas para calcular velocidad
        
    Returns:
        dict: {
            'phase': 'DORMANT'|'BIRTH'|'MATURE'|'DECLINE',
            'velocity': float,
            'acceleration': float
        }
    """
    if len(adx_history) < lookback + 2:
        return {'phase': 'UNKNOWN', 'velocity': 0, 'acceleration': 0}
    
    # Calcular velocidad (cambio por vela)
    recent = adx_history[-lookback:]
    velocity = (recent[-1] - recent[0]) / lookback
    
    # Calcular aceleración
    if len(adx_history) >= lookback * 2:
        prev_velocity = (adx_history[-lookback] - adx_history[-lookback*2]) / lookback
        acceleration = velocity - prev_velocity
    else:
        acceleration = 0
    
    # Clasificar fase
    if adx_current < 20:
        phase = 'DORMANT'  # Sin tendencia
    elif adx_current < 25 and velocity > 1.5:
        phase = 'BIRTH'    # Tendencia NACIENDO ✅
    elif 25 <= adx_current <= 40 and abs(velocity) < 1:
        phase = 'MATURE'   # Tendencia MADURA
    elif adx_current > 35 and velocity < -0.5:
        phase = 'DECLINE'  # Tendencia MURIENDO ❌
    else:
        phase = 'TRANSITION'
    
    return {
        'phase': phase,
        'velocity': velocity,
        'acceleration': acceleration
    }
def detect_di_divergence(di_plus_current, di_minus_current, 
                         di_plus_prev, di_minus_prev):
    """
    Detecta si DI+ y DI- están divergiendo o convergiendo
    
    Returns:
        str: 'DIVERGING'|'CONVERGING'|'STABLE'
    """
    spread_current = di_plus_current - di_minus_current
    spread_prev = di_plus_prev - di_minus_prev
    spread_change = abs(spread_current) - abs(spread_prev)
    
    if spread_change > 2:
        return 'DIVERGING'   # DI se separan → Fuerza creciendo
    elif spread_change < -2:
        return 'CONVERGING'  # DI se acercan → Fuerza muriendo
    else:
        return 'STABLE'
def detect_trend_phase(adx_data, momentum_data):
    """
    Clasifica la tendencia en EARLY/MID/LATE phase
    
    Args:
        adx_data: Dict de calculate_adx_with_memory
        momentum_data: Dict de calculate_momentum_with_memory
        
    Returns:
        dict: {
            'phase': 'EARLY'|'MID'|'LATE',
            'entry_allowed': bool,
            'bonus_penalty': int,
            'reasons': []
        }
    """
    # Calcular ciclo de vida del ADX
    lifecycle = calculate_adx_lifecycle(
        adx_data['adx_current'],
        adx_data.get('adx_history', [])
    )
    
    # Detectar divergencia DI
    di_divergence = detect_di_divergence(
        adx_data['di_plus_current'],
        adx_data['di_minus_current'],
        adx_data.get('di_plus_prev', adx_data['di_plus_current']),
        adx_data.get('di_minus_prev', adx_data['di_minus_current'])
    )
    
    # Momentum acelerando
    mom_strong = momentum_data.get('consecutive_count', 0) >= 3
    mom_color = momentum_data.get('color_current', 'neutral')
    
    phase = 'MID'
    entry_allowed = True
    bonus_penalty = 0
    reasons = []
    
    # ===== FASE EARLY (ENTRAR) =====
    if lifecycle['phase'] == 'BIRTH':
        if di_divergence == 'DIVERGING' and mom_strong:
            phase = 'EARLY'
            entry_allowed = True
            bonus_penalty = +30
            reasons.append("ADX naciendo desde bajo")
            reasons.append("DI+ separándose de DI-")
            reasons.append("Momentum acelerando")
        elif di_divergence == 'DIVERGING':
            phase = 'EARLY'
            bonus_penalty = +20
            reasons.append("ADX naciendo")
    
    # ===== FASE MID (EVALUAR) =====
    elif lifecycle['phase'] == 'MATURE':
        phase = 'MID'
        entry_allowed = True
        bonus_penalty = 0
        
        if di_divergence == 'CONVERGING':
            bonus_penalty = -15
            reasons.append("Cuidado: DI convergiendo")
    
    # ===== FASE LATE (NO ENTRAR) =====
    elif lifecycle['phase'] == 'DECLINE':
        phase = 'LATE'
        entry_allowed = False
        bonus_penalty = -60  # PENALTY SEVERA
        reasons.append("ADX cayendo desde pico")
        reasons.append("Tendencia MURIENDO")
        
        if di_divergence == 'CONVERGING':
            bonus_penalty = -80
            reasons.append("DI+ y DI- convergiendo")
    
    # ===== FASE DORMANT (SIN TENDENCIA) =====
    elif lifecycle['phase'] == 'DORMANT':
        phase = 'DORMANT'
        entry_allowed = False
        bonus_penalty = -40
        reasons.append("Sin tendencia (ADX < 20)")
    
    return {
        'phase': phase,
        'entry_allowed': entry_allowed,
        'bonus_penalty': bonus_penalty,
        'reasons': reasons,
        'lifecycle': lifecycle,
        'di_divergence': di_divergence
    }

def analyze_lime(momentum_data, value_history=None):
    """
    Analiza momentum LIME (verde explosivo - alcista acelerando)
    
    LIME = momentum > 0 AND momentum > prev_momentum
    
    Args:
        momentum_data: Dict de calculate_momentum_with_memory
        value_history: Lista de valores históricos de momentum
        
    Returns:
        dict: Análisis completo del LIME
    """
    if momentum_data.get('color_current') != 'lime':
        return {'is_active': False}
    
    age = momentum_data.get('consecutive_count', 1)
    value = momentum_data.get('value', 0)
    
    # Calcular intensidad vs promedio histórico
    if value_history and len(value_history) >= 20:
        positive_values = [v for v in value_history if v > 0]
        if positive_values:
            avg_positive = sum(positive_values) / len(positive_values)
            intensity = value / avg_positive if avg_positive != 0 else 1.0
        else:
            intensity = 1.0
    else:
        intensity = 1.0
    
    # Clasificar fase del LIME
    if age <= 2:
        phase = 'NASCENT'
        bonus_penalty = +5
        risk_level = 'LOW'
        recommendation = 'Momentum naciendo, buen timing'
    elif age <= 5 and intensity > 1.5:
        phase = 'EXPLOSIVE'
        bonus_penalty = +15
        risk_level = 'LOW'
        recommendation = 'Clímax alcista, momento óptimo'
    elif age <= 8:
        phase = 'CLIMAX'
        bonus_penalty = +10
        risk_level = 'MEDIUM'
        recommendation = 'Pico máximo, cuidado agotamiento'
    else:  # age >= 9
        phase = 'EXHAUSTED'
        bonus_penalty = -20
        risk_level = 'HIGH'
        recommendation = 'Momentum agotándose, entrada tardía'
    
    # Calcular riesgo de agotamiento
    exhaustion_risk = min(100, age * 10 + (20 if intensity < 1.0 else 0))
    
    return {
        'is_active': True,
        'age': age,
        'intensity': round(intensity, 2),
        'intensity_vs_avg': f"+{((intensity-1)*100):.0f}%" if intensity >= 1 else f"{((intensity-1)*100):.0f}%",
        'phase': phase,
        'bonus_penalty': bonus_penalty,
        'risk_level': risk_level,
        'exhaustion_risk': exhaustion_risk,
        'recommendation': recommendation
    }
def analyze_green(momentum_data, value_history=None):
    """
    Analiza momentum GREEN (verde estable - alcista sostenido)
    
    GREEN = momentum > 0 AND momentum <= prev_momentum
    
    Args:
        momentum_data: Dict de calculate_momentum_with_memory
        
    Returns:
        dict: Análisis completo del GREEN
    """
    if momentum_data.get('color_current') != 'green':
        return {'is_active': False}
    
    age = momentum_data.get('consecutive_count', 1)
    value = momentum_data.get('value', 0)
    color_history = momentum_data.get('color_history', [])
    
    # Detectar si viene de LIME (desaceleración)
    came_from_lime = len(color_history) > 1 and color_history[-2] == 'lime'
    
    # Clasificar fase del GREEN
    if age <= 3:
        phase = 'EARLY'
        bonus_penalty = +8
        if came_from_lime:
            bonus_penalty = -5  # Desaceleración desde LIME
            phase = 'DECELERATION'
    elif age <= 8:
        phase = 'MATURE'
        bonus_penalty = +12
    elif age <= 12:
        phase = 'LATE'
        bonus_penalty = +5
    else:  # age >= 13
        phase = 'WEAKENING'
        bonus_penalty = -10
    
    # Calcular estabilidad (basado en valor vs promedio)
    if value_history and len(value_history) >= 10:
        recent_avg = sum(value_history[-10:]) / 10
        stability = 1 - abs(value - recent_avg) / (abs(recent_avg) + 0.001)
        stability = max(0, min(1, stability))
    else:
        stability = 0.7
    
    # Probabilidades de transición
    green_to_lime_prob = max(0, 40 - age * 5)  # Menor probabilidad con edad
    green_to_red_risk = min(100, age * 3 + (30 if stability < 0.5 else 0))
    
    return {
        'is_active': True,
        'age': age,
        'stability': round(stability, 2),
        'phase': phase,
        'bonus_penalty': bonus_penalty,
        'came_from_lime': came_from_lime,
        'green_to_lime_prob': green_to_lime_prob,
        'green_to_red_risk': green_to_red_risk
    }

"""

# Matriz de bonus/penalties por transición
TRANSITION_MATRIX = {
    # Transiciones ALCISTAS (ENTRAR)
    ('red', 'green'): {'bonus': +20, 'type': 'REVERSAL', 'meaning': 'Reversión alcista'},
    ('maroon', 'green'): {'bonus': +25, 'type': 'STRONG_REVERSAL', 'meaning': 'Rebote fuerte'},
    ('green', 'lime'): {'bonus': +15, 'type': 'ACCELERATION', 'meaning': 'Aceleración alcista'},
    ('maroon', 'red'): {'bonus': +10, 'type': 'EASING', 'meaning': 'Presión bajista cediendo'},
    
    # Transiciones BAJISTAS (SALIR o SHORT)
    ('lime', 'green'): {'bonus': -10, 'type': 'DECELERATION', 'meaning': 'Desaceleración alcista'},
    ('green', 'red'): {'bonus': -30, 'type': 'REVERSAL', 'meaning': 'Reversión bajista'},
    ('lime', 'maroon'): {'bonus': -40, 'type': 'COLLAPSE', 'meaning': 'Colapso desde clímax'},
    ('green', 'maroon'): {'bonus': -35, 'type': 'STRONG_REVERSAL', 'meaning': 'Caída abrupta'},
    
    # Transiciones SHORT FAVORABLE
    ('green', 'red'): {'bonus': +20, 'type': 'SHORT_ENTRY', 'meaning': 'Entrada SHORT'},
    ('lime', 'red'): {'bonus': +25, 'type': 'SHORT_ENTRY', 'meaning': 'Entrada SHORT después de clímax'},
    
    # Sin cambio (neutral)
    ('lime', 'lime'): {'bonus': 0, 'type': 'CONTINUATION', 'meaning': 'Continúa explosivo'},
    ('green', 'green'): {'bonus': 0, 'type': 'STABLE', 'meaning': 'Estable verde'},
    ('red', 'red'): {'bonus': 0, 'type': 'STABLE', 'meaning': 'Estable rojo'},
    ('maroon', 'maroon'): {'bonus': 0, 'type': 'CONTINUATION', 'meaning': 'Continúa colapso'},
}
def detect_color_transition(color_history, lookback=5):
    """
    Detecta transiciones entre colores
    
    Args:
        color_history: Lista de colores (últimas N velas)
        lookback: Cuántas velas atrás buscar transición
        
    Returns:
        dict: Información de la transición
    """
    if len(color_history) < 2:
        return {'transition_detected': False}
    
    current_color = color_history[-1]
    
    # Buscar última transición
    transition_age = 0
    from_color = current_color
    
    for i in range(len(color_history) - 2, max(-1, len(color_history) - lookback - 1), -1):
        if color_history[i] != current_color:
            from_color = color_history[i]
            transition_age = len(color_history) - 1 - i
            break
    
    if from_color == current_color:
        # Sin transición reciente
        return {
            'transition_detected': False,
            'current_color': current_color,
            'same_color_count': len(color_history)
        }
    
    # Buscar info de transición
    transition_key = (from_color, current_color)
    transition_info = TRANSITION_MATRIX.get(transition_key, {
        'bonus': 0,
        'type': 'UNKNOWN',
        'meaning': f'{from_color} → {current_color}'
    })
    
    # Ajustar bonus según edad de transición
    # Transición muy vieja (>3 velas) pierde fuerza
    age_factor = max(0.3, 1 - (transition_age - 1) * 0.2)
    adjusted_bonus = int(transition_info['bonus'] * age_factor)
    
    return {
        'transition_detected': True,
        'from_color': from_color,
        'to_color': current_color,
        'transition_age': transition_age,
        'transition_type': transition_info['type'],
        'meaning': transition_info['meaning'],
        'raw_bonus': transition_info['bonus'],
        'adjusted_bonus': adjusted_bonus
    }
def analyze_momentum_complete(momentum_data, value_history=None):
    """
    Análisis COMPLETO del momentum: color + fase + transición
    
    Returns:
        dict: Análisis integrado con bonus/penalty final
    """
    from brain_momentum_colors import (
        analyze_lime, analyze_green, analyze_maroon, analyze_red
    )
    
    current_color = momentum_data.get('color_current', 'neutral')
    color_history = momentum_data.get('color_history', [])
    
    # 1. Analizar color actual
    if current_color == 'lime':
        color_analysis = analyze_lime(momentum_data, value_history)
    elif current_color == 'green':
        color_analysis = analyze_green(momentum_data, value_history)
    elif current_color == 'maroon':
        color_analysis = analyze_maroon(momentum_data, value_history)
    elif current_color == 'red':
        color_analysis = analyze_red(momentum_data, value_history)
    else:
        color_analysis = {'is_active': False, 'bonus_penalty': 0}
    
    # 2. Detectar transición
    transition = detect_color_transition(color_history)
    
    # 3. Calcular bonus/penalty total
    total_bonus = color_analysis.get('bonus_penalty', 0)
    if transition.get('transition_detected'):
        total_bonus += transition.get('adjusted_bonus', 0)
    
    return {
        'current_color': current_color,
        'color_analysis': color_analysis,
        'transition': transition,
        'total_bonus_penalty': total_bonus
    }
