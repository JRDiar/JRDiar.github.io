    def detect_live_pattern(self, df, trade_data):
        """Detecta patrones en vivo durante el trade (1M granular con confirmación)"""
        from brain_config import ENABLE_REVERSAL_DETECTOR
        
        if not ENABLE_REVERSAL_DETECTOR:
            # TEMPORALMENTE DESACTIVADO por falsos positivos
            return None
            
        try:
            # 0. PERIODO DE GRACIA (5 min)
            # Evitar pánico en los primeros minutos de volatilidad
            now = datetime.now(ARGENTINA_TZ)
            entry_time = trade_data['entry_time']
            duration_min = (now - entry_time).total_seconds() / 60
            
            if duration_min < 5:
                return None # Demasiado temprano para patrones de reversión
            
            # 0.5 CONFIRMACIÓN ANTI-RUIDO (1M) - CON BLOQUEO
            # Exigir 2+ velas contra-tendencia antes de buscar patrones de reversión
            direction = trade_data.get('direction', 'BULLISH')
            allow_reversal_patterns = False  # Por defecto, NO buscar reversión
            
            if len(df) >= 3:
                last_3_closes = df['close'].values[-3:]
                last_3_opens = df['open'].values[-3:]
                
                if direction == 'BULLISH':
                    # Contar velas rojas (bearish)
                    red_count = sum(1 for c, o in zip(last_3_closes, last_3_opens) if c < o)
                    
                    if red_count >= 2:
                        print(f"  ⚠️ 1M REVERSIÓN CONFIRMADA: {red_count}/3 velas rojas - habilitando Double Top detection")
                        allow_reversal_patterns = True  # Sí buscar reversión
                    else:
                        print(f"  ✅ 1M RUIDO: Solo {red_count}/3 velas rojas - ignorando reversión, buscando continuación")
                        
                else:  # BEARISH
                    # Contar velas verdes (bullish)
                    green_count = sum(1 for c, o in zip(last_3_closes, last_3_opens) if c > o)
                    
                    if green_count >= 2:
                        print(f"  ⚠️ 1M REVERSIÓN CONFIRMADA: {green_count}/3 velas verdes - habilitando Double Bottom detection")
                        allow_reversal_patterns = True
                    else:
                        print(f"  ✅ 1M RUIDO: Solo {green_count}/3 velas verdes - ignorando reversión, buscando continuación")
                    
                # 1. Flag Pattern (Continuación) - SIEMPRE se busca
                # Lógica simplificada: Consolidación después de impulso
                closes = df['close'].values
                if len(closes) > 10:
                    recent_volatility = df['close'].tail(5).std()
                    prev_volatility = df['close'].iloc[-10:-5].std()
                    
                    if recent_volatility < prev_volatility * 0.5:
                        return {'pattern': 'FLAG', 'action': 'EXTEND_TARGET', 'confidence': 85}
                
                # 2. Double Top (Reversión) - SOLO si hay confirmación de 2+ velas
                if allow_reversal_patterns:
                    # PROTECCIÓN: Si estamos en ganancia significativa, ignorar reversiones menores
                    current_price = df['close'].iloc[-1]
                    entry_price = trade_data['entry_price']
                    direction_mult = 1 if direction == 'BULLISH' else -1
                    current_pnl_pct = ((current_price - entry_price) / entry_price) * 100 * direction_mult
                    
                    if current_pnl_pct > 2.0:
                        print(f"  🛡️ PROTECCIÓN PNL: +{current_pnl_pct:.2f}% - Ignorando reversiones menores (deja correr)")
                        allow_reversal_patterns = False  # Deshabilitar búsqueda de reversión
                        # Continuar buscando otros patrones (FLAG, etc.)
                    
                    if allow_reversal_patterns:
                        highs = df['high'].rolling(3).max()
                        peaks = highs[highs == highs.rolling(20).max()]
                        
                        if len(peaks) >= 2:
                            last_two = peaks[-2:]
                            # Tolerancia aumentada de 0.5% a 1.5% (menos falsos positivos)
                            # Optimización: Evita salir en consolidaciones normales
                            if abs(last_two.iloc[0] - last_two.iloc[1]) / last_two.iloc[0] < 0.015:
                                
                                # CONFIRMACIÓN: Momentum debe haber girado
                                # Si seguimos con momentum verde fuerte, es solo una pausa, no un top
                                mom_data = calculate_momentum_with_memory(df)
                                mom_color = mom_data['color_current']
                                
                                confirmed = False
                                if direction == 'BULLISH' and mom_color not in ['green', 'lime']:
                                    confirmed = True # Momentum perdió fuerza alcista
                                elif direction == 'BEARISH' and mom_color not in ['red', 'maroon']:
                                    confirmed = True # Momentum perdió fuerza bajista
                                    
                                if confirmed:
                                    print(f"  ⚠️ DOUBLE_TOP CONFIRMADO: Momentum débil ({mom_color}) + 2 peaks similares")
                                    return {'pattern': 'DOUBLE_TOP', 'action': 'EXIT_NOW', 'confidence': 90}
                                else:
                                    print(f"  ✅ Posible Double Top IGNORADO: Momentum aún fuerte ({mom_color})")
                else:
                    # Saltamos Double Top porque no hay suficientes velas de reversión
                    pass

        except Exception as e:
            # print(f"Error detecting live pattern: {e}")

    def check_exit_signals(self, df, trade_data, duration_min):
        """
        Verifica 5 señales de salida inteligente
        
        Returns:
            dict: {'count': int, 'active': [list]}
        """
        signals_active = []
        
        # SEÑAL 1: Desaceleración (velas más chicas)
        try:
            ranges = (df['high'] - df['low']).values
            avg_range_10 = ranges[-10:].mean()
            avg_range_3 = ranges[-3:].mean()
            
            if avg_range_3 < avg_range_10 * DECELERATION_THRESHOLD:
                signals_active.append("Desaceleración")
        except:
            pass
        
        # SEÑAL 2: Volumen bajando
        try:
            volumes = df['volume'].values
            avg_vol_10 = volumes[-10:].mean()
            avg_vol_3 = volumes[-3:].mean()
            
            if avg_vol_3 < avg_vol_10 * VOLUME_DECLINE_THRESHOLD:
                signals_active.append("Volumen bajo")
        except:
            pass
        
        # SEÑAL 3: Divergencia oculta REAL (Normalizada)
        try:
            prices = df['close'].values
            # Slope normalizado (%)
            price_slope = ((prices[-1] - prices[-5]) / prices[-5]) * 100
            
            # Calcular momentum simplificado
            ema = df['close'].ewm(span=10, adjust=False).mean().values
            momentum_slope = ((ema[-1] - ema[-5]) / ema[-5]) * 100
            
            direction = trade_data['direction']
            if direction == 'BULLISH':
                # Precio sube pero momentum baja/plano
                if price_slope > 0.5 and momentum_slope < 0.2:
                    signals_active.append("Divergencia")
            elif direction == 'BEARISH':
                # Precio baja pero momentum sube/plano
                if price_slope < -0.5 and momentum_slope > -0.2:
                    signals_active.append("Divergencia")
        except:
            pass

        # SEÑAL 6: ADX debilitando (Tendencia muriendo)
        try:
            adx_data = calculate_adx_with_memory(df, period=14, lookback=5)
            adx_current = adx_data['adx_current']
            adx_prev = adx_data['adx_memory'][-2] if len(adx_data['adx_memory']) >= 2 else adx_current
            
            # Si ADX baja >5 puntos y está bajo
            if adx_current < adx_prev - 5 and adx_current < 30:
                signals_active.append("ADX debilitando")
        except:
            pass
        
        # SEÑAL 4: Llegada zona crítica (cerca de target)
        try:
            current_price = df['close'].iloc[-1]
            target = trade_data['target_price']
            distance_pct = abs((target - current_price) / current_price * 100)
            
            if distance_pct < CRITICAL_ZONE_DISTANCE_PCT:
                signals_active.append("Zona crítica")
        except:
            pass
        
        # SEÑAL 5: Protección Buffett (pérdida temprana)
        if duration_min < BUFFETT_EARLY_EXIT_TIME_MIN:
            current_price = df['close'].iloc[-1]
            entry_price = trade_data['entry_price']
            direction = trade_data['direction']
            
            if direction == 'BULLISH':
                pnl_pct = ((current_price - entry_price) / entry_price) * 100
            else:
                pnl_pct = ((entry_price - current_price) / entry_price) * 100
            
            if pnl_pct < -BUFFETT_EARLY_EXIT_LOSS_PCT:
                signals_active.append("Buffett (setup falló)")
        
        return {
            'count': len(signals_active),
            'active': signals_active
        }
    
    def close_trade(self, symbol, reason):
        """Cierra trade y registra resultado"""
        # Thread-safe: Verificar y extraer trade
        with self.trades_lock:
            if symbol not in self.active_trades:
                return
            
            trade_data = self.active_trades[symbol]
        
        # Fetch precio actual (MARKET ORDER: usar precio inmediato)
        exit_price = fetch_current_price(symbol)
        if not exit_price:
            # Fallback si falla ticker
            ohlcv = fetch_ohlcv(symbol, '5m', 10)
            exit_price = ohlcv[-1][4] if ohlcv else trade_data['entry_price']
        
        # Calcular PnL final
        entry_price = trade_data['entry_price']
        direction = trade_data['direction']
        
        if direction == 'BULLISH':
            pnl_pct = ((exit_price - entry_price) / entry_price) * 100
        else:
            pnl_pct = ((entry_price - exit_price) / entry_price) * 100
        
        pnl_usd = (self.capital_per_trade * pnl_pct / 100) * SIMULATE_LEVERAGE
        
        # Duración
        now = datetime.now(ARGENTINA_TZ)
        duration_min = (now - trade_data['entry_time']).total_seconds() / 60
        
        # Log
        emoji = "✅" if pnl_pct > 0 else "❌"
        print(f"\n{'='*80}")
        print(f"{emoji} SALIDA: {symbol}")
        print(f"   PnL: {pnl_pct:+.2f}% | ${pnl_usd:+.2f}")
        print(f"   Duración: {duration_min:.1f} min")
        print(f"   Razón: {reason}")
        print(f"{'='*80}\n")
        
        # Sheets
        sheets_manager.write_trade_exit(
            symbol, exit_price, pnl_pct, pnl_usd,
            duration_min, reason
        )
        
        # BRAIN_SALIDAS
        if "Salida inteligente" in reason:
            senales = reason.split(": ")[1] if ": " in reason else reason
            try:
                sheets_manager.write_salida_inteligente(
                    symbol, exit_price, pnl_pct, pnl_usd,
                    duration_min, senales, reason
                )
            except Exception as e:
                print(f"⚠️ Error escribiendo salida inteligente en Sheets: {e}")
        
        # Telegram
        notify_exit(symbol, pnl_pct, pnl_usd, duration_min, reason)
        
        # Thread-safe: Remover de activos y guardar en historial
        with self.trades_lock:
            # Remover de activos
            del self.active_trades[symbol]
            
            # Guardar en historial
            trade_data['exit_price'] = exit_price
            trade_data['exit_time'] = now
            trade_data['pnl_pct'] = pnl_pct
            trade_data['pnl_usd'] = pnl_usd
            trade_data['duration_min'] = duration_min
            trade_data['exit_reason'] = reason
            self.trade_history.append(trade_data)

            # Guardar en DB (SQLite) para reportes
            try:
                db_manager.insert_trade(
                    entry_time=trade_data['entry_time'].strftime("%Y-%m-%d %H:%M:%S"),
                    exit_time=now.strftime("%Y-%m-%d %H:%M:%S"),
                    symbol=symbol,
                    direction=trade_data['direction'],
                    entry_price=trade_data['entry_price'],
                    exit_price=exit_price,
                    pnl_pct=pnl_pct,
                    pnl_usd=pnl_usd,
                    duration_min=duration_min,
                    exit_reason=reason,
                    score=trade_data.get('score', 0)
                )
            except Exception as e:
                print(f"Error al guardar trade en DB: {e}")
        
        return pnl_usd
    
    def _check_recent_critical_zone_exit(self, symbol):
        """
        Verifica si hubo salida reciente por zona crítica
        
        Args:
            symbol: Símbolo a verificar
            
        Returns:
            dict con info o None
        """
        with self.trades_lock:
            # Buscar en historial reciente (últimos 15 min)
            now = datetime.now(ARGENTINA_TZ)
            for trade in reversed(self.trade_history):  # Más recientes primero
                if trade['symbol'] != symbol:
                    continue
                
                exit_time = trade.get('exit_time')
                if not exit_time:
                    continue
                
                minutes_ago = (now - exit_time).total_seconds() / 60
                
                # Solo considerar últimos 15 min
                if minutes_ago > 15:
                    break
                
                # Verificar si salió por "zona crítica"
                exit_reason = trade.get('exit_reason', '').lower()
                if 'zona cr' in exit_reason or 'crítica' in exit_reason or 'critica' in exit_reason:
                    return {
                        'minutes_ago': minutes_ago,
                        'exit_reason': trade.get('exit_reason'),
                        'exit_time': exit_time
                    }
        
        return None

    def monitor_stop_loss_thread(self):
        """Thread que verifica stop loss cada 2 segundos (OPTIMIZADO BATCH)"""
        while True:
            if not self.active_trades:
                time.sleep(2)
                continue
            
            # Thread-safe: Copiar lista de trades activos
            with self.trades_lock:
                trades_snapshot = list(self.active_trades.items())
                active_symbols = list(self.active_trades.keys())
            
            if not active_symbols:
                time.sleep(2)
                continue
                
            # OPTIMIZACIÓN: Fetch BATCH de precios (1 request en lugar de N)
            current_prices = fetch_multiple_tickers(active_symbols)
            
            for symbol, trade_data in trades_snapshot:
                current_price = current_prices.get(symbol)
                
                # Fallback si falla el batch (raro)
                if not current_price:
                    current_price = fetch_current_price(symbol)
                
                if not current_price:
                    continue
                
                direction = trade_data['direction']
                
                # Verificar stop loss
                if (direction == 'BULLISH' and current_price <= trade_data['stop_loss']) or \
                   (direction == 'BEARISH' and current_price >= trade_data['stop_loss']):
                    print(f"🚨 STOP LOSS ACTIVADO: {symbol} (${current_price:.8f})")
                    self.close_trade(symbol, "Stop loss")
            
            time.sleep(2)  # 2 segundos para evitar Rate Limit (Code 510)
