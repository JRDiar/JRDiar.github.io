"""
BRAIN SCENARIOS - Proyección de Escenarios Múltiples
Anticipa plan A, plan B con probabilidades
"""
from brain_calculator import detect_support_resistance_with_memory

def project_scenarios(symbol, current_price, ladder_analysis):
    """
    Proyecta escenarios múltiples con probabilidades (ROADMAP)
    Usa niveles de 4H/1H para definir el "Mapa" y 5M para la ejecución.
    
    Returns:
        {
            'primary': {scenario A con mayor probabilidad},
            'secondary': {scenario B alternativo},
            'current_bias': BULLISH/BEARISH/NEUTRAL
        }
    """
    # 1. Extraer niveles del Time Ladder (El Mapa Completo)
    a4h = ladder_analysis.get('4h', {})
    a1h = ladder_analysis.get('1h', {})
    a5m = ladder_analysis.get('5m', {})
    
    # Niveles Mayores (Roadmap)
    res_4h = a4h.get('sr', {}).get('resistances', [])
    sup_4h = a4h.get('sr', {}).get('supports', [])
    
    # Niveles Intermedios
    res_1h = a1h.get('sr', {}).get('resistances', [])
    sup_1h = a1h.get('sr', {}).get('supports', [])
    
    # Niveles Ejecución
    sr_5m = a5m.get('sr', {})
    zigzag_5m = a5m.get('zigzag', {})
    vp_5m = a5m.get('volume_profile', {})
    
    # Consolidar Resistencias (Ordenadas ascendente)
    all_res = sorted(list(set(res_4h + res_1h + sr_5m.get('resistances', []))))
    resistances = [r for r in all_res if r > current_price * 1.001] # Filtrar muy cercanas
    
    # Consolidar Soportes (Ordenados descendente)
    all_sup = sorted(list(set(sup_4h + sup_1h + sr_5m.get('supports', []))), reverse=True)
    supports = [s for s in all_sup if s < current_price * 0.999]
    
    pivot_highs = zigzag_5m.get('pivot_highs', [])
    pivot_lows = zigzag_5m.get('pivot_lows', [])
    poc = vp_5m.get('poc_current', current_price)
    
    # Determinar bias actual (Roadmap Direction)
    direction = a4h.get('direction', 'NEUTRAL')
    
    if direction == 'BULLISH':
        return project_bullish_scenarios(
            current_price, resistances, pivot_highs, 
            supports, pivot_lows, poc
        )
    elif direction == 'BEARISH':
        return project_bearish_scenarios(
            current_price, resistances, pivot_highs,
            supports, pivot_lows, poc
        )
    else:
        return {
            'primary': {'probability': 0, 'action': 'WAIT'},
            'secondary': {'probability': 0, 'action': 'WAIT'},
            'current_bias': 'NEUTRAL'
        }

def project_bullish_scenarios(price, resistances, highs, supports, lows, poc):
    """Escenarios para contexto alcista"""
    
    # Resistencia más cercana
    nearest_res = resistances[0] if resistances else price * 1.05
    dist_to_res = ((nearest_res - price) / price) * 100
    
    # HH más cercano
    next_hh = max([h for h in highs if h > price], default=price * 1.08)
    dist_to_hh = ((next_hh - price) / price) * 100
    
    # HL más cercano (soporte)
    last_hl = max([l for l in lows if l < price], default=price * 0.95)
    dist_to_hl = ((price - last_hl) / price) * 100
    
    # ESCENARIO A (Primario): Breakout
    if dist_to_res < 2.0:  # Resistencia cerca
        entry_price = nearest_res
        target_price = next_hh
        stop_price = last_hl
        
        risk = abs(entry_price - stop_price)
        reward = abs(target_price - entry_price)
        rr = reward / risk if risk > 0 else 0
        
        primary = {
            'probability': 60,
            'trigger': f'Breakout ${nearest_res:.4f} con volumen >1.5x',
            'target': f'${next_hh:.4f} (+{dist_to_hh:.1f}%)',
            'stop': f'${last_hl:.4f} (bajo HL)',
            'action': 'WAIT_BREAKOUT',
            'condition': f'SI rompe ${nearest_res:.4f}',
            # Numerical Data
            'entry_price': entry_price,
            'target_price': target_price,
            'stop_price': stop_price,
            'rr_ratio': round(rr, 2)
        }
        
        # Secondary (Pullback)
        entry_sec = poc
        target_sec = nearest_res
        stop_sec = last_hl # Conservative
        risk_sec = abs(entry_sec - stop_sec)
        reward_sec = abs(target_sec - entry_sec)
        rr_sec = reward_sec / risk_sec if risk_sec > 0 else 0
        
        secondary = {
            'probability': 40,
            'trigger': f'Rechazo en ${nearest_res:.4f}',
            'target': f'Pullback ${poc:.4f} (POC)',
            'stop': 'Esperar confirmación',
            'action': 'WAIT_PULLBACK',
            'condition': f'SI rechaza ${nearest_res:.4f}',
            # Numerical Data
            'entry_price': entry_sec,
            'target_price': target_sec,
            'stop_price': stop_sec,
            'rr_ratio': round(rr_sec, 2)
        }
    else:  # Espacio limpio
        entry_price = price
        target_price = nearest_res
        stop_price = last_hl
        
        risk = abs(entry_price - stop_price)
        reward = abs(target_price - entry_price)
        rr = reward / risk if risk > 0 else 0
        
        primary = {
            'probability': 75,
            'trigger': 'Entrada inmediata (espacio limpio)',
            'target': f'${nearest_res:.4f} (+{dist_to_res:.1f}%)',
            'stop': f'${last_hl:.4f} (-{dist_to_hl:.1f}%)',
            'action': 'ENTER_NOW',
            'condition': 'Espacio >2% hasta resistencia',
            # Numerical Data
            'entry_price': entry_price,
            'target_price': target_price,
            'stop_price': stop_price,
            'rr_ratio': round(rr, 2)
        }
        
        secondary = {
            'probability': 25,
            'trigger': f'Pullback a ${poc:.4f}',
            'target': 'Re-entrada en soporte',
            'stop': f'Bajo ${last_hl:.4f}',
            'action': 'WAIT_PULLBACK',
            'condition': 'SI retrocede sin romper estructura',
            'entry_price': poc,
            'target_price': price,
            'stop_price': last_hl,
            'rr_ratio': 0 # Indeterminado
        }
    
    return {
        'primary': primary,
        'secondary': secondary,
        'current_bias': 'BULLISH'
    }
