    score = 100
    velocity_factor = min(velocity_pct / 0.30, 0.70)
    
    if dist_pct < 0.15:
        penalty = 60 * (1 - velocity_factor * 0.5)  # Velocity reduce hasta 50%

        penalty = 40 * (1 - velocity_factor)
    
    return {'score': score - penalty, ...}
