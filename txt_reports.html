    # 3. VOLUME PROFILE: "Al tope" (Precio > POC) pero con ESPACIO
    # Usamos 1H para confirmar que estamos en zona de valor alta
    vp_1h = a1h.get('volume_profile', {})
    poc_1h = vp_1h.get('poc_current', 0)
    profile_high = vp_1h.get('profile_high', 0) # DATOS DISPONIBLES
    current_price = a1h.get('current_price', 0)
    
    # A. Precio sobre POC
    cond_vp_poc = current_price > poc_1h if poc_1h > 0 else False
    
    # B. Distancia al Profile High (Evitar comprar el techo)
    cond_vp_safe = True
    vp_block_reason = ""
    
    if profile_high > 0:
        # Calcular distancia en %
        # Si precio < high, dist es positiva (ej: 3%)
        # Si precio > high, dist es negativa (ej: -1% -> Breakout)
        dist_to_high = ((profile_high - current_price) / current_price) * 100
        
        # Solo bloqueamos si estamos DENTRO del perfil (dist > 0) y MUY CERCA del techo (< 2%)

            cond_vp_safe = False
            vp_block_reason = f"Price too close to VP High ({dist_to_high:.1f}% < 2%)"
    
    cond_vp = cond_vp_poc and cond_vp_safe
    
    if cond_adx and cond_mom_4h and cond_mom_1h:

            return {
                'detected': True,

                'reason': f"🔥 MARKET FORCE: ADX 1H({adx_val_1h:.1f} ↗️) + DI Spread ({di_spread:.1f}) + Mom 4H/1H ✅"
            }
        elif not cond_vp_safe:
             # Bloqueo específico por VP High
             return {
                'detected': False, 
                'score': 0, 
                'reason': f"⛔ Market Force BLOCKED: {vp_block_reason}"
            }

    return {'detected': False, 'score': 0, 'reason': ""}
