def _validate_explosion_entry(self, symbol, direction, force_result, ladder_data):
    """
    v5.2.1: TIME LADDER - Retorna SCORE, no boolean
    
    Velocity reduce penalty de espacio, no la elimina.
    """
    # ... obtener datos de zonas ...
    
    # Obtener velocity del force_result (si existe)
    velocity_pct = abs(force_result.get('velocity', 0))
    
    # Calcular factor de reducción por velocity
    from brain_config import VELOCITY_PENALTY_REDUCTION_MAX, SPACE_BASE_PENALTY
    velocity_factor = min(velocity_pct / 0.30, VELOCITY_PENALTY_REDUCTION_MAX)
    
    score = 100  # Score base
    warnings = []
    blocks = []
    
    # === EVALUAR ESPACIO CON TIME LADDER ===
    
    if direction == 'BULLISH' and vah > 0:
        dist_to_vah = abs(current_price - vah) / current_price * 100
        
        if dist_to_vah < 0.3:  # Muy cerca = BLOCK (no negociable)
            blocks.append(f'En resistencia VAH ({dist_to_vah:.1f}%)')
        elif dist_to_vah < MIN_DIST_TO_RESISTANCE:
            # Aplicar penalty REDUCIDA por velocity
            base_penalty = SPACE_BASE_PENALTY
            actual_penalty = base_penalty * (1 - velocity_factor)
            score -= actual_penalty
            
            if velocity_factor > 0:
                warnings.append(f'Cerca VAH ({dist_to_vah:.1f}%) - Penalty reducida {velocity_factor*100:.0f}%')
            else:
                warnings.append(f'Cerca VAH ({dist_to_vah:.1f}%) - Penalty completa')
    
    # === DECISIÓN FINAL POR SCORE ===
    
    if blocks:
        return {'can_enter': False, 'score': 0, 'reason': blocks[0]}
    
    if score >= 60:  # Threshold para entrar
        return {
            'can_enter': True, 
            'score': score,
            'reason': f'Score {score:.0f} (velocity redujo penalty)',
            'validation_type': 'TIME_LADDER'
        }
    else:
        return {
            'can_enter': False,
            'score': score,
            'reason': f'Score {score:.0f} insuficiente (mín 60)',
            'validation_type': 'LOW_SCORE'
        }
