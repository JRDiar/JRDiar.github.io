"""
TXT UPLOADER - BOT TRADING PHASE 3.5
Sube archivos .txt de análisis TradingView a GitHub Pages.
Genera HTML minimalista optimizado para lectura de Claude.

Ejecutar: python txt_uploader.py
"""

import os
from pathlib import Path
from datetime import datetime
from git import Repo

# ============================================================================
# CONFIGURACIÓN
# ============================================================================

TXT_SOURCE_PATH = Path(r"C:\Users\Joel\Trading automatizado\analisis_tradingview")
GITHUB_REPO_PATH = Path(r"C:\Users\Joel\Trading automatizado\JRDiar.github.io")
HTML_OUTPUT = GITHUB_REPO_PATH / "txt_reports.html"

# ============================================================================
# FUNCIONES PRINCIPALES
# ============================================================================

def read_txt_files():
    """Lee todos los archivos .txt de la carpeta ordenados alfabéticamente."""
    print(f"\n📂 Leyendo archivos .txt desde: {TXT_SOURCE_PATH}")
    
    if not TXT_SOURCE_PATH.exists():
        print(f"  ❌ Carpeta no existe: {TXT_SOURCE_PATH}")
        return []
    
    # Buscar archivos .txt
    txt_files = sorted(TXT_SOURCE_PATH.glob("*.txt"))
    
    if not txt_files:
        print(f"  ⚠️  No se encontraron archivos .txt")
        return []
    
    print(f"  ✅ Encontrados {len(txt_files)} archivos .txt")
    
    # Leer contenido completo
    files_content = []
    for txt_path in txt_files:
        try:
            with open(txt_path, 'r', encoding='utf-8') as f:
                content = f.read()
            files_content.append({
                'filename': txt_path.name,
                'content': content,
                'chars': len(content)
            })
            print(f"  ✅ {txt_path.name}: {len(content):,} caracteres")
        except Exception as e:
            print(f"  ❌ Error leyendo {txt_path.name}: {e}")
    
    return files_content


def generate_html(files_content):
    """Genera HTML minimalista con contenido de archivos .txt."""
    print(f"\n🌐 Generando HTML: {HTML_OUTPUT}")
    
    timestamp = datetime.now().strftime('%Y-%m-%d %H:%M:%S')
    total_chars = sum(f['chars'] for f in files_content)
    
    # Construir HTML
    html_parts = []
    
    # Header
    html_parts.append('<!DOCTYPE html>')
    html_parts.append('<html lang="es">')
    html_parts.append('<head>')
    html_parts.append('  <meta charset="UTF-8">')
    html_parts.append('  <meta name="viewport" content="width=device-width, initial-scale=1.0">')
    html_parts.append('  <title>TXT Reports - Bot Phase 3.5</title>')
    html_parts.append('  <style>')
    html_parts.append('    body { font-family: monospace; margin: 20px; background: #1e1e1e; color: #d4d4d4; }')
    html_parts.append('    h1 { color: #4ec9b0; }')
    html_parts.append('    .meta { color: #808080; margin-bottom: 20px; }')
    html_parts.append('    .separator { color: #569cd6; font-weight: bold; margin: 30px 0 10px 0; }')
    html_parts.append('    pre { background: #252526; padding: 15px; border-left: 3px solid #007acc; white-space: pre-wrap; word-wrap: break-word; }')
    html_parts.append('    .footer { margin-top: 30px; padding-top: 20px; border-top: 1px solid #404040; color: #808080; }')
    html_parts.append('  </style>')
    html_parts.append('</head>')
    html_parts.append('<body>')
    
    # Título
    html_parts.append('  <h1>📋 Comunicación Claude - Bot Phase 3.5</h1>')
    html_parts.append(f'  <div class="meta">Última actualización: {timestamp}</div>')
    
    # Contenido de archivos
    if files_content:
        for file_data in files_content:
            html_parts.append(f'  <div class="separator">{"=" * 80}</div>')
            html_parts.append(f'  <div class="separator">{file_data["filename"]}</div>')
            html_parts.append(f'  <div class="separator">{"=" * 80}</div>')
            html_parts.append('  <pre>')
            # Escapar HTML básico
            safe_content = file_data['content'].replace('&', '&amp;').replace('<', '&lt;').replace('>', '&gt;')
            html_parts.append(safe_content)
            html_parts.append('  </pre>')
    else:
        html_parts.append('  <div class="meta">⚠️ Sin archivos para procesar</div>')
    
    # Footer
    html_parts.append('  <div class="footer">')
    html_parts.append(f'    Total archivos: {len(files_content)}<br>')
    html_parts.append(f'    Total caracteres: {total_chars:,}<br>')
    html_parts.append(f'    Generado: {timestamp}')
    html_parts.append('  </div>')
    
    html_parts.append('</body>')
    html_parts.append('</html>')
    
    # Guardar archivo
    html_content = '\n'.join(html_parts)
    
    try:
        with open(HTML_OUTPUT, 'w', encoding='utf-8') as f:
            f.write(html_content)
        print(f"  ✅ HTML generado: {len(html_content):,} caracteres")
        print(f"  📊 Total archivos procesados: {len(files_content)}")
        print(f"  📊 Total caracteres contenido: {total_chars:,}")
        return True
    except Exception as e:
        print(f"  ❌ Error guardando HTML: {e}")
        return False


def push_to_github():
    """Pushea cambios a GitHub Pages."""
    print(f"\n📤 Pusheando a GitHub Pages...")
    
    try:
        repo = Repo(GITHUB_REPO_PATH)
        
        # Add txt_reports.html
        repo.index.add(['txt_reports.html'])
        print(f"  ✅ git add txt_reports.html")
        
        # Commit
        timestamp = datetime.now().strftime('%Y-%m-%d %H:%M:%S')
        commit_message = f"Update txt reports - {timestamp}"
        repo.index.commit(commit_message)
        print(f"  ✅ git commit: {commit_message}")
        
        # Push
        origin = repo.remote(name='origin')
        origin.push('main')
        print(f"  ✅ git push completado")
        
        # URL con timestamp para cachebust
        url_timestamp = datetime.now().strftime('%Y%m%d%H%M%S')
        url = f"https://jrdiar.github.io/txt_reports.html?v={url_timestamp}"
        print(f"\n🌐 Reportes publicados en: {url}")
        print(f"   (Esperar 1-2 minutos para que GitHub Pages actualice)")
        
        return True
        
    except Exception as e:
        print(f"  ❌ Error pusheando a GitHub: {e}")
        print(f"  ℹ️  El HTML fue generado localmente pero no se subió a GitHub")
        return False


# ============================================================================
# MAIN
# ============================================================================

def main():
    """Función principal."""
    print("=" * 80)
    print("TXT UPLOADER - BOT TRADING PHASE 3.5")
    print("=" * 80)
    
    # Paso 1: Leer archivos .txt
    files_content = read_txt_files()
    
    # Paso 2: Generar HTML
    html_success = generate_html(files_content)
    
    if not html_success:
        print("\n❌ Error generando HTML. Abortando.")
        return
    
    # Paso 3: Subir a GitHub
    git_success = push_to_github()
    
    # Resumen final
    print("\n" + "=" * 80)
    if git_success:
        print("✅ PROCESO COMPLETO")
        print(f"📄 HTML generado: {HTML_OUTPUT}")
        print(f"🌐 URL: https://jrdiar.github.io/txt_reports.html")
    else:
        print("⚠️  PROCESO PARCIAL")
        print(f"📄 HTML generado: {HTML_OUTPUT}")
        print(f"❌ No se pudo subir a GitHub")
    print("=" * 80)


if __name__ == "__main__":
    main()
