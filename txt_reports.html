    def check_exit_signals(self, df, trade_data, duration_min):
        """
        Verifica 5 señales de salida inteligente
        
        Returns:
            dict: {'count': int, 'active': [list]}
        """
        signals_active = []
        
        # SEÑAL 1: Desaceleración (velas más chicas)
        try:
            ranges = (df['high'] - df['low']).values
            avg_range_10 = ranges[-10:].mean()
            avg_range_3 = ranges[-3:].mean()
            
            if avg_range_3 < avg_range_10 * DECELERATION_THRESHOLD:
                signals_active.append("Desaceleración")
        except:
            pass
        
        # SEÑAL 2: Volumen bajando
        try:
            volumes = df['volume'].values
            avg_vol_10 = volumes[-10:].mean()
            avg_vol_3 = volumes[-3:].mean()
            
            if avg_vol_3 < avg_vol_10 * VOLUME_DECLINE_THRESHOLD:
                signals_active.append("Volumen bajo")
        except:
            pass
        
        # SEÑAL 3: Divergencia oculta (precio sube pero momentum flat)
        try:
            prices = df['close'].values
            price_slope = (prices[-1] - prices[-5]) / 5
            
            # Calcular momentum simplificado
            ema = df['close'].ewm(span=10, adjust=False).mean().values
            momentum_slope = (ema[-1] - ema[-5]) / 5
            
            direction = trade_data['direction']
            if direction == 'BULLISH' and price_slope > 0 and abs(momentum_slope) < 0.001:
                signals_active.append("Divergencia")
            elif direction == 'BEARISH' and price_slope < 0 and abs(momentum_slope) < 0.001:
                signals_active.append("Divergencia")
        except:
            pass
        
        # SEÑAL 4: Llegada zona crítica (cerca de target)
        try:
            current_price = df['close'].iloc[-1]
            target = trade_data['target_price']
            distance_pct = abs((target - current_price) / current_price * 100)
            
            if distance_pct < CRITICAL_ZONE_DISTANCE_PCT:
                signals_active.append("Zona crítica")
        except:
            pass
        
        # SEÑAL 5: Protección Buffett (pérdida temprana)
        if duration_min < BUFFETT_EARLY_EXIT_TIME_MIN:
            current_price = df['close'].iloc[-1]
            entry_price = trade_data['entry_price']
            direction = trade_data['direction']
            
            if direction == 'BULLISH':
                pnl_pct = ((current_price - entry_price) / entry_price) * 100
            else:
                pnl_pct = ((entry_price - current_price) / entry_price) * 100
            
            if pnl_pct < -BUFFETT_EARLY_EXIT_LOSS_PCT:
                signals_active.append("Buffett (setup falló)")
        
        return {
            'count': len(signals_active),
            'active': signals_active
        }

    def monitor_stop_loss_thread(self):
        """Thread que verifica stop loss cada 2 segundos"""
        while True:
            if not self.active_trades:
                time.sleep(2)
                continue
            
            # Thread-safe: Copiar lista de trades activos
            with self.trades_lock:
                trades_snapshot = list(self.active_trades.items())
            
            for symbol, trade_data in trades_snapshot:
                # Fetch precio actual (solo último dato)
                ohlcv = fetch_ohlcv(symbol, '5m', 2)
                if not ohlcv:
                    continue
                
                current_price = ohlcv[-1][4]
                direction = trade_data['direction']
                
                # Verificar stop loss
                if (direction == 'BULLISH' and current_price <= trade_data['stop_loss']) or \
                   (direction == 'BEARISH' and current_price >= trade_data['stop_loss']):
                    print(f"🚨 STOP LOSS ACTIVADO: {symbol} (${current_price:.8f})")
                    self.close_trade(symbol, "Stop loss")
            
            time.sleep(2)
    
    def start_stop_loss_monitor(self):
        """Inicia thread de monitoreo stop loss"""
        if not self.stop_loss_thread or not self.stop_loss_thread.is_alive():
            self.stop_loss_thread = threading.Thread(target=self.monitor_stop_loss_thread, daemon=True)
            self.stop_loss_thread.start()
            print("✅ Thread stop loss iniciado (verificación cada 2s)")
    
    def reanalyze_breakeven_exits(self):
        """Re-analiza monedas que salieron en breakeven"""
        if not self.breakeven_exited_symbols:
            return
        
        now = datetime.now(ARGENTINA_TZ)
        symbols_to_remove = []
        
        for symbol, exit_time in self.breakeven_exited_symbols.items():
            # Solo re-analizar después de 5 minutos
            if (now - exit_time).total_seconds() < 300:
                continue
            
            print(f"🔄 Re-analizando {symbol} (salió en breakeven)...")
            
            from brain_analyzer import analyze_coin_complete
            analysis = analyze_coin_complete(symbol)
            
            if analysis and analysis['viable']:
                print(f"✅ {symbol} sigue viable - agregando a watchlist")
                watchlist_manager.add_to_watchlist(symbol, analysis)
            else:
                print(f"❌ {symbol} ya no viable")
            
            symbols_to_remove.append(symbol)
        
        # Limpiar procesados
        for symbol in symbols_to_remove:
            del self.breakeven_exited_symbols[symbol]
