# VELOCITY DETECTION SYSTEM (Time Ladder v5.2.5)
# ===================================================================
import time
class PriceMemory:
    """Memoria en RAM de precios para calcular velocity real."""
    
    def __init__(self):
        self.history = {}  # {symbol: [(timestamp, price), ...]}
    
    def update(self, symbol, current_price):
        """Agregar precio actual a memoria"""
        timestamp = time.time()
        
        if symbol not in self.history:
            self.history[symbol] = []
        
        self.history[symbol].append((timestamp, current_price))
        
        # Mantener solo últimos 5 minutos
        cutoff = timestamp - (5 * 60)
        self.history[symbol] = [
            (t, p) for t, p in self.history[symbol] if t > cutoff
        ]
    
    def get_velocity(self, symbol, minutes=3):
        """Calcula cambio % en últimos N minutos."""
        if symbol not in self.history or len(self.history[symbol]) < 2:
            return 0.0
        
        now = time.time()
        target_time = now - (minutes * 60)
        
        current_price = self.history[symbol][-1][1]
        
        past_price = None
        for timestamp, price in self.history[symbol]:
            if timestamp >= target_time:
                past_price = price
                break
        
        if past_price is None or past_price == 0:
            return 0.0
        
        velocity = ((current_price - past_price) / past_price) * 100
        return velocity
# Instancia global
_price_memory = PriceMemory()
def calculate_velocity_3m(symbol):
    """Función helper para usar desde sniper."""
    return _price_memory.get_velocity(symbol, minutes=3)
def update_price_memory(symbol, current_price):
    """Actualizar memoria con precio actual."""
    _price_memory.update(symbol, current_price)
