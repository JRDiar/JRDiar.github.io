    def _frozen_calculate_zigzag(self, df, percent=None, threshold_pct=5):

        # Aceptar ambos nombres de parámetro

                    direction *= -1
                    pivots.append(i)
                    values.append(close[i])
                    last_pivot = close[i]

        if len(values) >= 2:
            last_high = None
            last_low = None
            for i in range(1, len(values)):
                curr_val = values[i]
                prev_val = values[i-1]
                is_high = False
                is_low = False
                if curr_val > prev_val:
                    is_high = True
                else:
                    is_low = True
                type_str = ""
                if is_high:
                    if last_high is None: type_str = "H"
                    elif curr_val > last_high: type_str = "HH"
                    elif curr_val < last_high: type_str = "LH"
                    else: type_str = "DH"
                    last_high = curr_val
                elif is_low:

                    elif curr_val > last_low: type_str = "HL"
                    elif curr_val < last_low: type_str = "LL"
                    else: type_str = "DL"
                    last_low = curr_val

        last_structure = "Indefinido"

            last_structure = f"{pivot_types[-2]} -> {pivot_types[-1]}"

    def _frozen_calculate_adx(self, df, period=14):
        """COPIA FIEL de brain_calculator.py - Wilder's Smoothing"""
        # Nota: Simplificado para el auditor, pero mantiene la matemática exacta del cálculo final
        n = len(df)
        if n < period + 1:
            return {'adx_current': 0}
            
        high = df['high'].values
        low = df['low'].values
        close = df['close'].values
        
        tr = np.zeros(n)
        dm_plus = np.zeros(n)
        dm_minus = np.zeros(n)
        
        for i in range(1, n):
            tr[i] = max(high[i] - low[i], abs(high[i] - close[i-1]), abs(low[i] - close[i-1]))
            up = high[i] - high[i-1]
            down = low[i-1] - low[i]
            dm_plus[i] = up if (up > down and up > 0) else 0
            dm_minus[i] = down if (down > up and down > 0) else 0
            
        # Wilder's smoothing
        atr = np.zeros(n)
        smooth_plus = np.zeros(n)
        smooth_minus = np.zeros(n)
        
        # Init
        atr[period] = tr[1:period+1].mean()
        smooth_plus[period] = dm_plus[1:period+1].mean()
        smooth_minus[period] = dm_minus[1:period+1].mean()
        
        for i in range(period+1, n):
            atr[i] = (atr[i-1] * (period-1) + tr[i]) / period
            smooth_plus[i] = (smooth_plus[i-1] * (period-1) + dm_plus[i]) / period
            smooth_minus[i] = (smooth_minus[i-1] * (period-1) + dm_minus[i]) / period
            
        di_plus = np.zeros(n)
        di_minus = np.zeros(n)
        dx = np.zeros(n)
        
        with np.errstate(divide='ignore', invalid='ignore'):
            mask = atr != 0
            di_plus[mask] = 100 * smooth_plus[mask] / atr[mask]
            di_minus[mask] = 100 * smooth_minus[mask] / atr[mask]
            dx_mask = (di_plus + di_minus) != 0
            dx[dx_mask] = 100 * np.abs(di_plus[dx_mask] - di_minus[dx_mask]) / (di_plus[dx_mask] + di_minus[dx_mask])
            
        # ADX Smoothing
        adx = np.zeros(n)
        if n > 2*period:
            adx[2*period-1] = dx[period:2*period].mean()
            for i in range(2*period, n):
                adx[i] = (adx[i-1] * (period-1) + dx[i]) / period
                
        return {'adx_current': float(adx[-1]) if not np.isnan(adx[-1]) else 0}
