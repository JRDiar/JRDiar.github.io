"""
Test Pattern Matcher contra telemetría histórica
"""
import sys
import os
import json
import sqlite3

# Añadir root project path para imports
sys.path.append(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

from brain_pattern_matcher import pattern_matcher
from brain_config import DATA_DIR

# Ruta a la base de datos de telemetría
DB_PATH = os.path.join(DATA_DIR, 'brain_rechazos.db')

def run_test():
    if not os.path.exists(DB_PATH):
        print(f"❌ Error: No se encuentra la base de datos en {DB_PATH}")
        return

    print(f"\n🧪 TESTING PATTERN MATCHER (desde DB)")
    print(f"📂 Base de datos: {DB_PATH}")
    
    try:
        conn = sqlite3.connect(DB_PATH)
        conn.row_factory = sqlite3.Row
        
        # Consultar últimos 50 registros con JSON válido
        cursor = conn.execute("""
            SELECT symbol, ladder_json, timestamp, decision 
            FROM telemetry_ladder 
            ORDER BY id DESC 
            LIMIT 50
        """)
        rows = cursor.fetchall()
        
        print(f"📊 Registros analizados: {len(rows)}")
        print("=" * 60)

        matches = []

        for row in rows:
            symbol = row['symbol']
            try:
                ladder_data = json.loads(row['ladder_json'])
            except json.JSONDecodeError:
                continue

            # Simular setup con datos históricos
            # Nota: ladder_json a veces guarda estructura completa o por keys
            # Intentamos reconstruir el setup esperado por el matcher
            
            current_setup = {
                '1h': ladder_data.get('1h', {}),
                '15m': ladder_data.get('15m', {}),
                '5m': ladder_data.get('5m', {}),
                'grid': ladder_data.get('grid', {}) # Si existe en el JSON
            }
            
            # Si no hay datos de Grid en el JSON (versiones viejas), intentar inferir o saltar
            if not current_setup['grid']:
                # Intento de compatibilidad: buscar en 5m si hay datos de grid anidados
                if 'grid_data' in ladder_data:
                    current_setup['grid'] = ladder_data['grid_data']
            
            match = pattern_matcher.find_best_match(symbol, current_setup)
            
            if match['similarity'] > 70:
                matches.append({
                    'symbol': symbol,
                    'time': row['timestamp'],
                    'similarity': match['similarity'],
                    'dna': match['dna_name']
                })
                print(f"[OK] {symbol} [{row['timestamp']}]: {match['similarity']:.1f}% match con {match['dna_name']}")

        # INYECCIÓN DE PRUEBA SINTÉTICA (Alineada con ADN BEAT/USDT Real)
        # NOTA: El ADN de BEAT tiene Momentum=REVERSIÓN y ADX=MODERADO/FUERTE.
        # Simulamos ese comportamiento exacto para probar el motor de comparación.
        synthetic_match = {
            '15m': {
                'adx': {
                    'adx_current': 22.0, 
                    'adx_trend': 'estable',
                    'behavior': 'MODERADO' # Match DNA
                },
                'momentum': {
                    'color_current': 'lime', 
                    'consecutive_count': 1, 
                    'pattern': 'REVERSIÓN_RECIENTE' # Match DNA
                },
                'swing': {'size_pct': 0.8, 'direction': 'UP'}
            },
            '5m': {
                'adx': {
                    'adx_current': 32.0, 
                    'adx_trend': 'estable',
                    'behavior': 'FUERTE_ESTABLE' # Match DNA
                },
                'momentum': {
                    'color_current': 'lime', 
                    'consecutive_count': 2,
                    'pattern': 'REVERSIÓN_RECIENTE'
                },
                'grid': {'ancho_pct': 26.0, 'direction': 'BULLISH'},
                'swing': {'direction': 'UP'}
            },
            '1h': {
                'adx': {'adx_current': 20.0, 'behavior': 'MODERADO'},
                'momentum': {'color_current': 'lime', 'pattern': 'REVERSIÓN_RECIENTE'},
                'swing': {'direction': 'UP'}
            },
            'grid': {'direction': 'BULLISH', 'ancho_pct': 26.0}
        }
        
        # Test unitario aislado
        print("\n[TEST] TEST SINTÉTICO (Validación de Lógica):")
        match_synth = pattern_matcher.find_best_match("SYNTHETIC_TEST", synthetic_match)
        if match_synth['similarity'] > 80:
             print(f"[OK] LÓGICA CONFIRMADA: Setup sintético logró {match_synth['similarity']:.1f}% match con {match_synth['dna_name']}")
        else:
             print(f"[FAIL] FALLO SINTÉTICO: Score bajo ({match_synth['similarity']:.1f}%)")
             
        # También probamos el "Fallback" (sin behaviors explícitos)
        synthetic_match_fallback = {
            '15m': {
                'adx': {'adx_current': 24.0, 'adx_trend': 'creciendo'}, # Should detect 'EMERGIENDO'
                'momentum': {'color_current': 'lime', 'consecutive_count': 5},
                'swing': {'direction': 'UP'}
            },
            '5m': {
                'adx': {'adx_current': 24.0},
                'momentum': {'color_current': 'lime'},
                'grid': {'direction': 'BULLISH'}
            },
            '1h': {'adx': {'adx_current': 24.0}, 'momentum': {'color_current': 'lime'}},
            'grid': {'direction': 'BULLISH'}
        }
        print("\n[TEST] TEST SINTÉTICO 2 (Fallback Logic):")
        match_fallback = pattern_matcher.find_best_match("SYNTHETIC_FALLBACK", synthetic_match_fallback)
        print(f"   Score Fallback: {match_fallback['similarity']:.1f}%")

        print("=" * 60)
        
    except Exception as e:
        print(f"[ERROR] Error durante el test: {e}")
    finally:
        if 'conn' in locals():
            conn.close()

if __name__ == "__main__":
    run_test()
