def validate_adx_separation(ladder, direction, timeframe='5m'):
    """
    Valida ADX usando RANGOS GRADUALES (Time Ladder).
    
    Filosofía:
    - ADX >40: FUERZA EXTREMA (bypass todo)
    - ADX 30-40: FUERZA ALTA (bonus alto)
    - ADX 25-30: FUERZA MODERADA (bonus medio)
    - ADX 20-25: TENDENCIA EMERGENTE (bonus bajo)
    - ADX <20: SIN FUERZA (penalty)
    """
    result = {
        'separated': False,
        'score_adjustment': 0,
        'reason': '',
        'adx_value': 0,
        'di_spread': 0,
        'is_explosive': False,
        'force_state': 'CHOPPY'
    }
    
    tf_data = ladder.get(timeframe, {})
    adx_data = tf_data.get('adx', {})
    
    adx_val = adx_data.get('adx_current', 0)
    di_plus = adx_data.get('di_plus_current', 0)
    di_minus = adx_data.get('di_minus_current', 0)
    adx_trend = adx_data.get('adx_trend', 'estable')
    
    result['adx_value'] = adx_val
    result['di_spread'] = abs(di_plus - di_minus)
    
    # 1. CLASIFICACIÓN POR RANGOS
    if adx_val >= 40:
        result['force_state'] = 'EXPLOSIVE'
        result['is_explosive'] = True
        base_score = 50
    elif adx_val >= 30:
        result['force_state'] = 'STRONG'
        base_score = 35
    elif adx_val >= 25:
        result['force_state'] = 'MODERATE'
        base_score = 25
    elif adx_val >= 20:
        result['force_state'] = 'EMERGING'
        base_score = 15
    else:
        result['force_state'] = 'CHOPPY'
        base_score = -30
    
    # 2. VALIDACIÓN DIRECCIONAL
    if direction == 'BULLISH':
        is_aligned = di_plus > di_minus
        spread_ok = di_plus > di_minus + 5
    else:
        is_aligned = di_minus > di_plus
        spread_ok = di_minus > di_plus + 5
    
    # 3. CÁLCULO FINAL
    if not is_aligned:
        # Dirección incorrecta = penalty duro
        result['separated'] = False
        result['score_adjustment'] = -100
        result['reason'] = f"⛔ DI contrario (ADX {adx_val:.1f})"
        result['is_explosive'] = False
        return result
    
    # 4. BONUSES POR CALIDAD
    final_score = base_score
    
    # Bonus separación
    if spread_ok:
        final_score += 10
        result['separated'] = True
        result['reason'] = f"✅ ADX {adx_val:.1f} ({result['force_state']}) + Spread OK"
    else:
        result['separated'] = False
        final_score -= 5
        result['reason'] = f"⚠️ ADX {adx_val:.1f} ({result['force_state']}) sin separación"
    
    # Bonus tendencia
    if adx_trend == 'creciendo':
        final_score += 5
    
    result['score_adjustment'] = final_score
    return result
