import pandas as pd
import numpy as np
from brain_config import GRID_LOOKBACK, GRID_COMPRESSION_BASE, GRID_COMPRESSION_MULTIPLIERS
from brain_calculator import calculate_atr
class MarketGridNavigator:
    def navigate(self, symbol, df_5m, ladder):
        if df_5m is None or len(df_5m) < 30:
            return {'decision': 'BLOCK', 'reason': 'No Data', 'score': -100, 'grid_data': {}}
        
        # 1. Grid & Volatilidad
        grid = self._build_grid(df_5m)
        vol_class = self._classify_volatility(symbol, grid['atr_pct'])
        grid['vol_class'] = vol_class
        grid['threshold'] = GRID_COMPRESSION_BASE * GRID_COMPRESSION_MULTIPLIERS.get(vol_class, 1.0)
        
        # 2. Puntero & Fuerza
        pointer = self._calc_pointer(df_5m, ladder)
        force = self._get_force(ladder)
        
        return self._decide(grid, pointer, force, df_5m)
    
    def _build_grid(self, df):
        ceiling = df['high'].rolling(GRID_LOOKBACK).max().iloc[-1]
        floor = df['low'].rolling(GRID_LOOKBACK).min().iloc[-1]
        
        atr_data = calculate_atr(df, period=14)
        atr = atr_data.get('atr_current', 0)
        close = df['close'].iloc[-1]
        
        width_atr = (ceiling - floor) / atr if atr > 0 else 0
        atr_pct = (atr / close) * 100 if close > 0 else 0
        
        return {
            'ceiling': ceiling, 'floor': floor, 
            'width_atr': width_atr, 'atr': atr, 'atr_pct': atr_pct
        }
    def _classify_volatility(self, symbol, atr_pct):
        # Clasificación simple basada en ATR % recent
        if atr_pct > 1.0: return 'high_vol'
        if atr_pct < 0.3: return 'low_vol'
        return 'medium_vol'
    def _calc_pointer(self, df, ladder):
        # (Misma lógica v6.4: Velocity + Momentum 15m)
        closes = df['close'].values
        velocity = closes[-1] - closes[-3] if len(closes) > 3 else 0
        mom_color = ladder.get('15m', {}).get('momentum', {}).get('color_current', 'neutral')
        
        direction = "NEUTRAL"
        if mom_color in ['green', 'lime']: direction = "UP"
        elif mom_color in ['red', 'maroon']: direction = "DOWN"
            
        return {'direction': direction, 'velocity': velocity}
    def _get_force(self, ladder):
        adx_15m = ladder.get('15m', {}).get('adx', {})
        return {'adx': adx_15m.get('adx_current', 0)}
    def _decide(self, grid, pointer, force, df):
        score = 0
        reasons = []
        
        # 1. COMPRESIÓN DINÁMICA
        if grid['width_atr'] < grid['threshold']:
            score -= 80 # Soft Block masivo
            reasons.append(f"⛔ CAJA LATERAL: {grid['width_atr']:.1f} < {grid['threshold']:.1f} ATRs")
        
        # 2. ALINEACIÓN PUNTERO
        current_price = df['close'].iloc[-1]
        if pointer['direction'] == 'UP':
            if current_price >= grid['ceiling'] * 0.998:
                score += 20; reasons.append("🔨 Grid: Rompiendo Techo")
            else:
                score += 10
        elif pointer['direction'] == 'DOWN':
            if current_price <= grid['floor'] * 1.002:
                score += 20; reasons.append("🔨 Grid: Rompiendo Piso")
            else:
                score += 10
                
        # 3. CONTRADICCIÓN (Penalización)
        if (pointer['velocity'] > 0 and pointer['direction'] == 'DOWN') or \
           (pointer['velocity'] < 0 and pointer['direction'] == 'UP'):
            score -= 30
            reasons.append("⚠️ Grid: Contradicción Puntero/Velocidad")
        # Decision
        decision = "GO" if score >= 30 else "BLOCK" if score <= -50 else "WAIT"
        
        return {
            'decision': decision, 'score': score, 
            'reasons': reasons, 'grid_data': grid
        }
