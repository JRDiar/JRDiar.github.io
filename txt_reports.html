def fetch_ohlcv(symbol, timeframe='5m', limit=100):
    """
    Obtiene datos OHLCV para símbolo y timeframe
    
    Args:
        symbol: Símbolo (ej: 'BTC/USDT:USDT')
        timeframe: Timeframe (1d, 4h, 1h, 15m, 5m)
        limit: Número de velas
    
    Returns:
        list: OHLCV [[timestamp, open, high, low, close, volume], ...]
        None: Si error
    """
    try:
        ohlcv = exchange.fetch_ohlcv(symbol, timeframe, limit=limit)
        
        if not ohlcv or len(ohlcv) < 10:
            return None
            
        return ohlcv
        
    except Exception as e:
        print(f"❌ Error fetch_ohlcv {symbol} {timeframe}: {e}")
        return None


def fetch_markets():
    """
    Obtiene lista de mercados disponibles
    
    Returns:
        dict: Markets {symbol: market_info}
        None: Si error
    """
    try:
        markets = exchange.load_markets()
        return markets
    except Exception as e:
        print(f"❌ Error fetch_markets: {e}")

def get_futures_symbols():
    """
    Obtiene lista de símbolos futuros USDT
    
    Returns:
        list: ['BTC/USDT:USDT', 'ETH/USDT:USDT', ...]
        None: Si error
    """
    try:
        markets = fetch_markets()
        if not markets:
            return None
        
        # Filtrar solo futuros USDT
        symbols = [
            symbol for symbol, market in markets.items()
            if market.get('type') == 'swap' 
            and symbol.endswith(':USDT')
            and symbol not in BLACKLIST_SYMBOLS  # <--- Filtro de seguridad
        ]
        
        return sorted(symbols)
        
    except Exception as e:
        print(f"❌ Error get_futures_symbols: {e}")
        return None


def get_top_symbols_by_volume(limit=100):
    """
    Obtiene símbolos ordenados por volumen 24h (top movers)
    
    Args:
        limit: Cantidad de símbolos a retornar
    
    Returns:
        list: Símbolos ordenados por volumen descendente
    """
    try:
        tickers = exchange.fetch_tickers()
        
        # Filtrar y ordenar por volumen
        volume_list = []
        for symbol, ticker in tickers.items():
            if ':USDT' in symbol and ticker.get('quoteVolume'):
                volume_list.append((symbol, ticker['quoteVolume']))
        
        # Ordenar por volumen descendente
        volume_list.sort(key=lambda x: x[1], reverse=True)
        
        # Retornar top N
        return [symbol for symbol, _ in volume_list[:limit]]
        
    except Exception as e:
        print(f"❌ Error obteniendo top symbols: {e}")
        return []


def fetch_current_price(symbol, max_retries=3):
    """
    Obtiene precio actual (ticker) con retry y backoff para rate limits
    
    Args:
        symbol: Símbolo (ej: 'BTC/USDT:USDT')
        max_retries: Número máximo de reintentos
    
    Returns:
        float: Precio actual
        None: Si error
    """
    for attempt in range(max_retries):
        try:
            ticker = exchange.fetch_ticker(symbol)
            return float(ticker['last'])
        except Exception as e:
            error_str = str(e).lower()
            if '510' in str(e) or 'too frequent' in error_str or 'rate' in error_str:
                # Rate limit - esperar y reintentar
                wait_time = (attempt + 1) * 2  # 2, 4, 6 segundos
                print(f"⏳ Rate limit, esperando {wait_time}s...")
                time.sleep(wait_time)
            else:
                print(f"❌ Error fetch_current_price {symbol}: {e}")
                return None
    print(f"❌ Max retries alcanzado para {symbol}")
    return None


def fetch_multiple_tickers(symbols, max_retries=3):
    """
    Obtiene precios actuales para múltiples símbolos con retry.
    Optimizado para evitar Rate Limits (Code 510).
    
    Args:
        symbols: Lista de símbolos (ej: ['BTC/USDT:USDT', 'ETH/USDT:USDT'])
        max_retries: Número máximo de reintentos

        dict: {symbol: price, ...}
    """

            else:
                print(f"❌ Error fetch_multiple_tickers: {e}")
                return {}
    print(f"❌ Max retries alcanzado para fetch_multiple_tickers")

# 15M WARNINGS SYSTEM - v5.4.0 "5M Decide, 15M Advierte"

def generate_15m_warnings(data_15m: dict, direction_5m: str) -> dict:
    """
    15M revisa contexto macro y genera warnings LEVES para 5M.
    No decide, solo advierte.
    
    Args:
        data_15m: Return de analyze_15m()
        direction_5m: Direction decidida por 5M
        
    Returns:
        {
            'total_penalty': int (0-105 máximo teórico),
            'warnings': list[str],
            'should_hard_veto': bool (solo casos extremos)
        }
    """
    from brain_config import (
        ENABLE_15M_WARNINGS,
        WARNING_15M_ADX_DECAY,
        WARNING_15M_MOMENTUM_OLD,
        WARNING_15M_DIRECTION,
        WARNING_15M_LATERAL,
        WARNING_15M_NO_STRUCTURE,
        WARNING_15M_ADX_WEAK,
        WARNING_15M_SPREAD_MIN
    )
    
    if not ENABLE_15M_WARNINGS:
        return {'total_penalty': 0, 'warnings': [], 'should_hard_veto': False}
    
    if not data_15m:
        return {'total_penalty': 0, 'warnings': [], 'should_hard_veto': False}
    
    penalties = []
    warnings = []
    
    # Extraer datos de 15M
    adx_15m = data_15m.get('adx', {})
    adx_current = adx_15m.get('adx_current', 0)
    adx_trend = adx_15m.get('adx_trend', 'estable')
    spread_15m = adx_15m.get('spread', 0)
    
    momentum_15m = data_15m.get('momentum', {})
    mom_age = momentum_15m.get('consecutive_count', 0)
    
    direction_15m = data_15m.get('direction', 'LATERAL')
    lateral_penalty_15m = data_15m.get('lateral_penalty', 0)
    
    # ZigZag para estructura
    zigzag_15m = data_15m.get('zigzag_granular', {})
    
    # ================================================
    # HARD VETO: Solo para spread muy bajo
    # ================================================
    if abs(spread_15m) < WARNING_15M_SPREAD_MIN:
        print(f"   🚫 15M HARD VETO: Spread {spread_15m:.1f} < {WARNING_15M_SPREAD_MIN} (inoperable)")
        return {
            'total_penalty': 0,
            'warnings': [f'Spread 15M {spread_15m:.1f} < {WARNING_15M_SPREAD_MIN} (inoperable)'],
            'should_hard_veto': True
        }
    
    # ================================================
    # WARNING 1: ADX decreciendo
    # ================================================
    if adx_trend == 'decreciendo':
        penalties.append(WARNING_15M_ADX_DECAY)
        warnings.append(f'ADX 15M decreciendo ({adx_current:.1f})')
    
    # ================================================
    # WARNING 2: Momentum agotado (>12 velas)
    # ================================================
    if mom_age > 12:
        penalties.append(WARNING_15M_MOMENTUM_OLD)
        warnings.append(f'Momentum 15M agotado ({mom_age}v)')
    
    # ================================================
    # WARNING 3: Dirección contradice
    # ================================================
    if direction_15m != direction_5m and direction_15m != 'LATERAL':
        penalties.append(WARNING_15M_DIRECTION)
        warnings.append(f'15M dice {direction_15m}, 5M dice {direction_5m}')
    
    # ================================================
    # WARNING 4: LATERAL detectado (con penalty alta)
    # ================================================
    if lateral_penalty_15m > 30:
        penalties.append(WARNING_15M_LATERAL)
        warnings.append(f'15M lateral penalty -{lateral_penalty_15m}')
    
    # ================================================
    # WARNING 5: Sin estructura (choppy)
    # ================================================
    structure = zigzag_15m.get('current_trend', 'UNKNOWN')
    if structure == 'LATERAL':
        penalties.append(WARNING_15M_NO_STRUCTURE)
        warnings.append('15M sin estructura clara')
    
    # ================================================
    # WARNING 6: ADX muy débil (<20)
    # ================================================
    if adx_current < 20:
        penalties.append(WARNING_15M_ADX_WEAK)
        warnings.append(f'ADX 15M débil ({adx_current:.1f})')
    
    total_penalty = sum(penalties)
    
    if total_penalty > 0:
        print(f"   ⚠️ 15M WARNINGS: {', '.join(warnings)} → penalty total -{total_penalty}")
    else:
        print(f"   ✅ 15M: Sin warnings (contexto macro OK)")
    
    return {
        'total_penalty': total_penalty,
        'warnings': warnings,
        'should_hard_veto': False
    }

def analyze_temporal_context(symbol):
    """
    Analiza el contexto temporal reciente (últimos 60 min) usando la DB.
    Busca estabilidad y consistencia.
    """
    try:
        history = db_manager.get_recent_context(symbol, limit_minutes=60)
        if not history or len(history) < 10:
            return {'stability': 'DESCONOCIDO', 'bonus': 0}
        
        # 1. Verificar cambios de dirección (inestabilidad)
        directions = [row['direction'] for row in history]
        flips = 0
        for i in range(1, len(directions)):
            if directions[i] != directions[i-1]:
                flips += 1
        
        # 2. Verificar consistencia de ADX
        adx_values = [row['adx_15m'] for row in history]
        adx_rising = 0
        for i in range(1, len(adx_values)):
            if adx_values[i] > adx_values[i-1]:
                adx_rising += 1
        
        # Evaluación
        bonus = 0
        stability = 'ESTABLE'
        
        if flips > 3:
            stability = 'INESTABLE'
            bonus -= 10  # Penalización por inestabilidad
        elif flips == 0:
            stability = 'MUY_ESTABLE'
            bonus += 10  # Bonus por tendencia sostenida
            
        if adx_rising > len(adx_values) * 0.7:
            bonus += 5  # Bonus si ADX sube la mayoría del tiempo
            
        return {'stability': stability, 'bonus': bonus, 'flips': flips}
        
    except Exception as e:
        print(f"⚠️ Error analizando contexto temporal: {e}")
        return {'stability': 'ERROR', 'bonus': 0}

def analyze_trend_evolution(symbol):
    """
    ANÁLISIS DE EVOLUCIÓN (TIME LADDER / ROADMAP)
    Analiza velas de 1M para detectar tendencias explosivas en tiempo real.
    Permite 'bypassear' filtros estrictos si la moneda está explotando ahora mismo.
    """
    try:
        # Buscar 30 velas de 1M (media hora de evolución)
        ohlcv = fetch_ohlcv(symbol, '1m', 30)
        if not ohlcv or len(ohlcv) < 15:
            return {'is_explosive': False, 'score': 0, 'reason': 'Sin datos 1M'}

        df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
        
        # 1. Análisis de ADX en 1M (Velocidad de tendencia)
        adx_data = calculate_adx_with_memory(df, period=14, lookback=5)
        adx_current = adx_data['adx_current']
        adx_history = adx_data['adx_history']
        
        # 2. Análisis de Precio (Escalera de precios)
        closes = df['close'].values
        price_velocity = calculate_velocity(df, lookback=5)
        
        # DETECCIÓN DE EXPLOSIÓN
        is_explosive = False
        score = 0
        reason = []

        # Criterio A: ADX Disparándose (Roadmap de fuerza)
        # Si ADX subió más de 5 puntos en las últimas 5 velas 1M
        if len(adx_history) >= 5:
            adx_change = adx_history[-1] - adx_history[0]
            if adx_change > 3.0 and adx_current > 20:
                score += 40
                reason.append(f"ADX 1M disparándose (+{adx_change:.1f})")
                if adx_current > 25:
                    is_explosive = True

        # Criterio B: Escalera de Precios (ZigZag alcista agresivo)
        # Verificar si las últimas 3 velas son verdes y con cuerpos crecientes o fuertes
        last_3_closes = closes[-3:]
        last_3_opens = df['open'].values[-3:]
        is_green_ladder = all(c > o for c, o in zip(last_3_closes, last_3_opens))
        
        if is_green_ladder and price_velocity['velocity_current'] > 0:
            score += 30
            reason.append("Escalera de precios 1M alcista")
            if price_velocity['trend'] == 'acelerando':
                is_explosive = True
                score += 20

        # Criterio C: Volumen Climático
        volumes = df['volume'].values
        vol_avg = volumes[:-1].mean()
        if volumes[-1] > vol_avg * 2:
            score += 20
            reason.append("Volumen 1M climático")

        # Decisión final de explosividad
        if score >= 70:
            is_explosive = True
        
        return {
            'is_explosive': is_explosive,
            'score': score,
            'adx_1m': adx_current,
            'reason': ", ".join(reason),
            'timestamp': datetime.now().isoformat()
        }

    except Exception as e:
        print(f"⚠️ Error en analyze_trend_evolution: {e}")
        return {'is_explosive': False, 'score': 0, 'reason': 'Error'}

def _extract_score(score_value, default=0):
    """
    Recursively extract integer score from potentially nested dict.
    
    Args:
        score_value: The value to extract (can be int, float, dict, or None)
        default: Default value if extraction fails
    
    Returns:
        int: Extracted integer score
    """
    if score_value is None:
        return default
    if isinstance(score_value, (int, float)):
        return int(score_value)
    if isinstance(score_value, dict):
        # Recursively extract from nested dict
        return _extract_score(score_value.get('score', default), default)
    # Fallback for any other type
    return default

            pass
    
    if atr > 0:

        if direction == 'BULLISH':
            stop_loss = entry_price - stop_loss_dist
        else:
            stop_loss = entry_price + stop_loss_dist

            else: stop_loss = entry_price + min_dist

        target_price = entry_price + target_dist
    else:
        target_price = entry_price - target_dist

    rr_ratio = target_dist / risk if risk > 0 else 0
    tiempo_proyectado_min = 15

    return {
        'entry_price': entry_price,
        'stop_loss': stop_loss,
        'target_price': target_price,
        'rr_ratio': rr_ratio,
        'tiempo_proyectado_min': tiempo_proyectado_min,

    }

    def validate_entry(self, symbol, direction, phase_result, force_result, ladder_data):
        """

    def _validate_explosion_entry(self, symbol, direction, force_result, ladder_data):
        """
        v5.2.1: TIME LADDER PURO - Retorna SCORE, no can_enter.

            return {'score': 0, 'reason': 'No price data', 'velocity_factor': 0}
        
        score = 100
        penalties = []
        
        # === VELOCITY FACTOR ===
        velocity_pct = abs(force_result.get('velocity', 0))

        print(f"   🏎️ Velocity: {velocity_pct:.3f}% → Factor: {velocity_factor:.2f}")

    def _check_rebound_confirmed(self, symbol, direction, ladder_data):
        """

        1. Piso formado (LL o soporte tocado)
        2. Precio empieza a subir (últimas 3 velas)

        adx_trend = adx_data.get('adx_trend', 'estable')
        pivot_types = zigzag_data.get('pivot_types', [])
        last_pivot = pivot_types[-1] if pivot_types else 'UNKNOWN'

        # 2. Precio subiendo (últimas 3 velas)
        if len(price_values) >= 3:
            if direction == 'BULLISH':
                conditions['2_precio_subiendo'] = price_values[-1] > price_values[-3]
