    def pre_filter_weekly_candles(self):
        """
        PRE-FILTRO: Rechaza monedas con <10 velas semanales
        
        Returns:
            list: Símbolos que pasan filtro
        """
        print(f"\n{'='*80}")
        print("🔍 FASE 0: PRE-FILTRO VELAS SEMANALES")
        print(f"{'='*80}")
        
        all_symbols = get_futures_symbols()
        if not all_symbols:
            print("❌ Error obteniendo símbolos")
            return []
        
        total_symbols = len(all_symbols)
        print(f"📊 Total símbolos: {total_symbols}")
        
        filtered = []
        rejected = []
        pre_filtro_batch = []  # Acumular para batch writing
        
        print("Analizando velas semanales...")
        
        for i, symbol in enumerate(all_symbols):
            if i % 100 == 0:
                print(f"  Progreso: {i}/{total_symbols} ({i/total_symbols*100:.1f}%)")
            
            try:
                ohlcv = fetch_ohlcv(symbol, '1w', 15)
                if not ohlcv:
                    rejected.append((symbol, 0))
                    continue
                
                # Contar velas reales
                real_candles = 0
                for candle in ohlcv:
                    volume = candle[5]
                    high = candle[2]
                    low = candle[3]
                    
                    if volume > 0 and high != low:
                        real_candles += 1
                
                if real_candles >= MIN_WEEKLY_CANDLES:
                    filtered.append(symbol)
                    print(f"  ✅ {symbol}: {real_candles} velas semanales - APROBADO")
                    # Acumular para batch
                    timestamp = datetime.now(ARGENTINA_TZ).strftime("%Y-%m-%d %H:%M:%S")
                    pre_filtro_batch.append([timestamp, symbol, real_candles, "APROBADO"])
                else:
                    rejected.append((symbol, real_candles))
                    print(f"  ❌ {symbol}: {real_candles} velas semanales - RECHAZADO")
                    # Acumular para batch
                    timestamp = datetime.now(ARGENTINA_TZ).strftime("%Y-%m-%d %H:%M:%S")
                    pre_filtro_batch.append([timestamp, symbol, real_candles, "RECHAZADO"])
            
            except Exception as e:
                rejected.append((symbol, 0))
        
        print(f"\n✅ FILTRADAS: {len(filtered)}/{total_symbols} ({len(filtered)/total_symbols*100:.1f}%)")
        print(f"❌ RECHAZADAS: {len(rejected)} (<{MIN_WEEKLY_CANDLES} velas semanales)")
        
        # Escribir batch en Google Sheets
        if pre_filtro_batch:
            print(f"\n📊 Escribiendo {len(pre_filtro_batch)} registros a BRAIN_PRE_FILTRO...")
            sheets_manager.write_batch(SHEET_BRAIN_PRE_FILTRO, pre_filtro_batch)
            print("✅ Batch escritura completada")
        
        return filtered
    
    def analyze_all_coins(self):
        """
        FASE 1: Análisis contextual completo de todas las monedas filtradas
        
        Returns:
            list: [(symbol, analysis_data), ...] solo monedas viables
        """
        print(f"\n{'='*80}")
        print("🧠 FASE 1: ANÁLISIS CONTEXTUAL MULTI-TIMEFRAME")
        print(f"{'='*80}")
        
        if not self.filtered_symbols:
            print("⚠️ No hay símbolos filtrados")
            return []
        
        total = len(self.filtered_symbols)
        analyzed = []
        viable_count = 0
        
        print(f"Analizando {total} monedas (1D→4H→1H→15M→5M)...\n")
        
        for i, symbol in enumerate(self.filtered_symbols):
            # Mostrar progreso cada 20 monedas (más frecuente)
            if i % 20 == 0:
                print(f"\n📊 Progreso: {i}/{total} ({i/total*100:.1f}%) - Viables: {viable_count}")
            
            try:
                print(f"\n🔍 Analizando: {symbol}")
                
                analysis = analyze_coin_complete(symbol)
                
                if analysis and analysis['viable']:
                    analyzed.append((symbol, analysis))
                    viable_count += 1
                    
                    score = analysis['score_data']['score']
                    clasificacion = analysis['score_data']['clasificacion']
                    direction = analysis['analysis_5m']['direction']
                    
                    print(f"  ✅ {symbol}: {score}/100 ({clasificacion}) - {direction}")
                    
                    # Prints detallados de análisis multi-timeframe
                    if analysis['analysis_4h']:
                        print(f"     4H: {analysis['analysis_4h']['direction']} ({analysis['analysis_4h']['strength']}) | ADX: {analysis['analysis_4h']['adx']['adx_current']:.1f}")
                    if analysis['analysis_1h']:
                        print(f"     1H: {analysis['analysis_1h']['direction']} ({analysis['analysis_1h']['strength']}) | ADX: {analysis['analysis_1h']['adx']['adx_current']:.1f} | Edad: {analysis['analysis_1h']['momentum']['consecutive_count']}v")
                    if analysis['analysis_15m']:
                        print(f"     15M: Edad {analysis['analysis_15m']['momentum']['consecutive_count']}v | EMAs: {analysis['analysis_15m']['emas'].get('classification', 'N/A')}")
                    if analysis['analysis_5m']:
                        print(f"     5M: Vel: {analysis['analysis_5m']['velocity']['velocity_current']:.6f} | Acel: {analysis['analysis_5m']['acceleration']['acceleration_current']:.6f}")
                else:
                    # Moneda no viable - mostrar razón
                    print(f"  ⚠️ {symbol}: No cumple criterios mínimos")
                    if analysis and isinstance(analysis, dict):
                        if analysis.get('analysis_4h'):
                            print(f"     4H: {analysis['analysis_4h']['direction']} ({analysis['analysis_4h']['strength']})")
                        if analysis.get('score_data'):
                            print(f"     Score: {analysis['score_data']['score']}/100 (< 70 requerido)")
            
            except Exception as e:
                print(f"  ❌ {symbol}: ERROR al analizar")
                print(f"     Error técnico: {str(e)}")
                continue
        
        print(f"\n✅ ANÁLISIS COMPLETADO: {viable_count} monedas viables de {total}")
        
        return analyzed
    
    def update_watchlist(self, analyzed_coins):
        """
        FASE 2: Actualiza watchlist con monedas analizadas
        
        Args:
            analyzed_coins: List de (symbol, analysis_data)
        """
        print(f"\n{'='*80}")
        print("📋 FASE 2: ACTUALIZACIÓN WATCHLIST")
        print(f"{'='*80}")
        
        watchlist_manager.update_watchlist_from_analysis(analyzed_coins)
        
        watchlist = watchlist_manager.get_watchlist()
        
        if watchlist:
            print(f"\n📊 WATCHLIST ACTUAL ({len(watchlist)} monedas):")
            for symbol, data in watchlist[:10]:  # Mostrar top 10
                print(f"  {symbol}: {data['score']} ({data['clasificacion']}) - {data['direction']}")
                if data['triggers']:
                    print(f"    → Target: ${data['triggers']['target_price']:.8f} | R:R: {data['triggers']['rr_ratio']:.2f}")
            
            # Escribir toda la watchlist en BRAIN_WATCHLIST
            print(f"\n💾 Escribiendo watchlist a Google Sheets...")
            for symbol, data in watchlist:
                # Obtener datos necesarios
                score = data['score']
                direction = data['direction']
                
                # EMAs distance desde analysis_snapshot
                emas_distance = 0.0
                if 'analysis_snapshot' in data and data['analysis_snapshot']:
                    emas_distance = data['analysis_snapshot'].get('emas_separation_15m', 0.0)
                
                # Clasificar estructura según score
                if score >= 80:
                    structure = "SANA"
                elif score >= 70:
                    structure = "REGULAR"
                else:
                    structure = "MALA"
                
                # Clasificar edad según momentum_age_5m
                edad = "MADURA"
                if 'analysis_snapshot' in data and data['analysis_snapshot']:
                    momentum_age_5m = data['analysis_snapshot'].get('momentum_age_5m', 10)
                    if momentum_age_5m <= 5:
                        edad = "JOVEN"
                
                # Obtener datos de triggers (tiempo proyectado y R/R ratio)
                tiempo_proyectado_min = 0
                rr_ratio = 0.0
                if 'triggers' in data and data['triggers']:
                    tiempo_proyectado_min = data['triggers'].get('tiempo_proyectado_min', 0)
                    rr_ratio = data['triggers'].get('rr_ratio', 0.0)
                
                # Escribir a Sheet
                sheets_manager.write_watchlist(symbol, score, direction, emas_distance, structure, edad, tiempo_proyectado_min, rr_ratio)
        else:
            print("⚠️ Watchlist vacía")
    
    def monitor_and_trade(self):
        """
        FASE 3: Monitorea watchlist y trades activos
        """
        print(f"\n{'='*80}")
        print("👁️ FASE 3: MONITOREO ACTIVO")
        print(f"{'='*80}")
        
        # Limpiar watchlist de monedas estancadas (TTL)
        removed = watchlist_manager.cleanup_stale_symbols(max_age_minutes=30, max_checks=20)
        if removed > 0:
            watchlist_count = len(watchlist_manager.get_watchlist())
            print(f"🔄 Watchlist renovada: {watchlist_count} monedas activas (removidas {removed})")
        
        # Verificar si watchlist quedó vacía
        watchlist = watchlist_manager.get_watchlist()
        if len(watchlist) == 0:
            print(f"\n⚠️ Watchlist vacía - Saliendo a re-análisis...")
            return  # Retorna para forzar re-análisis completo
        
        # Monitorear trades activos
        if trade_manager.active_trades:
            print(f"\n📈 TRADES ACTIVOS: {len(trade_manager.active_trades)}")
            trade_manager.monitor_active_trades()
        
        # Verificar triggers en watchlist
        if trade_manager.can_open_trade():
            
            print(f"\n🔍 VERIFICANDO TRIGGERS ({len(watchlist)} monedas en watchlist)...")
            
            for symbol, data in watchlist:
                if not trade_manager.can_open_trade():
                    break
                
                print(f"  🔍 Verificando: {symbol} (Score: {data['score']})")
                
                # Verificar trigger (ahora retorna dict)
                trigger_result = trade_manager.check_trigger_activated(symbol, data)
                
                if trigger_result and trigger_result['activated']:
                    # Resetear contador (casi se activó)
                    watchlist_manager.reset_check_counter(symbol)
                    
                    # Escribir BRAIN_TRIGGERS
                    sheets_manager.write_trigger(
                        symbol,
                        "ADX+MOMENTUM+VOLUMEN",
                        trigger_result['adx'],
                        trigger_result['momentum_age'],
                        trigger_result['volume_ratio']
                    )
                    
                    # CALCULAR Y ESCRIBIR BRAIN_PROYECCIONES
                    ohlcv_fresh = fetch_ohlcv(symbol, '5m', 50)
                    if ohlcv_fresh and len(ohlcv_fresh) >= 20:
                        df_fresh = pd.DataFrame(ohlcv_fresh, columns=['timestamp','open','high','low','close','volume'])
                        prices = df_fresh['close'].values
                        
                        velocity = calculate_velocity(prices)
                        acceleration = calculate_acceleration(prices)
                        current_price = prices[-1]
                        
                        if 'triggers' in data and data['triggers']:
                            sheets_manager.write_proyeccion(
                                symbol,
                                current_price,
                                velocity,
                                acceleration,
                                data['triggers']['tiempo_proyectado_min'],
                                data['triggers']['rr_ratio'],
                                data['triggers']['target_price'],
                                data['triggers']['stop_loss']
                            )
                    
                    # Entrar al trade
                    trade_manager.enter_trade(symbol, data)
                else:
                    # Incrementar contador de checks sin trigger
                    watchlist_manager.increment_check_counter(symbol)
                    print(f"     ⏳ Trigger no activado aún")
        
        # Re-analizar breakeven exits
        trade_manager.reanalyze_breakeven_exits()
    
    def run(self):
        """Loop principal del bot"""
        self.running = True
        self.print_banner()
        
        # Notificar inicio
        if ENABLE_TELEGRAM:
            notify_bot_start(CAPITAL_INICIAL, SIMULATE_LEVERAGE)
        
        # Pre-filtro inicial
        print("🚀 Iniciando pre-filtro...")
        self.filtered_symbols = self.pre_filter_weekly_candles()
        
        if not self.filtered_symbols:
            print("❌ No hay símbolos para analizar. Abortando.")
            return
        
        # Iniciar thread de stop loss
        trade_manager.start_stop_loss_monitor()
        
        # Análisis inicial completo
        print("\n🧠 Iniciando análisis contextual completo...")
        analyzed_coins = self.analyze_all_coins()
        
        # Watchlist inicial
        self.update_watchlist(analyzed_coins)
        self.last_full_analysis = datetime.now(ARGENTINA_TZ)
        self.last_mini_scan = None  # Tracking para mini-scans
        
        print(f"\n{'='*80}")
        print("✅ INICIALIZACIÓN COMPLETA - Entrando en loop de monitoreo")
        print(f"{'='*80}\n")
        
        # Loop principal
        try:
            while self.running:
                self.cycle_count += 1
                now = datetime.now(ARGENTINA_TZ)
                
                print(f"\n{'─'*80}")
                print(f"🔄 CICLO #{self.cycle_count} - {now.strftime('%H:%M:%S')}")
                print(f"{'─'*80}")
                
                # Monitoreo y trading
                self.monitor_and_trade()
                
                # Si watchlist vacía → re-analizar TODO inmediatamente
                watchlist_current = watchlist_manager.get_watchlist()
                if len(watchlist_current) == 0:
                    print("\n⚡ Watchlist vacía → Re-analizando mercado completo...")
                    analyzed_coins = self.analyze_all_coins()
                    self.update_watchlist(analyzed_coins)
                
                # Mini-scan cada 30 minutos
                if not self.last_mini_scan or (now - self.last_mini_scan).total_seconds() >= 1800:  # 30min
                    print(f"\n{'='*80}")
                    print("🔍 MINI-SCAN: Buscando nuevas oportunidades...")
                    print(f"{'='*80}")
                    
                    # Obtener top 100 símbolos por volumen
                    from exchange_connection import get_top_symbols_by_volume
                    top_symbols = get_top_symbols_by_volume(limit=100)
                    
                    if top_symbols:
                        print(f"📊 Analizando top {len(top_symbols)} símbolos por volumen 24h...")
                        
                        analyzed_mini = []
                        for i, symbol in enumerate(top_symbols[:100], 1):
                            # Progress cada 25
                            if i % 25 == 0:
                                print(f"  Progreso: {i}/100...")
                            
                            analysis = analyze_coin_complete(symbol)
                            if analysis and analysis['viable']:
                                analyzed_mini.append((symbol, analysis))
                        
                        # Actualizar watchlist con nuevas
                        if analyzed_mini:
                            print(f"✅ {len(analyzed_mini)} nuevas monedas → watchlist")
                            self.update_watchlist(analyzed_mini)
                        else:
                            print("⚠️ Sin nuevas oportunidades")
                    else:
                        print("❌ Error obteniendo top symbols por volumen")
                    
                    self.last_mini_scan = now
                
                # ANÁLISIS CONTINUO - Sin delay artificial
                import time
                time.sleep(5)  # Solo 5s para no saturar API
        
        except KeyboardInterrupt:
            print("\n\n🛑 Bot detenido por usuario")
            self.stop()
    
    def stop(self):
        """Detiene el bot"""
        self.running = False
        
        print("\n" + "=" * 80)
        print("📊 RESUMEN SESIÓN")
        print("=" * 80)
        
        # Stats trades
        total_trades = len(trade_manager.trade_history)
        if total_trades > 0:
            wins = sum(1 for t in trade_manager.trade_history if t['pnl_pct'] > 0)
            losses = total_trades - wins
            win_rate = (wins / total_trades) * 100
            
            avg_pnl = sum(t['pnl_pct'] for t in trade_manager.trade_history) / total_trades
            total_pnl_usd = sum(t['pnl_usd'] for t in trade_manager.trade_history)
            
            print(f"Total trades: {total_trades}")
            print(f"Wins: {wins} | Losses: {losses}")
            print(f"Win rate: {win_rate:.1f}%")
            print(f"Avg PnL: {avg_pnl:+.2f}%")
            print(f"Total PnL: ${total_pnl_usd:+.2f}")
            
            # Guardar stats en Sheet
            best_trade = max(t['pnl_pct'] for t in trade_manager.trade_history)
            worst_trade = min(t['pnl_pct'] for t in trade_manager.trade_history)
            
            sheets_manager.write_stats(
                total_trades, wins, losses, win_rate,
                avg_pnl, best_trade, worst_trade
            )
        
        else:
            print("No se ejecutaron trades")
        
        print("=" * 80)
        print("👋 Hasta pronto!")
        print("=" * 80 + "\n")
