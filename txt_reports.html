PASO 1: CREAR brain_vision_unified.py (v3.0 FINAL)
"""
================================================================================
BRAIN VISION UNIFIED v3.0 - EVALUACIÓN GRADUAL DE CONFIANZA
================================================================================
Joel: "El bot debe decidir como trader humano, no como calculadora"
NO USA:
- Thresholds fijos (> 30, < 1, >= 3)
- Labels binarios ('ACELERANDO', 'BLOQUEADO', 'FRESH')
- Pattern matching (if A and B: ENTER)
USA:
- Valores numéricos RAW de trayectorias
- Evaluación gradual de confianza
- Factores múltiples acumulativos
================================================================================
"""
from brain_movie_engine import MovieEngine
from brain_grid import grid
from brain_trend_exhaustion import analyze_trend_exhaustion
from brain_frames import get_frame_storage
from brain_force_filter import force_filter
from exchange_connection import fetch_ohlcv
import pandas as pd
class UnifiedVision:
    """
    Cerebro que evalúa CONFIANZA basada en trayectorias.
    NO etiqueta. NO usa thresholds binarios.
    """
    
    def __init__(self):
        self.movie_engine = MovieEngine()
    
    # =========================================================================
    # FASE 1: RECOPILAR FRAMES
    # =========================================================================
    
    def gather_intelligence(self, symbol, ladder_analysis):
        """Recopila FRAMES de módulos."""
        
        direction = ladder_analysis.get('15m', {}).get('direction', 'LATERAL')
        
        # Frames (historia temporal)
        frames_15m = list(get_frame_storage(symbol, '15m').get_frames(10))
        frames_5m = list(get_frame_storage(symbol, '5m').get_frames(10))
        
        # Exhaustion (espacio actual)
        exhaustion_result = analyze_trend_exhaustion(symbol, direction)
        
        # Grid (posición)
        ohlcv = fetch_ohlcv(symbol, '5m', 50)
        if ohlcv:
            df = pd.DataFrame(ohlcv, columns=['timestamp','open','high','low','close','volume'])
            grid_result = grid.analyze(symbol, df, ladder_analysis)
        else:
            grid_result = {'status': 'ERROR'}
        
        # Force
        force_result = force_filter.has_force(symbol, ladder_analysis)
        
        # ZigZag
        zigzag = ladder_analysis.get('15m', {}).get('zigzag', {})
        
        return {
            'frames': {'15m': frames_15m, '5m': frames_5m},
            'exhaustion': exhaustion_result,
            'grid': grid_result,
            'force': force_result,
            'zigzag': zigzag,
            'direction': direction
        }
    
    # =========================================================================
    # FASE 2: EXTRAER TRAYECTORIAS (VALORES RAW)
    # =========================================================================
    
    def extract_trajectories(self, intelligence):
        """
        Extrae valores numéricos RAW de frames.
        NO etiqueta. Solo extrae números.
        """
        frames_15m = intelligence.get('frames', {}).get('15m', [])
        
        if len(frames_15m) < 5:
            return {'error': 'INSUFICIENTES_FRAMES', 'count': len(frames_15m)}
        
        recent = frames_15m[-5:]
        
        trajectories = {}
        
        # ---- ADX ----
        adx_values = []
        for f in recent:
            adx = f.get('adx', {})
            adx_values.append(adx.get('value', 0) if isinstance(adx, dict) else 0)
        
        trajectories['adx_values'] = adx_values
        trajectories['adx_start'] = adx_values[0] if adx_values else 0
        trajectories['adx_end'] = adx_values[-1] if adx_values else 0
        trajectories['adx_delta'] = trajectories['adx_end'] - trajectories['adx_start']
        
        # Calcular cambios consecutivos
        if len(adx_values) >= 2:
            changes = [adx_values[i+1] - adx_values[i] for i in range(len(adx_values)-1)]
            trajectories['adx_changes'] = changes

    # =========================================================================
    # FASE 3: EVALUAR CONFIANZA (GRADUAL)
    # =========================================================================
    
    def evaluate_confidence(self, traj, direction):
        """
        Evalúa confianza de forma GRADUAL.
        
        NO usa thresholds binarios.
        Cada factor aporta confianza proporcionalmente.
        """
        
        if traj.get('error'):
            return {
                'total_confidence': 0,
                'factors': [],
                'reason': f"Error: {traj.get('error')}"
            }
        
        confidence_factors = []
        
        # ================================================================
        # FACTOR 1: ADX ACELERACIÓN
        # ================================================================
        # Si ADX sube → confianza aumenta
        # Si ADX baja → confianza disminuye
        
        adx_delta = traj.get('adx_delta', 0)
        adx_end = traj.get('adx_end', 0)
        
        # Confianza por aceleración (max 25 puntos)
        # adx_delta de 0→10 = 0→20 puntos
        # adx_delta de 10→20 = 20→25 puntos (tope)
        if adx_delta > 0:
            accel_conf = min(adx_delta * 2, 25)
            confidence_factors.append(('ADX acelerando', accel_conf, f'+{adx_delta:.1f}'))
        elif adx_delta < -3:
            # Penaliza si ADX cae mucho
            decel_conf = max(adx_delta * 2, -30)
            confidence_factors.append(('ADX muriendo', decel_conf, f'{adx_delta:.1f}'))
        
        # Bonus por ADX alto SOLO si está subiendo
        if adx_end > 30 and adx_delta > 0:
            high_conf = min((adx_end - 30) / 2, 10)
            confidence_factors.append(('ADX fuerte', high_conf, f'{adx_end:.0f}'))
        
        # ================================================================
        # FACTOR 2: VELAS CRECIENDO
        # ================================================================
        # Si velas crecen → fuerza real
        # Si velas achican → fuerza muriendo
        
        candle_ratio = traj.get('candle_ratio', 1)
        
        if candle_ratio > 1.2:
            # Velas creciendo 20%+ = confianza
            candle_conf = min((candle_ratio - 1) * 25, 20)
            confidence_factors.append(('Velas creciendo', candle_conf, f'{candle_ratio:.2f}x'))
        elif candle_ratio < 0.8:
            # Velas achicando 20%+ = problema
            shrink_conf = max((candle_ratio - 1) * 30, -25)
            confidence_factors.append(('Velas achicándose', shrink_conf, f'{candle_ratio:.2f}x'))
        
        # ================================================================
        # FACTOR 3: MOMENTUM SOSTENIDO
        # ================================================================
        # Momentum age creciendo = impulso sostenido
        
        momentum_delta = traj.get('momentum_delta', 0)
        momentum_end = traj.get('momentum_end', 0)
        
        if momentum_delta > 0 and momentum_end >= 3:
            mom_conf = min(momentum_end * 2, 15)
            confidence_factors.append(('Momentum sostenido', mom_conf, f'{momentum_end}v'))
        
        # ================================================================
        # FACTOR 4: ESPACIO LIBRE
        # ================================================================
        # Room amplio = confianza
        # Room estrecho = cuidado
        
        room_pct = traj.get('room_pct', 0)
        exhaustion_velocity = traj.get('exhaustion_velocity', 0)
        
        # Room estático
        if room_pct > 3:
            room_conf = min(room_pct * 3, 15)
            confidence_factors.append(('Espacio amplio', room_conf, f'{room_pct:.1f}%'))
        elif room_pct < 1.5:
            tight_conf = (room_pct - 1.5) * 20  # Negativo
            confidence_factors.append(('Espacio estrecho', tight_conf, f'{room_pct:.1f}%'))
        
        # Exhaustion velocity (dinámico)
        # Si exhaustion explota rápido = problema
        if exhaustion_velocity > 20:
            expl_conf = min(exhaustion_velocity * -1.5, -25)
            confidence_factors.append(('Exhaustion explosiva', expl_conf, f'+{exhaustion_velocity:.0f}%'))
        elif exhaustion_velocity < -10:
            # Exhaustion bajando = espacio abriéndose
            opening_conf = min(abs(exhaustion_velocity) * 0.8, 12)
            confidence_factors.append(('Espacio abriéndose', opening_conf, f'{exhaustion_velocity:.0f}%'))
        
        # ================================================================
        # FACTOR 5: ESTRUCTURA ZIGZAG
        # ================================================================
        pivot_types = traj.get('pivot_types', [])
        
        if len(pivot_types) >= 2:
            last_two = pivot_types[-2:]
            
            if direction == 'BULLISH':
                if last_two[-1] in ['HH', 'HL']:
                    confidence_factors.append(('Estructura sana', 15, 'HH/HL'))
                elif 'LH' in last_two or 'LL' in last_two:
                    confidence_factors.append(('Estructura rota', -20, 'LH/LL'))
            elif direction == 'BEARISH':
                if last_two[-1] in ['LL', 'LH']:
                    confidence_factors.append(('Estructura sana', 15, 'LL/LH'))
                elif 'HH' in last_two or 'HL' in last_two:
                    confidence_factors.append(('Estructura rota', -20, 'HH/HL'))
        
        # ================================================================
        # FACTOR 6: POSICIÓN EN GRID
        # ================================================================
        position_pct = traj.get('position_pct', 50)
        
        if direction == 'BULLISH':
            if position_pct < 30:
                # Cerca soporte = ideal para long
                pos_conf = (30 - position_pct) / 3
                confidence_factors.append(('Cerca soporte', pos_conf, f'{position_pct:.0f}%'))
            elif position_pct > 80:
                # Cerca techo = peligro
                top_conf = (position_pct - 80) * -1.5
                confidence_factors.append(('Cerca techo', top_conf, f'{position_pct:.0f}%'))
        
        # ================================================================
        # TOTAL
        # ================================================================
        total_confidence = sum(c for _, c, _ in confidence_factors)
        
        # Clamp 0-100
        total_confidence = max(0, min(100, total_confidence))
        
        return {
            'total_confidence': total_confidence,
            'factors': confidence_factors
        }
    
    # =========================================================================
    # FASE 4: DECISIÓN CON EXPLICACIÓN
    # =========================================================================
    
    def make_decision(self, confidence_eval, traj, direction):
        """
        Decide basado en confianza total.
        Genera explicación HUMANA.
        """
        
        total = confidence_eval['total_confidence']
        factors = confidence_eval['factors']
        
        # Construir explicación
        positive_factors = [f for f in factors if f[1] > 0]
        negative_factors = [f for f in factors if f[1] < 0]
        
        # Explicación de factores positivos
        pos_text = ", ".join([f"{name} {detail}" for name, _, detail in positive_factors[:3]])
        
        # Explicación de factores negativos
        neg_text = ", ".join([f"{name} {detail}" for name, _, detail in negative_factors[:3]])
        
        # Decisión
        if total >= 60:
            decision = 'ENTER'
            reason = f"🎯 CONFIANZA {total:.0f}%: {pos_text}"
            if neg_text:
                reason += f" | Alertas: {neg_text}"
        else:
            decision = 'WAIT'
            if total < 35:
                reason = f"🔴 CONFIANZA BAJA {total:.0f}%: {neg_text if neg_text else 'Señales mixtas'}"
            else:
                reason = f"⏸️ CONFIANZA MEDIA {total:.0f}%: Requiere más confirmación"
                if pos_text:
                    reason += f" | Positivos: {pos_text}"
        
        return {
            'decision': decision,
            'confidence': total,
            'reason': reason,
            'factors': factors,
            'trajectories': {
                'adx_delta': traj.get('adx_delta', 0),
                'candle_ratio': traj.get('candle_ratio', 1),
                'exhaustion_velocity': traj.get('exhaustion_velocity', 0),
                'room_pct': traj.get('room_pct', 0)
            }
        }
