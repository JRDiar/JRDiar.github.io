"""
================================================================================
BRAIN VISION UNIFIED v2.0 - VISIÓN DE PELÍCULA (TRAYECTORIAS)
================================================================================
Joel: "Ver la película = ver MOVIMIENTO, no fotos estáticas"
ANTES (v1.0):
- Etiquetas: explosion='NACIENDO', runway='AMPLIO'
- Pattern matching: if A and B and C: ENTER
AHORA (v2.0):
- Trayectorias: adx_motion='ACELERANDO' (calculado de frames)
- Pensamiento: ¿La situación MEJORA o EMPEORA?
================================================================================
"""
from brain_movie_engine import MovieEngine
from brain_grid import grid
from brain_trend_exhaustion import analyze_trend_exhaustion
from brain_frames import get_frame_storage
from brain_force_filter import force_filter
from exchange_connection import fetch_ohlcv
import pandas as pd
class UnifiedVision:
    """
    Cerebro que VE LA PELÍCULA (trayectorias en movimiento).
    NO usa estados estáticos.
    """
    
    def __init__(self):
        self.movie_engine = MovieEngine()
    
    # =========================================================================
    # FASE 1: RECOPILAR FRAMES (LA PELÍCULA)
    # =========================================================================
    
    def gather_intelligence(self, symbol, ladder_analysis):
        """
        Recopila FRAMES de todos los módulos.
        La clave es tener HISTORIA, no solo valores actuales.
        """
        
        direction = ladder_analysis.get('15m', {}).get('direction', 'LATERAL')
        
        # 1. FRAMES - La película real (últimos 5-10 frames)
        frames_15m = list(get_frame_storage(symbol, '15m').get_frames(10))
        frames_5m = list(get_frame_storage(symbol, '5m').get_frames(10))
        
        # 2. EXHAUSTION actual (para room_pct)
        exhaustion_result = analyze_trend_exhaustion(symbol, direction)
        
        # 3. GRID actual (para posición en diagonal)
        ohlcv = fetch_ohlcv(symbol, '5m', 50)
        if ohlcv:
            df = pd.DataFrame(ohlcv, columns=['timestamp','open','high','low','close','volume'])
            grid_result = grid.analyze(symbol, df, ladder_analysis)
        else:
            grid_result = {'status': 'ERROR'}
        
        # 4. FORCE actual
        force_result = force_filter.has_force(symbol, ladder_analysis)
        
        # 5. ZIGZAG (estructura)
        zigzag = ladder_analysis.get('15m', {}).get('zigzag', {})
        
        return {
            'frames': {
                '15m': frames_15m,
                '5m': frames_5m
            },
            'exhaustion': exhaustion_result,
            'grid': grid_result,
            'force': force_result,
            'zigzag': zigzag,
            'direction': direction
        }
    
    # =========================================================================
    # FASE 2: VER LA PELÍCULA (ANALIZAR TRAYECTORIAS)
    # =========================================================================
    
    def see_the_movie(self, intelligence):
        """
        Ve la PELÍCULA analizando TRAYECTORIAS.
        
        NO etiqueta estados estáticos.
        SÍ analiza cómo CAMBIAN los valores frame a frame.
        """
        vision = {}
        
        frames_15m = intelligence.get('frames', {}).get('15m', [])
        
        # Validar suficientes frames
        if len(frames_15m) < 5:
            return {
                'error': 'INSUFICIENTES_FRAMES',
                'message': f'Solo {len(frames_15m)} frames, necesito 5+'
            }
        
        # Usar últimos 5 frames para análisis
        recent_frames = frames_15m[-5:]
        
        # ================================================================
        # A. TRAYECTORIA DE ADX (¿acelerando o muriendo?)
        # ================================================================
        adx_values = []
        for f in recent_frames:
            adx = f.get('adx', {})
            if isinstance(adx, dict):
                adx_values.append(adx.get('value', 0))
            else:
                adx_values.append(0)
        
        # Calcular cambios frame a frame
        if len(adx_values) >= 2:
            adx_changes = [adx_values[i+1] - adx_values[i] for i in range(len(adx_values)-1)]
            
            # Analizar tendencia
            positive_changes = sum(1 for c in adx_changes if c > 0.5)
            negative_changes = sum(1 for c in adx_changes if c < -0.5)
            
            if all(c > 0.5 for c in adx_changes):
                vision['adx_motion'] = 'ACELERANDO_RÁPIDO'
            elif positive_changes >= 3:
                vision['adx_motion'] = 'CRECIENDO'
            elif all(c < -0.5 for c in adx_changes):
                vision['adx_motion'] = 'MURIENDO_RÁPIDO'
            elif negative_changes >= 3:
                vision['adx_motion'] = 'DECRECIENDO'
            else:
                vision['adx_motion'] = 'LATERAL'
            
            # Velocidad total
            vision['adx_velocity'] = sum(adx_changes)
            vision['adx_current'] = adx_values[-1] if adx_values else 0
        else:
            vision['adx_motion'] = 'DESCONOCIDO'
            vision['adx_velocity'] = 0
        
        # ================================================================
        # B. TRAYECTORIA DE VELAS (¿creciendo o achicándose?)
        # ================================================================
        candle_sizes = []
        for f in recent_frames:
            ohlc = f.get('ohlc', {})
            high = ohlc.get('high', 0)
            low = ohlc.get('low', 0)
            if high and low:
                candle_sizes.append(abs(high - low))
        
        if len(candle_sizes) >= 3:
            # Comparar últimas 3 velas
            last_three = candle_sizes[-3:]
            
            if last_three[2] > last_three[1] > last_three[0]:
                vision['candle_motion'] = 'CRECIENDO'
            elif last_three[2] < last_three[1] < last_three[0]:
                vision['candle_motion'] = 'ACHICÁNDOSE'
            else:
                vision['candle_motion'] = 'MIXTO'
        else:
            vision['candle_motion'] = 'DESCONOCIDO'
        
        # ================================================================
        # C. TRAYECTORIA DE MOMENTUM (¿mejorando o empeorando?)
        # ================================================================
        momentum_ages = []
        for f in recent_frames:
            mom = f.get('momentum', {})
            if isinstance(mom, dict):
                momentum_ages.append(mom.get('age', 0))
        
        if len(momentum_ages) >= 2:
            mom_velocity = momentum_ages[-1] - momentum_ages[0]
            
            if mom_velocity > 0:
                vision['momentum_motion'] = 'CRECIENDO'
            elif mom_velocity < 0:
                vision['momentum_motion'] = 'RESETEÁNDOSE'
            else:
                vision['momentum_motion'] = 'ESTABLE'
        else:
            vision['momentum_motion'] = 'DESCONOCIDO'
        
        # ================================================================
        # D. ESPACIO LIBRE (room_pct y su tendencia)
        # ================================================================
        exhaustion = intelligence.get('exhaustion', {})
        room_pct = exhaustion.get('room_pct', 0)
        exhaustion_pct = exhaustion.get('exhaustion_pct', 50)
        moved_pct = exhaustion.get('moved_pct', 0)
        
        vision['room_current'] = room_pct
        vision['exhaustion_current'] = exhaustion_pct
        vision['moved_current'] = moved_pct
        
        # Calcular velocidad de exhaustion si tenemos frames con ese dato
        exhaustion_values = []
        for f in recent_frames:
            exh = f.get('exhaustion_pct')
            if exh is not None:
                exhaustion_values.append(exh)
        
        if len(exhaustion_values) >= 2:
            exhaustion_velocity = exhaustion_values[-1] - exhaustion_values[0]
            
            if exhaustion_velocity > 30:
                vision['exhaustion_motion'] = 'EXPLOSIÓN_RÁPIDA'
            elif exhaustion_velocity > 10:
                vision['exhaustion_motion'] = 'SUBIENDO'
            elif exhaustion_velocity < -10:
                vision['exhaustion_motion'] = 'MEJORANDO'
            else:
                vision['exhaustion_motion'] = 'ESTABLE'
            
            vision['exhaustion_velocity'] = exhaustion_velocity
        else:
            # Si no tenemos historia, usar valor actual
            if exhaustion_pct > 80:
                vision['exhaustion_motion'] = 'MUY_ALTO'
            elif exhaustion_pct < 30:
                vision['exhaustion_motion'] = 'FRESCO'
            else:
                vision['exhaustion_motion'] = 'DESCONOCIDO'
            vision['exhaustion_velocity'] = 0
        
        # ================================================================
        # E. ESTRUCTURA ZIGZAG
        # ================================================================
        zigzag = intelligence.get('zigzag', {})
        pivot_types = zigzag.get('pivot_types', [])
        direction = intelligence.get('direction', 'LATERAL')
        
        if len(pivot_types) >= 2:
            last_two = pivot_types[-2:]
            
            if direction == 'BULLISH':
                if last_two[-1] in ['HH', 'HL']:
                    vision['structure'] = 'SANA'
                elif 'LH' in last_two or 'LL' in last_two:
                    vision['structure'] = 'DETERIORANDO'
                else:
                    vision['structure'] = 'AMBIGUA'
            elif direction == 'BEARISH':
                if last_two[-1] in ['LL', 'LH']:
                    vision['structure'] = 'SANA'
                elif 'HH' in last_two or 'HL' in last_two:
                    vision['structure'] = 'DETERIORANDO'
                else:
                    vision['structure'] = 'AMBIGUA'
            else:
                vision['structure'] = 'LATERAL'
        else:
            vision['structure'] = 'INSUFICIENTE'
        
        # ================================================================
        # F. POSICIÓN EN GRID
        # ================================================================
        grid_data = intelligence.get('grid', {})
        diagonal = grid_data.get('diagonal_data', {})
        position_pct = diagonal.get('price_position_pct', 50)
        
        vision['position_pct'] = position_pct
        vision['grid_status'] = grid_data.get('status', 'UNKNOWN')
        
        # ================================================================
        # G. FUERZA
        # ================================================================
        force = intelligence.get('force', {})
        vision['has_force'] = force.get('has_force', False)
        vision['force_level'] = force.get('force_level', 'NONE')
        
        return vision
    
    # =========================================================================
    # FASE 3: PENSAR COMO TRADER (NO PATTERN MATCHING)
    # =========================================================================
    
    def make_human_decision(self, vision, intelligence):
        """
        Piensa como Joel pensaba.
        
        NO pregunta: "¿Matchea el pattern?"
        SÍ pregunta: "¿La situación MEJORA o EMPEORA?"
        """
        
        # Si hubo error en análisis
        if vision.get('error'):
            return {
                'decision': 'WAIT',
                'confidence': 30,
                'reason': f"⚠️ {vision.get('message', 'Error en análisis')}",
                'archetype': 'ERROR'
            }
        
        # ================================================================
        # EXTRAER TRAYECTORIAS
        # ================================================================
        adx_motion = vision.get('adx_motion', 'DESCONOCIDO')
        candle_motion = vision.get('candle_motion', 'DESCONOCIDO')
        momentum_motion = vision.get('momentum_motion', 'DESCONOCIDO')
        exhaustion_motion = vision.get('exhaustion_motion', 'DESCONOCIDO')
        structure = vision.get('structure', 'AMBIGUA')
        
        room_current = vision.get('room_current', 0)
        exhaustion_current = vision.get('exhaustion_current', 50)
        position_pct = vision.get('position_pct', 50)
        has_force = vision.get('has_force', False)
        
        # ================================================================
        # PENSAR: ¿LA SITUACIÓN MEJORA O EMPEORA?
        # ================================================================
        
        # 1. ¿La FUERZA está construyéndose o muriendo?
        force_is_building = (
            adx_motion in ['ACELERANDO_RÁPIDO', 'CRECIENDO'] and
            candle_motion in ['CRECIENDO', 'MIXTO']
        )
        
        force_is_dying = (
            adx_motion in ['MURIENDO_RÁPIDO', 'DECRECIENDO'] or
            candle_motion == 'ACHICÁNDOSE'
        )
        
        # 2. ¿El ESPACIO está abriéndose o cerrándose?
        space_is_opening = (
            exhaustion_motion == 'MEJORANDO' or
            (exhaustion_current < 40 and room_current > 2)
        )
        
        space_is_closing = (
            exhaustion_motion in ['EXPLOSIÓN_RÁPIDA', 'SUBIENDO', 'MUY_ALTO'] or
            room_current < 1 or
            position_pct > 85  # Cerca del techo
        )
        
        # 3. ¿La ESTRUCTURA es sana?
        structure_is_healthy = (structure == 'SANA')
        structure_is_broken = (structure == 'DETERIORANDO')
        
        # ================================================================
        # DECISIÓN BASADA EN CONTEXTO COMPLETO
        # ================================================================
        
        # ----- CASO 1: TODO MEJORA (como ASR +2.00%) -----
        if force_is_building and space_is_opening and structure_is_healthy:
            return {
                'decision': 'ENTER',
                'confidence': 90,
                'reason': f"🎯 CONTEXTO IDEAL: Fuerza naciendo ({adx_motion}), espacio abriéndose ({exhaustion_motion}), estructura sana",
                'archetype': 'IDEAL_CONTEXT',
                'story': {
                    'adx': adx_motion,
                    'candles': candle_motion,
                    'exhaustion': exhaustion_motion,
                    'structure': structure
                }
            }
        
        # ----- CASO 2: Fuerza naciendo pero espacio cerrándose (como AKE -1.78%) -----
        if force_is_building and space_is_closing:
            return {
                'decision': 'WAIT',
                'confidence': 85,
                'reason': f"⚠️ EXPLOSIÓN RÁPIDA: Fuerza sí ({adx_motion}) pero espacio cerrándose ({exhaustion_motion}) - compraría techo",
                'archetype': 'BUYING_TOP',
                'story': {
                    'adx': adx_motion,
                    'exhaustion': exhaustion_motion,
                    'room': room_current
                }
            }
        
        # ----- CASO 3: Fuerza muriendo (como DGRAM -3.00%) -----
        if force_is_dying:
            return {
                'decision': 'WAIT',
                'confidence': 80,
                'reason': f"🔴 FUERZA MURIENDO: ADX {adx_motion}, velas {candle_motion} - explosión terminó",
                'archetype': 'DYING_FORCE',
                'story': {
                    'adx': adx_motion,
                    'candles': candle_motion
                }
            }
        
        # ----- CASO 4: Fuerza activa con espacio OK -----
        if has_force and not space_is_closing and structure_is_healthy:
            return {
                'decision': 'ENTER',
                'confidence': 75,
                'reason': f"📈 TENDENCIA ACTIVA: Fuerza presente, espacio OK, estructura sana",
                'archetype': 'TREND_CONTINUATION',
                'story': {
                    'adx': adx_motion,
                    'exhaustion': exhaustion_motion
                }
            }
        
        # ----- CASO 5: Estructura rota -----
        if structure_is_broken:
            return {
                'decision': 'WAIT',
                'confidence': 75,
                'reason': f"📉 ESTRUCTURA ROTA: ZigZag deteriorándose",
                'archetype': 'BROKEN_STRUCTURE',
                'story': {'structure': structure}
            }
        
        # ----- CASO 6: Sin espacio aunque fuerza OK -----
        if space_is_closing and not force_is_building:
            return {
                'decision': 'WAIT',
                'confidence': 70,
                'reason': f"🚫 SIN ESPACIO: Room {room_current:.1f}%, exhaustion {exhaustion_current:.0f}%",
                'archetype': 'NO_RUNWAY',
                'story': {
                    'room': room_current,
                    'exhaustion': exhaustion_current
                }
            }
        
        # ----- CASO 7: Contexto ambiguo -----
        return {
            'decision': 'WAIT',
            'confidence': 50,
            'reason': f"❓ CONTEXTO AMBIGUO: ADX {adx_motion}, exhaustion {exhaustion_motion}, estructura {structure}",
            'archetype': 'AMBIGUOUS',
            'story': {
                'adx': adx_motion,
                'exhaustion': exhaustion_motion,
                'structure': structure
            }
        }
    
    # =========================================================================
    # FUNCIÓN PRINCIPAL
    # =========================================================================
    
    def decide_with_unified_vision(self, symbol, ladder_analysis):
        """
        Punto de entrada principal.
        """
        # 1. Recopilar FRAMES
        intelligence = self.gather_intelligence(symbol, ladder_analysis)
        
        # 2. Ver la PELÍCULA (analizar trayectorias)
        vision = self.see_the_movie(intelligence)
        
        # 3. PENSAR (no pattern matching)
        decision = self.make_human_decision(vision, intelligence)
        
        # 4. Agregar datos
        decision['vision'] = vision
        decision['intelligence_summary'] = {
            'exhaustion_pct': intelligence['exhaustion'].get('exhaustion_pct', 50),
            'room_pct': intelligence['exhaustion'].get('room_pct', 0),
            'force_level': intelligence['force'].get('force_level', 'NONE'),
            'grid_status': intelligence['grid'].get('status', 'UNKNOWN'),
            'frames_count': len(intelligence['frames'].get('15m', []))
        }
        
        return decision
# =============================================================================
# FUNCIÓN DE CONVENIENCIA
# =============================================================================
def decide_unified(symbol, ladder_analysis):
    """Wrapper para uso fácil."""
    unified = UnifiedVision()
    return unified.decide_with_unified_vision(symbol, ladder_analysis)
