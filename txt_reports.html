def check_btc_alignment(score=0, strict=None):
    """Verifica si BTC está estable. Ahora CONDICIONAL por score.
    
    Args:
        score: Score de la moneda analizada
        strict: Si True, bloquea siempre. Si False/None, solo bloquea scores bajos
    
    Returns:
        bool: True si OK para operar, False si debe bloquearse
    """
    if strict is None:
        strict = BTC_ALIGNMENT_STRICT  # Usar config
    
    try:
        btc_ohlcv = fetch_ohlcv('BTC/USDT:USDT', '15m', 10)
        if not btc_ohlcv or len(btc_ohlcv) < 5:
            return True  # Sin datos, no bloqueamos
        
        df_btc = pd.DataFrame(btc_ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
        last_5_closes = df_btc['close'].tail(5).values
        btc_change_pct = ((last_5_closes[-1] - last_5_closes[0]) / last_5_closes[0]) * 100
        
        # Threshold más suave: -3% en vez de -2%
        if btc_change_pct < -3.0:
            # Si score es alto (>70), permitir operar igual (desacoplamiento)
            if not strict and score >= 70:
                print(f"🔓 BTC cayendo {btc_change_pct:.2f}% PERO score {score} alto. Permitiendo.")
                return True
            else:
                print(f"⚠️ BTC cayendo {btc_change_pct:.2f}% en 15M. Protección activada.")
                return False
        return True
    except Exception as e:
        print(f"⚠️ Error verificando BTC: {e}")
        return True  # Error = no bloquear

def analyze_4h(symbol):
    """
    4H: Define contexto dominante y clima de mercado
    
    NUNCA RECHAZA - Solo informa dirección y fuerza
    
    Args:
        symbol (str): Par trading (ej: 'BTC/USDT')
    
    Returns:
        dict: Análisis completo con direction, strength, indicadores
        None: Solo si no hay datos (caso extremo)
    """
    # Fetch datos
    ohlcv = fetch_ohlcv(symbol, '4h', 100)
    if not ohlcv or len(ohlcv) < 50:
        return None

    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    current_price = float(df['close'].iloc[-1])
    
    # ===== ADX con memoria =====
    adx_data = calculate_adx_with_memory(df, period=14, lookback=10)
    
    # ADX ANTICIPATORIO: Proyectar evolución
    adx_now = adx_data['adx_current']
    adx_trend = adx_data['adx_trend']
    
    # Calcular velocidad ADX (puntos por vela)
    if len(adx_data['adx_history']) >= 5:
        adx_velocity = (adx_data['adx_history'][-1] - adx_data['adx_history'][0]) / 5
    else:
        adx_velocity = 0.0
    
    # Proyección 3 velas adelante (12 horas en 4H)
    adx_proyectado = adx_now + (adx_velocity * 3)
    
    # ===== Determinar dirección base por dominancia =====
    if adx_data['dominance'] == 'DI+':
        direction = 'BULLISH'
    elif adx_data['dominance'] == 'DI-':
        direction = 'BEARISH'
    else:
        direction = 'LATERAL'

    # ===== Evaluar fuerza ADX =====
    strength = 'WEAK'  # Inicializar en WEAK
    
    # CASO 1: ADX ya >25 (perfecto)
    if adx_now >= 25:
        strength = 'STRONG'
    elif adx_now >= 17 and adx_velocity > 1.5:
        strength = 'MODERATE'
    else:
        strength = 'WEAK'

    if adx_trend == 'decreciendo' and adx_velocity < -0.8:
        strength = 'WEAK'

    if direction != 'LATERAL' and abs(adx_data['spread']) < 5.0:
        strength = 'WEAK'

    momentum_data = calculate_momentum_with_memory(df, lookback=10)
    # FIX: Aceptar 'lime' como bullish y 'maroon' como bearish
    expected_colors = ['green', 'lime'] if direction == 'BULLISH' else ['red', 'maroon']

    if direction != 'LATERAL' and momentum_data['color_current'] not in expected_colors:
        direction = 'LATERAL'

    if momentum_data['consecutive_count'] < 3:
        strength = 'WEAK'

    rsi_data = calculate_rsi_with_memory(df, period=14, lookback=10)
    emas_data = calculate_emas_with_memory(df, periods=[9, 21, 50, 200], lookback=10)

    if ENABLE_4H_STRUCTURE:
        structure = detect_price_structure(df, lookback=20)
    else:
        structure = {'structure': 'N/A'}

    if ENABLE_4H_ZIGZAG:
        zigzag = calculate_zigzag(df, threshold_pct=3.0)
    else:
        zigzag = {'zigzag': 'N/A'}

    # ROADMAP: Detectar Soportes y Resistencias Mayores
    sr_data = detect_support_resistance_with_memory(df, lookback=50)
    
    # VOLUME PROFILE (Exponer para Market Force)
    vp_data = calculate_volume_profile_with_memory(df, lookback=200)

    print(f"📊 4H | {symbol} | {direction} {strength} | ADX:{adx_data['adx_current']:.1f} DI+:{adx_data['di_plus_current']:.1f} DI-:{adx_data['di_minus_current']:.1f} | Mom:{momentum_data['consecutive_count']}v | Estructura:{structure['structure']}")

    score_4h = 0
    if strength == 'STRONG':
        score_4h += 20
    elif strength == 'MODERATE':
        score_4h += 10
    else:
        score_4h += 0
    if adx_trend == 'creciendo':
        score_4h += 5
    if momentum_data['consecutive_count'] >= 5:
        score_4h += 5
    if abs(adx_data['spread']) > 7:
        score_4h += 5

    return {
        'direction': direction,
        'strength': strength,
        'score_4h': score_4h,
        'adx': adx_data,
        'momentum': momentum_data,
        'rsi': rsi_data,
        'emas': emas_data,
        'structure': structure,
        'structure': structure,
        'zigzag': zigzag,
        'zigzag': zigzag,
        'sr': sr_data, # ROADMAP DATA
        'volume_profile': vp_data, # EXPOSED FOR MARKET FORCE
        'current_price': current_price,
        'current_price': current_price,
        'context': f"4H {direction} {strength} - ADX {adx_data['adx_current']:.1f}",
        'timestamp': datetime.now().isoformat()

def analyze_15m(symbol, strict=None):
    """15M: Timing de entrada - Ahora con FUZZY SCORING"""
    # Auto-detectar modo si no se especifica
    if strict is None:
        strict = not ENABLE_INTELLIGENT_MODE  # Intelligent mode = No estricto
    
    ohlcv = fetch_ohlcv(symbol, '15m', 100)
    if not ohlcv or len(ohlcv) < 50:
        return None

    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    current_price = float(df['close'].iloc[-1])
    
    # --- ANTI-PUMP MEJORADO (Más inteligente) ---
    vol_avg = df['volume'].iloc[:-1].mean()
    last_vol = df['volume'].iloc[-1]
    last_candle_pct = abs((df['close'].iloc[-1] - df['open'].iloc[-1]) / df['open'].iloc[-1]) * 100
    
    # Solo rechazar si volumen > 15x Y vela > 20% (pump claro)
    if last_vol > vol_avg * 15 and last_candle_pct > 20:
        print(f"⚠️ PUMP EXTREMO en {symbol}: Vol {last_vol/vol_avg:.1f}x + Vela {last_candle_pct:.1f}%. Rechazando.")
        return None
    # -----------------------------------------------

    adx_data = calculate_adx_with_memory(df, period=14, lookback=10)
    momentum_data = calculate_momentum_with_memory(df, lookback=10)
    rsi_data = calculate_rsi_with_memory(df, period=14, lookback=10)
    emas_data = calculate_emas_with_memory(df, periods=[9, 21, 50], lookback=10)
    
    # Determinar dirección siempre (no rechazar)
    if adx_data['dominance'] == 'DI+':
        direction = 'BULLISH'
    elif adx_data['dominance'] == 'DI-':
        direction = 'BEARISH'
    else:
        direction = 'LATERAL'
    
    expected_colors = ['green', 'lime'] if direction == 'BULLISH' else ['red', 'maroon']
    
    # ===== MODO ESTRICTO (Solo en modo cascada tradicional) =====
    if strict:
        # Rechazos HARD solo si es estricto
        if adx_data['adx_current'] < 17:  # Reducido de 20 → 17
            return None
        if abs(adx_data['spread']) < 2.0:  # Reducido de 3.0 → 2.0
            return None
        if momentum_data['color_current'] not in expected_colors:
            return None
        if momentum_data['consecutive_count'] < 1:  # Reducido de 2 → 1
            return None
    
    # ===== FUZZY SCORING (Siempre calcular) =====
    score_15m = 0
    
    # ADX: Gradiente (no binario)
    adx_current = adx_data['adx_current']
    if adx_current >= 25:
        score_15m += 15
    elif adx_current >= 20:
        score_15m += 10
    elif adx_current >= 17:
        score_15m += 5
    elif adx_current >= 14:
        score_15m += 2
    # Si ADX < 14, no suma pero no rechaza
    
    # Spread DI: Claridad direccional
    spread = abs(adx_data['spread'])
    if spread >= 7:
        score_15m += 10
    elif spread >= 4:
        score_15m += 7
    elif spread >= 2:
        score_15m += 4
    elif spread >= 1:
        score_15m += 2
    
    # Momentum alineado
    if momentum_data['color_current'] in expected_colors:
        score_15m += 8
        if momentum_data['consecutive_count'] >= 3:
            score_15m += 5
        elif momentum_data['consecutive_count'] >= 2:
            score_15m += 3
    
    # RSI razonable
    rsi = rsi_data['rsi_current']
    if rsi and 40 <= rsi <= 70:
        score_15m += 5
    elif rsi and 30 <= rsi <= 80:
        score_15m += 2
    
    # EMAs alineadas
    if emas_data.get('emas_aligned', False):
        score_15m += 5
    
    # Tendencia ADX
    if adx_data['adx_trend'] == 'creciendo':
        score_15m += 3
    
    # ===== DETERMINAR STRENGTH =====
    if adx_current >= 25 and spread >= 5:
        strength = 'STRONG'
    elif adx_current >= 20 and spread >= 3:
        strength = 'MODERATE'
    elif adx_current >= 17:
        strength = 'WEAK'
    else:
        strength = 'VERY_WEAK'
    
    print(f"      🎯 15M | {symbol} | {direction} {strength} | ADX:{adx_current:.1f} Spread:{spread:.1f} Mom:{momentum_data['consecutive_count']}v | Score15M:{score_15m}")
    
    return {
        'direction': direction,
        'strength': strength,
        'score_15m': score_15m,
        'current_price': current_price,
        'adx': adx_data,
        'momentum': momentum_data,
        'rsi': rsi_data,
        'emas': emas_data,
        'context': f"15M {direction} {strength} - Score {score_15m}",
        'timestamp': datetime.now().isoformat()
    }

def analyze_temporal_context(symbol):
    """
    Analiza el contexto temporal reciente (últimos 60 min) usando la DB.
    Busca estabilidad y consistencia.
    """
    try:
        history = db_manager.get_recent_context(symbol, limit_minutes=60)
        if not history or len(history) < 10:
            return {'stability': 'DESCONOCIDO', 'bonus': 0}
        
        # 1. Verificar cambios de dirección (inestabilidad)
        directions = [row['direction'] for row in history]
        flips = 0
        for i in range(1, len(directions)):
            if directions[i] != directions[i-1]:
                flips += 1
        
        # 2. Verificar consistencia de ADX
        adx_values = [row['adx_15m'] for row in history]
        adx_rising = 0
        for i in range(1, len(adx_values)):
            if adx_values[i] > adx_values[i-1]:
                adx_rising += 1
        
        # Evaluación
        bonus = 0
        stability = 'ESTABLE'
        
        if flips > 3:
            stability = 'INESTABLE'
            bonus -= 10  # Penalización por inestabilidad
        elif flips == 0:
            stability = 'MUY_ESTABLE'
            bonus += 10  # Bonus por tendencia sostenida
            
        if adx_rising > len(adx_values) * 0.7:
            bonus += 5  # Bonus si ADX sube la mayoría del tiempo
            
        return {'stability': stability, 'bonus': bonus, 'flips': flips}
        
    except Exception as e:
        print(f"⚠️ Error analizando contexto temporal: {e}")
        return {'stability': 'ERROR', 'bonus': 0}

def _extract_score(score_value, default=0):
    """
    Recursively extract integer score from potentially nested dict.
    
    Args:
        score_value: The value to extract (can be int, float, dict, or None)
        default: Default value if extraction fails
    
    Returns:
        int: Extracted integer score
    """
    if score_value is None:
        return default
    if isinstance(score_value, (int, float)):
        return int(score_value)
    if isinstance(score_value, dict):
        # Recursively extract from nested dict
        return _extract_score(score_value.get('score', default), default)
    # Fallback for any other type
    return default

def calculate_zigzag(df, percent=None, threshold_pct=None):
    """
    Calcula los puntos ZigZag y la ESTRUCTURA DE MERCADO (HH, HL, LH, LL).
    
    Args:
        percent/threshold_pct: Porcentaje mínimo de reversión.
        
    Returns:

    pct = threshold_pct if threshold_pct is not None else (percent if percent is not None else 5)
    
    if len(df) < 2:
        return {'pivots': [], 'values': [], 'pivot_types': [], 'last_structure': "N/A"}
        
    close = df['close'].values
    pivots = [0]
    values = [close[0]]
    last_pivot = close[0]
    direction = 0  # 0: indefinido, 1: arriba, -1: abajo
    
    # 1. Identificar Pivotes
    for i in range(1, len(close)):
        change = (close[i] - last_pivot) / last_pivot * 100
        if direction == 0:
            if abs(change) >= pct:
                direction = 1 if change > 0 else -1
                pivots.append(i)
                values.append(close[i])
                last_pivot = close[i]
        else:
            if (direction == 1 and close[i] < last_pivot * (1 - pct/100)) or \
               (direction == -1 and close[i] > last_pivot * (1 + pct/100)):
                direction *= -1
                pivots.append(i)
                values.append(close[i])
                last_pivot = close[i]

    # 2. Clasificar Estructura (HH, HL, LH, LL)
    pivot_types = ["Start"] * len(values)
    if len(values) >= 2:
        last_high = None
        last_low = None
        
        # Determinar si el primer movimiento fue High o Low
        # Si values[1] > values[0], el 1 es un High
        
        for i in range(1, len(values)):
            curr_val = values[i]
            prev_val = values[i-1]
            
            is_high = False
            is_low = False
            
            # En ZigZag, los puntos alternan.
            # Si values[i] > values[i-1], entonces 'i' es un pico (High)
            # Si values[i] < values[i-1], entonces 'i' es un valle (Low)
            
            if curr_val > prev_val:
                is_high = True
            else:
                is_low = True
                
            type_str = ""
            if is_high:
                if last_high is None: type_str = "H"
                elif curr_val > last_high: type_str = "HH" # Higher High
                elif curr_val < last_high: type_str = "LH" # Lower High
                else: type_str = "DH" # Double High
                last_high = curr_val
            elif is_low:
                if last_low is None: type_str = "L"
                elif curr_val > last_low: type_str = "HL" # Higher Low
                elif curr_val < last_low: type_str = "LL" # Lower Low
                else: type_str = "DL" # Double Low
                last_low = curr_val
            
            pivot_types[i] = type_str

    # 3. Resumen de última estructura
    last_structure = "Indefinido"
    if len(pivot_types) >= 2:
        # Tomar los últimos 2 tipos significativos
        last_structure = f"{pivot_types[-2]} -> {pivot_types[-1]}"

    return {
        'pivots': pivots, 
        'values': values, 
        'pivot_types': pivot_types, 
        'last_structure': last_structure
    }

def calculate_linear_regression_slope(values):
    """
    Calcula pendiente de regresión lineal simple
    
    Args:
        values: Lista de valores (últimas N velas)
        
    Returns:
        float: Pendiente (slope)
    """
    n = len(values)
    if n < 2:
        return 0.0
    
    x = np.arange(n)
    y = np.array(values)
    
    # y = mx + b → m = (n*Σxy - Σx*Σy) / (n*Σx² - (Σx)²)
    x_mean = np.mean(x)
    y_mean = np.mean(y)
    
    numerator = np.sum((x - x_mean) * (y - y_mean))
    denominator = np.sum((x - x_mean) ** 2)
    
    if denominator == 0:
        return 0.0
    
    slope = numerator / denominator
    return slope


def classify_trend(slope, threshold=0.3):

    Returns:
        str: 'creciendo', 'decreciendo', 'estable'
    """

        return 'estable'


def calculate_velocity(df, lookback=10):
    """
    Calcula velocidad de precio con memoria temporal
    
    Args:
        df: DataFrame con columna 'close'
        lookback: Velas a analizar
        
    Returns:
        dict: {
            'velocity_current': float (cambio precio por vela),
            'velocity_history': [list últimos lookback],
            'trend': 'acelerando'/'desacelerando'/'constante'
        }
    """
    if len(df) < lookback + 1:
        return {
            'velocity_current': 0.0,
            'velocity_history': [],
            'trend': 'constante'
        }
    
    prices = df['close'].values
    velocities = []
    
    # Calcular velocidad para cada par de velas
    for i in range(len(prices) - lookback, len(prices)):
        if i > 0:
            vel = prices[i] - prices[i-1]
            velocities.append(vel)
    
    velocity_current = velocities[-1] if velocities else 0.0
    
    # Calcular tendencia (velocidad creciendo/decreciendo)
    if len(velocities) >= 3:
        # Pendiente de las velocidades
        x = list(range(len(velocities)))
        y = velocities
        n = len(x)
        
        sum_x = sum(x)
        sum_y = sum(y)
        sum_xy = sum(x[i] * y[i] for i in range(n))
        sum_x2 = sum(x[i] ** 2 for i in range(n))
        
        slope = (n * sum_xy - sum_x * sum_y) / (n * sum_x2 - sum_x ** 2) if (n * sum_x2 - sum_x ** 2) != 0 else 0
        
        if slope > 0.0001:
            trend = 'acelerando'
        elif slope < -0.0001:
            trend = 'desacelerando'
        else:
            trend = 'constante'
    else:
        trend = 'constante'
    
    return {
        'velocity_current': velocity_current,
        'velocity_history': velocities,
        'trend': trend
    }

def calculate_adx_with_memory(df, period=14, lookback=10):
    """
    ADX con Wilder's Smoothing EXACTO del Bot 3.5.
    Retorna dict con valores actuales + memoria temporal.
    """
    df = df.copy()
    n = len(df)
    
    if n < period + 1:
        return {
            'adx_current': 0.0,
            'adx_history': [],
            'adx_trend': 'insuficiente',
            'di_plus_current': 0.0,
            'di_plus_history': [],
            'di_plus_trend': 'insuficiente',
            'di_minus_current': 0.0,
            'di_minus_history': [],
            'di_minus_trend': 'insuficiente',
            'dominance': 'neutral',
            'spread': 0.0
        }
    
    high = df['high'].values
    low = df['low'].values
    close = df['close'].values
    
    # 1. Calcular True Range y Directional Movements
    tr = np.zeros(n)
    dm_plus = np.zeros(n)
    dm_minus = np.zeros(n)
    
    for i in range(1, n):
        prev_close = close[i-1]
        prev_high = high[i-1]
        prev_low = low[i-1]
        
        # True Range (máximo de 3 valores)
        tr1 = high[i] - low[i]
        tr2 = abs(high[i] - prev_close)
        tr3 = abs(low[i] - prev_close)
        tr[i] = max(tr1, tr2, tr3)
        
        # Directional Movement
        move_up = high[i] - prev_high
        move_down = prev_low - low[i]
        
        if move_up > move_down and move_up > 0:
            dm_plus[i] = move_up
        else:
            dm_plus[i] = 0
            
        if move_down > move_up and move_down > 0:
            dm_minus[i] = move_down
        else:
            dm_minus[i] = 0
    
    # 2. Wilder's Smoothing MANUAL
    smoothed_tr = np.zeros(n)
    smoothed_dm_plus = np.zeros(n)
    smoothed_dm_minus = np.zeros(n)
    
    for i in range(n):
        if i < period:
            if i == period - 1:
                # Primera suavización: suma de los primeros 'period' valores
                smoothed_tr[i] = tr[:i+1].sum()
                smoothed_dm_plus[i] = dm_plus[:i+1].sum()
                smoothed_dm_minus[i] = dm_minus[:i+1].sum()
            else:
                smoothed_tr[i] = np.nan
                smoothed_dm_plus[i] = np.nan
                smoothed_dm_minus[i] = np.nan
        else:
            # Wilder's smoothing: prev - (prev/period) + current
            smoothed_tr[i] = smoothed_tr[i-1] - (smoothed_tr[i-1] / period) + tr[i]
            smoothed_dm_plus[i] = smoothed_dm_plus[i-1] - (smoothed_dm_plus[i-1] / period) + dm_plus[i]
            smoothed_dm_minus[i] = smoothed_dm_minus[i-1] - (smoothed_dm_minus[i-1] / period) + dm_minus[i]
    
    # 3. Calcular DI+ y DI- (con protección contra división por cero)
    di_plus = np.zeros(n)
    di_minus = np.zeros(n)
    for i in range(n):
        if smoothed_tr[i] != 0 and not np.isnan(smoothed_tr[i]):
            di_plus[i] = (smoothed_dm_plus[i] / smoothed_tr[i]) * 100
            di_minus[i] = (smoothed_dm_minus[i] / smoothed_tr[i]) * 100
        else:
            di_plus[i] = np.nan
            di_minus[i] = np.nan
    
    # 4. Calcular DX y ADX
    dx = np.zeros(n)
    for i in range(n):
        denominator = di_plus[i] + di_minus[i]
        if denominator != 0 and not np.isnan(denominator):
            dx[i] = np.abs(di_plus[i] - di_minus[i]) / denominator * 100
        else:
            dx[i] = np.nan
    
    # ADX es la media móvil simple de DX
    adx = np.zeros(n)
    for i in range(n):
        if i < period * 2 - 1:
            adx[i] = np.nan
        elif i == period * 2 - 1:
            # Calcular media ignorando NaN
            valid_dx = dx[period:i+1]
            valid_dx = valid_dx[~np.isnan(valid_dx)]
            if len(valid_dx) > 0:
                adx[i] = np.nanmean(valid_dx)
            else:
                adx[i] = np.nan
        else:
            # Continuar con Wilder's smoothing para ADX también
            if not np.isnan(adx[i-1]) and not np.isnan(dx[i]):
                adx[i] = (adx[i-1] * (period - 1) + dx[i]) / period
            else:
                adx[i] = np.nan
    
    # 5. Extraer historia y calcular tendencias
    start_idx = max(0, n - lookback)
    
    adx_history = [float(x) for x in adx[start_idx:n] if not np.isnan(x)]
    di_plus_history = [float(x) for x in di_plus[start_idx:n] if not np.isnan(x)]
    di_minus_history = [float(x) for x in di_minus[start_idx:n] if not np.isnan(x)]
    
    adx_current = float(adx[-1]) if not np.isnan(adx[-1]) else 0.0
    di_plus_current = float(di_plus[-1]) if not np.isnan(di_plus[-1]) else 0.0
    di_minus_current = float(di_minus[-1]) if not np.isnan(di_minus[-1]) else 0.0
    
    # 6. Determinar tendencias
    def get_trend(history):
        if len(history) < 3:
            return 'insuficiente'
        if history[-1] > history[-2] > history[-3]:
            return 'creciendo'
        elif history[-1] < history[-2] < history[-3]:
            return 'decreciendo'
        else:
            return 'estable'
    
    adx_trend = get_trend(adx_history)
    di_plus_trend = get_trend(di_plus_history)
    di_minus_trend = get_trend(di_minus_history)
    
    # 7. Dominancia y spread
    spread = abs(di_plus_current - di_minus_current)
    
    if di_plus_current > di_minus_current + 5:
        dominance = 'DI+'
    elif di_minus_current > di_plus_current + 5:
        dominance = 'DI-'
    else:
        dominance = 'neutral'
    
    return {
        'adx_current': adx_current,
        'adx_history': adx_history,
        'adx_trend': adx_trend,
        'di_plus_current': di_plus_current,
        'di_plus_history': di_plus_history,
        'di_plus_trend': di_plus_trend,
        'di_minus_current': di_minus_current,
        'di_minus_history': di_minus_history,
        'di_minus_trend': di_minus_trend,
        'dominance': dominance,
        'spread': spread
    }

def calculate_momentum_with_memory(df, lookback=10):
    """
    LazyBear Momentum EXACTO del 3.5 con regresión lineal manual.
    Devuelve color, historia, fuerza y valores de bandas.
    Requiere columnas: 'high', 'low', 'close'
    """
    df = df.copy()
    n = len(df)
    
    if n < max(BB_LENGTH, KC_LENGTH) + 2:
        return {
            'color_current': 'neutral',
            'color_history': [],
            'consecutive_count': 0,
            'strength': 'débil',
            'value': 0.0,
            'squeeze_active': False
        }
    
    source = df['close']
    high = df['high']
    low = df['low']
    
    # 1. Bollinger Bands
    basis = source.rolling(BB_LENGTH).mean()
    dev = BB_MULT * source.rolling(BB_LENGTH).std()
    upperBB = basis + dev
    lowerBB = basis - dev
    
    # 2. Keltner Channels
    ma = source.rolling(KC_LENGTH).mean()
    
    if USE_TRUE_RANGE:
        prev_close = source.shift(1)
        tr1 = high - low
        tr2 = abs(high - prev_close)
        tr3 = abs(low - prev_close)
        true_range = pd.concat([tr1, tr2, tr3], axis=1).max(axis=1)
        range_ma = true_range.rolling(KC_LENGTH).mean()
    else:
        range_ma = (high - low).rolling(KC_LENGTH).mean()
    
    upperKC = ma + range_ma * KC_MULT
    lowerKC = ma - range_ma * KC_MULT
    
    # 3. Squeeze Detection (NO bloquear, solo marcar)
    sqzOn = (lowerBB > lowerKC) & (upperBB < upperKC)
    squeeze_active = bool(sqzOn.iloc[-1])
    # NOTA: Continuar calculando momentum aunque haya squeeze
    # El squeeze indica consolidación pero el momentum sigue siendo válido
    
    # 4. Regresión Lineal MANUAL (LazyBear exacto)
    highest_high = high.rolling(KC_LENGTH).max()
    lowest_low = low.rolling(KC_LENGTH).min()
    avg_hl = (highest_high + lowest_low) / 2
    sma_close = source.rolling(KC_LENGTH).mean()
    avg_combined = (avg_hl + sma_close) / 2
    
    momentum_values = []
    for i in range(len(df)):
        if i < KC_LENGTH - 1:
            momentum_values.append(np.nan)
        else:
            start_idx = i - KC_LENGTH + 1
            end_idx = i + 1
            y_values = (source.iloc[start_idx:end_idx] - avg_combined.iloc[start_idx:end_idx]).values
            x_values = np.arange(KC_LENGTH)
            x_mean = np.mean(x_values)
            y_mean = np.mean(y_values)
            
            numerator = np.sum((x_values - x_mean) * (y_values - y_mean))
            denominator = np.sum((x_values - x_mean) ** 2)
            
            if denominator != 0:
                slope = numerator / denominator
                val = slope * (KC_LENGTH - 1 - x_mean) + y_mean
                momentum_values.append(val)
            else:
                momentum_values.append(0)
    
    momentum = pd.Series(momentum_values)
    prev_momentum = momentum.shift(1)
    
    # 5. Colores LazyBear (lime/green/red/maroon)
    color_conditions = []
    for i in range(len(momentum)):
        if pd.isna(momentum.iloc[i]) or pd.isna(prev_momentum.iloc[i]):
            color_conditions.append('neutral')
        elif momentum.iloc[i] > 0:
            if momentum.iloc[i] > prev_momentum.iloc[i]:
                color_conditions.append('lime')
            else:
                color_conditions.append('green')
        else:
            if momentum.iloc[i] < prev_momentum.iloc[i]:
                color_conditions.append('red')
            else:
                color_conditions.append('maroon')
    
    # 6. Historia y fuerza
    start_idx = max(0, n - lookback)
    color_history = color_conditions[start_idx:n]
    color_current = color_conditions[-1]
    
    consecutive_count = 1
    for i in range(len(color_history) - 2, -1, -1):
        if color_history[i] == color_current:
            consecutive_count += 1
        else:
            break
    
    if consecutive_count >= 5:
        strength = 'fuerte'
    elif consecutive_count >= 3:
        strength = 'moderada'
    else:
        strength = 'débil'
    
    return {
        'color_current': color_current,
        'color_history': color_history,
        'consecutive_count': consecutive_count,
        'strength': strength,
        'value': float(momentum.iloc[-1]),
        'squeeze_active': squeeze_active
    }

def calculate_rsi_with_memory(df, period=14, lookback=10):
    """
    Calcula RSI con memoria temporal (Wilder's Smoothing exacto).
    Devuelve dict con valor actual, historia y tendencia.
    """
    close = df['close'].values
    n = len(close)
    if n < period + 1:
        return {
            'rsi_current': 0.0,
            'rsi_history': [],
            'rsi_trend': 'insuficiente'
        }
    deltas = np.diff(close)
    seed = deltas[:period]
    up = seed[seed >= 0].sum() / period
    down = -seed[seed < 0].sum() / period
    rs = up / down if down != 0 else 0
    rsi = np.zeros(n)
    rsi[:period] = np.nan
    rsi[period] = 100. - 100. / (1. + rs)
    up_avg = up
    down_avg = down
    for i in range(period + 1, n):
        delta = deltas[i - 1]
        upval = max(delta, 0)
        downval = -min(delta, 0)
        up_avg = (up_avg * (period - 1) + upval) / period
        down_avg = (down_avg * (period - 1) + downval) / period
        rs = up_avg / down_avg if down_avg != 0 else 0
        rsi[i] = 100. - 100. / (1. + rs)
    start_idx = max(0, n - lookback)
    rsi_history = [float(x) for x in rsi[start_idx:n] if not np.isnan(x)]
    rsi_current = float(rsi[-1]) if not np.isnan(rsi[-1]) else 0.0
    # Tendencia simple
    if len(rsi_history) >= 3:
        if rsi_history[-1] > rsi_history[-2] > rsi_history[-3]:
            trend = 'creciendo'
        elif rsi_history[-1] < rsi_history[-2] < rsi_history[-3]:
            trend = 'decreciendo'
        else:
            trend = 'estable'
    else:
        trend = 'insuficiente'
    return {
        'rsi_current': rsi_current,
        'rsi_history': rsi_history,
        'rsi_trend': trend
    }

    aligned = all(emas[periods[i]] > emas[periods[i+1]] for i in range(len(periods)-1))
    # Clasificación simple
    if aligned:
        classification = 'CERCANAS'
    else:
        classification = 'SEPARADAS'
    return {
        'emas': emas,
        'emas_history': emas_history,
        'emas_aligned': aligned,
        'classification': classification
    }

def detect_price_structure(df, lookback=20, pivot_window=5):
    """
    Detecta estructura de precio: pivotes altos/bajos (HH/HL/LH/LL).
    Retorna dict con listas de pivotes y estructura dominante.
    """
    highs = df['high'].values[-lookback:]
    lows = df['low'].values[-lookback:]
    closes = df['close'].values[-lookback:]
    pivot_highs = []
    pivot_lows = []

    for i in range(pivot_window, len(highs) - pivot_window):
        window = highs[i - pivot_window:i + pivot_window + 1]
        if highs[i] == max(window):
            pivot_highs.append(highs[i])
        window = lows[i - pivot_window:i + pivot_window + 1]
        if lows[i] == min(window):
            pivot_lows.append(lows[i])

    # Clasificación simple de estructura
    structure = 'INDETERMINADO'
    if len(pivot_highs) >= 2 and len(pivot_lows) >= 2:
        if pivot_highs[-1] > pivot_highs[-2] and pivot_lows[-1] > pivot_lows[-2]:
            structure = 'BULLISH'
        elif pivot_highs[-1] < pivot_highs[-2] and pivot_lows[-1] < pivot_lows[-2]:
            structure = 'BEARISH'
        else:
            structure = 'LATERAL'

    return {
        'pivot_highs': pivot_highs,
        'pivot_lows': pivot_lows,
        'structure': structure
    }

def detect_support_resistance_with_memory(df, lookback=50, num_clusters=5):
    """
    Detección de soportes y resistencias por clustering de precios.
    Retorna dict con listas de resistencias y soportes ordenadas por cercanía.
    """
    closes = df['close'].values[-lookback:]
    if len(closes) < 10:
        return {'resistances': [], 'supports': []}

    # Clustering simple por bins de precio
    min_price = min(closes)
    max_price = max(closes)
    bin_size = (max_price - min_price) / num_clusters if num_clusters > 0 else 1
    bins = [min_price + i * bin_size for i in range(num_clusters + 1)]
    counts = [0] * num_clusters
    levels = [[] for _ in range(num_clusters)]

    for price in closes:
        for i in range(num_clusters):
            if bins[i] <= price < bins[i+1]:
                counts[i] += 1
                levels[i].append(price)
                break

    # Soportes: bins con más toques en la parte baja
    # Resistencias: bins con más toques en la parte alta
    supports = []
    resistances = []
    for i in range(num_clusters):
        if counts[i] >= 2:  # Al menos 2 toques para considerar nivel
            avg_level = sum(levels[i]) / len(levels[i])
            if avg_level < (min_price + max_price) / 2:
                supports.append(avg_level)
            else:
                resistances.append(avg_level)

    # Ordenar por cercanía al precio actual
    current_price = closes[-1]
    supports = sorted(supports, key=lambda x: abs(current_price - x))
    resistances = sorted(resistances, key=lambda x: abs(current_price - x))

    return {
        'resistances': resistances,
        'supports': supports
    }


def calculate_time_to_target(current_price, target_price, velocity, 
                             acceleration=0.0, candle_duration_min=5):
    """
    Proyecta tiempo estimado para alcanzar target
    
    Args:
        current_price: Precio actual
        target_price: Precio objetivo
        velocity: Velocidad actual (precio/vela)
        acceleration: Aceleración (cambio velocidad)
        candle_duration_min: Duración vela (default 5min para 5M)
    
    Returns:
        int: Minutos estimados para alcanzar target (0-120 max)
    """
    try:
        if velocity == 0 or velocity is None:
            return None
        
        distance = abs(target_price - current_price)
        
        # Si aceleración positiva, tiempo menor
        if acceleration > 0:
            effective_velocity = velocity * (1 + acceleration / 100)
        else:
            effective_velocity = velocity
        
        if effective_velocity <= 0:
            return None
        
        # Velas estimadas
        estimated_candles = distance / abs(effective_velocity) if effective_velocity != 0 else float('inf')
        
        # Convertir a minutos
        time_minutes = estimated_candles * candle_duration_min
        
        # Limitar a máximo 120 minutos (2 horas)
        return min(int(time_minutes), 120) if time_minutes < float('inf') else None
    
    except Exception as e:
        print(f"❌ Error calculate_time_to_target: {e}")
        return None


def calculate_volume_profile_with_memory(df, num_levels=10, lookback=50):
    """
    Calcula Volume Profile (Perfil de Volumen)
    
    Args:
        df: DataFrame con columnas ['close', 'volume']
        num_levels: Cantidad de niveles de precio
        lookback: Velas a analizar
    
    Returns:
        dict: {
            'poc_current': float (price of control - máximo volumen),
            'poc_history': [últimos values],
            'high_volume_nodes': [precios con alto volumen],
            'low_volume_nodes': [precios con bajo volumen]
        }
    """
    try:
        if len(df) < lookback:
            lookback = len(df)
        
        # Últimas N velas
        subset = df.tail(lookback).copy()
        
        # Crear bins de precio
        min_price = subset['close'].min()
        max_price = subset['close'].max()
        
        if min_price == max_price:
            return {
                'poc_current': float(min_price),
                'poc_history': [float(min_price)],
                'high_volume_nodes': [float(min_price)],
                'low_volume_nodes': []
            }
        
        # Crear niveles
        bins = np.linspace(min_price, max_price, num_levels)
        
        # Contar volumen en cada nivel
        volume_per_level = {}
        for i, price_level in enumerate(bins):
            tolerance = (max_price - min_price) / num_levels / 2
            # Velas que cerraron cerca de este nivel
            mask = (subset['close'] >= price_level - tolerance) & \
                   (subset['close'] <= price_level + tolerance)
            vol = subset.loc[mask, 'volume'].sum()
            volume_per_level[price_level] = vol
        
        # Point of Control (mayor volumen)
        poc = max(volume_per_level, key=volume_per_level.get)
        max_volume = volume_per_level[poc]
        min_volume = min(volume_per_level.values())
        
        # Nodos de alto/bajo volumen
        high_volume_threshold = min_volume + (max_volume - min_volume) * 0.7
        low_volume_threshold = min_volume + (max_volume - min_volume) * 0.3
        
        high_volume_nodes = [price for price, vol in volume_per_level.items() if vol > high_volume_threshold]
        low_volume_nodes = [price for price, vol in volume_per_level.items() if vol < low_volume_threshold]
        
        return {
            'poc_current': float(poc),
            'poc_history': [float(poc)],
            'high_volume_nodes': [float(p) for p in high_volume_nodes],
            'low_volume_nodes': [float(p) for p in low_volume_nodes],
            'profile_high': float(max(high_volume_nodes)) if high_volume_nodes else float(max_price),
            'profile_low': float(min(high_volume_nodes)) if high_volume_nodes else float(min_price)
        }
    
    except Exception as e:
        print(f"❌ Error calculate_volume_profile_with_memory: {e}")
        return {
            'poc_current': 0.0,
            'poc_history': [],
            'high_volume_nodes': [],
            'low_volume_nodes': [],
            'profile_high': 0.0,
            'profile_low': 0.0
        }

def calculate_bollinger_bands(df, period=20, std_dev=2):
    """
    Calcula Bandas de Bollinger.
    Returns dict with upper, middle, lower bands.
    """
    try:
        close = df['close']
        middle = close.rolling(window=period).mean()
        std = close.rolling(window=period).std()
        upper = middle + (std * std_dev)
        lower = middle - (std * std_dev)
        
        return {
            'upper': upper.fillna(0).tolist(),
            'middle': middle.fillna(0).tolist(),
            'lower': lower.fillna(0).tolist()
        }
    except Exception as e:
        print(f"❌ Error calculate_bollinger_bands: {e}")
        return {'upper': [], 'middle': [], 'lower': []}


def calculate_atr(df, period=14):
    """
    Calcula el Average True Range (ATR).
    
    Args:
        df: DataFrame con columnas 'high', 'low', 'close'
        period: Período para el ATR (default 14)
        
    Returns:
        float: Valor actual del ATR
    """
    try:
        high = df['high']
        low = df['low']
        close = df['close']
        
        # Calculate True Range
        tr1 = high - low
        tr2 = abs(high - close.shift(1))
        tr3 = abs(low - close.shift(1))
        
        tr = pd.concat([tr1, tr2, tr3], axis=1).max(axis=1)
        
        # Calculate ATR using rolling mean (Simple Moving Average of TR)
        # Note: Wilder's smoothing is more standard for ATR, but SMA is often used too.
        # Using Wilder's Smoothing to match standard TradingView ATR
        atr = tr.ewm(alpha=1/period, adjust=False).mean()
        
        return {
            'atr_current': float(atr.iloc[-1]) if not pd.isna(atr.iloc[-1]) else 0.0,
            'atr_history': atr.tail(10).tolist()
        }
        
    except Exception as e:
        print(f"❌ Error calculate_atr: {e}")
        return {
            'atr_current': 0.0,
            'atr_history': []
        }
