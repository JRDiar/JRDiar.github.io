def orchestrate_intelligence(symbol):

    Fases:
    1. Percepción (Patrones + Time Ladder)

def make_pattern_based_decision(patterns, ladder, signals, symbol, current_price, recent_history=None):
    """
    Núcleo de Decisión v5.0 (Corregido)
    Integra: Física (Wrecking Ball), Estructura (ZigZag), Zonas (S/R), Memoria.
    """
    score = 0
    reasons = []
    block_trade = False

    # --- 1. DATOS DE ENTRADA ---
    pattern_dir = patterns['confidence'].get('direction', 'NEUTRAL')
    force_dir = signals.get('market_force', {}).get('direction', 'NEUTRAL')
    
    # Prioridad: Patrón > Fuerza
    # Prioridad: Patrón > Fuerza
    trade_dir = pattern_dir if pattern_dir in ['BULLISH', 'BEARISH'] else force_dir
    
    # Extraer datos de timeframes
    a15m = ladder.get('15m', {}) or {}
    a5m = ladder.get('5m', {})
    if not a5m: return {'action': 'WAIT', 'confidence': 0, 'reasons': ['No data 5M']}
    
    # --- 2. FÍSICA (WRECKING BALL - Lógica de Compensación) ---
    adx_val = signals.get('adx_val', 0)
    velocity_info = a5m.get('velocity', {})
    
    # 1. Analizar la calidad del movimiento
    trend_state = velocity_info.get('trend', 'estable')
    is_accelerating = trend_state in ['acelerando', 'aumentando']
    
    # 2. Definir el Umbral de Resistencia (La "Vara" a saltar)
    # UMBRAL REALISTA: ADX 30 base (alcanzable)
    required_force = 30  # Reducido de 45

    # BONUS DINÁMICO
    if is_accelerating:
        required_force -= 5  # 30 → 25 con aceleración
        
    # BONUS MOMENTUM MADURO
    if ladder.get('15m', {}).get('momentum', {}).get('consecutive_count', 0) >= 3:
        required_force -= 3  # Bonus adicional

    # 3. Evaluación Final (Gradiente)
    is_wrecking_ball = adx_val > required_force
    
    if is_wrecking_ball:
        reasons.append(f"☄️ FÍSICA: Inercia suficiente (ADX {adx_val:.1f} > Req {required_force})")

    # --- 3. ESTRUCTURA (ZIGZAG) ---
    zz_data = a5m.get('zigzag', {})
    pivot_types = zz_data.get('pivot_types', [])
    last_pivot = pivot_types[-1] if pivot_types else 'UNKNOWN'
    
    if trade_dir == 'BEARISH':
        if last_pivot == 'HL': 
            score -= 50
            reasons.append("⛔ ESTRUCTURA: ZigZag Higher Low (Posible reversión alcista)")
            if not is_wrecking_ball: block_trade = True
        elif last_pivot == 'LH': score += 20; reasons.append("✅ ESTRUCTURA: ZigZag Lower High")
        elif last_pivot == 'LL': score += 10; reasons.append("✅ ESTRUCTURA: ZigZag Lower Low")
            
    elif trade_dir == 'BULLISH':
        if last_pivot == 'LH': 
            score -= 50
            reasons.append("⛔ ESTRUCTURA: ZigZag Lower High (Posible reversión bajista)")
            if not is_wrecking_ball: block_trade = True
        elif last_pivot == 'HL': score += 20; reasons.append("✅ ESTRUCTURA: ZigZag Higher Low")
        elif last_pivot == 'HH': score += 10; reasons.append("✅ ESTRUCTURA: ZigZag Higher High")

    # --- 4. GEOGRAFÍA (SOPORTES/RESISTENCIAS) ---
    sr_data = a5m.get('sr', {})
    
    if trade_dir == 'BEARISH':
        supports = sr_data.get('supports', [])
        near = [s for s in supports if s < current_price]
        if near:
            nearest = max(near)
            dist = (current_price - nearest) / current_price * 100
            if dist < 1.5:
                if is_wrecking_ball: score += 15; reasons.append(f"🔨 ROMPIENDO SOPORTE ({dist:.2f}%)")
                else: score -= 100; reasons.append(f"🧱 MURO: Vendiendo en Soporte ({dist:.2f}%)"); block_trade = True
    
    elif trade_dir == 'BULLISH':
        resistances = sr_data.get('resistances', [])
        near = [r for r in resistances if r > current_price]
        if near:
            nearest = min(near)
            dist = (nearest - current_price) / current_price * 100
            if dist < 1.5:
                if is_wrecking_ball: score += 15; reasons.append(f"🔨 ROMPIENDO RESISTENCIA ({dist:.2f}%)")
                else: score -= 100; reasons.append(f"🧱 MURO: Comprando en Resistencia ({dist:.2f}%)"); block_trade = True

    # --- 5. VALIDACIÓN DE FATIGA (GOMA ELÁSTICA) ---
    dist_ema_pct = a5m.get('dist_ema_pct', 0)
    rsi_val = a5m.get('rsi', {}).get('rsi_current', 50)
    
    EXTREME_RSI_HIGH = 80
    EXTREME_RSI_LOW = 20
    
    if trade_dir == 'BULLISH':
        # Caso Long: Precio muy arriba de la media
        if dist_ema_pct > 5.0:
            score -= 100
            reasons.append(f"🛑 FATIGA EXTREMA: +{dist_ema_pct:.2f}% vs EMA21")
            # NO bloquear, solo penalizar fuerte (Time Ladder)
        elif dist_ema_pct > 3.5:
            score -= 50
            reasons.append(f"⚠️ FATIGA ALTA: +{dist_ema_pct:.2f}% vs EMA21")
        elif dist_ema_pct > 2.5:
            score -= 25
            reasons.append(f"⚠️ FATIGA MODERADA: +{dist_ema_pct:.2f}% vs EMA21")
            
        # RSI Check
        if rsi_val > EXTREME_RSI_HIGH:
            score -= 50
            reasons.append(f"⚠️ FATIGA: RSI Sobrecomprado ({rsi_val:.1f})")
            if not is_wrecking_ball: block_trade = True 
            
    elif trade_dir == 'BEARISH':
        # Caso Short: Precio muy abajo de la media
        if dist_ema_pct < -5.0:
            score -= 100
            reasons.append(f"🛑 FATIGA EXTREMA: {dist_ema_pct:.2f}% vs EMA21")
            # NO bloquear, solo penalizar fuerte (Time Ladder)
        elif dist_ema_pct < -3.5:
            score -= 50
            reasons.append(f"⚠️ FATIGA ALTA: {dist_ema_pct:.2f}% vs EMA21")
        elif dist_ema_pct < -2.5:
            score -= 25
            reasons.append(f"⚠️ FATIGA MODERADA: {dist_ema_pct:.2f}% vs EMA21")
            
        # RSI Check
        if rsi_val < EXTREME_RSI_LOW:
            score -= 50
            reasons.append(f"⚠️ FATIGA: RSI Sobrevendido ({rsi_val:.1f})")
            if not is_wrecking_ball: block_trade = True

    # --- 6. MEMORIA (PERSISTENCIA) ---
    if recent_history and not block_trade:
        attempts = 0
        for row in recent_history:
            # Detectar si intentamos romper este muro antes
            rec_reasons = str(row[3])
            if "MURO" in rec_reasons or "Soporte" in rec_reasons or "Resistencia" in rec_reasons:
                attempts += 1
        
        if attempts > 0:
            if is_wrecking_ball or signals.get('market_force', {}).get('detected'):
                score += min(attempts * 5, 15)
                reasons.append(f"🧠 MEMORIA: Persistencia ({attempts} intentos previos)")

    # --- 7. SCORE FINAL ---
    if signals.get('market_force', {}).get('detected'):
        score += 40
        reasons.append(signals['market_force']['reason'])
    
    if patterns.get('best_pattern'):
        score += patterns['best_pattern']['confidence'] * 0.3
        reasons.append(f"Patrón: {patterns['best_pattern']['type']}")

    # Ajustar por GRID
    if 'grid_result' in locals() and grid_result.get('score'):
        final_score = score + grid_result['score']
        print(f"  📐 Grid ajuste: {grid_result['score']:+d} → Total: {final_score}")
    else:
        final_score = score

    final_score = max(0, min(100, final_score))
    
    # Proyección de Escenarios (si vale la pena)
    scenarios = None
    if final_score >= 50 and not block_trade:
        try:
            from brain_scenarios import project_scenarios
            scenarios = project_scenarios(symbol, current_price, ladder)
        except: pass

    if block_trade:
        return {'action': 'WAIT', 'confidence': 0, 'reasons': [f"🚫 BLOQUEO: {reasons[-1]}"] + reasons}

    # UMBRAL DINÁMICO (Time Ladder)
    MIN_THRESHOLD = 50  # Base realista

    # Ajustar según fuerza
    max_adx = signals.get('market_force', {}).get('max_adx', 0)

    if max_adx > 35:
        MIN_THRESHOLD = 45  # Con fuerza extrema
    elif max_adx > 25:
        MIN_THRESHOLD = 50  # Con fuerza moderada
    else:
        MIN_THRESHOLD = 60  # Sin fuerza, exigir precisión
    
    if final_score >= MIN_THRESHOLD:
        return {
            'action': 'ENTER',
            'confidence': final_score,
            'reasons': reasons,
            'strategy': determine_strategy(patterns, ladder, scenarios),
            'pattern': patterns.get('best_pattern', {}).get('type', 'STRUCTURE')
        }
    else:
        return {
            'action': 'WAIT',
            'confidence': final_score,
            'reasons': reasons,
            'missing': MIN_THRESHOLD - final_score
        }

def determine_strategy(patterns, ladder, scenarios=None):
    """
    Define Take Profit y Stop Loss.
    Usa Proyección Dinámica (Escenarios) si están disponibles.
    """
    pattern_type = patterns.get('best_pattern', {}).get('type', 'STANDARD')
    
    # Valores Base
    stop_mult = 2.0
    target_mult = 2.0
    position = 0.75
    
    # Ajuste por Patrón
    if 'DOUBLE' in pattern_type or 'BOS' in pattern_type:
        target_mult = 2.5
        position = 0.5
    elif 'FLAG' in pattern_type:
        target_mult = 3.0
        position = 1.0

    # Ajuste por Escenario (Proyección)
    if scenarios:
        primary = scenarios.get('primary', {})
        prob = primary.get('probability', 0)
        rr = primary.get('rr_ratio', 0)
        
        if prob > 60 and rr > 2.0:
            target_mult = rr # Usar target real del gráfico
            position = min(position + 0.25, 1.0)
        elif prob < 50:
            position *= 0.8 # Reducir mano

    return {
        'type': 'SCENARIO' if scenarios else 'PATTERN',
        'stop_multiplier': stop_mult,
        'target_multiplier': target_mult,
        'position_size': position
    }

def check_market_force(ladder):
    """
    Verifica FUERZA DE MERCADO (Blindada v5.0).
    Reglas: ADX > 25, Spread > 5, Momentum Alineado.
    """
    a1h = ladder.get('1h') or {}
    if not a1h: return {'detected': False, 'reason': "No data 1H"}

    adx_data = a1h.get('adx', {})
    mom_data = a1h.get('momentum', {})
    
    adx_val = adx_data.get('adx_current', 0)
    di_plus = adx_data.get('di_plus_current', 0)
    di_minus = adx_data.get('di_minus_current', 0)
    mom_color = mom_data.get('color_current', 'gray')
    
    force_direction = 'NEUTRAL'
    if di_plus > di_minus: force_direction = 'BULLISH'; spread = di_plus - di_minus
    else: force_direction = 'BEARISH'; spread = di_minus - di_plus
        
    if adx_val < 25: return {'detected': False, 'reason': f"ADX débil ({adx_val:.1f})"}
    if spread < 5.0: return {'detected': False, 'reason': f"Spread sucio ({spread:.1f})"}
    
    aligned = False
    if force_direction == 'BULLISH' and mom_color in ['green', 'lime']: aligned = True
    elif force_direction == 'BEARISH' and mom_color in ['red', 'maroon']: aligned = True
    
    if not aligned: return {'detected': False, 'reason': f"Divergencia Mom/ADX"}

    return {
        'detected': True,
        'direction': force_direction,
        'reason': f"💪 FUERZA: ADX {adx_val:.1f} {force_direction}"
    }

def check_adx_alignment(ladder):
    """Verifica si ADX apoya la tendencia en TFs clave"""
    a1h = ladder.get('1h') or {}
    a15m = ladder.get('15m') or {}
    
    def get_val(data):
        if isinstance(data, dict):
            return data.get('adx_current', 0)
        return float(data) if data else 0

    adx_1h = get_val(a1h.get('adx'))
    adx_15m = get_val(a15m.get('adx'))

    # Alineación: Al menos uno debe mostrar fuerza real
    return adx_1h > 20 or adx_15m > 20

def check_momentum_propagation(ladder):
    """Verifica si el momentum se propaga de 4H a 15M"""
    dirs = []
    for tf in ['4h', '1h', '15m']:
        analysis = ladder.get(tf)
        if analysis and isinstance(analysis, dict):
            dirs.append(analysis.get('direction', 'NEUTRAL'))
            
    # Si tenemos 3 direcciones y son iguales (y no LATERAL)
    if len(dirs) >= 3 and len(set(dirs)) == 1 and dirs[0] in ['BULLISH', 'BEARISH']:
        return True
    return False

def check_volume_confirmation(ladder):
    """Verifica si hay volumen confirmando en 5M"""
    a5m = ladder.get('5m') or {}
    vp_data = a5m.get('volume_profile', {})
    
    # Si no hay perfil, no bloqueamos (neutral)
    if not vp_data: return True
    
    # Lógica simple: Si estamos sobre el POC en subida, o bajo el POC en bajada
    return True
