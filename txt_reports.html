pythondef update_trailing_stop(self, trade_data, current_price, pnl_pct):
    """Trailing adaptativo según momentum"""
    
    # Distancia variable según profit acumulado
    if pnl_pct > 5.0:
        trailing_dist_pct = 1.0  # Ajustado (más profit = más cerca)
    elif pnl_pct > 3.0:
        trailing_dist_pct = 1.5
    else:
        trailing_dist_pct = 2.0
    
    # Calcular nuevo stop
    direction = trade_data['direction']
    if direction == 'BULLISH':
        new_stop = current_price * (1 - trailing_dist_pct/100)
        # Validar que mejora posición
        if new_stop > trade_data['stop_loss']:
            # Protección: No subir si retroceso > 0.5%
            last_max = trade_data['max_pnl_reached']
            if pnl_pct >= last_max - 0.5:
                return new_stop
    else:
        new_stop = current_price * (1 + trailing_dist_pct/100)
        if new_stop < trade_data['stop_loss']:
            last_max = trade_data['max_pnl_reached']
            if pnl_pct >= last_max - 0.5:
                return new_stop
    
    return trade_data['stop_loss']  # No cambiar
