def execute_entry(symbol, decision):
    # RECÁLCULO OBLIGATORIO
    print(f"   🔄 Recalculando score antes de entrada...")
    
    from brain_analyzer import analyze_15m
    
    # Recalcular con datos frescos
    direction = decision.get('direction')
    fresh_result = analyze_15m(symbol, direction)
    
    # Verificar si está bloqueado
    if fresh_result.get('blocked'):
        print(f"   🚫 ENTRADA BLOQUEADA: {fresh_result.get('block_reason')}")
        return None
    
    # Verificar score mínimo
    fresh_score = fresh_result.get('score_15m', 0)
    if fresh_score < 55:
        print(f"   ❌ Score insuficiente: {fresh_score} < 55")
        return None
    
    # Proceder con score fresco
    enter_trade(symbol, direction, price, fresh_score)
