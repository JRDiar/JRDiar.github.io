def analyze_volume_profile_context(symbol, timeframe='15m'):
    """
    Analiza posición precio vs Volume Profile

    DETECTA:
    - ¿Precio cerca límite superior VP? (peligro techo)
    - ¿Precio cerca límite inferior VP? (peligro suelo)
    - ¿En zona alta liquidez (POC)? (zona segura)
    - ¿En zona baja liquidez (gap)? (zona peligrosa)

    Returns:
        dict: {
            'position': 'EXTREMO_SUPERIOR'/'EXTREMO_INFERIOR'/'ZONA_POC'/'ZONA_GAP',
            'distance_to_limit_pct': float,
            'safe_to_trade': bool,
            'warning': str o None
        }
    """
    ohlcv = fetch_ohlcv(symbol, timeframe, 200)
    if not ohlcv or len(ohlcv) < 100:
        return None

    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    current_price = float(df['close'].iloc[-1])

    # Volume Profile completo (últimas 100 velas)
    vp_data = calculate_volume_profile_with_memory(df, num_levels=30, lookback=100)

    poc = vp_data['poc_current']
    high_volume_nodes = vp_data['high_volume_nodes']
    low_volume_nodes = vp_data['low_volume_nodes']

    # Límites VP
    max_price = df['high'].tail(100).max()
    min_price = df['low'].tail(100).min()

    # Distancia a límites
    distance_to_top = ((max_price - current_price) / current_price) * 100
    distance_to_bottom = ((current_price - min_price) / current_price) * 100

    # Clasificar posición
    if distance_to_top < 1.5:  # <1.5% del techo
        position = 'EXTREMO_SUPERIOR'
        safe = False
        warning = f"Precio a {distance_to_top:.1f}% del límite VP superior"
    elif distance_to_bottom < 1.5:
        position = 'EXTREMO_INFERIOR'
        safe = False
        warning = f"Precio a {distance_to_bottom:.1f}% del límite VP inferior"
    elif abs(current_price - poc) / current_price * 100 < 2.0:
        position = 'ZONA_POC'
        safe = True
        warning = None
    elif any(abs(current_price - node) / current_price * 100 < 1.0 for node in low_volume_nodes):
        position = 'ZONA_GAP'
        safe = False
        warning = "Precio en zona baja liquidez (gap VP)"
    else:
        position = 'ZONA_NORMAL'
        safe = True
        warning = None

    return {
        'position': position,
        'distance_to_top_pct': round(distance_to_top, 2),
        'distance_to_bottom_pct': round(distance_to_bottom, 2),
        'safe_to_trade': safe,
        'warning': warning,
        'poc': round(poc, 6)
    }

def analyze_support_resistance_touches(symbol, timeframe='4h'):
    """
    Cuenta cuántas veces precio tocó soportes/resistencias

    CRÍTICO: Si soporte tocado 10+ veces = zona MUY fuerte (probable rebote)
    Ejemplo: SUPRA tocó soporte múltiples veces → entrar SHORT ahí = suicidio

    Returns:
        dict: {
            'nearest_support': {
                'price': float,
                'touches': int,
                'strength': 'DÉBIL'/'MODERADO'/'FUERTE'/'MUY_FUERTE',
                'distance_pct': float
            },
            'nearest_resistance': {...},
            'danger_zone': bool
        }
    """
    ohlcv = fetch_ohlcv(symbol, timeframe, 200)
    if not ohlcv or len(ohlcv) < 100:
        return None

    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    current_price = float(df['close'].iloc[-1])

    # SR con lookback largo (100 velas = 400 horas = 16 días)
    sr_data = detect_support_resistance_with_memory(df, num_clusters=8, lookback=100)

    supports = sr_data['supports']
    resistances = sr_data['resistances']

    # Función auxiliar: contar toques
    def count_touches(level, tolerance_pct=1.5):
        """Cuenta cuántas velas tocaron este nivel"""
        touches = 0
        for i in range(len(df)):
            low = df['low'].iloc[i]
            high = df['high'].iloc[i]
            # Toque = si vela cruzó nivel ±tolerance
            if level == 0:
                continue
            if abs((low - level) / level * 100) <= tolerance_pct or \
               abs((high - level) / level * 100) <= tolerance_pct:
                touches += 1
        return touches

    # Analizar soporte más cercano
    nearest_support_info = None
    if supports:
        nearest_support = supports[0]  # Ya viene ordenado por cercanía
        touches = count_touches(nearest_support)
        distance_pct = ((current_price - nearest_support) / current_price) * 100

        if touches >= 10:
            strength = 'MUY_FUERTE'
        elif touches >= 6:
            strength = 'FUERTE'
        elif touches >= 3:
            strength = 'MODERADO'
        else:
            strength = 'DÉBIL'

        nearest_support_info = {
            'price': round(nearest_support, 6),
            'touches': touches,
            'strength': strength,
            'distance_pct': round(distance_pct, 2)
        }

    # Analizar resistencia más cercana
    nearest_resistance_info = None
    if resistances:
        nearest_resistance = resistances[0]  # Ya viene ordenado por cercanía
        touches = count_touches(nearest_resistance)
        distance_pct = ((nearest_resistance - current_price) / current_price) * 100

        if touches >= 10:
            strength = 'MUY_FUERTE'
        elif touches >= 6:
            strength = 'FUERTE'
        elif touches >= 3:
            strength = 'MODERADO'
        else:
            strength = 'DÉBIL'

        nearest_resistance_info = {
            'price': round(nearest_resistance, 6),
            'touches': touches,
            'strength': strength,
            'distance_pct': round(distance_pct, 2)
        }

    # ¿Zona peligrosa?
    danger_zone = False
    if nearest_support_info and nearest_support_info['distance_pct'] < 2.0 and nearest_support_info['touches'] >= 8:
        danger_zone = True  # Muy cerca de soporte fuerte = no SHORT
    if nearest_resistance_info and nearest_resistance_info['distance_pct'] < 2.0 and nearest_resistance_info['touches'] >= 8:
        danger_zone = True  # Muy cerca de resistencia fuerte = no LONG

    return {
        'nearest_support': nearest_support_info,
        'nearest_resistance': nearest_resistance_info,
        'danger_zone': danger_zone
    }

def detect_structure_deterioration(symbol, timeframe='1h'):
    """
    Detecta si estructura precio se está deteriorando

    PATRÓN: HH → HL → LL = debilidad progresiva
    Ejemplo: Alto 100 → rebote 95 → nuevo alto 98 (más bajo) → siguiente rebote será <95

    Returns:
        dict: {
            'deteriorating': bool,
            'pattern': 'HH_HL_LL' o 'STRONG' o 'UNCLEAR',
            'confidence': float (0-100)
        }
    """
    ohlcv = fetch_ohlcv(symbol, timeframe, 100)
    if not ohlcv or len(ohlcv) < 50:
        return None

    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])

    # Detectar pivots significativos
    structure = detect_price_structure(df, lookback=50, pivot_window=3)

    pivot_highs = structure['pivot_highs']
    pivot_lows = structure['pivot_lows']

    if len(pivot_highs) < 3 or len(pivot_lows) < 3:
        return {
            'deteriorating': False,
            'pattern': 'UNCLEAR',
            'confidence': 0
        }

    # Últimos 3 máximos
    last_3_highs = pivot_highs[-3:]
    # Últimos 3 mínimos
    last_3_lows = pivot_lows[-3:]

    # Patrón deterioro alcista: HH → HL (máximo más bajo) + LL
    if last_3_highs[2] < last_3_highs[1] < last_3_highs[0]:  # Máximos bajando
        if last_3_lows[2] < last_3_lows[1]:  # Mínimos también bajando
            return {
                'deteriorating': True,
                'pattern': 'HH_HL_LL',
                'confidence': 85
            }

    # Patrón fuerte alcista: HH + HL consecutivos
    if last_3_highs[2] > last_3_highs[1] > last_3_highs[0]:
        if last_3_lows[2] > last_3_lows[1] > last_3_lows[0]:
            return {
                'deteriorating': False,
                'pattern': 'STRONG_BULLISH',
                'confidence': 90
            }

    # Patrón fuerte bajista: LH + LL consecutivos
    if last_3_highs[2] < last_3_highs[1] < last_3_highs[0]:
        if last_3_lows[2] < last_3_lows[1] < last_3_lows[0]:
            return {
                'deteriorating': False,
                'pattern': 'STRONG_BEARISH',
                'confidence': 90
            }

    return {
        'deteriorating': False,
        'pattern': 'MIXED',
        'confidence': 40
    }

def detect_divergence(symbol, timeframe='15m'):
    """
    Detecta divergencia ADX vs precio

    DIVERGENCIA BAJISTA: Precio sube + ADX baja = debilidad (peligro)
    DIVERGENCIA ALCISTA: Precio baja + ADX sube = fuerza oculta

    Returns:
        dict: {
            'divergence_detected': bool,
            'type': 'BEARISH'/'BULLISH'/'NONE',
            'strength': 'FUERTE'/'MODERADA'/'DÉBIL'
        }
    """
    ohlcv = fetch_ohlcv(symbol, timeframe, 50)
    if not ohlcv or len(ohlcv) < 30:
        return None

    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])

    adx_data = calculate_adx_with_memory(df, period=14, lookback=20)

    # Últimas 10 velas: tendencia precio vs tendencia ADX
    prices = df['close'].tail(10).tolist()
    adx_values = adx_data['adx_history'][-10:]

    # Precio subiendo?
    price_rising = prices[-1] > prices[0]
    price_change_pct = ((prices[-1] - prices[0]) / prices[0]) * 100

    # ADX subiendo?
    adx_rising = adx_values[-1] > adx_values[0]
    adx_change = adx_values[-1] - adx_values[0]

    # Divergencia bajista: precio sube + ADX baja
    if price_rising and not adx_rising and price_change_pct > 2.0 and adx_change < -3:
        return {
            'divergence_detected': True,
            'type': 'BEARISH',
            'strength': 'FUERTE' if abs(adx_change) > 5 else 'MODERADA'
        }

    # Divergencia alcista: precio baja + ADX sube
    if not price_rising and adx_rising and price_change_pct < -2.0 and adx_change > 3:
        return {
            'divergence_detected': True,
            'type': 'BULLISH',
            'strength': 'FUERTE' if adx_change > 5 else 'MODERADA'
        }

    return {
        'divergence_detected': False,
        'type': 'NONE',
        'strength': 'N/A'
    }

def deep_analysis_complete(symbol, analysis_4h, analysis_1h, analysis_15m):
    """
    Análisis profundo completo (2-5 minutos)

    Args:
        symbol: Par trading
        analysis_4h, 1h, 15m: Análisis previos (ya calculados)

    Returns:
        dict: {
            'probability': int (0-100%),
            'safe_to_trade': bool,
            'warnings': [lista de advertencias],
            'strengths': [lista de fortalezas],
            'vp_context': dict,
            'sr_context': dict,
            'structure_analysis': dict,
            'divergence': dict
        }
    """
    print(f"\n🔬 ANÁLISIS PROFUNDO: {symbol}")

    warnings = []
    strengths = []
    probability = 50  # Base

    # 1. Volume Profile
    vp_ctx = analyze_volume_profile_context(symbol, '15m')
    if vp_ctx:
        if not vp_ctx['safe_to_trade']:
            probability -= 20
            warnings.append(vp_ctx['warning'])
        else:
            probability += 5
            if vp_ctx['position'] == 'ZONA_POC':
                strengths.append("Precio en zona alta liquidez (POC)")

    # 2. SR multi-touch
    sr_ctx = analyze_support_resistance_touches(symbol, '4h')
    if sr_ctx:
        if sr_ctx['danger_zone']:
            probability -= 25
            if sr_ctx['nearest_support'] and sr_ctx['nearest_support']['touches'] >= 8:
                warnings.append(f"Soporte MUY fuerte {sr_ctx['nearest_support']['touches']} toques a {sr_ctx['nearest_support']['distance_pct']:.1f}%")
            if sr_ctx['nearest_resistance'] and sr_ctx['nearest_resistance']['touches'] >= 8:
                warnings.append(f"Resistencia MUY fuerte {sr_ctx['nearest_resistance']['touches']} toques a {sr_ctx['nearest_resistance']['distance_pct']:.1f}%")
        else:
            probability += 10
            strengths.append("SR libre sin obstáculos fuertes")

    # 3. Deterioro estructura
    structure_ctx = detect_structure_deterioration(symbol, '1h')
    if structure_ctx:
        if structure_ctx['deteriorating']:
            probability -= 15
            warnings.append(f"Estructura deteriorándose: {structure_ctx['pattern']}")
        elif structure_ctx['pattern'] in ['STRONG_BULLISH', 'STRONG_BEARISH']:
            probability += 10
            strengths.append(f"Estructura fuerte: {structure_ctx['pattern']}")

    # 4. Divergencias
    div_ctx = detect_divergence(symbol, '15m')
    if div_ctx and div_ctx['divergence_detected']:
        if div_ctx['type'] == 'BEARISH':
            probability -= 15
            warnings.append(f"Divergencia bajista {div_ctx['strength']}")
        elif div_ctx['type'] == 'BULLISH':
            probability += 10
            strengths.append(f"Divergencia alcista {div_ctx['strength']}")

    # CONTEXTUAL OVERRIDE: Setup explosivo ignora warnings menores
    if (analysis_4h['adx']['adx_current'] >= 30 and  # Era 45
        analysis_4h['momentum']['consecutive_count'] >= 6 and  # Era 8
        analysis_1h['adx']['adx_current'] >= 25):  # Era 40
        
        print(f"  🚀 SETUP EXPLOSIVO DETECTADO (ADX 4H={analysis_4h['adx']['adx_current']:.1f})")
        
        if vp_ctx and 'gap' in str(vp_ctx.get('warning', '')).lower():
            probability += 25
            if vp_ctx['warning'] in warnings:
                warnings.remove(vp_ctx['warning'])
            strengths.append("Gap alcista en impulso explosivo")
            print(f"  ✅ Gap reinterpretado (+25%)")
        
        # VP extremos contextuales
        direction = analysis_4h['direction']
        if vp_ctx:
            vp_pos = vp_ctx.get('position', '')
            if vp_pos == 'EXTREMO_SUPERIOR' and direction == 'BULLISH':
                probability += 15
                strengths.append("VP extremo superior favorable para LONG")
            elif vp_pos == 'EXTREMO_INFERIOR' and direction == 'BEARISH':
                probability += 15
                strengths.append("VP extremo inferior favorable para SHORT")
        
        if structure_ctx and structure_ctx.get('pattern') in ['MIXED', 'HH_HL_LL']:
            probability += 10
            warnings = [w for w in warnings if 'estructura' not in w.lower()]
            strengths.append("Breakout rompiendo estructura")
            print(f"  ✅ Estructura reinterpretada (+10%)")
        
        probability = min(100, probability)
        safe_to_trade = probability >= 40

    # Limitar probabilidad 0-100
    probability = max(0, min(100, probability))

    # Decisión final: permitir warnings menores si probabilidad es alta
    safe_to_trade = probability >= 55 or (probability >= 40 and len(warnings) <= 1)

    print(f"   Probabilidad: {probability}%")
    if warnings:
        for w in warnings:
            print(f"   ⚠️ {w}")
    if strengths:
        for s in strengths:
            print(f"   ✅ {s}")

    return {
        'probability': probability,
        'safe_to_trade': safe_to_trade,
        'warnings': warnings,
        'strengths': strengths,
        'vp_context': vp_ctx if vp_ctx else {},
        'sr_context': sr_ctx if sr_ctx else {},
        'structure_analysis': structure_ctx if structure_ctx else {},
        'divergence': div_ctx if div_ctx else {}
    }

    def detect_live_pattern(self, df, trade_data):
        """Detecta patrones en vivo durante el trade (1M granular con confirmación)"""
        from brain_config import ENABLE_REVERSAL_DETECTOR
        
        if not ENABLE_REVERSAL_DETECTOR:
            # TEMPORALMENTE DESACTIVADO por falsos positivos
            return None
            
        try:
            # 0. PERIODO DE GRACIA (5 min)
            # Evitar pánico en los primeros minutos de volatilidad
            now = datetime.now(ARGENTINA_TZ)
            entry_time = trade_data['entry_time']
            duration_min = (now - entry_time).total_seconds() / 60
            
            if duration_min < 5:
                return None # Demasiado temprano para patrones de reversión
            
            # 0.5 CONFIRMACIÓN ANTI-RUIDO (1M) - CON BLOQUEO
            # Exigir 2+ velas contra-tendencia antes de buscar patrones de reversión
            direction = trade_data.get('direction', 'BULLISH')
            allow_reversal_patterns = False  # Por defecto, NO buscar reversión
            
            if len(df) >= 3:
                last_3_closes = df['close'].values[-3:]
                last_3_opens = df['open'].values[-3:]
                
                if direction == 'BULLISH':
                    # Contar velas rojas (bearish)
                    red_count = sum(1 for c, o in zip(last_3_closes, last_3_opens) if c < o)
                    
                    if red_count >= 2:
                        print(f"  ⚠️ 1M REVERSIÓN CONFIRMADA: {red_count}/3 velas rojas - habilitando Double Top detection")
                        allow_reversal_patterns = True  # Sí buscar reversión
                    else:
                        print(f"  ✅ 1M RUIDO: Solo {red_count}/3 velas rojas - ignorando reversión, buscando continuación")
                        
                else:  # BEARISH
                    # Contar velas verdes (bullish)
                    green_count = sum(1 for c, o in zip(last_3_closes, last_3_opens) if c > o)
                    
                    if green_count >= 2:
                        print(f"  ⚠️ 1M REVERSIÓN CONFIRMADA: {green_count}/3 velas verdes - habilitando Double Bottom detection")
                        allow_reversal_patterns = True
                    else:
                        print(f"  ✅ 1M RUIDO: Solo {green_count}/3 velas verdes - ignorando reversión, buscando continuación")
                    
                # 1. Flag Pattern (Continuación) - SIEMPRE se busca
                # Lógica simplificada: Consolidación después de impulso
                closes = df['close'].values
                if len(closes) > 10:
                    recent_volatility = df['close'].tail(5).std()
                    prev_volatility = df['close'].iloc[-10:-5].std()
                    
                    if recent_volatility < prev_volatility * 0.5:
                        return {'pattern': 'FLAG', 'action': 'EXTEND_TARGET', 'confidence': 85}
                
                # 2. Double Top (Reversión) - SOLO si hay confirmación de 2+ velas
                if allow_reversal_patterns:
                    # PROTECCIÓN: Si estamos en ganancia significativa, ignorar reversiones menores
                    current_price = df['close'].iloc[-1]
                    entry_price = trade_data['entry_price']
                    direction_mult = 1 if direction == 'BULLISH' else -1
                    current_pnl_pct = ((current_price - entry_price) / entry_price) * 100 * direction_mult
                    
                    if current_pnl_pct > 2.0:
                        print(f"  🛡️ PROTECCIÓN PNL: +{current_pnl_pct:.2f}% - Ignorando reversiones menores (deja correr)")
                        allow_reversal_patterns = False  # Deshabilitar búsqueda de reversión
                        # Continuar buscando otros patrones (FLAG, etc.)
                    
                    if allow_reversal_patterns:
                        highs = df['high'].rolling(3).max()
                        peaks = highs[highs == highs.rolling(20).max()]
                        
                        if len(peaks) >= 2:
                            last_two = peaks[-2:]
                            # Tolerancia aumentada de 0.5% a 1.5% (menos falsos positivos)
                            # Optimización: Evita salir en consolidaciones normales
                            if abs(last_two.iloc[0] - last_two.iloc[1]) / last_two.iloc[0] < 0.015:
                                
                                # CONFIRMACIÓN: Momentum debe haber girado
                                # Si seguimos con momentum verde fuerte, es solo una pausa, no un top
                                mom_data = calculate_momentum_with_memory(df)
                                mom_color = mom_data['color_current']
                                
                                confirmed = False
                                if direction == 'BULLISH' and mom_color not in ['green', 'lime']:
                                    confirmed = True # Momentum perdió fuerza alcista
                                elif direction == 'BEARISH' and mom_color not in ['red', 'maroon']:
                                    confirmed = True # Momentum perdió fuerza bajista
                                    
                                if confirmed:
                                    print(f"  ⚠️ DOUBLE_TOP CONFIRMADO: Momentum débil ({mom_color}) + 2 peaks similares")
                                    return {'pattern': 'DOUBLE_TOP', 'action': 'EXIT_NOW', 'confidence': 90}
                                else:
                                    print(f"  ✅ Posible Double Top IGNORADO: Momentum aún fuerte ({mom_color})")
                else:
                    # Saltamos Double Top porque no hay suficientes velas de reversión
                    pass
                
            return None
        except Exception as e:
            # print(f"Error detecting live pattern: {e}")
            return None
