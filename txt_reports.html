def _extract_score(score_value, default=0):
    """
    Recursively extract integer score from potentially nested dict.
    
    Args:
        score_value: The value to extract (can be int, float, dict, or None)
        default: Default value if extraction fails
    
    Returns:
        int: Extracted integer score
    """
    if score_value is None:
        return default
    if isinstance(score_value, (int, float)):
        return int(score_value)
    if isinstance(score_value, dict):
        # Recursively extract from nested dict
        return _extract_score(score_value.get('score', default), default)
    # Fallback for any other type
    return default

def _extract_adx_value(adx_data):
    """Safely extract ADX value from dict or number."""
    if isinstance(adx_data, (int, float)):
        return float(adx_data)
    if isinstance(adx_data, dict):
        return float(adx_data.get('adx_current', 0))
    return 0.0

    def pre_filter_weekly_candles(self):
        """
        PRE-FILTRO: Rechaza monedas con <10 velas diarias
        
        Returns:
            list: Símbolos que pasan filtro
        """
        print(f"\n{'='*80}")
        print("🔍 FASE 0: PRE-FILTRO VELAS DIARIAS")
        print(f"{'='*80}")
        
        all_symbols = get_futures_symbols()
        if not all_symbols:
            print("❌ Error obteniendo símbolos")
            return []
        
        total_symbols = len(all_symbols)
        print(f"📊 Total símbolos: {total_symbols}")
        
        filtered = []
        rejected = []
        pre_filtro_batch = []  # Acumular para batch writing
        
        print("Analizando velas diarias...")
        
        for i, symbol in enumerate(all_symbols):
            if i % 100 == 0:
                print(f"  Progreso: {i}/{total_symbols} ({i/total_symbols*100:.1f}%)")
            
            try:
                # Chequeo de prioridad
                if symbol in PRIORITY_SYMBOLS:
                    print(f"  🌟 {symbol}: SÍMBOLO PRIORITARIO - APROBADO AUTOMÁTICAMENTE")
                    filtered.append(symbol)
                    timestamp = datetime.now(ARGENTINA_TZ).strftime("%Y-%m-%d %H:%M:%S")
                    pre_filtro_batch.append([timestamp, symbol, "N/A", "APROBADO (PRIORIDAD)"])
                    continue

                ohlcv = fetch_ohlcv(symbol, '1d', 50)
                if not ohlcv:
                    rejected.append((symbol, 0))
                    print(f"  ❌ {symbol}: 0 velas diarias - RECHAZADO (sin datos)")
                    timestamp = datetime.now(ARGENTINA_TZ).strftime("%Y-%m-%d %H:%M:%S")
                    pre_filtro_batch.append([timestamp, symbol, 0, "RECHAZADO"])
                    continue
                
                # Contar velas reales
                total_candles = len(ohlcv)
                real_candles = 0
                for candle in ohlcv:
                    volume = candle[5]
                    high = candle[2]
                    low = candle[3]
                    
                    if volume > 0 and high != low:
                        real_candles += 1

                if real_candles >= MIN_DAILY_CANDLES:
                    filtered.append(symbol)
                    print(f"  ✅ {symbol}: {real_candles} velas diarias - APROBADO")
                    timestamp = datetime.now(ARGENTINA_TZ).strftime("%Y-%m-%d %H:%M:%S")
                    pre_filtro_batch.append([timestamp, symbol, real_candles, "APROBADO"])
                else:
                    rejected.append((symbol, real_candles))
                    if real_candles == 0 and total_candles > 0:
                        print(f"  ❌ {symbol}: 0/{total_candles} velas válidas - RECHAZADO (velas sin volumen)")
                    else:
                        print(f"  ❌ {symbol}: {real_candles}/{total_candles} velas válidas - RECHAZADO (mín {MIN_DAILY_CANDLES})")
                    timestamp = datetime.now(ARGENTINA_TZ).strftime("%Y-%m-%d %H:%M:%S")
                    pre_filtro_batch.append([timestamp, symbol, real_candles, "RECHAZADO"])
            
            except Exception as e:
                rejected.append((symbol, 0))
        
        print(f"\n✅ FILTRADAS: {len(filtered)}/{total_symbols} ({len(filtered)/total_symbols*100:.1f}%)")
        print(f"❌ RECHAZADAS: {len(rejected)} (<{MIN_DAILY_CANDLES} velas diarias)")
        
        # Escribir batch en Google Sheets
        if pre_filtro_batch:
            print(f"\n📊 Escribiendo {len(pre_filtro_batch)} registros a BRAIN_PRE_FILTRO...")
            sheets_manager.write_batch(SHEET_BRAIN_PRE_FILTRO, pre_filtro_batch)
            print("✅ Batch escritura completada")
        
        return filtered

                    emas_distance = data['analysis_snapshot'].get('emas_separation_15m', 0.0)
                
                # Clasificar estructura según score
                if score >= 80:
                    structure = "SANA"
                elif score >= 70:
                    structure = "REGULAR"
                else:
                    structure = "MALA"
                
                # Clasificar edad según momentum_age_5m
                edad = "MADURA"
                if 'analysis_snapshot' in data and data['analysis_snapshot']:
                    momentum_age_5m = data['analysis_snapshot'].get('momentum_age_5m', 10)
                    if momentum_age_5m <= 5:
                        edad = "JOVEN"
                
                # Obtener datos de triggers (tiempo proyectado y R/R ratio)
                tiempo_proyectado_min = 0
                rr_ratio = 0.0
                if 'triggers' in data and data['triggers']:
                    tiempo_proyectado_min = data['triggers'].get('tiempo_proyectado_min', 0)
                    rr_ratio = data['triggers'].get('rr_ratio', 0.0)
                
                # Escribir a Sheet
                sheets_manager.write_watchlist(symbol, score, direction, emas_distance, structure, edad, tiempo_proyectado_min, rr_ratio)

                            acceleration = acceleration_data['acceleration_current']

                                triggers = data['triggers']
                                sheets_manager.write_proyeccion(
                                    symbol,
                                    current_price,
                                    velocity,
                                    acceleration,
                                    triggers.get('tiempo_proyectado_min', 0),
                                    triggers.get('rr_ratio', 0),
                                    triggers.get('target_price', 0),
                                    triggers.get('stop_loss', 0)
                                )
                    except Exception as e:
                        print(f"⚠️ Error escribiendo en Sheets (no crítico): {e}")
                    
                    # Entrar al trade
                    trade_manager.enter_trade(symbol, data)
                else:
                    # Incrementar contador de checks sin trigger
                    watchlist_manager.increment_check_counter(symbol)
                    print(f"     ⏳ Trigger no activado aún")
        
        # Re-analizar breakeven exits
        trade_manager.reanalyze_breakeven_exits()

                self.monitor_and_trade()
                
                # Si watchlist vacía → re-analizar TODO inmediatamente
                watchlist_current = watchlist_manager.get_watchlist()
                if len(watchlist_current) == 0:

                # Mini-scan cada 30 minutos
                if not self.last_mini_scan or (now - self.last_mini_scan).total_seconds() >= 1800:  # 30min
                    print(f"\n{'='*80}")

                        print(f"📊 Analizando top {len(top_symbols)} símbolos por volumen 24h...")

                            # Progress cada 25

                                print(f"  Progreso: {i}/100...")
