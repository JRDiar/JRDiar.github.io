f_ichimoku() =>
    tenkan = math.avg(ta.highest(high, ichi_tenkan_period), ta.lowest(low, ichi_tenkan_period))
    kijun = math.avg(ta.highest(high, ichi_kijun_period), ta.lowest(low, ichi_kijun_period))
    senkou_a = math.avg(tenkan, kijun)
    senkou_b = math.avg(ta.highest(high, ichi_senkou_b_period), ta.lowest(low, ichi_senkou_b_period))

f_support_resistance() =>
    high_pivot = ta.pivothigh(high, sr_left_bars, sr_right_bars)
    low_pivot = ta.pivotlow(low, sr_left_bars, sr_right_bars)
    var float resistance = na
    var float support = na
    if not na(high_pivot)
        resistance := high_pivot
    if not na(low_pivot)
        support := low_pivot
    short_vol = ta.ema(volume, 5)
    long_vol = ta.ema(volume, 10)
    vol_osc = 100 * (short_vol - long_vol) / long_vol
    res_break_vol = ta.crossover(close, resistance) and vol_osc > sr_volume_threshold
    sup_break_vol = ta.crossunder(close, support) and vol_osc > sr_volume_threshold
    bull_wick = ta.crossover(close, resistance) and (open - low) > (close - open)
    bear_wick = ta.crossunder(close, support) and (open - close) < (high - open)
    res_break_wick = bull_wick and not res_break_vol
    sup_break_wick = bear_wick and not sup_break_vol
    [resistance, support, res_break_vol, sup_break_vol, res_break_wick, sup_break_wick]

f_multi_vp(length, bins, poc_mult, show_style = true) =>
    if barstate.islast
        var values = array.new<float>()
        var vol_bins = array.new<float>(bins)
        var tot_vol = array.new<float>()
        var delta_vol = array.new<float>()
        var bin_levels = array.new<line>()
        var bin_labels = array.new<label>()
        values.clear()
        vol_bins.fill(0)
        tot_vol.clear()
        delta_vol.clear()
        if bin_levels.size() > 0
            for i = 0 to bin_levels.size() - 1
                line.delete(bin_levels.get(i))
            bin_levels.clear()
        if bin_labels.size() > 0
            for i = 0 to bin_labels.size() - 1
                label.delete(bin_labels.get(i))
            bin_labels.clear()
        for i = 0 to length - 1
            values.push(high[i])
            values.push(low[i])
        float max_val = values.max()
        float min_val = values.min()
        float step = (max_val - min_val) / bins
        for i = 0 to length - 1
            c = close[i]
            o = open[i]
            vol = volume[i]
            tot_vol.push(vol)
            delta_vol.push(c > o ? vol : -vol)
            for j = 0 to bins - 1
                lower = min_val + step * j
                upper = lower + step
                if c <= upper + step * 1.5 and c >= lower - step * 1.5
                    vol_bins.set(j, vol_bins.get(j) + vol)
        var poly = array.new<chart.point>()
        poly.clear()
        bool poc_found = false
        float max_vol = vol_bins.max()
        for i = 0 to bins - 1
            lower = min_val + step * i
            upper = lower + step
            mid = math.avg(upper, lower)
            vol = int(vol_bins.get(i) / max_vol * 25)
            start = bar_index - length
            end_point = start + vol
            poly.push(chart.point.from_index(i == 0 ? start : i == bins - 1 ? start : end_point + 1, mid))
            if vol_bins.get(i) == max_vol and not poc_found
                poc_found := true
                bin_labels.push(label.new(start, mid, str.tostring(vol_bins.get(i), format.volume), style=label.style_label_right, color=color.new(mvp_poc_color, 25)))
                bin_levels.push(line.new(start, mid, start + int(length / poc_mult), mid, color=mvp_poc_color, width=3))
        if show_style
            bin_levels.push(line.new(start, max_val, bar_index, max_val, color=chart.fg_color))
            bin_levels.push(line.new(start, min_val, bar_index, min_val, color=chart.fg_color))
            bin_levels.push(line.new(start, max_val, start, min_val, color=chart.fg_color))
            bin_labels.push(label.new(start, max_val, "Total\n" + str.tostring(tot_vol.sum(), format.volume), style=label.style_label_down, color=color.blue, size=size.small))
            bin_labels.push(label.new(start, min_val, str.tostring(delta_vol.sum(), format.volume) + "\nDelta", style=label.style_label_up, size=size.small, color=delta_vol.sum() < 0 ? mvp_delta_neg : mvp_delta_pos))
        polyline.delete(polyline.new(poly, line_color=#00e67700, fill_color=mvp_profile_color)[1])

f_delta_vp() =>
    var max_min = array.new<float>()
    var vol_plus = array.new<float>()
    var vol_minus = array.new<float>()
    var boxes = array.new<box>()
    var labels = array.new<label>()
    step = 0.
    bins = 0
    atr_val = ta.atr(200)
    if barstate.islast
        max_min.clear()
        if boxes.size() > 0
            for b in boxes
                box.delete(b)
            boxes.clear()
        if labels.size() > 0
            for l in labels
                label.delete(l)
            labels.clear()
        for i = 0 to dvp_lookback - 1
            h = high[i]
            l = low[i]
            max_min.push(h)
            max_min.push(l)
        bins := int((max_min.max() - max_min.min()) / atr_val)
        bins := math.max(bins, 20)
        bins := math.min(bins, 150)
        step := (max_min.max() - max_min.min()) / bins
        vol_plus.clear()
        vol_minus.clear()
        for i = 0 to bins - 1
            vol_plus.push(0.0)
            vol_minus.push(0.0)
        for i = 0 to dvp_lookback - 1
            h = high[i]
            l = low[i]
            c = close[i]
            o = open[i]
            v = volume[i]
            for k = 0 to bins - 1
                loww = max_min.min() + step * k
                mid = loww + step / 2
                if h > mid and l < mid
                    if c > o
                        vol_plus.set(k, vol_plus.get(k) + v)
                    else
                        vol_minus.set(k, vol_minus.get(k) + v)
        for i = 0 to bins - 1
            loww = max_min.min() + step * i
            highh = loww + step
            vp = vol_plus.get(i)
            vm = vol_minus.get(i)
            value_plus = vol_plus.sum() > 0 ? int(vp / vol_plus.sum() * 200) : 0
            value_minus = vol_minus.sum() > 0 ? int(vm / vol_minus.sum() * 200) : 0
            max_plus = vp == vol_plus.max()
            max_minus = vm == vol_minus.max()
            col_plus = color.from_gradient(vp, vol_plus.min(), vol_plus.max(), color.new(dvp_color_plus, 90), dvp_color_plus)
            col_minus = color.from_gradient(vm, vol_minus.min(), vol_minus.max(), color.new(dvp_color_minus, 90), dvp_color_minus)
            sum_bars = value_plus + value_minus
            col_sum = color.from_gradient(sum_bars, 0, 60, color.rgb(189, 189, 189, 100), #787b8699)
            b1 = box.new(bar_index - dvp_lookback, highh, bar_index, loww, max_plus ? dvp_poc_plus : max_minus ? dvp_poc_minus : color.new(chart.bg_color, 50), bgcolor=col_sum)
            b2 = box.new(bar_index + 50, highh, bar_index + 50 - value_plus, loww,chart.bg_color, bgcolor=max_plus ? dvp_poc_plus : col_plus, text=vp != 0 ? "+" + str.tostring(vp, format.volume) : "", text_color=chart.fg_color, text_size=size.small,text_halign=text.align_right)
            b3 = box.new(bar_index + 50, highh, bar_index + 50 + value_minus, loww, chart.bg_color,bgcolor=max_minus ? dvp_poc_minus : col_minus, text=vm != 0 ? "-" + str.tostring(vm, format.volume) : "", text_color=chart.fg_color, text_size=size.small,text_halign=text.align_left)
            boxes.push(b1)
            boxes.push(b2)
            boxes.push(b3)
        total_vol = vol_plus.sum() + vol_minus.sum()
        delta_calc = vol_minus.sum() > 0 ? ((vol_plus.sum() - vol_minus.sum()) / vol_minus.sum()) * 100 : 0
        l1 = label.new(bar_index + 50, max_min.max(),
             "Delta: " + str.tostring(delta_calc, format.percent),
             color=delta_calc > 0 ? dvp_color_plus : dvp_color_minus,
             textcolor=chart.fg_color,
             style=label.style_label_down)
        l2 = label.new(bar_index + 50, max_min.min(),
             "Total: " + str.tostring(total_vol, format.volume),
             color=color.gray,
             textcolor=chart.fg_color,
             style=label.style_label_up)
        labels.push(l1)
        labels.push(l2)

f_mtpc_select_auto_tf() =>
    timeframe.isseconds ? '120': timeframe.isminutes ? '1440': timeframe.isdaily ? 'W': timeframe.isweekly ? 'M': timeframe.ismonthly and timeframe.multiplier < 3 ? '3M': '12M'

f_mtpc_draw_box(left, right, top, bottom, diff) =>
    drawnBox = box.new(left, top, right, bottom)
    if diff < 0
        box.set_border_color(drawnBox, mtpc_border_fall)
        box.set_bgcolor(drawnBox, mtpc_fall_color)
    else
        box.set_border_color(drawnBox, mtpc_border_growth)
        box.set_bgcolor(drawnBox, mtpc_growth_color)
    drawnBox

f_mtpc() =>
    var prevBarIndex = bar_index
    tf_to_use = mtpc_auto ? f_mtpc_select_auto_tf() : mtpc_timeframe
    [previousHigh, previousLow, previousClose, previousOpen, prePreviousClose] = request.security(syminfo.tickerid, tf_to_use, [high[1], low[1], close[1], open[1], close[2]], lookahead=barmerge.lookahead_on)
    [currentHigh, currentLow, currentClose, currentOpen, prevClose] = request.security(syminfo.tickerid, tf_to_use, [high, low, close, open, close[1]])
    isNewPeriod = ta.change(time(tf_to_use))
    drawCurrent = barstate.islast and not isNewPeriod
    secHigh = drawCurrent ? currentHigh : previousHigh
    secPrevClose = drawCurrent ? prevClose : prePreviousClose
    secLow = drawCurrent ? currentLow : previousLow
    diff = drawCurrent ? currentClose - currentOpen : previousClose - previousOpen
    left = prevBarIndex
    right = drawCurrent ? bar_index : bar_index[1]
    [ashiHigh, ashiLow] = request.security(ticker.heikinashi(syminfo.tickerid), tf_to_use, [high, low])
    float top = na
    float bottom = na
    if mtpc_calc_method == "High/Low Range"
        top := secHigh
        bottom := secLow
    else if mtpc_calc_method == "True Range"
        top := math.max(secHigh, secPrevClose)
        bottom := math.min(secLow, secPrevClose)
    else if mtpc_calc_method == "Heikin Ashi Range"
        top := ashiHigh
        bottom := ashiLow
    if isNewPeriod
        prevBarIndex := bar_index
        f_mtpc_draw_box(left, right, top, bottom, diff)
    var box prevBox = na
    if barstate.islast
        if isNewPeriod
            prevBox := na
        else
            box.delete(prevBox)
            prevBox := f_mtpc_draw_box(left, right, top, bottom, diff)
    var box prevBox1 = na
    var float prevTop = na
    var float prevBottom = na
    if array.size(box.all) > 1
        prevBox1 := array.get(box.all, array.size(box.all) - 1)
        prevTop := box.get_top(prevBox1)
        prevBottom := box.get_bottom(prevBox1)
    [prevTop, prevBottom]

f_mtf_convert_tf(tf) =>
    time_in = timeframe.in_seconds(tf) / 60 / 60
    time_in >= 24 ? tf : time_in >= 1 ? str.tostring(time_in) + "h" : tf + "m"

f_mtf_request(src1, src2, src3, src4, src5, tf) =>
    float chartTFInMinutes = timeframe.in_seconds("")
    float inputTFInMinutes = timeframe.in_seconds(tf)
    ss1 = src1
    ss2 = src2
    ss3 = src3
    ss4 = src4
