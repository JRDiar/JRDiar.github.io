1. brain_grid.py - El Nuevo Mapa Dinámico
Este es el corazón del cambio. Calcula el canal diagonal (piso y techo) usando matemáticas de regresión lineal (como lo hacen los indicadores de TradingView).

"""
brain_grid.py - Generador de Grid Dinámico (Regresión Lineal)
=============================================================
Calcula soportes y resistencias DIAGONALES en tiempo real.
Define el "campo de juego" dinámico del bot.
"""

import numpy as np
import pandas as pd
from scipy.stats import linregress

class GridManager:
    def __init__(self):
        self.cached_grid = {}

    def calculate_grid(self, df):
        """
        Calcula el canal dinámico usando Regresión Lineal.
        Devuelve: {
            'soporte_diagonal': float,
            'resistencia_diagonal': float,
            'linea_media': float,
            'pendiente': float (para saber si la tendencia general es alcista/bajista)
        }
        """
        if df is None or len(df) < 50:
            return {'valid': False}

        # Configuración del Grid (Ajustable)
        LOOKBACK_PERIOD = 50  # Cuántas velas atrás mirar para trazar la línea
        DESVIACION_STD = 2.0  # Ancho del canal (2 desviaciones estándar es lo común)

        # Usamos los datos recientes
        recent_df = df.tail(LOOKBACK_PERIOD).copy()
        
        # Preparamos ejes X (tiempo/índice) e Y (precio)
        x = np.arange(len(recent_df))
        y = recent_df['close'].values

        # Calcular Regresión Lineal (La línea central)
        slope, intercept, r_value, p_value, std_err = linregress(x, y)

        # Calcular la línea media teórica para el último punto (el precio actual)
        last_x = x[-1]
        linea_media_actual = slope * last_x + intercept

        # Calcular Desviación Estándar (La anchura del canal)
        # Calculamos la distancia de cada precio a la línea media
        regression_line = slope * x + intercept
        residuals = y - regression_line
        std_deviation = np.std(residuals)

        # Calcular Techo y Piso Dinámicos
        techo_diagonal = linea_media_actual + (std_deviation * DESVIACION_STD)
        piso_diagonal = linea_media_actual - (std_deviation * DESVIACION_STD)

        # Proyectar al futuro (opcional, para la siguiente vela)
        # future_x = last_x + 1
        # future_mean = slope * future_x + intercept

        result = {
            'valid': True,
            'soporte_diagonal': piso_diagonal,
            'resistencia_diagonal': techo_diagonal,
            'linea_media': linea_media_actual,
            'pendiente': slope,
            'ancho_canal_pct': ((techo_diagonal - piso_diagonal) / piso_diagonal) * 100
        }
        
        # Guardar en cache
        self.cached_grid = result
        return result

# Instancia global
grid_manager = GridManager()

        # 1. Obtener datos del Grid Dinámico (Calculados en brain_grid)
        grid = data_15m.get('grid', {})
        if not grid.get('valid', False):
            # Si no hay grid, operamos con precaución o bloqueamos
            return {'can_enter': True, 'reason': 'Grid no disponible (Modo Ciego)', 'validation_type': 'CAUTION'}

        techo = grid.get('resistencia_diagonal')
        piso = grid.get('soporte_diagonal')
        media = grid.get('linea_media')
        ancho_canal = grid.get('ancho_canal_pct')

        # 2. Lógica para LONG (Compras)
        if direction == 'BULLISH':
            # A) REGLA DEL TECHO: ¿Estamos muy cerca del techo?
            distancia_al_techo = ((techo - current_price) / current_price) * 100
            
            # Mínimo 0.8% de espacio libre requerido (ajustable en config)
            MIN_ESPACIO = 0.8 

            if distancia_al_techo < MIN_ESPACIO:
                msg = f"⛔ BLOQUEADO: Sin espacio. Techo a {distancia_al_techo:.2f}%. Mínimo req: {MIN_ESPACIO}%"
                print(f"   {msg}")
                return {'can_enter': False, 'reason': msg, 'validation_type': 'NO_RUNWAY'}

            # B) REGLA DE LA ZONA: Solo comprar en la mitad inferior del canal
            # Si el precio está por encima de la línea media, es zona de venta, no de compra.

                # Excepción: Si la fuerza es NUCLEAR (rompimiento masivo), permitimos entrar arriba

                    msg = f"⛔ BLOQUEADO: Precio en zona premium (arriba de media). Esperar retroceso."
                    print(f"   {msg}")

"""
brain_trend_birth.py - Detector de Oportunidades
================================================
Modificado para detectar REBOTES en soportes dinámicos
en lugar de rompimientos tardíos.
"""

def detect_opportunity(df, grid_data):
    """
    Busca la configuración perfecta: 
    1. Precio toca/cerca del soporte diagonal.
    2. Vela verde (o patrón de giro) aparece.
    """
    if df is None or len(df) < 2: return {'action': 'WAIT'}
    
    current_close = df['close'].iloc[-1]
    prev_close = df['close'].iloc[-2]
    
    # Datos del Grid
    soporte = grid_data.get('soporte_diagonal', 0)
    linea_media = grid_data.get('linea_media', 0)
    
    if soporte == 0: return {'action': 'WAIT'}

    # 1. ¿Estamos en "Zona de Compra"? (Cerca del soporte)
    # Definimos "Cerca" como estar dentro del 1.5% del soporte
    distancia_soporte = ((current_close - soporte) / soporte) * 100
    
    is_in_buy_zone = distancia_soporte < 1.5 and current_close > soporte * 0.98 # No queremos que esté MUY por debajo (caída libre)

    # 2. ¿Hay señal de giro? (Vela Verde tras caída o RSI sobrevendido)
    is_green_candle = current_close > df['open'].iloc[-1]
    
    # (Opcional) Usar ZigZag para confirmar que es un Low
    # is_local_low = ... (lógica zigzag)

    if is_in_buy_zone and is_green_candle:
        return {
            'action': 'ENTER_LONG',
            'reason': f'Rebote en Soporte Diagonal (Dist: {distancia_soporte:.2f}%)',
            'quality': 'HIGH'
        }
        
    return {'action': 'WAIT', 'reason': 'Esperando zona de compra'}
