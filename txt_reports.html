"""

ANÁLISIS INTELIGENTE MULTI-CAPA (Simula Mente Humana)
================================================================================

Este módulo implementa una arquitectura de análisis paralelo que evalúa
todas las temporalidades simultáneamente y toma decisiones holísticas,
similar a cómo un trader humano "ve el panorama completo".
"""

from brain_analyzer import (
    analyze_1d, analyze_4h, analyze_1h, analyze_15m, analyze_5m,
    analyze_trend_evolution
)
from brain_config import *

def _extract_score(score_value, default=0):
    """
    Recursively extract integer score from potentially nested dict.
    
    Args:
        score_value: The value to extract (can be int, float, dict, or None)
        default: Default value if extraction fails
    
    Returns:
        int: Extracted integer score
    """
    if score_value is None:
        return default
    if isinstance(score_value, (int, float)):
        return int(score_value)
    if isinstance(score_value, dict):
        # Recursively extract from nested dict
        return _extract_score(score_value.get('score', default), default)
    # Fallback for any other type
    return default


def _extract_adx_value(adx_data):
    """Safely extract ADX value from dict or number."""
    if isinstance(adx_data, (int, float)):
        return float(adx_data)
    if isinstance(adx_data, dict):
        return float(adx_data.get('adx_current', 0))
    return 0.0

def analyze_all_timeframes_parallel(symbol):
    """
    Analiza TODAS las temporalidades en PARALELO (no cascada).
    
    Args:
        symbol (str): Símbolo a analizar
    
    Returns:
        dict: {
            'context': {temporalidad: análisis},
            'bullish_count': int,
            'strong_count': int,
            'opportunities': []
        }
    """
    
    # ANÁLISIS PARALELO (todas las temporalidades a la vez)
    context = {
        '1D': analyze_1d(symbol) if ENABLE_ANALYSIS_1D else None,
        '4H': analyze_4h(symbol),
        '1H': analyze_1h(symbol),
        '15M': analyze_15m(symbol, strict=False), # IMPORTANTE: No filtrar por ADX<20 aquí
        '5M': analyze_5m(symbol),
        '1M': analyze_trend_evolution(symbol)  # Capa de 1 minuto
    }
    
    # CONTAR SEÑALES
    bullish_count = 0
    bearish_count = 0
    strong_count = 0  # Temporalidades con ADX > 20
    explosive_count = 0
    
    opportunities = []  # Lista de oportunidades detectadas
    
    for tf, analysis in context.items():
        if analysis is None:
            continue
        
        # Contar direcciones
        if tf == '1M':
            # Caso especial: evolución 1M
            if analysis.get('is_explosive'):
                explosive_count += 1
                bullish_count += 1
                evo_score = _extract_score(analysis.get('evolution_score', 0), 0)
                opportunities.append({
                    'type': 'SCALPING_1M',
                    'reason': f"1M Explosivo: {analysis.get('reason', 'N/A')}",
                    'confidence': 'ALTA' if evo_score > 40 else 'MEDIA'
                })
        else:
            # Temporalidades normales
            direction = analysis.get('direction', 'LATERAL')
            
            if direction == 'BULLISH':
                bullish_count += 1
            elif direction == 'BEARISH':
                bearish_count += 1
            
            # Contar fuerza (ADX)
            adx_raw = analysis.get('adx', 0)
            adx = _extract_adx_value(adx_raw)
            
            if adx > 20:
                strong_count += 1
                
                # Detectar oportunidades por temporalidad
                if tf == '15M' and adx > 25:
                    opportunities.append({
                        'type': 'SWING_15M',
                        'reason': f"15M fuerte (ADX {adx:.1f})",
                        'confidence': 'ALTA'
                    })
                elif tf == '1H' and adx > 30:
                    opportunities.append({
                        'type': 'POSITION_1H',
                        'reason': f"1H muy fuerte (ADX {adx:.1f})",
                        'confidence': 'MUY_ALTA'
                    })
    
    return {
        'context': context,
        'bullish_count': bullish_count,
        'bearish_count': bearish_count,
        'strong_count': strong_count,
        'explosive_count': explosive_count,
        'opportunities': opportunities
    }


def decide_intelligently(symbol):
    """
    Toma decisión INTELIGENTE basada en el contexto completo.
    Simula la mente humana: "¿Cuántas temporalidades dicen SÍ?"
    
    Args:
        symbol (str): Símbolo a evaluar
    
    Returns:
        dict: {
            'action': 'OPERAR' | 'NO_OPERAR',
            'trade_type': 'SCALPING' | 'SWING' | 'POSITION',
            'reason': str,
            'confidence': 'BAJA' | 'MEDIA' | 'ALTA' | 'MUY_ALTA',
            'context_summary': str,
            'full_analysis': dict (para debugging)
        }
    """
    
    # PASO 1: Analizar todas las temporalidades
    multi_tf = analyze_all_timeframes_parallel(symbol)
    
    bullish = multi_tf['bullish_count']
    bearish = multi_tf['bearish_count']
    strong = multi_tf['strong_count']
    explosive = multi_tf['explosive_count']
    opportunities = multi_tf['opportunities']
    
    # PASO 2: Validación de seguridad (1D caída libre)
    context_1d = multi_tf['context']['1D']
    if context_1d:
        # Si 1D está en caída libre (3+ velas rojas + RSI < 30), advertencia
        if context_1d.get('direction') == 'BEARISH' and context_1d.get('rsi', 50) < 30:
            # Pero si hay explosión en 1M, podemos considerar scalping
            if not (explosive > 0 and strong >= 2):
                return {
                    'action': 'NO_OPERAR',
                    'trade_type': None,
                    'reason': '1D en caída libre (RSI < 30)',
                    'confidence': 'N/A',
                    'context_summary': f"Bullish: {bullish}/6 | Strong: {strong}/6",
                    'full_analysis': multi_tf
                }
    
    # PASO 3: LÓGICA DE DECISIÓN INTELIGENTE
    
    # REGLA 1: Explosión en 1M + 2+ temporalidades fuertes
    if explosive > 0 and strong >= 2:
        return {
            'action': 'OPERAR',
            'trade_type': 'SCALPING',
            'reason': f"Explosión 1M + {strong} temporalidades fuertes",
            'confidence': 'ALTA',
            'context_summary': f"Bullish: {bullish}/6 | Strong: {strong}/6 | Explosivo: SÍ",
            'full_analysis': multi_tf
        }
    
    # REGLA 1.5: RESCATE DE TENDENCIA 1H (Rescue Rule)
    # Si 1H es MUY FUERTE (ADX > 30) y 5M acompaña, entramos aunque falten datos o 4H sea lateral.
    # Esto soluciona el caso de "subida fuerte de 1 hora" que el bot ignoraba por falta de consenso.
    context_1h = multi_tf['context'].get('1H', {})
    context_5m = multi_tf['context'].get('5M', {})
    
    if context_1h and context_5m:
        adx_1h = _extract_adx_value(context_1h.get('adx', 0))
        dir_1h = context_1h.get('direction', 'NEUTRAL')
        dir_5m = context_5m.get('direction', 'NEUTRAL')
        
        if dir_1h == 'BULLISH' and adx_1h >= 30 and dir_5m == 'BULLISH':
             return {
                'action': 'OPERAR',
                'trade_type': 'SCALPING', # Entramos como scalping por seguridad
                'reason': f"Rescate Tendencia 1H Fuerte (ADX {adx_1h:.1f})",
                'confidence': 'MEDIA',
                'context_summary': f"1H Strong Bullish | Bullish: {bullish}/6",
                'full_analysis': multi_tf
            }

    # REGLA 2: 4+ temporalidades alcistas + 3+ fuertes (Consenso Estándar)
    # Se relaja a 3 alcistas si hay mucha fuerza (Strong >= 3) para cubrir casos de 4H lateral
    if (bullish >= 4 and strong >= 3) or (bullish >= 3 and strong >= 4):
        # Determinar tipo de trade según dónde está la fuerza
        context_15m = multi_tf['context']['15M']
        context_1h = multi_tf['context']['1H']
        
        adx_15m = _extract_adx_value(context_15m.get('adx', 0)) if context_15m else 0
        adx_1h = _extract_adx_value(context_1h.get('adx', 0)) if context_1h else 0
        
        if context_15m and adx_15m > 25:
            trade_type = 'SWING'
        elif context_1h and adx_1h > 30:
            trade_type = 'POSITION'
        else:
            trade_type = 'SWING'  # Por defecto
        
        return {
            'action': 'OPERAR',
            'trade_type': trade_type,
            'reason': f"{bullish} temporalidades alcistas, {strong} con fuerza",
            'confidence': 'ALTA',
            'context_summary': f"Bullish: {bullish}/6 | Strong: {strong}/6",
            'full_analysis': multi_tf
        }
    
    # REGLA 3: 5+ temporalidades alcistas (aunque no todas sean fuertes)
    if bullish >= 5:
        return {
            'action': 'OPERAR',
            'trade_type': 'SWING',
            'reason': f"{bullish} temporalidades alcistas (consenso masivo)",
            'confidence': 'MEDIA',
            'context_summary': f"Bullish: {bullish}/6 | Strong: {strong}/6",
            'full_analysis': multi_tf
        }
    
    # REGLA 4: Si no cumple nada, rechazar
    return {
        'action': 'NO_OPERAR',
        'trade_type': None,
        'reason': f"Solo {bullish} temporalidades alcistas y {strong} fuertes (insuficiente)",
        'confidence': 'N/A',
        'context_summary': f"Bullish: {bullish}/6 | Strong: {strong}/6",
        'full_analysis': multi_tf
    }
