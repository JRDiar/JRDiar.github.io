    """Motor de detección de patrones reales"""

        }
        
    def detect_all_patterns(self, symbol):

        results = {

    def scan_patterns(self, symbol, tf):

        df = fetch_ohlcv(symbol, tf, 200)
        
        patterns_found = []
        
        # 1. ESTRUCTURA DE MERCADO

    def detect_market_structure(self, df):
        """HH/HL o LH/LL - La base de todo"""
        
        highs = []
        lows = []
        
        # Encontrar pivotes con ZigZag
        for i in range(2, len(df)-2):
            if df['high'][i] > df['high'][i-1] and df['high'][i] > df['high'][i+1]:
                highs.append((i, df['high'][i]))
            if df['low'][i] < df['low'][i-1] and df['low'][i] < df['low'][i+1]:
                lows.append((i, df['low'][i]))
        
        if len(highs) < 2 or len(lows) < 2:
            return {'valid': False}
            
        # Detectar estructura
        last_highs = [h[1] for h in highs[-2:]]
        last_lows = [l[1] for l in lows[-2:]]
        
        # UPTREND: HH + HL
        if last_highs[-1] > last_highs[-2] and last_lows[-1] > last_lows[-2]:
            return {
                'valid': True,
                'type': 'UPTREND',
                'strength': 90,
                'action': 'BUY_CONTINUATION'
            }
            
        # DOWNTREND: LH + LL
        elif last_highs[-1] < last_highs[-2] and last_lows[-1] < last_lows[-2]:
            return {
                'valid': True,
                'type': 'DOWNTREND',
                'strength': 90,
                'action': 'SELL_CONTINUATION'
            }
            
        # CAMBIO DE ESTRUCTURA (BOS)
        elif last_lows[-1] < last_lows[-2] and last_highs[-1] > last_highs[-2]:
            return {
                'valid': True,
                'type': 'BOS_BULLISH',
                'strength': 85,
                'action': 'BUY_REVERSAL'
            }
            
        return {'valid': False}
    
    def detect_double_patterns(self, df):
        """Doble techo/suelo - Reversión clásica"""
        
        # Buscar dos máximos/mínimos similares
        recent_highs = df['high'][-50:].nlargest(5)
        recent_lows = df['low'][-50:].nsmallest(5)
        
        # Doble techo
        if len(recent_highs) >= 2:
            if abs(recent_highs.iloc[0] - recent_highs.iloc[1]) / recent_highs.iloc[0] < 0.005:
                return {
                    'found': True,
                    'type': 'DOUBLE_TOP',
                    'price_level': recent_highs.iloc[0],
                    'confidence': 75,
                    'action': 'PREPARE_SHORT'
                }
                
        # Doble suelo
        if len(recent_lows) >= 2:
            if abs(recent_lows.iloc[0] - recent_lows.iloc[1]) / recent_lows.iloc[0] < 0.005:
                return {
                    'found': True,
                    'type': 'DOUBLE_BOTTOM',
                    'price_level': recent_lows.iloc[0],
                    'confidence': 75,
                    'action': 'PREPARE_LONG'
                }
                
        return {'found': False}
    
    def detect_triangles(self, df):
        """Triángulos - Los más confiables"""
        
        # Líneas de tendencia con regresión
        highs = df['high'][-30:]
        lows = df['low'][-30:]
        
        # Pendientes
        high_slope = np.polyfit(range(len(highs)), highs, 1)[0]
        low_slope = np.polyfit(range(len(lows)), lows, 1)[0]
        
        # Triángulo ascendente
        if abs(high_slope) < 0.001 and low_slope > 0:
            return {
                'type': 'ASCENDING_TRIANGLE',
                'breakout_level': highs.max(),
                'confidence': 85,
                'action': 'BUY_ON_BREAKOUT'
            }
            
        # Triángulo descendente
        elif abs(low_slope) < 0.001 and high_slope < 0:
            return {
                'type': 'DESCENDING_TRIANGLE',
                'breakdown_level': lows.min(),
                'confidence': 85,
                'action': 'SELL_ON_BREAKDOWN'
            }
            
        # Triángulo simétrico
        elif abs(high_slope) > 0 and abs(low_slope) > 0 and high_slope * low_slope < 0:
            return {
                'type': 'SYMMETRIC_TRIANGLE',
                'confidence': 70,
                'action': 'WAIT_BREAKOUT'
            }
            
        return {'type': 'none'}
    
    def detect_flags(self, df):
        """Banderas - Continuación de tendencia"""
        
        # Detectar impulso fuerte seguido de consolidación
        last_20 = df[-20:]
        prev_20 = df[-40:-20]
        
        # Cambio de precio en los períodos
        impulse = (prev_20['close'].iloc[-1] - prev_20['close'].iloc[0]) / prev_20['close'].iloc[0]
        consolidation = (last_20['high'].max() - last_20['low'].min()) / last_20['close'].mean()
        
        # Flag alcista
        if impulse > 0.05 and consolidation < 0.02:
            return {
                'valid': True,
                'type': 'BULL_FLAG',
                'confidence': 80,
                'target': last_20['close'].iloc[-1] * (1 + impulse),
                'action': 'BUY_CONTINUATION'
            }
            
        # Flag bajista
        elif impulse < -0.05 and consolidation < 0.02:
            return {
                'valid': True,
                'type': 'BEAR_FLAG',
                'confidence': 80,
                'target': last_20['close'].iloc[-1] * (1 + impulse),
                'action': 'SELL_CONTINUATION'
            }
            
        return {'valid': False}
    
    def calculate_pattern_confluence(self, all_patterns):
        """Confluencia entre TFs"""
        score = 0
        
        # Contar patrones bullish vs bearish
        bullish = 0
        bearish = 0
        
        for tf, patterns in all_patterns.items():
            weight = {'15m': 0.2, '1h': 0.4, '4h': 0.4}[tf]
            
            for pattern in patterns:
                if 'BUY' in pattern.get('action', ''):
                    bullish += pattern.get('confidence', 0) * weight
                elif 'SELL' in pattern.get('action', ''):
                    bearish += pattern.get('confidence', 0) * weight
                    
        # Retornar dirección dominante
        if bullish > bearish * 1.5:
            return {'direction': 'BULLISH', 'strength': bullish}
        elif bearish > bullish * 1.5:
            return {'direction': 'BEARISH', 'strength': bearish}
        else:
            return {'direction': 'NEUTRAL', 'strength': 0}

def make_pattern_based_decision(patterns, ladder, signals):
    """
    Decisión basada en confluencia REAL
    """
    
    score = 0
    reasons = []
    
    # 1. Patrón principal detectado
    if patterns['best_pattern']:
        pattern = patterns['best_pattern']
        score += pattern['confidence'] * 0.4
        reasons.append(f"Pattern: {pattern['type']}")
        
    # 2. Estructura de mercado
    structure_aligned = 0
    for tf, analysis in ladder.items():
        if analysis and analysis.get('direction') == patterns['confidence']['direction']:
            structure_aligned += 1
            
    if structure_aligned >= 3:
        score += 30
        reasons.append(f"Structure aligned {structure_aligned}/4")
        
    # 3. Indicadores matemáticos
    if signals['adx_aligned']:
        score += 15
        reasons.append("ADX aligned")
        
    if signals['momentum_propagating']:
        score += 15
        reasons.append("Momentum propagating")
        
    if signals['volume_confirming']:
        score += 10
        reasons.append("Volume confirming")
        
    # DECISIÓN FINAL
    if score >= 70:
        return {
            'action': 'ENTER',
            'confidence': score,
            'pattern': patterns['best_pattern']['type'] if patterns['best_pattern'] else None,
            'reasons': reasons,
            'strategy': determine_strategy(patterns, ladder)
        }
    else:
        return {
            'action': 'WAIT',
            'confidence': score,
            'missing': 70 - score,
            'reasons': reasons
        }
