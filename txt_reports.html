def detect_momentum_drive(self, df, timeframe):
    """
    PATRÓN DE FUERZA PURA
    No requiere geometría, solo fuerza matemática brutal
    """
    adx_data = calculate_adx_with_memory(df, period=14, lookback=10)
    mom_data = calculate_momentum_with_memory(df, lookback=10)
    
    # Condiciones "Fuerza Bestial"
    adx_brutal = adx_data['adx_current'] > 30
    adx_acelerando = adx_data['adx_trend'] == 'creciendo'
    momentum_sostenido = mom_data['consecutive_count'] >= 5
    spread_claro = abs(adx_data['spread']) > 10
    
    if adx_brutal and adx_acelerando and momentum_sostenido and spread_claro:
        # Determinar dirección
        direction = 'BULLISH' if adx_data['dominance'] == 'DI+' else 'BEARISH'
        
        # Confianza proporcional a la fuerza
        confidence = min(95, 60 + (adx_data['adx_current'] - 30))
        
        return {
            'found': True,
            'type': 'MOMENTUM_DRIVE',
            'confidence': int(confidence),
            'action': f'BUY_FORCE' if direction == 'BULLISH' else 'SELL_FORCE',
            'strength': adx_data['adx_current'],
            'timeframe': timeframe
        }
    
    return {'found': False}
