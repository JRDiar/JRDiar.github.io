def orchestrate_intelligence(symbol):
    """
    Orquesta TODO el bot con inteligencia real de patrones.
    Combina:
    1. Detección de Patrones (Triángulos, Flags, Estructura)
    2. Análisis Time Ladder (Indicadores matemáticos)
    3. Confluencia Real
    """
    print(f"\n🧠 ORCHESTRATING INTELLIGENCE: {symbol}")
    
    # 1. DETECTAR PATRONES (La nueva "visión" del bot)
    print("   Scanning patterns...")
    patterns_result = pattern_engine.detect_all_patterns(symbol)
    
    if patterns_result['best_pattern']:
        p = patterns_result['best_pattern']
        print(f"   ✨ PATTERN FOUND: {p['type']} in {p['tf']} (Conf: {p['confidence']})")
    
    # 2. ANÁLISIS TIME LADDER (Los "sensores" del bot)
    print("   Analyzing Time Ladder...")
    ladder_analysis = {
        '4h': analyze_4h(symbol),
        '1h': analyze_1h(symbol),
        '15m': analyze_15m(symbol, strict=False),
        '5m': analyze_5m(symbol)
    }
    
    # 3. INDICADORES MATEMÁTICOS (Validación cruzada)
    # Obtener ADX promedio para visualización (Safe Access fix)
    a4h = ladder_analysis.get('4h') or {}
    a1h = ladder_analysis.get('1h') or {}
    a15m = ladder_analysis.get('15m') or {}
    
    adx_4h = a4h.get('adx', {}).get('adx_current', 0)
    adx_1h = a1h.get('adx', {}).get('adx_current', 0)
    adx_15m = a15m.get('adx', {}).get('adx_current', 0)
    adx_avg = (adx_4h + adx_1h + adx_15m) / 3 if (adx_4h or adx_1h or adx_15m) else 0
    
    math_signals = {
        'adx_aligned': check_adx_alignment(ladder_analysis),
        'momentum_propagating': check_momentum_propagation(ladder_analysis),
        'volume_confirming': check_volume_confirmation(ladder_analysis),
        'market_force': check_market_force(ladder_analysis), # NUEVO: Detectar Fuerza Pura
        'adx_val': adx_avg  # Para visualización
    }
    
    # 4. DECISIÓN FINAL
    # Extraer symbol y current_price para proyección de escenarios
    symbol_for_decision = symbol
    current_price_for_decision = ladder_analysis.get('5m', {}).get('current_price', 0)
    
    decision = make_pattern_based_decision(
        patterns_result, ladder_analysis, math_signals, 
        symbol_for_decision, current_price_for_decision
    )

    # --- REPORTE INTELIGENTE ---
    print(f"\n🔍 REPORTE INTELIGENTE: {symbol}")
    print(f"   1. ¿QUÉ VEO? (Percepción)")
    pat_str = patterns_result['best_pattern']['type'] if patterns_result['best_pattern'] else "❌ NINGUNO"
    print(f"      - Patrón Detectado: {pat_str}")
    
    # ADX VISUAL
    if ENABLE_ADX_VISUAL:
        adx_val = math_signals.get('adx_val', 0)
        bars = int(adx_val / 5)  # 25 -> 5 barras
        visual_bar = "|" * bars + "." * (10 - bars)
        print(f"      - Fuerza ADX:       [{visual_bar}] {adx_val:.1f} ({'✅ ALINEADA' if math_signals['adx_aligned'] else '⚠️ DÉBIL'})")
    else:
        print(f"      - Fuerza ADX:       {'✅ ALINEADA' if math_signals['adx_aligned'] else '⚠️ DÉBIL'}")
        
    print(f"      - Momentum:         {'✅ PROPAGANDO' if math_signals['momentum_propagating'] else '⚠️ NO'}")
    
    print(f"   2. ¿QUÉ PENSÉ? (Decisión)")
    print(f"      - Acción: {decision['action']} (Confianza: {decision['confidence']})")
    print(f"      - Razones: {', '.join(decision['reasons'])}")
    
    if decision['action'] != 'ENTER':
        print(f"   3. ¿QUÉ FALTÓ?")
        print(f"      - Puntos faltantes: {decision.get('missing', 0)}")
    print("-" * 60)
    
    # --- TELEMETRÍA (HUELLA DIGITAL) ---
    # Guardar estado completo para análisis forense
    try:
        db_manager.insert_telemetry(
            symbol, 
            'OPERAR' if decision['action'] == 'ENTER' else 'NO_OPERAR',
            decision['confidence'],
            ladder_analysis
        )
    except Exception as e:
        print(f"⚠️ Error guardando telemetría: {e}")

    # Formatear para consumo del bot_main
    if decision['action'] == 'ENTER':
        return {
            'action': 'OPERAR',
            'trade_type': decision['strategy']['type'],
            'confidence': decision['confidence'],
            'pattern': decision.get('pattern'),
            'reasons': decision['reasons'],
            'full_analysis': {
                'ladder_data': ladder_analysis,
                'patterns': patterns_result
            },
            'position_size': decision['strategy']['position_size']
        }
    else:
        return {
            'action': 'NO_OPERAR',
            'confidence': decision['confidence'],
            'reason': f"Low confidence ({decision['confidence']}): {', '.join(decision['reasons'][:2])}",
            'full_analysis': {
                'ladder_data': ladder_analysis,
                'patterns': patterns_result
            }
        }

def make_pattern_based_decision(patterns, ladder, signals, symbol, current_price):
    """Decisión basada en confluencia REAL + ESCENARIOS"""
    score = 0
    reasons = []
    
    # 0. DEFINIR MULTIPLICADOR DE MODO
    penalty_multiplier = 1.0  # Default (CONSERVATIVE)
    if TRADING_MODE == 'AGGRESSIVE':
        penalty_multiplier = 0.6
    elif TRADING_MODE == 'BALANCED':
        penalty_multiplier = 0.8
    
    # ==============================================================================
    # 0.5. BLOQUEO TEMPRANO DE DIRECCIÓN (Optimización)
    # ==============================================================================
    dir_4h = ladder.get('4h', {}).get('direction', 'NEUTRAL')
    
    # A. 4H MANDA (LÓGICA GRADIENTE):
    # 1. Si es BEARISH -> FUERA SIEMPRE (No operamos contra tendencia mayor)
    # v4.9.0: PURE SCORING REPLACEMENT (No Hard Blocks)
    if dir_4h == 'BEARISH':
        adx_4h = ladder.get('4h', {}).get('adx', {}).get('adx_current', 0)
        if adx_4h > 40:
            score -= 60
            reasons.append(f"⚠️ 4H BEARISH Extremo (ADX {adx_4h:.1f})")
        elif adx_4h > 25:
            score -= 40
            reasons.append(f"⚠️ 4H BEARISH Fuerte (ADX {adx_4h:.1f})")
        else:
            score -= 20
            reasons.append("⚠️ 4H BEARISH Débil - Posible giro")

    elif dir_4h in ['LATERAL', 'NEUTRAL']:
        # Penalizar incertidumbre, pero NO bloquear si hay fuerza menor
        score -= 15
        reasons.append(f"⚠️ 4H {dir_4h} (Lack of trend)")

    # B. DEMASIADO RUIDO: Diagnóstico de Lateralidad
    valid_dirs = [
        ladder.get('4h', {}).get('direction'),
        ladder.get('1h', {}).get('direction'),
        ladder.get('15m', {}).get('direction'),
        ladder.get('5m', {}).get('direction')
    ]
    lateral_count = valid_dirs.count('LATERAL')
    
    if lateral_count >= 3:
        score -= 30
        reasons.append(f"⚠️ High Noise (LATERAL in {lateral_count} TFs)")
        
    # C. DIRECCIÓN DEL PATRÓN Y CLARIDAD
    pattern_dir = patterns['confidence'].get('direction', 'NEUTRAL')
    
    non_lateral = [d for d in valid_dirs if d in ['BULLISH', 'BEARISH']]
    
    if len(non_lateral) < 2:
        score -= 25
        reasons.append('⚠️ Low Clarity (Insuficientes TFs con dirección)')
    
    # ==============================================================================
    # MARKET FORCE OVERRIDE MODE
    # ==============================================================================
    market_force = signals.get('market_force', {'detected': False})
    entry_threshold = 65  # Default
    
    if market_force['detected']:
        # Con Market Force, relajar filtros moderadamente (AJUSTADO)
        penalty_multiplier *= 0.7  # Penalizaciones al 70% (más conservador)
        entry_threshold = 55        # Threshold rebajado de 65 a 55 (más estricto)
        reasons.append("🔥 MARKET FORCE MODE: Relaxed filters activated")
        print(f"   ⚡ Market Force Override: Threshold {entry_threshold}, Penalties {penalty_multiplier}x")
    
    # ==============================================================================
    # 1. FUERZA DE MERCADO (The "Human" First Step)
    # ==============================================================================
    if market_force['detected']:
        score += 40 # Boost masivo
        reasons.append(market_force['reason'])
    
    # NUEVO: Prioridad a Patrones Geométricos (Solicitud Usuario)
    # Si hay un patrón claro (>70%), lo priorizamos sobre conteos difusos
    if patterns['best_pattern'] and patterns['best_pattern']['confidence'] > 70:
        score += 25
        reasons.append(f"✨ High Confidence Pattern: {patterns['best_pattern']['type']}")
        print(f"   ✨ Pattern Priority: {patterns['best_pattern']['type']} (Conf: {patterns['best_pattern']['confidence']})")
    
    # ==============================================================================
    
    # ==============================================================================
    # 1.5. CONTEXTO DE ENTRADA (Si Market Force detectado)
    # ==============================================================================
    if market_force['detected']:
        entry_context = detect_entry_context(ladder, pattern_dir, current_price, market_force_active=True)
        
        # CRÍTICO: Bloqueo total si context score muy bajo
        # AJUSTE: Con ADX extremo (>40), rebajar threshold a 15
        adx_avg = signals.get('adx_val', 0)
        min_context_score = 15 if adx_avg > 40 else 20
        
        if entry_context['score_bonus'] < min_context_score:
            # En lugar de rechazar directamente, consultar al Sniper
            print(f"   ⚠️ Entry context débil (score:{entry_context['score_bonus']}), consultando Sniper...")
            
            from brain_think import sniper_checklist
            direction_force = market_force['direction']
            detected_pattern = patterns['best_pattern']['type'] if patterns['best_pattern'] else None
            sniper_verdict = sniper_checklist(symbol, direction_force, pattern_type=detected_pattern, ladder=ladder)
            
            if sniper_verdict['ready'] and sniper_verdict['total_score'] >= 70:
                # Sniper aprueba: confiar en su criterio
                print(f"   ✅ SNIPER OVERRIDE: Aprueba ({sniper_verdict['total_score']}), ignorando entry_context bajo")
                # Continuar con el flujo normal, no rechazar
            else:
                # Sniper también rechaza: bloquear
                print(f"   ❌ Sniper también rechaza (score:{sniper_verdict['total_score']})")
                return {
                    'action': 'WAIT',
                    'confidence': 0,
                    'reasons': [f'❌ Entry context+Sniper blocked: {entry_context["reasons"]}'],
                    'scenarios': None,
                    'missing': 100
                }
        
        if entry_context['valid']:
            score += entry_context['score_bonus']
            reasons.extend(entry_context['reasons'])
            print(f"   🎯 Entry context VALIDATED (+{entry_context['score_bonus']} pts)")
        else:
            # Market Force detectado pero contexto NO ideal
            # Penalización más fuerte para esperar mejor momento
            score -= 20  # Aumentado de 10 a 20
            reasons.append("⏸️ Market Force detected, waiting for better entry context")
            reasons.extend(entry_context['reasons'])
            print("   ⚠️ Waiting for ideal entry context (penalty -20)...")
    
    # ==============================================================================
    # 2. PATRONES (The "Entry" Signal)
    # ==============================================================================
    if patterns['best_pattern']:
        pattern = patterns['best_pattern']
        score += pattern['confidence'] * 0.4
        reasons.append(f"Pattern: {pattern['type']} ({pattern['tf']})")
        
        # Si hay fuerza, relajamos requisitos de patrón
        if market_force['detected']:
             # Ya tiene boost por fuerza, pero confirmamos que el patrón apoya
             pass
    elif market_force['detected']:
        # Si NO hay patrón pero hay FUERZA BRUTAL
        score += 20 # Compensar falta de patrón geométrico
        reasons.append("Force compensates pattern")
    
    # Confluencia de patrones
    confluence_bonus = 0
    if patterns['best_pattern']:
        pattern_count = sum(1 for tf in patterns['patterns'].values() if len(tf) > 0)
        if pattern_count >= 2:
            confluence_bonus = 10
            reasons.append(f"Pattern Confluence (+{confluence_bonus})")
    score += confluence_bonus
    
    # ==============================================================================
    # 3. PROYECCIÓN DE ESCENARIOS (The "Plan" - Minimizar Pérdidas)
    # ==============================================================================
    # Proyectamos SIEMPRE si hay mínima viabilidad (score > 20) para ver el R/R
    scenarios = None
    if score > 20:
        try:
            from brain_scenarios import project_scenarios
            scenarios = project_scenarios(symbol, current_price, ladder)
            
            # Evaluar Escenario Primario
            primary = scenarios.get('primary', {})
            rr = primary.get('rr_ratio', 0)
            prob = primary.get('probability', 0)
            
            # BONUS/PENALTY por R/R
            if rr >= 2.0:
                score += 10
                reasons.append(f"✅ Great R/R ({rr:.1f})")
            elif rr >= 1.5:
                score += 5
                reasons.append(f"✅ Good R/R ({rr:.1f})")
            elif rr < 1.0:
                score -= 20 # Penalización fuerte: Riesgo > Beneficio
                reasons.append(f"❌ Bad R/R ({rr:.1f})")
            
            # BONUS/PENALTY por Probabilidad del Escenario
            if prob < 50:
                score -= 15
                reasons.append(f"⚠️ Low Scenario Prob ({prob}%)")
                
        except Exception as e:
            print(f"⚠️ Error projecting scenarios: {e}")
            reasons.append("Scenario Error")

    # ==============================================================================
    # 4. ALINEACIÓN Y CONFIRMACIÓN (The "Checks")
    # ==============================================================================
    
    # Estructura de mercado
    pattern_dir = patterns['confidence']['direction']
    structure_aligned = 0
    for tf, analysis in ladder.items():
        if analysis and analysis.get('direction') == pattern_dir:
            structure_aligned += 1
            
    if structure_aligned >= 3:
        score += 30
        reasons.append(f"Trend aligned {structure_aligned}/4")
    elif structure_aligned >= 2:
        score += 15
        reasons.append(f"Trend aligned {structure_aligned}/4")

    # Indicadores matemáticos
    if signals['adx_aligned']:
        score += 15
        reasons.append("ADX aligned")
    
    if signals['momentum_propagating']:
        score += 15
        reasons.append("Momentum propagating")
        
    if signals['volume_confirming']:
        score += 10
        reasons.append("Volume confirming")

    # ==============================================================================
    # 5. SMART CONTEXT LOGIC (Penalizaciones)
    # ==============================================================================
    a5m = ladder.get('5m', {})
    if a5m:
        # 0. VP HIGH DANGER ZONE (Evitar comprar techo)
        vp_data = a5m.get('volume_profile', {})
        profile_high = vp_data.get('profile_high', 0)
        
        if profile_high > 0 and current_price > 0:
            dist_to_vp_high = ((profile_high - current_price) / current_price) * 100
            
            # Solo penalizar si estamos DENTRO del perfil (dist > 0)
            # Si dist < 0 es breakout (ya manejado en otro lado o positivo)
            if 0 < dist_to_vp_high < 1.5:
                score -= 35
                reasons.append(f"⛔ DANGER: Too close to VP High ({dist_to_vp_high:.1f}%)")
            elif 1.5 <= dist_to_vp_high < 3.0:
                score -= 15
                reasons.append(f"⚠️ Caution: Near VP High ({dist_to_vp_high:.1f}%)")

        # A. DYNAMIC OVEREXTENSION
        dist_ema = abs(a5m.get('dist_ema_pct', 0))
        atr_pct = a5m.get('atr_pct', 1.0)
        extension_ratio = dist_ema / atr_pct if atr_pct > 0 else 0
        
        if extension_ratio > 4.0:
            score -= int(15 * penalty_multiplier)
            reasons.append(f"⚠️ Overextended ({extension_ratio:.1f} ATRs)")
        elif extension_ratio > 2.0:
            score -= int(10 * penalty_multiplier)
            reasons.append(f"⚠️ Extended ({extension_ratio:.1f} ATRs)")
        elif extension_ratio < 0.5:
            score += 5
            reasons.append("In Value Zone")

        # B. ZONE AWARENESS
        sr_data = a5m.get('sr', {})
        if pattern_dir == 'BULLISH':
            resistances = sr_data.get('resistances', [])
            if resistances:
                nearest_res = resistances[0]
                dist_to_res = (nearest_res - current_price) / current_price * 100
                
                if 0 < dist_to_res < 1.0:
                    # NUEVO: Interpretar según Market Force
                    if market_force['detected']:
                        # CON FUERZA: Resistencia cercana = Breakout opportunity
                        score += 10
                        reasons.append(f"🚀 Breakout w/Force (+{dist_to_res:.2f}%)")
                        print(f"   🎯 Breakout opportunity detected at +{dist_to_res:.2f}%")

        elif pattern_dir == 'BEARISH':
            supports = sr_data.get('supports', [])
            if supports:
                nearest_sup = supports[0]
                dist_to_sup = (current_price - nearest_sup) / current_price * 100
                if 0 < dist_to_sup < 1.0:
                    score -= int(35 * penalty_multiplier)
                    reasons.append(f"⚠️ Support below (-{dist_to_sup:.2f}%)")

        # C. RSI TIMING
        rsi_val = a5m.get('rsi', {}).get('rsi_current', 50)
        mom_strength = a5m.get('momentum', {}).get('strength', 'débil')
        if pattern_dir == 'BULLISH' and rsi_val > 75 and mom_strength != 'fuerte':
            score -= int(15 * penalty_multiplier)
            reasons.append(f"⚠️ RSI Overbought ({rsi_val:.0f})")
        elif pattern_dir == 'BEARISH' and rsi_val < 25 and mom_strength != 'fuerte':
            score -= int(15 * penalty_multiplier)
            reasons.append(f"⚠️ RSI Oversold ({rsi_val:.0f})")

    # D. TIMING ÓPTIMO (Flags)
    if ENABLE_PULLBACK_CHECK and not market_force['detected']:
        # CAMBIO: Solo aplicar si NO hay Market Force
        poc = a5m.get('volume_profile', {}).get('poc_current', 0)
        dist_to_poc = abs(current_price - poc) / current_price * 100 if poc > 0 else 0
        if extension_ratio > 3.0 and dist_to_poc > 2.0:
            score -= int(40 * penalty_multiplier)
            reasons.append(f"⏸️ Pullback needed ({dist_to_poc:.1f}% from POC)")
    elif ENABLE_PULLBACK_CHECK and market_force['detected']:
        # Con Market Force, NO exigir pullback
        reasons.append("✅ Pullback check BYPASSED by Market Force")
        print("   ⚡ Pullback requirement waived (Market Force active)")

    if ENABLE_BREAKOUT_CHECK and pattern_dir == 'BULLISH':
        resistances = sr_data.get('resistances', [])
        if resistances:
            nearest_res = resistances[0]
            dist_to_res = (nearest_res - current_price) / current_price * 100
            if 0 < dist_to_res < 1.0:
                score -= int(30 * penalty_multiplier)
                reasons.append(f"⛔ Breakout Blocked (+{dist_to_res:.2f}%)")

    if ENABLE_HH_PENALTY and pattern_dir == 'BULLISH':
        recent_highs = a5m.get('zigzag', {}).get('pivot_highs', [])
        if recent_highs:
            hh = max(recent_highs)
            if current_price < hh:
                dist_to_hh = ((hh - current_price) / current_price * 100)
                if dist_to_hh < 0.2:
                    # NUEVO: Validar si es continuación de tendencia fuerte
                    adx_trend_4h = ladder.get('4h', {}).get('adx', {}).get('adx_trend', 'estable')
                    
                    if market_force['detected'] and adx_trend_4h == 'creciendo':
                        # Tendencia ACELERANDO = Permitir new highs
                        score += 5
                        reasons.append("✅ New HH in accelerating trend (continuation)")
                        print(f"   📈 New high allowed: ADX trend {adx_trend_4h}")
                    else:
                        # Sin aceleración = Peligro buying top
                        score -= int(40 * penalty_multiplier)
                        reasons.append(f"⛔ Buying Top (Dist to HH: {dist_to_hh:.2f}%)")

    if ENABLE_CLEAN_SPACE_BONUS and pattern_dir == 'BULLISH':
        resistances = sr_data.get('resistances', [])
        recent_highs = a5m.get('zigzag', {}).get('pivot_highs', [])
        res_clear = not resistances or ((resistances[0] - current_price) / current_price * 100 > 3.0)
        hh_clear = True
        if recent_highs:
            hh = max(recent_highs)
            if current_price < hh and ((hh - current_price) / current_price * 100) <= 3.0:
                hh_clear = False
        if res_clear and hh_clear:
            score += 10
            reasons.append("✨ Clean Space (>3% room)")

    # Clamp score
    score = max(0, min(100, score))

    # ==============================================================================
    # COMPENSACIÓN DE FUERZA EXTREMA (Market Force Override)
    # ==============================================================================
    # Si ADX es brutal (>35), compensar penalizaciones de "precisión milimétrica"
    # FILOSOFÍA: La fuerza crea los breakouts, el precio sigue a la fuerza
    if market_force['detected']:
        adx_avg = signals.get('adx_val', 0)
        if adx_avg > 35:
            compensation = 0
            comp_reasons = []
            
            # Compensar "Buying Top" / "Dist to HH" si hay fuerza extrema
            if any('Buying Top' in r or 'Dist to HH' in r for r in reasons):
                compensation += 25
                comp_reasons.append("Buying Top → Breakout Setup")
            
            # Compensar "Overextended" si momentum es sostenido
            if any('Overextended' in r or 'Extension' in r for r in reasons):
                compensation += 15
                comp_reasons.append("Extension → Trend Strength")
            
            # Compensar "Breakout Blocked" (contradicción: resistencia cerca)
            if any('Breakout Blocked' in r for r in reasons):
                compensation += 20
                comp_reasons.append("Breakout Blocked → Force Will Break")
            
            if compensation > 0:
                score += compensation
                reasons.append(f"⚡ FORCE OVERRIDE: {', '.join(comp_reasons)} (+{compensation})")
                print(f"   🔥 Force Compensation: ADX {adx_avg:.1f} overrides precision penalties (+{compensation})")
    
    # Re-Clamp score tras compensación
    score = max(0, min(100, score))

    # ==============================================================================
    # ⚠️ THRESHOLD CHECK MOVED TO LINE ~930 (after all v4.8.0 validations)
    # ==============================================================================
    
    # --- v4.7.0: HARD BLOCK MOMENTUM 5M (Caso Recall) ---
    # Si el momentum de ejecución (5M) es contrario, BLOQUEAR SIEMPRE.
    mom_5m = ladder.get('5m', {}).get('momentum', {})
    mom_color_5m = mom_5m.get('color_current', 'neutral')
    
    direction_final = pattern_dir if pattern_dir in ['BULLISH', 'BEARISH'] else market_force.get('direction', 'NEUTRAL')

    if direction_final == 'BULLISH' and mom_color_5m in ['red', 'maroon']:
        return {
            'action': 'WAIT',
            'confidence': 0,
            'reasons': [f'❌ MOMENTUM 5M CONTRARIO ({mom_color_5m}) - Esperar giro'],
            'scenarios': scenarios,
            'missing': 100
        }
    elif direction_final == 'BEARISH' and mom_color_5m in ['green', 'lime']:
        return {

        return {
            'type': 'REVERSAL',
            'stop_multiplier': 1.5,
            'target_multiplier': 2.0,
            'position_size': 0.5
        }
    # Continuación
    elif 'FLAG' in pattern_type or 'TRIANGLE' in pattern_type or 'TREND' in pattern_type or 'MOMENTUM' in pattern_type:
        return {
            'type': 'CONTINUATION',
            'stop_multiplier': 2.0,
            'target_multiplier': 3.0,

        }
    
    # Default
    return {
        'type': 'STANDARD',
        'stop_multiplier': 2.0,
        'target_multiplier': 2.0,
        'position_size': 0.75
    }

def detect_entry_context(ladder, pattern_dir, current_price, market_force_active=False):
    """
    Detecta el CONTEXTO ideal de entrada después de Market Force.
    Busca el "pantallazo" que el trader ve antes de entrar.
    
    VERSIÓN MEJORADA: Validaciones más estrictas basadas en resultados reales.
    
    Args:
        ladder: Dict con análisis de todos los timeframes
        pattern_dir: Dirección del patrón ('BULLISH'/'BEARISH')
        current_price: Precio actual
        market_force_active: Si True, relaja ciertos requisitos (Buying Top, Momentum)
    
    Returns:
        dict: {
            'valid': bool,
            'score_bonus': int (0-30),
            'reasons': [str]
        }
    """
    context_score = 0
    context_reasons = []
    
    a5m = ladder.get('5m', {})
    a15m = ladder.get('15m', {})
    
    if not a5m or not a15m:
        return {'valid': False, 'score_bonus': 0, 'reasons': ['⚠️ Missing 5M/15M data']}
    
    # 1. PULLBACK MÍNIMO (5-15% desde último high)
    recent_highs = a5m.get('zigzag', {}).get('pivot_highs', [])
    pullback_pct = 0
    
    if recent_highs and pattern_dir == 'BULLISH':
        last_high = max(recent_highs)
        pullback_pct = ((last_high - current_price) / last_high) * 100
        
        # DEFINIR UMBRAL DE "BUYING TOP"
        # Si hay fuerza, toleramos estar mucho más cerca del high (breakout inminente)
        buying_top_threshold = 0.5 if market_force_active else 2.0
        
        if 5 <= pullback_pct <= 15:
            # IDEAL: Pullback moderado, no profundo
            context_score += 10
            context_reasons.append(f"✅ Healthy pullback ({pullback_pct:.1f}%)")
        elif pullback_pct < buying_top_threshold:
            # NUEVO: Muy cerca del high = PELIGRO (Buying Top)
            # El usuario se quejó de esto específicamente en BRETT
            context_reasons.append(f"❌ Buying Top (Only {pullback_pct:.1f}% from High)")
            return {'valid': False, 'score_bonus': 0, 'reasons': context_reasons}
        elif pullback_pct < 5:
            # Pullback mínimo
            if market_force_active:
                # Con fuerza, esto es señal de fortaleza (flag apretada)
                context_score += 5
                context_reasons.append(f"🚀 Strong trend/Tight flag ({pullback_pct:.1f}%)")
            else:
                context_reasons.append(f"⚠️ Minimal pullback ({pullback_pct:.1f}%)")
        elif pullback_pct > 20:
            # Demasiado profundo
            context_reasons.append(f"⚠️ Deep correction ({pullback_pct:.1f}%)")
    
    # 2. MOMENTUM RECUPERÁNDOSE (VERSIÓN MÁS ESTRICTA)
    momentum_5m = a5m.get('momentum', {})
    consecutive = momentum_5m.get('consecutive_count', 0)
    color = momentum_5m.get('color_current', 'red')
    
    # AJUSTE: Con fuerza extrema, aceptar incluso 1 vela verde
    a1h = ladder.get('1h', {})
    adx_1h = a1h.get('adx', {}).get('adx_current', 0)
    
    if adx_1h > 40:
        required_candles = 1  # Fuerza brutal: 1 vela es suficiente
    elif market_force_active:
        required_candles = 2
    else:
        required_candles = 3
    
    if pattern_dir == 'BULLISH':
        if color in ['green', 'lime'] and consecutive >= required_candles:
            # NUEVO: Mínimo 3 velas (antes 2), o 2 si hay Market Force
            context_score += 10
            context_reasons.append(f"✅ Momentum recovering ({consecutive}v green)")
        elif consecutive < required_candles:
            # BLOQUEO si momentum insuficiente
            context_reasons.append(f"❌ Insufficient momentum ({consecutive}v, need {required_candles}+)")
            return {'valid': False, 'score_bonus': 0, 'reasons': context_reasons}
        elif color in ['red', 'maroon']:
            context_reasons.append("❌ Momentum still red")
            return {'valid': False, 'score_bonus': 0, 'reasons': context_reasons}
    
    # 3. VOLUMEN INCREMENTANDO
    vp_data = a5m.get('volume_profile', {})
    vol_ratio = vp_data.get('volume_ratio', 0)
    
    if vol_ratio >= 1.2:
        context_score += 5
        context_reasons.append(f"✅ Volume increasing ({vol_ratio:.1f}x)")
    elif vol_ratio < 0.8:
        context_reasons.append(f"⚠️ Volume declining ({vol_ratio:.1f}x)")
    
    # 4. ZONA DE FUERZA 15M (ADX > 25)
    adx_15m = a15m.get('adx', {})
    adx_val_15m = adx_15m.get('adx_current', 0)
    
    if adx_val_15m >= 25:
        context_score += 10
        context_reasons.append(f"✅ 15M Force Zone (ADX {adx_val_15m:.1f})")
    else:
        context_reasons.append(f"⚠️ Weak 15M ADX ({adx_val_15m:.1f})")
    
    # NUEVO: 5. VALIDAR DISTANCIA A STOP LOSS
    sr_data = a5m.get('sr', {})
    nearest_support = sr_data.get('supports', [None])[0] if sr_data.get('supports') else None
    
    if nearest_support and pattern_dir == 'BULLISH':
        stop_distance_pct = ((current_price - nearest_support) / current_price) * 100
        
        if stop_distance_pct > 2.5:
            # Stop demasiado lejos
            context_score -= 5
            context_reasons.append(f"⚠️ Stop too far ({stop_distance_pct:.1f}%)")
        elif stop_distance_pct < 1.0:
            # Stop muy cerca = PELIGRO (entra justo antes de stop)
            context_reasons.append(f"❌ Stop too close ({stop_distance_pct:.1f}%), danger zone")
            return {'valid': False, 'score_bonus': 0, 'reasons': context_reasons}
    
    # Validación Final MÁS ESTRICTA
    valid = context_score >= 20  # NUEVO: Aumentado de 15 a 20
    
    return {
        'valid': valid,
        'score_bonus': context_score if valid else 0,
        'reasons': context_reasons
    }
