"""
brain_adx_plus.py - Análisis Granular DI+ (Directional Index Plus)
===================================================================
Responsabilidad: Analizar la línea VERDE del ADX (DI+)
Para LONG: DI+ debe DOMINAR
Para SHORT: DI+ debe estar SUBORDINADO
Autor: Joel Rodríguez Diarte
Fecha: 2025-12-21
"""
import pandas as pd
import numpy as np
def analyze_di_plus(df, lookback=14):
    """
    Analiza DI+ de forma granular.
    
    Args:
        df: DataFrame con OHLCV
        lookback: Período del indicador
        
    Returns:
        dict: {
            'value': float,           # Valor actual DI+
            'trend': str,             # 'creciendo', 'decreciendo', 'estable'
            'velocity': float,        # Velocidad de cambio pts/vela
            'history': list,          # Últimos 5 valores
            'phase': str,             # 'nacimiento', 'climax', 'muerte'
            'strength': str,          # 'fuerte', 'moderado', 'debil'
            'valid': bool
        }
    """
    try:
        if len(df) < lookback + 5:
            return _default_result()
        
        # Calcular DI+
        high = df['high']
        low = df['low']
        close = df['close']
        
        # True Range
        tr1 = high - low
        tr2 = abs(high - close.shift(1))
        tr3 = abs(low - close.shift(1))
        tr = pd.concat([tr1, tr2, tr3], axis=1).max(axis=1)
        atr = tr.rolling(lookback).mean()
        
        # +DM (Directional Movement Plus)
        up_move = high - high.shift(1)
        down_move = low.shift(1) - low
        
        plus_dm = pd.Series(0.0, index=df.index)
        plus_dm[(up_move > down_move) & (up_move > 0)] = up_move
        
        # +DI
        plus_di = 100 * (plus_dm.rolling(lookback).mean() / atr)
        
        # Valores
        current = float(plus_di.iloc[-1])
        prev_1 = float(plus_di.iloc[-2])
        prev_3 = float(plus_di.iloc[-4])
        prev_5 = float(plus_di.iloc[-6]) if len(plus_di) > 5 else current
        
        # Trend (últimas 3 velas)
        delta = current - prev_3
        if delta > 2:
            trend = 'creciendo'
        elif delta < -2:
            trend = 'decreciendo'
        else:
            trend = 'estable'
        
        # Velocity (pts por vela)
        velocity = delta / 3
        
        # History
        history = [float(plus_di.iloc[-i]) for i in range(1, 6)]
        
        # Phase (basado en cambio desde 5 velas)
        delta_5 = current - prev_5
        if delta_5 > 5 and current > 25:
            phase = 'climax'
        elif delta_5 > 2:
            phase = 'nacimiento'
        elif delta_5 < -3:
            phase = 'muerte'
        else:
            phase = 'estable'
        
        # Strength
        if current > 35:
            strength = 'fuerte'
        elif current > 25:
            strength = 'moderado'
        else:
            strength = 'debil'
        
        return {
            'value': round(current, 2),
            'trend': trend,
            'velocity': round(velocity, 2),
            'history': history,
            'phase': phase,
            'strength': strength,
            'valid': True
        }
        
    except Exception as e:
        print(f"❌ Error analyze_di_plus: {e}")
        return _default_result()
def validate_di_plus_for_long(di_plus_data, di_minus_value):
    """
    Valida DI+ para entrada LONG.
    
    Requisitos:
    - DI+ > DI- por al menos 5 puntos
    - DI+ no decreciendo rápido
    - DI+ > 20 (mínimo de fuerza)
    
    Returns:
        dict: {valid, score, block_reason}
    """
    if not di_plus_data['valid']:
        return {'valid': False, 'score': 0, 'block_reason': 'DI+ inválido'}
    
    value = di_plus_data['value']
    trend = di_plus_data['trend']
    spread = value - di_minus_value
    
    # BLOQUEO: DI+ no domina (spread < 5)
    if spread < 5:
        return {
            'valid': False,
            'score': -30,
            'block_reason': f'DI+ no domina: spread {spread:.1f} < 5'
        }
    
    # BLOQUEO: DI+ muy débil
    if value < 20:
        return {
            'valid': False,
            'score': -20,
            'block_reason': f'DI+ débil: {value:.1f} < 20'
        }
    
    # ADVERTENCIA: DI+ decreciendo
    if trend == 'decreciendo':
        return {
            'valid': True,
            'score': -15,
            'block_reason': f'DI+ decreciendo'
        }
    
    # BONUS: DI+ fuerte y creciendo
    score = 0
    if spread >= 15:
        score += 20
    elif spread >= 10:
        score += 10
    
    if trend == 'creciendo':
        score += 10
    
    if di_plus_data['phase'] == 'climax':
        score += 15
    
    return {
        'valid': True,
        'score': score,
        'block_reason': None
    }

📦 MÓDULO 3: brain_adx_value.py
"""
brain_adx_value.py - Análisis Granular ADX Value (Línea Negra)
==============================================================
Responsabilidad: Analizar el ADX (fuerza de la tendencia)
ADX > 25 = Tendencia fuerte
ADX < 20 = Lateral/débil
ADX creciendo = Tendencia ganando fuerza
ADX decreciendo = Tendencia perdiendo fuerza
Autor: Joel Rodríguez Diarte
Fecha: 2025-12-21
"""
import pandas as pd
import numpy as np
def analyze_adx_value(df, lookback=14):
    """
    Analiza ADX (valor absoluto) de forma granular.
    
    Returns:
        dict: {
            'value': float,
            'trend': str,
            'velocity': float,
            'lifecycle': str,      # 'nacimiento', 'expansion', 'climax', 'muerte'
            'strength': str,
            'history': list,
            'valid': bool
        }
    """
    try:
        if len(df) < lookback + 10:
            return _default_result()
        
        high = df['high']
        low = df['low']
        close = df['close']
        
        # True Range
        tr1 = high - low
        tr2 = abs(high - close.shift(1))
        tr3 = abs(low - close.shift(1))
        tr = pd.concat([tr1, tr2, tr3], axis=1).max(axis=1)
        atr = tr.rolling(lookback).mean()
        
        # +DM / -DM
        up_move = high - high.shift(1)
        down_move = low.shift(1) - low
        
        plus_dm = pd.Series(0.0, index=df.index)
        plus_dm[(up_move > down_move) & (up_move > 0)] = up_move
        
        minus_dm = pd.Series(0.0, index=df.index)
        minus_dm[(down_move > up_move) & (down_move > 0)] = down_move
        
        # DI+ / DI-
        plus_di = 100 * (plus_dm.rolling(lookback).mean() / atr)
        minus_di = 100 * (minus_dm.rolling(lookback).mean() / atr)
        
        # ADX
        dx = 100 * abs(plus_di - minus_di) / (plus_di + minus_di)
        adx = dx.rolling(lookback).mean()
        
        # Valores
        current = float(adx.iloc[-1])
        prev_1 = float(adx.iloc[-2])
        prev_3 = float(adx.iloc[-4])
        prev_5 = float(adx.iloc[-6])
        prev_10 = float(adx.iloc[-11]) if len(adx) > 10 else current
        
        # Trend (últimas 3 velas)
        delta_3 = current - prev_3
        if delta_3 > 1.5:
            trend = 'creciendo'
        elif delta_3 < -1.5:
            trend = 'decreciendo'
        else:
            trend = 'estable'
        
        # Velocity
        velocity = delta_3 / 3
        
        # Lifecycle (basado en cambios históricos)
        delta_5 = current - prev_5
        delta_10 = current - prev_10
        
        if current < 20:
            lifecycle = 'lateral'
        elif delta_10 > 10 and delta_5 > 3:
            lifecycle = 'nacimiento'  # ADX subiendo fuerte
        elif current > 40 and delta_5 > 0:
            lifecycle = 'climax'      # ADX alto y aún subiendo
        elif current > 35 and delta_5 < -2:
            lifecycle = 'muerte'      # ADX alto pero cayendo
        elif delta_5 > 1:
            lifecycle = 'expansion'   # ADX subiendo moderado
        else:
            lifecycle = 'estable'
        
        # Strength
        if current > 50:
            strength = 'brutal'
        elif current > 35:
            strength = 'fuerte'
        elif current > 25:
            strength = 'moderado'
        elif current > 20:
            strength = 'debil'
        else:
            strength = 'muy_debil'
        
        # History
        history = [float(adx.iloc[-i]) for i in range(1, 8)]
        
        return {
            'value': round(current, 2),
            'trend': trend,
            'velocity': round(velocity, 2),
            'lifecycle': lifecycle,
            'strength': strength,
            'history': history,
            'valid': True
        }
        
    except Exception as e:
        print(f"❌ Error analyze_adx_value: {e}")
        return _default_result()
def validate_adx_for_entry(adx_data, brutal_force=False):
    """
    Valida ADX para cualquier entrada (LONG o SHORT).
    
    Requisitos:
    - ADX > 20 (mínimo)
    - ADX no en fase 'muerte' (a menos que brutal_force)
    - ADX no en 'lateral'
    """
    if not adx_data['valid']:
        return {'valid': False, 'score': 0, 'block_reason': 'ADX inválido'}
    
    value = adx_data['value']
    trend = adx_data['trend']
    lifecycle = adx_data['lifecycle']
    strength = adx_data['strength']
    
    # BLOQUEO: ADX muy bajo (lateral)
    if value < 18:
        return {
            'valid': False,
            'score': -40,
            'block_reason': f'ADX {value:.1f} < 18 (lateral)'
        }
    
    # BLOQUEO: ADX decreciendo sin brutal_force
    if lifecycle == 'muerte' and not brutal_force:
        return {
            'valid': False,
            'score': -30,
            'block_reason': f'ADX {value:.1f} muriendo'
        }
    
    # ADVERTENCIA: ADX débil
    if 18 <= value < 22:
        return {
            'valid': True,
            'score': -10,
            'block_reason': f'ADX {value:.1f} débil'
        }
    
    # BONUS según strength
    score = 0
    
    if strength == 'brutal':
        score += 30
    elif strength == 'fuerte':
        score += 20
    elif strength == 'moderado':
        score += 10
    
    if lifecycle == 'climax':
        score += 15
    elif lifecycle == 'nacimiento':
        score += 20  # Mejor entrar en nacimiento
    elif lifecycle == 'expansion':
        score += 10
    
    if trend == 'creciendo':
        score += 10
    
    return {
        'valid': True,
        'score': score,
        'block_reason': None
    }

"""
brain_momentum_green.py - Análisis Granular Momentum Verde
==========================================================
Responsabilidad: Analizar barras VERDES del momentum (lime, green)
Colores verdes = momentum ALCISTA
- lime = verde brillante (momentum fuerte alcista)
- green = verde oscuro (momentum moderado alcista)
Fases:
- Nacimiento: velas 1-3 (óptimo)
- Clímax: velas 4-8 (bueno)
- Madurez: velas 9-12 (aceptable)
- Agotamiento: velas 13+ (peligro)
Autor: Joel Rodríguez Diarte
Fecha: 2025-12-21
"""
def analyze_momentum_green(momentum_data):
    """
    Analiza momentum verde de forma granular.
    
    Args:
        momentum_data: Dict del calculate_momentum_with_memory
        
    Returns:
        dict: {
            'is_green': bool,
            'color': str,
            'age': int,
            'phase': str,
            'phase_score': int,
            'intensity': float,
            'squeeze_on': bool,
            'valid': bool
        }
    """
    try:
        color = momentum_data.get('color_current', 'gray')
        age = momentum_data.get('consecutive_count', 0)
        intensity = momentum_data.get('intensity', 0)
        squeeze_on = momentum_data.get('squeeze_on', False)
        
        # ¿Es verde?
        is_green = color in ['lime', 'green']
        
        if not is_green:
            return {
                'is_green': False,
                'color': color,
                'age': age,
                'phase': 'no_green',
                'phase_score': 0,
                'intensity': intensity,
                'squeeze_on': squeeze_on,
                'valid': True
            }
        
        # Determinar fase según edad
        if age <= 3:
            phase = 'nacimiento'
            phase_score = 25  # Óptimo para entrar
        elif age <= 8:
            phase = 'climax'
            phase_score = 20  # Muy bueno
        elif age <= 12:
            phase = 'madurez'
            phase_score = 5   # Aceptable
        elif age <= 15:
            phase = 'tardio'
            phase_score = -15 # Cuidado
        else:
            phase = 'agotamiento'
            phase_score = -30 # Peligro
        
        # Bonus por color lime (más fuerte que green)
        if color == 'lime':
            phase_score += 5
        
        # Bonus por squeeze (compresión liberándose)
        if squeeze_on:
            phase_score += 10
        
        return {
            'is_green': True,
            'color': color,
            'age': age,
            'phase': phase,
            'phase_score': phase_score,
            'intensity': intensity,
            'squeeze_on': squeeze_on,
            'valid': True
        }
        
    except Exception as e:
        print(f"❌ Error analyze_momentum_green: {e}")
        return _default_result()
def validate_green_for_long(green_data):
    """
    Valida momentum verde para entrada LONG.
    
    Requisitos:
    - Debe ser verde (lime o green)
    - Fase no debe ser agotamiento
    """
    if not green_data['valid']:
        return {'valid': False, 'score': 0, 'block_reason': 'Momentum inválido'}
    
    # BLOQUEO: No es verde
    if not green_data['is_green']:
        return {
            'valid': False,
            'score': -40,
            'block_reason': f'Momentum {green_data["color"]} no es verde para LONG'
        }
    
    # BLOQUEO: Agotamiento
    if green_data['phase'] == 'agotamiento':
        return {
            'valid': False,
            'score': -35,
            'block_reason': f'Momentum verde AGOTADO ({green_data["age"]} velas)'
        }

"""
brain_momentum_red.py - Análisis Granular Momentum Rojo
=======================================================
Responsabilidad: Analizar barras ROJAS del momentum (red, maroon)
Colores rojos = momentum BAJISTA
- red = rojo brillante (momentum fuerte bajista)
- maroon = rojo oscuro (momentum moderado bajista)
Fases:
- Nacimiento: velas 1-3 (óptimo para SHORT)
- Clímax: velas 4-8 (bueno)
- Madurez: velas 9-12 (aceptable)
- Agotamiento: velas 13+ (peligro)
Autor: Joel Rodríguez Diarte
Fecha: 2025-12-21
"""
def analyze_momentum_red(momentum_data):
    """
    Analiza momentum rojo de forma granular.
    
    Args:
        momentum_data: Dict del calculate_momentum_with_memory
        
    Returns:
        dict: {
            'is_red': bool,
            'color': str,
            'age': int,
            'phase': str,
            'phase_score': int,
            'intensity': float,
            'squeeze_on': bool,
            'valid': bool
        }
    """
    try:
        color = momentum_data.get('color_current', 'gray')
        age = momentum_data.get('consecutive_count', 0)
        intensity = momentum_data.get('intensity', 0)
        squeeze_on = momentum_data.get('squeeze_on', False)
        
        # ¿Es rojo?
        is_red = color in ['red', 'maroon']
        
        if not is_red:
            return {
                'is_red': False,
                'color': color,
                'age': age,
                'phase': 'no_red',
                'phase_score': 0,
                'intensity': intensity,
                'squeeze_on': squeeze_on,
                'valid': True
            }
        
        # Determinar fase según edad
        if age <= 3:
            phase = 'nacimiento'
            phase_score = 25
        elif age <= 8:
            phase = 'climax'
            phase_score = 20
        elif age <= 12:
            phase = 'madurez'
            phase_score = 5
        elif age <= 15:
            phase = 'tardio'
            phase_score = -15
        else:
            phase = 'agotamiento'
            phase_score = -30
        
        # Bonus por color red (más fuerte que maroon)
        if color == 'red':
            phase_score += 5
        
        # Bonus por squeeze
        if squeeze_on:
            phase_score += 10
        
        return {
            'is_red': True,
            'color': color,
            'age': age,
            'phase': phase,
            'phase_score': phase_score,
            'intensity': intensity,
            'squeeze_on': squeeze_on,
            'valid': True
        }
        
    except Exception as e:
        print(f"❌ Error analyze_momentum_red: {e}")
        return _default_result()
