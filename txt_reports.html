class MovieAnalyzer:
    @staticmethod
    def calculate_velocity(values):
        """
        Velocity = (final - initial) / num_frames
        
        Args:
            values: list[float] - Serie temporal
        
        Returns:
            float - Velocity (units per frame)
        """
        if len(values) < 2:
            return 0.0
        return (values[-1] - values[0]) / (len(values) - 1)
    
    @staticmethod
    def calculate_acceleration(values):
        """
        Acceleration = velocity_second_half - velocity_first_half
        
        Args:
            values: list[float] - Serie temporal
        
        Returns:
            float - Acceleration
        """
        if len(values) < 4:
            return 0.0
        
        mid = len(values) // 2
        first_half = values[:mid]
        second_half = values[mid:]
        
        vel_first = MovieAnalyzer.calculate_velocity(first_half)
        vel_second = MovieAnalyzer.calculate_velocity(second_half)
        
        return vel_second - vel_first
    
    @staticmethod
    def analyze_frames(frames):
        """
        Calcula velocities y accelerations para TODOS los indicators.
        
        Args:
            frames: list[dict] - Frames completos
        
        Returns:
            dict - {
                'velocities': {...},
                'accelerations': {...}
            }
        """
        result = {
            'velocities': {},
            'accelerations': {}
        }
        
        # Extract time series
        adx_values = [f['indicators']['adx'] for f in frames]
        exhaustion_values = [f['indicators']['exhaustion_pct'] for f in frames]
        room_values = [f['indicators']['room_pct'] for f in frames]
        spread_values = [f['indicators']['spread'] for f in frames]
        grid_values = [f['indicators'].get('grid_position', 50) for f in frames]
        
        # Calculate velocities
        result['velocities']['adx'] = {
            'value': MovieAnalyzer.calculate_velocity(adx_values),
            'interpretation': MovieAnalyzer._interpret_velocity('adx', ...)
        }
        
        result['velocities']['exhaustion'] = {
            'value': MovieAnalyzer.calculate_velocity(exhaustion_values),
            'interpretation': MovieAnalyzer._interpret_velocity('exhaustion', ...)
        }
        
        # ... mismo para room, spread, grid
        
        # Calculate accelerations
        result['accelerations']['adx'] = {
            'value': MovieAnalyzer.calculate_acceleration(adx_values),
            'interpretation': MovieAnalyzer._interpret_acceleration('adx', ...)
        }
        
        # ... mismo para otros
        
        return result
    
    @staticmethod
    def _interpret_velocity(indicator, value):
        """
        Interpreta velocity en términos humanos.
        """
        if indicator == 'adx':
            if value > 2:
                return 'CRECIENDO_RAPIDO'
            elif value > 0.5:
                return 'CRECIENDO'
            elif value < -2:
                return 'CAYENDO_RAPIDO'
            elif value < -0.5:
                return 'CAYENDO'
            else:
                return 'ESTABLE'
        
        elif indicator == 'exhaustion':
            if value > 5:
                return 'QUEMANDOSE_BRUTAL'
            elif value > 2:
                return 'QUEMANDOSE_RAPIDO'
            elif value < -2:
                return 'RECUPERANDOSE'
            else:
                return 'ESTABLE'
        
        # ... etc para otros indicators
Key Point: Interpretations son DESCRIPTIVAS, NO decisionales

class EventDetector:
    @staticmethod
    def detect_explosion_start(frames):
        """
        Detecta cuándo empezó la explosión.
        
        Criteria:
        - exhaustion < 30% (FRESH)
        - ADX velocity > 0 (creciendo)
        - Candle size creciendo
        
        Returns:
            timestamp or None
        """
        for i in range(1, len(frames)):
            current = frames[i]
            previous = frames[i-1]
            
            if (current['indicators']['exhaustion_pct'] < 30 and
                current['indicators']['adx'] > previous['indicators']['adx'] and
                current['price']['high'] - current['price']['low'] > 
                previous['price']['high'] - previous['price']['low']):
                
                return current['timestamp']
        
        return None
    
    @staticmethod
    def detect_price_peak(frames):
        """
        Frame con precio más alto.
        """
        if not frames:
            return None
        
        max_frame = max(frames, key=lambda f: f['price']['high'])
        return max_frame['timestamp']
    
    @staticmethod
    def detect_bot_analysis_time(frames, entry_time):
        """
        Estima cuándo el bot "decidió" analizar.
        
        Típicamente ~2min antes de entry.
        """
        # Simple heuristic: entry_time - 2min
        return entry_time - timedelta(minutes=2)
    
    @staticmethod
    def calculate_timing_metrics(events, trade_data):
        """
        Calcula métricas de timing.
        
        Returns:
            dict - {
                'entry_timing': 'EARLY'/'PERFECT'/'LATE',
                'entry_delay_min': int,
                'exit_timing': 'EARLY'/'PERFECT'/'LATE',
                'exit_early_min': int
            }
        """
        metrics = {}
        
        explosion_start = events.get('explosion_detected')
        entry_time = trade_data['entry_time']
        price_peak = events.get('price_peak')
        exit_time = trade_data['exit_time']
        
        # Entry timing
        if explosion_start:
            delay = (entry_time - explosion_start).total_seconds() / 60
            metrics['entry_delay_min'] = int(delay)
            
            if delay < 5:
                metrics['entry_timing'] = 'EARLY'
            elif delay < 15:
                metrics['entry_timing'] = 'PERFECT'
            else:
                metrics['entry_timing'] = 'LATE'
        
        # Exit timing
        if price_peak:
            early = (price_peak - exit_time).total_seconds() / 60
            metrics['exit_early_min'] = int(early)
            
            if early > 10:
                metrics['exit_timing'] = 'EARLY'
            elif early > -5:
                metrics['exit_timing'] = 'PERFECT'
            else:
                metrics['exit_timing'] = 'LATE'
        
        return metrics

class TradeMovie:
    def __init__(self, trade_data, frames, velocities, accelerations, events):
        self.trade_data = trade_data
        self.frames = frames
        self.velocities = velocities
        self.accelerations = accelerations
        self.events = events
    
    def generate_json(self):
        """
        Genera JSON completo con toda la info.
        """
        return {
            'meta': self._generate_meta(),
            'timeline': self._generate_timeline(),
            'frames': self._generate_frames(),
            'velocities': self.velocities,
            'accelerations': self.accelerations,
            'analysis': self._generate_analysis()
        }
    
    def _generate_meta(self):
        """Meta info del trade."""
        return {
            'trade_id': self.trade_data['trade_id'],
            'symbol': self.trade_data['symbol'],
            'outcome': self.trade_data['outcome'],
            'pnl_pct': self.trade_data['pnl_pct'],
            'pnl_usd': self.trade_data.get('pnl_usd', 0),
            'entry_time': self.trade_data['entry_time'].isoformat(),
            'exit_time': self.trade_data['exit_time'].isoformat(),
            'duration_min': int((self.trade_data['exit_time'] - 
                                self.trade_data['entry_time']).total_seconds() / 60),
            'direction': self.trade_data['direction']
        }
    
    def _generate_timeline(self):
        """Timeline con eventos marcados."""
        return {
            'recording_start': self.frames[0]['timestamp'].isoformat(),
            'explosion_detected': self.events.get('explosion_detected', '').isoformat() if self.events.get('explosion_detected') else None,
            'bot_analyzed': self.events.get('bot_analyzed', '').isoformat() if self.events.get('bot_analyzed') else None,
            'entry': self.trade_data['entry_time'].isoformat(),
            'price_peak': self.events.get('price_peak', '').isoformat() if self.events.get('price_peak') else None,
            'exit': self.trade_data['exit_time'].isoformat(),
            'recording_end': self.frames[-1]['timestamp'].isoformat()
        }
    
    def _generate_frames(self):
        """
        Frames con relative_min agregado.
        """
        entry_time = self.trade_data['entry_time']
        
        formatted_frames = []
        for frame in self.frames:
            relative_min = int((frame['timestamp'] - entry_time).total_seconds() / 60)
            
            formatted_frames.append({
                'timestamp': frame['timestamp'].isoformat(),
                'relative_min': relative_min,
                'price': frame['price'],
                'indicators': frame['indicators']
            })
        
        return formatted_frames
    
    def _generate_analysis(self):
        """
        Análisis crítico: binary decisions vs time ladder.
        """
        analysis = {}
        
        # Timing metrics
        timing = EventDetector.calculate_timing_metrics(self.events, self.trade_data)
        analysis.update(timing)
        
        # Binary decisions analysis
        analysis['binary_decisions'] = self._analyze_binary_decisions()
        
        # Time Ladder verdict
        analysis['time_ladder_would_say'] = self._time_ladder_verdict()
        
        # Root cause
        analysis['root_cause'] = self._identify_root_cause()
        
        return analysis
    
    def _analyze_binary_decisions(self):
        """
        Identifica decisiones binarias que se tomaron.
        """
        decisions = []
        
        # Entry frame
        entry_frame = self._get_frame_at_time(self.trade_data['entry_time'])
        if not entry_frame:
            return decisions
        
        exh = entry_frame['indicators']['exhaustion_pct']
        room = entry_frame['indicators']['room_pct']
        grid = entry_frame['indicators'].get('grid_position', 50)
        
        # Exhaustion threshold
        if exh < 85:
            exh_vel = self.velocities['exhaustion']['value']
            decisions.append(
                f"Exhaustion {exh:.0f}% < 85% threshold → APPROVED "
                f"(pero velocity +{exh_vel:.1f} {'era CRÍTICA' if exh_vel > 5 else 'era alta'})"
            )
        
        # Room threshold
        if room > 0.5:
            room_vel = self.velocities['room']['value']
            decisions.append(
                f"Room {room:.1f}% > 0.5% threshold → APPROVED "
                f"(pero velocity {room_vel:.2f} {'cerrando rápido' if room_vel < -0.1 else 'OK'})"
            )
        
        # Grid threshold
        if grid < 75:
            grid_vel = self.velocities.get('grid_position', {}).get('value', 0)
            decisions.append(
                f"Grid {grid:.0f}% < 75% threshold → APPROVED "
                f"(pero velocity +{grid_vel:.1f} acercándose)" if grid_vel > 0 else
                f"Grid {grid:.0f}% < 75% threshold → APPROVED (velocity OK)"
            )
        
        return decisions
    
    def _time_ladder_verdict(self):
        """
        Qué diría Time Ladder viendo las trayectorias.
        """
        exh_vel = self.velocities['exhaustion']['value']
        room_vel = self.velocities['room']['value']
        adx_vel = self.velocities['adx']['value']
        
        # Criterios Time Ladder
        red_flags = []
        
        if exh_vel > 5:
            red_flags.append("Exhaustion quemándose brutal")
        if room_vel < -0.2:
            red_flags.append("Room cerrando rápido")
        if adx_vel < -1:
            red_flags.append("ADX cayendo")
        
        if len(red_flags) >= 2:
            return f"WAIT - {', '.join(red_flags)}"
        elif len(red_flags) == 1:
            return f"CAUTION - {red_flags[0]}"
        else:
            return "ENTER - Trayectorias sostenibles"
    
    def _identify_root_cause(self):
        """
        Identifica causa raíz del outcome.
        """
        if self.trade_data['outcome'] == 'WIN':
            return "N/A (trade exitoso)"
        
        # Para losses, analizar
        exh_vel = self.velocities['exhaustion']['value']
        room_vel = self.velocities['room']['value']
        
        if exh_vel > 5 and room_vel < -0.2:
            return "BINARY_THINKING - Thresholds aprobaron pero trayectorias insostenibles"
        elif self.analysis['entry_timing'] == 'LATE':
            return "LATE_ENTRY - Entró después de la explosión"
        elif self.analysis['exit_timing'] == 'EARLY':
            return "EARLY_EXIT - Salió antes del pico"
        else:
            return "UNKNOWN - Requiere análisis manual"
    
    def _get_frame_at_time(self, target_time):
        """Helper para encontrar frame en timestamp específico."""
        for frame in self.frames:
            if abs((frame['timestamp'] - target_time).total_seconds()) < 60:
                return frame
        return None

import os
import json
from datetime import datetime, timedelta
class VARRecorder:
    def __init__(self, output_dir='trade_recordings'):
        self.output_dir = output_dir
        self.frame_recorder = FrameRecorder()
        self.movie_analyzer = MovieAnalyzer()
        self.event_detector = EventDetector()
        
        # Create output directory
        os.makedirs(output_dir, exist_ok=True)
    
    def on_trade_closed(self, trade_data):
        """
        Main entry point - called when trade closes.
        
        Args:
            trade_data: dict - {
                'trade_id': int,
                'symbol': str,
                'outcome': 'WIN'/'LOSS',
                'pnl_pct': float,
                'pnl_usd': float,
                'entry_time': datetime,
                'exit_time': datetime,
                'direction': 'BULLISH'/'BEARISH'
            }
        """
        try:
            print(f"\n🎬 VAR Recording trade #{trade_data['trade_id']} {trade_data['symbol']}...")
            
            # 1. Define timerange
            start_time = trade_data['entry_time'] - timedelta(minutes=30)
            end_time = trade_data['exit_time'] + timedelta(minutes=10)
            
            # 2. Fetch frames
            frames = self.frame_recorder.fetch_frames(
                trade_data['symbol'],
                start_time,
                end_time
            )
            
            if not frames:
                print(f"   ⚠️ No frames available for {trade_data['symbol']}")
                return
            
            print(f"   📹 Recorded {len(frames)} frames")
            
            # 3. Calculate velocities & accelerations
            analysis = self.movie_analyzer.analyze_frames(frames)
            
            # 4. Detect events
            events = {
                'explosion_detected': self.event_detector.detect_explosion_start(frames),
                'price_peak': self.event_detector.detect_price_peak(frames),
                'bot_analyzed': self.event_detector.detect_bot_analysis_time(
                    frames, trade_data['entry_time']
                )
            }
            
            # 5. Reconstruct movie
            movie = TradeMovie(
                trade_data,
                frames,
                analysis['velocities'],
                analysis['accelerations'],
                events
            )
            
            # 6. Generate JSON
            movie_json = movie.generate_json()
            
            # 7. Save to file
            filename = self._generate_filename(trade_data)
            filepath = os.path.join(self.output_dir, filename)
            
            with open(filepath, 'w', encoding='utf-8') as f:
                json.dump(movie_json, f, indent=2, ensure_ascii=False)
            
            print(f"   ✅ Saved: {filename}")
            
        except Exception as e:
            print(f"   ❌ VAR Error: {str(e)}")
            # NO crashear el backtest - solo logear
    
    def _generate_filename(self, trade_data):
        """
        Genera nombre de archivo descriptivo.
        
        Format: trade_XXX_SYMBOL_OUTCOME_DATE.json
        Example: trade_142_1000CAT_LOSS_20260103.json
        """
        trade_id = str(trade_data['trade_id']).zfill(3)
        symbol = trade_data['symbol'].replace('/', '').replace(':', '')
        outcome = trade_data['outcome']
        date = trade_data['exit_time'].strftime('%Y%m%d')
        
        return f"trade_{trade_id}_{symbol}_{outcome}_{date}.json"
