def _check_coherence(self, q5m, q1h, q4h, p5m, p1h, p4h):
        """
        v5.5: Time Ladder - Sin returns REJECT, solo penalties graduales.
        Retorna (coherence, verdict, confidence) donde confidence = score penalty.
        """
        excellent = ['EXCELLENT', 'GOOD']
        poor = ['POOR', 'BAD', 'DYING']
        explosion = ['EXPLOSION_NASCENT', 'EXPLOSION_SUSTAINED', 'EXPLOSION_CLIMAX']
        dying = ['EXPLOSION_FADING', 'FORCE_DYING', 'FORCE_WEAK']
        
        # Empezar con confianza base
        confidence = 50
        coherence = 'MIXED'
        
        # === PENALTIES GRADUALES (Time Ladder) ===
        
        # 5M pobre = penalty fuerte (es el timeframe de entrada)
        if q5m in poor:
            confidence -= 35
            coherence = 'WEAK_5M'
        
        # 5M dying phase = penalty adicional
        if p5m in dying:
            confidence -= 15
            if coherence == 'MIXED':
                coherence = 'DYING_5M'
        
        # 1H pobre = penalty moderada
        if q1h in poor:
            confidence -= 20
            if 'WEAK' not in coherence:
                coherence = 'WEAK_1H'
        
        # 5M + 1H ambos pobres = penalty agravada
        if q5m in poor and q1h in poor:
            confidence -= 10  # Extra por doble debilidad
            coherence = 'ALL_WEAK'
        
        # === BONUSES POR ALINEACIÓN ===
        
        # 5M excelente = bonus
        if q5m in excellent:
            confidence += 20
            coherence = 'GOOD_5M'
        
        # Todos alineados = bonus grande
        if q5m in excellent and q1h in excellent:
            confidence += 25
            if q4h in excellent:
                coherence = 'PERFECT_ALIGNMENT'
                confidence = 95
            else:
                coherence = 'MOSTLY_ALIGNED'
        
        # 5M excelente pero 1H muriendo = penalty (caso JCT)
        if q5m in excellent and q1h in poor:
            confidence -= 30
            coherence = 'CONFLICTING'
        
        # === CASO ESPECIAL: Sin datos ===
        if q5m == 'UNKNOWN' and q1h == 'UNKNOWN':
            confidence = 40  # Penalizar por falta de datos
            coherence = 'NO_MOVIE_DATA'
        elif q5m == 'UNKNOWN' or q1h == 'UNKNOWN':
            # Uno sin datos - usar el que sí tenemos
            if q5m in poor:
                confidence -= 30
                coherence = 'WEAK_5M_PARTIAL'
            elif q1h in poor:
                confidence -= 15
                coherence = 'WEAK_1H_PARTIAL'
            else:
                coherence = 'PARTIAL_DATA'
        
        # Limitar confianza entre 5 y 95
        confidence = max(5, min(95, confidence))
        
        # Determinar verdict basado en confianza (NO bloqueo duro)
        if confidence >= 70:
            verdict = 'STRONG_APPROVE'
        elif confidence >= 50:
            verdict = 'APPROVE'
        elif confidence >= 30:
            verdict = 'CAUTION'
        else:
            verdict = 'WEAK'  # NO es REJECT, sigue el flujo
        
        return coherence, verdict, confidence
