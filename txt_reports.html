import pandas as pd
import numpy as np
from brain_config import GRID_LOOKBACK, GRID_COMPRESSION_BASE, GRID_COMPRESSION_MULTIPLIERS
from brain_calculator import calculate_atr
class MarketGridNavigator:
    def navigate(self, symbol, df_5m, ladder):
        if df_5m is None or len(df_5m) < 30:
            return {'decision': 'WAIT', 'score': -100, 'reasons': ['No Data'], 'grid_data': {}}
        
        # 1. Grid
        grid = self._build_grid(df_5m)
        vol_class = self._classify_volatility(df_5m, grid['atr_current'])
        grid['vol_class'] = vol_class
        grid['threshold'] = GRID_COMPRESSION_BASE * GRID_COMPRESSION_MULTIPLIERS.get(vol_class, 1.0)
        
        # 2. Datos
        pointer = self._calc_pointer(df_5m, ladder)
        force = self._get_force(ladder)  # FIX #2
        
        # 3. Decisión
        return self._decide(grid, pointer, force, df_5m)
    
    def _build_grid(self, df):
        ceiling = df['high'].rolling(GRID_LOOKBACK).max().iloc[-1]
        floor = df['low'].rolling(GRID_LOOKBACK).min().iloc[-1]
        atr_data = calculate_atr(df, period=14)
        atr = atr_data.get('atr_current', 0)
        width_atr = (ceiling - floor) / atr if atr > 0 else 0
        return {'ceiling': ceiling, 'floor': floor, 'width_atr': width_atr, 'atr_current': atr}
    def _classify_volatility(self, df, atr_current):
        closes = df['close'].tail(5)
        avg_price = closes.mean()
        avg_atr_pct = (atr_current / avg_price) * 100 if avg_price > 0 else 0
        if avg_atr_pct > 1.0: return 'high_vol'
        if avg_atr_pct < 0.3: return 'low_vol'
        return 'medium_vol'
    def _calc_pointer(self, df, ladder):
        closes = df['close'].values
        velocity = closes[-1] - closes[-3] if len(closes) > 3 else 0
        mom_color = ladder.get('15m', {}).get('momentum', {}).get('color_current', 'neutral')
        direction = "NEUTRAL"
        if mom_color in ['green', 'lime']: direction = "UP"
        elif mom_color in ['red', 'maroon']: direction = "DOWN"
        return {'direction': direction, 'velocity': velocity}
    def _get_force(self, ladder):
        # FIX #2: Extraer fuerza internamente
        return {
            'adx': ladder.get('15m', {}).get('adx', {}).get('adx_current', 0),
            'mom_color': ladder.get('15m', {}).get('momentum', {}).get('color_current', 'neutral')
        }
    def _decide(self, grid, pointer, force, df):
        score = 0
        reasons = []
        current_price = df['close'].iloc[-1]
        
        # 1. PENALTY: Compresión (FIX #1: NO RETURN BINARIO)
        if grid['width_atr'] < grid['threshold']:
            score -= 100
            reasons.append(f"⛔ LATERAL ({grid['width_atr']:.1f} ATRs)")
        
        # 2. PENALTY: Sin fuerza
        if force['adx'] < 22:
            score -= 50
            reasons.append(f"⚠️ ADX débil ({force['adx']:.1f})")
        
        # 3. SCORING: Posición (FIX #4: Scoring aumentado)
        denominator = (grid['ceiling'] - grid['floor']) or 0.0001
        position = (current_price - grid['floor']) / denominator
        
        # Caso A: Breakout/Breakdown
        if position > 0.85 and pointer['direction'] == 'UP':
            score += 50
            reasons.append("🚀 Breakout techo")
            # Validar momentum
            if force['mom_color'] not in ['green', 'lime']:
                score -= 40
                reasons.append("⚠️ Momentum contrario")
        
        elif position < 0.15 and pointer['direction'] == 'DOWN':
            score += 50
            reasons.append("🚀 Breakdown piso")
            if force['mom_color'] not in ['red', 'maroon']:
                score -= 40
                reasons.append("⚠️ Momentum contrario")
        
        # Caso B: Zona Neutral
        elif 0.3 <= position <= 0.7:
            score += 25
            reasons.append("✅ Zona neutral")
        
        # Caso C: Contra Estructura
        elif position > 0.7 and pointer['direction'] == 'DOWN':
            score -= 50
            reasons.append("⚠️ SHORT cerca techo")
        
        elif position < 0.3 and pointer['direction'] == 'UP':
            score -= 50
            reasons.append("⚠️ LONG cerca piso")
        
        # FIX #5: Threshold estricto
        decision = "GO" if score >= 35 else "WAIT"
        return {'decision': decision, 'score': score, 'reasons': reasons, 'grid_data': grid}
