pythondef analyze_microstructure(df):
    # Buying/Selling Pressure
    buying_pressure = (df['close'] - df['low']) / (df['high'] - df['low'])
    selling_pressure = (df['high'] - df['close']) / (df['high'] - df['low'])
    
    # Delta Volume
    delta = buying_pressure.mean() - selling_pressure.mean()
    
    return {
        'delta': delta,
        'trend': 'BULLISH' if delta > 0.2 else 'BEARISH',
        'strength': abs(delta) * 100
    }

pythondef detect_manipulation(df):
    signals = []
    
    # Volume spike anormal
    if df['volume'].iloc[-1] > df['volume'].mean() * 10:
        signals.append('VOLUME_SPIKE_10X')
    
    # Precio sin volumen
    price_change = df['close'].pct_change().iloc[-1]
    if abs(price_change) > 0.05 and df['volume'].iloc[-1] < df['volume'].mean() * 0.5:
        signals.append('PRICE_WITHOUT_VOLUME')
    
    # Wicks largos
    wick_ratio = (df['high'] - df['close']) / (df['close'] - df['open'])
    if wick_ratio > 3:
        signals.append('LONG_WICKS')
    
    return {
        'is_manipulated': len(signals) >= 2,
        'signals': signals
    }
