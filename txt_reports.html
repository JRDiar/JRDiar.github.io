    def check_rebote_conditions(self, current_analysis, watch_data):
        """Valida condiciones de rebote con datos frescos"""
        current_price = current_analysis['analysis_5m']['current_price']
        zona_objetivo = watch_data.get('zona_objetivo', 0)
        
        # 1. Precio en zona (±1.5%)
        dist_to_zone = abs(current_price - zona_objetivo) / zona_objetivo * 100
        if dist_to_zone > 1.5:
            return {'ready': False, 'reason': f"Price far from zone ({dist_to_zone:.1f}%)"}
            
        # 2. ADX 15M Creciendo
        a15m = current_analysis.get('analysis_15m', {})
        adx_trend = a15m.get('adx', {}).get('adx_trend', 'estable')
        if adx_trend != 'creciendo':
             return {'ready': False, 'reason': f"ADX 15M not growing ({adx_trend})"}
             
        # 3. Momentum 15M Confirmado
        mom_15m = a15m.get('momentum', {})
        if mom_15m.get('color_current') not in ['green', 'lime'] or mom_15m.get('consecutive_count', 0) < 2:
