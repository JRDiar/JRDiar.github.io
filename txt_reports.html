"""
BRAIN TREND EXHAUSTION v3
=========================
Mide agotamiento de tendencia + distancia a S/R
"""
import pandas as pd
from exchange_connection import fetch_ohlcv
from brain_calculator import calculate_adx_with_memory, calculate_momentum_with_memory
def analyze_trend_exhaustion(symbol, direction, df_5m=None, df_15m=None):
    """
    Analiza exhaustión combinando 5M (crítico) + 15M (contexto).
    """
    # Fetch si no se proporcionan
    if df_5m is None:
        ohlcv = fetch_ohlcv(symbol, '5m', 100)
        if not ohlcv:
            return {'exhaustion_pct': 50, 'room_pct': 2, 'penalty': 0, 'phase': 'UNKNOWN'}
        df_5m = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    
    current_price = float(df_5m['close'].iloc[-1])
    
    # S/R de últimas 20 velas
    high_20 = df_5m['high'].rolling(20).max().iloc[-1]
    low_20 = df_5m['low'].rolling(20).min().iloc[-1]
    range_20 = high_20 - low_20
    
    # Indicadores
    adx_data = calculate_adx_with_memory(df_5m, period=14, lookback=10)
    mom_data = calculate_momentum_with_memory(df_5m, lookback=10)
    
    adx = adx_data.get('adx_current', 25)
    adx_vel = adx_data.get('adx_velocity', 0)
    mom_age = mom_data.get('consecutive_count', 0)
    
    exhaustion_score = 0
    details = {}
    
    # === 1. DISTANCIA A S/R (MÁS IMPORTANTE) ===
    if direction == 'BULLISH':
        dist_to_resistance = ((high_20 - current_price) / current_price) * 100
        dist_from_support = ((current_price - low_20) / low_20) * 100
        room_pct = dist_to_resistance
        moved_pct = dist_from_support
    else:
        dist_to_support = ((current_price - low_20) / current_price) * 100
        dist_from_resistance = ((high_20 - current_price) / high_20) * 100
        room_pct = dist_to_support
        moved_pct = dist_from_resistance
    
    details['room_pct'] = room_pct
    details['moved_pct'] = moved_pct
    
    # Penalty por poco espacio
    if room_pct < 0.3:
        exhaustion_score += 40  # Casi bloqueado
    elif room_pct < 0.5:
        exhaustion_score += 30
    elif room_pct < 1.0:
        exhaustion_score += 20
    elif room_pct < 2.0:
        exhaustion_score += 10
    
    # === 2. RATIO MOVED/REMAINING ===
    if room_pct > 0:
        ratio = moved_pct / room_pct
        if ratio > 4:
            exhaustion_score += 25
        elif ratio > 2:
            exhaustion_score += 15
        elif ratio > 1:
            exhaustion_score += 5
        details['move_ratio'] = ratio
    
    # === 3. ADX LIFECYCLE ===
    if adx > 45 and adx_vel < 0:
        exhaustion_score += 15
        details['adx_phase'] = 'PEAKED'
    elif adx > 35:
        exhaustion_score += 8
        details['adx_phase'] = 'STRONG'
    elif adx < 20:
        details['adx_phase'] = 'WEAK'
    else:
        details['adx_phase'] = 'DEVELOPING'
    
    # === 4. MOMENTUM AGE ===
    if mom_age >= 8:
        exhaustion_score += 15
    elif mom_age >= 5:
        exhaustion_score += 8
    
    # === RESULTADO ===
    exhaustion_pct = min(100, exhaustion_score)
    
    if exhaustion_pct >= 60:
        phase = 'EXHAUSTED'
        penalty = -35
    elif exhaustion_pct >= 40:
        phase = 'MATURE'
        penalty = -20
    elif exhaustion_pct >= 20:
        phase = 'DEVELOPING'
        penalty = -10
    else:
        phase = 'FRESH'
        penalty = 0
    
    return {
        'exhaustion_pct': exhaustion_pct,
        'room_pct': room_pct,
        'moved_pct': moved_pct,
        'phase': phase,
        'penalty': penalty,
        'details': details
    }
def get_dynamic_sl_distance(exhaustion_result, base_sl_pct=2.0):
    """
    Calcula SL dinámico basado en room disponible.
    
    Si hay mucho room → SL más amplio (dejar desarrollar)
    Si poco room → SL más ajustado (proteger)
    """
    room = exhaustion_result.get('room_pct', 2.0)
    
    if room > 3.0:
        # Mucho espacio → SL amplio, dejar correr
        return min(base_sl_pct * 1.5, 3.0)
    elif room > 1.5:
        # Espacio moderado
        return base_sl_pct
    elif room > 0.5:
        # Poco espacio → SL ajustado
        return base_sl_pct * 0.7
    else:
        # Casi bloqueado → SL muy ajustado
        return base_sl_pct * 0.5
def get_trailing_distance(exhaustion_result, pnl_pct, base_trail=2.0):
    """
    Trailing dinámico: se aprieta cerca de S/R.
    """
    room = exhaustion_result.get('room_pct', 2.0)
    
    # Base según profit
    if pnl_pct > 3.0:
        trail = 1.0
    elif pnl_pct > 1.5:
        trail = 1.5
    elif pnl_pct > 0.5:
        trail = 2.0
    else:
        trail = 2.5
    
    # Ajustar según cercanía a S/R
    if room < 0.5:
        trail = min(trail, 0.5)  # Muy apretado cerca de S/R
    elif room < 1.0:
        trail = min(trail, 1.0)
    
    return trail
