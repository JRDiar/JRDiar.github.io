    def scan_cycle(self) -> Optional[dict]:
        """
        Ejecuta un ciclo de escaneo.
        
        Returns:
            dict: Setup Priority para entrar, o None si no hay
        """
        self.cycle_count += 1
        start_time = time.time()
        
        # Limpiar colas
        self.priority_queue.clear()
        self.developing_queue.clear()
        self._clean_rejected_cache()
        
        # ================================================================
        # FASE 1: FULL SCAN (solo cada 30s)
        # ================================================================
        
        should_full_scan = (
            self.last_full_scan is None or
            (datetime.now() - self.last_full_scan).total_seconds() >= self.full_scan_interval
        )
        
        if should_full_scan:
            # Full scan puede retornar entry inmediato
            immediate_entry = self._do_full_scan()
            
            if immediate_entry:
                print(f"\n  >>> ENTRADA INMEDIATA DESDE FULL SCAN <<<")
                return immediate_entry  # Retornar al main.py para ejecutar
        else:
            secs_since = (datetime.now() - self.last_full_scan).total_seconds()
            print(f"\n  [CYCLE {self.cycle_count}] Skip full scan ({secs_since:.0f}s < {self.full_scan_interval}s)")
        
        # ================================================================
        # FASE 2: RE-ANALIZAR WATCHLIST (cada ciclo)
        # ================================================================
        
        print(f"\n  [WATCHLIST] Re-analizando {len(self.watchlist)} monedas...")
        
        priority_found = None
        symbols_to_remove = []
        
        for symbol, data in self.watchlist.items():
            try:
                # >>> USAR PHYSICS (no holistic) para evitar conflicto de thresholds
                analysis = self._analyze_physics(symbol)
                
                if analysis is None:
                    # Fallback a holistic si physics falla
                    analysis = self._analyze_symbol_quick(symbol)
                
                if analysis is None:
                    continue
                
                # Actualizar datos en watchlist
                data['last_analysis'] = analysis
                data['checks'] += 1
                
                # Clasificar resultado
                classification = analysis['classification']
                
                if classification == 'PRIORITY':
                    # Setup perfecto encontrado
                    self.priority_queue.append(analysis)
                    
                    # Confirmar con multi-timeframe
                    if self._confirm_with_mtf(symbol, analysis):
                        priority_found = analysis
                        print(f"    ⭐ {symbol} → PRIORITY + MTF OK → ENTRAR")
                    else:
                        print(f"    ⭐ {symbol} → PRIORITY (sin MTF)")
                        
                elif classification == 'DEVELOPING':
                    self.developing_queue.append(analysis)
                    print(f"    🔄 {symbol} → DEVELOPING ({data['checks']} checks)")
                    
                elif classification == 'REJECTED':
                    symbols_to_remove.append(symbol)
                    self.rejected_cache[symbol] = datetime.now()
                    print(f"    ❌ {symbol} → REJECTED (cooldown {self.reject_cooldown}min)")
                    
            except Exception as e:
                logger.debug(f"Error analizando {symbol}: {e}")
        
        # Remover rejected de watchlist
        for symbol in symbols_to_remove:
            del self.watchlist[symbol]
        
        # ================================================================
        # RESUMEN
        # ================================================================
        
        duration = time.time() - start_time
        
        print(f"\n  {'='*50}")
        print(f"  CYCLE {self.cycle_count} | {duration:.1f}s")
        print(f"  Watchlist: {len(self.watchlist)} | Priority: {len(self.priority_queue)} | Developing: {len(self.developing_queue)}")
        print(f"  {'='*50}")
        
        return priority_found

    def _sniper_loop(self):
        """Thread 2: Sniper (Foreground) - Revisa watchlist cada 10s."""
        logger.info("🎯 Sniper thread iniciado")
        while not self.stop_threads.is_set():
            try:
                with self.watchlist_lock:
                    watchlist_snapshot = dict(self.watchlist)
                
                if not watchlist_snapshot:
                    self.stop_threads.wait(timeout=self.SNIPER_CHECK_INTERVAL)
                    continue
                
                print(f"\n[{datetime.now().strftime('%H:%M:%S')}] 🎯 Sniper: Revisando {len(watchlist_snapshot)} monedas calientes...")
                priority_found = False
                
                for symbol, info in watchlist_snapshot.items():
                    if info.get('classification') not in ['PRIORITY', 'DEVELOPING']:
                        continue
                    
                    quick_check = self._quick_watchlist_check(symbol, info)
                    if quick_check['classification'] == 'PRIORITY':
                        print(f"  ✅ {symbol:12s} │ PRIORITY detectado")
                        priority_found = True
                        mtf_result = self._confirm_with_mtf(symbol, quick_check)
                        if mtf_result: # Simplificado para usar el retorno booleano o dict de confirm_with_mtf
                            print(f"  🎯 {symbol:12s} │ MTF confirmado → Señalizar")
                            self._signal_trade_opportunity(symbol, quick_check)

                if not priority_found:
                    print(f"  ⏳ Sin setups perfectos aún")
                self.stop_threads.wait(timeout=self.SNIPER_CHECK_INTERVAL)
            except Exception as e:
                logger.error(f"Error en sniper loop: {e}")
                self.stop_threads.wait(timeout=5)
        logger.info("🎯 Sniper thread detenido")

    def _quick_watchlist_check(self, symbol: str, cached_info: dict) -> dict:
        """Chequeo rápido sin fetch pesado."""
        try:
            ticker = self.connector.exchange.fetch_ticker(symbol)
            if not ticker: return {'classification': 'UNKNOWN'}
            
            price_change_5m = ticker.get('percentage', 0)
            if abs(price_change_5m) < 0.1:
                return {'classification': 'DEVELOPING', 'reason': 'Se lateralizó'}
            
            # Verificar dirección
            cached_direction = cached_info.get('last_analysis', {}).get('velocity_per_min', 0) # Usar last_analysis
            cached_dir_sign = 1 if cached_direction > 0 else -1
            current_dir_sign = 1 if price_change_5m > 0 else -1
            
            if cached_dir_sign != current_dir_sign:
                return {'classification': 'REJECTED', 'reason': 'Invirtió dirección'}
                
            velocity = cached_info.get('last_analysis', {}).get('velocity_per_min', 0)
            has_runway = cached_info.get('last_analysis', {}).get('has_runway', False)
            
            if abs(velocity) > 0 and has_runway:
                # Retornar una estructura que parezca analysis
                base_analysis = cached_info.get('last_analysis', {}).copy()
                base_analysis['classification'] = 'PRIORITY'
                base_analysis['_immediate_entry'] = True
                return base_analysis
            else:
                # Logging: qué le falta para ser PRIORITY
                missing = []
                if abs(velocity) <= 0: missing.append('sin velocity')
                if not has_runway: missing.append('sin runway')
                logger.debug(f"🔄 {symbol:12s} DEVELOPING - Falta: {', '.join(missing) if missing else 'verificando setup'}")
                return {'classification': 'DEVELOPING', 'reason': 'Falta runway o velocity'}
        except Exception as e:
            logger.error(f"Error quick check {symbol}: {e}")
            return {'classification': 'UNKNOWN'}

    def _signal_trade_opportunity(self, symbol: str, analysis: dict):
        """Señaliza trade a main.py."""
        opportunity = {'symbol': symbol, 'timestamp': datetime.now(), 'analysis': analysis}
        with self.watchlist_lock:
            if not hasattr(self, 'pending_opportunities'):
                self.pending_opportunities = []
            self.pending_opportunities.append(opportunity)
        logger.info(f"🎯 Oportunidad señalizada: {symbol}")

    def start_dual_speed(self):
        """Inicia threads."""
        self.stop_threads.clear()
        self.deep_scan_thread = threading.Thread(target=self._deep_scanner_loop, name="DeepScanner", daemon=True)
        self.deep_scan_thread.start()
        self.sniper_thread = threading.Thread(target=self._sniper_loop, name="Sniper", daemon=True)
        self.sniper_thread.start()
        logger.info("✅ Arquitectura dual-speed activada")

    def stop_dual_speed(self):
        """Detiene threads."""
        logger.info("Deteniendo threads...")
        self.stop_threads.set()
        if self.deep_scan_thread: self.deep_scan_thread.join(timeout=5)
        if self.sniper_thread: self.sniper_thread.join(timeout=5)
        logger.info("✅ Threads detenidos")

    # CONFIRMACIÓN MULTI-TIMEFRAME

    def _confirm_with_mtf(self, symbol: str, analysis: dict) -> bool:
        """
        Confirma Priority con Multi-Timeframe correcto.
        
        LÓGICA:
        - 1D = REFERENCIA (evitar correcciones de movimiento mayor)
        - 4H = DEFINE TENDENCIA (la dirección real, 8-12h de contexto)
        - 1H, 15M, 5M = Deben ALINEARSE con 4H
        
        Returns:
            bool: True si todo alineado correctamente
        """
        try:
            sym_short = symbol.replace('/USDT:USDT', '')
            
            # ============================================================
            # PASO 1: CONTEXTO DIARIO (1D) - ¿Estamos en tendencia o corrección?
            # ============================================================
            ohlcv_1d = self.connector.exchange.fetch_ohlcv(symbol, '1d', limit=100)
            
            if ohlcv_1d and len(ohlcv_1d) >= 7:
                closes_1d = [c[4] for c in ohlcv_1d]
                highs_1d = [c[2] for c in ohlcv_1d]
                lows_1d = [c[3] for c in ohlcv_1d]
                
                # Tendencia 1D: comparar semana actual vs anterior
                avg_week_old = np.mean(closes_1d[-14:-7]) if len(closes_1d) >= 14 else np.mean(closes_1d[:7])
                avg_week_new = np.mean(closes_1d[-7:])
                
                trend_1d = 'UP' if avg_week_new > avg_week_old else 'DOWN'
                
                # ¿Es una corrección? (precio cayó desde máximo reciente)
                recent_high = max(highs_1d[-7:])
                current_price = closes_1d[-1]
                correction_pct = (recent_high - current_price) / recent_high * 100
                
                is_correction = correction_pct > 5  # >5% de caída desde máximo
                
                if is_correction and trend_1d == 'DOWN':
                    print(f"    [MTF] {sym_short}: ❌ 1D en corrección ({correction_pct:.1f}% desde máximo)")
                    return False
            else:
                trend_1d = 'UNKNOWN'
                is_correction = False
            
            # ============================================================
            # PASO 2: TENDENCIA 4H - LA QUE DEFINE TODO
            # ============================================================
            ohlcv_4h = self.connector.exchange.fetch_ohlcv(symbol, '4h', limit=100)
            
            if not ohlcv_4h or len(ohlcv_4h) < 6:
                print(f"    [MTF] {sym_short}: ⚠️ Sin datos 4H")
                return False
            
            closes_4h = [c[4] for c in ohlcv_4h]
            
            # Tendencia 4H (últimas 8-12 horas = 2-3 velas de 4h)
            trend_4h = 'UP' if closes_4h[-1] > closes_4h[-3] else 'DOWN'
            
            # Si 4H es bajista, NO entrar long
            if trend_4h == 'DOWN':
                print(f"    [MTF] {sym_short}: ❌ 4H BAJISTA - no entrar long")
                return False
            
            # ============================================================
            # PASO 3: ALINEACIÓN 1H, 15M, 5M con 4H
            # ============================================================
            
            # 1H
            ohlcv_1h = self.connector.exchange.fetch_ohlcv(symbol, '1h', limit=100)
            if ohlcv_1h and len(ohlcv_1h) >= 4:
                closes_1h = [c[4] for c in ohlcv_1h]
                trend_1h = 'UP' if closes_1h[-1] > closes_1h[-3] else 'DOWN'
            else:
                trend_1h = trend_4h  # Asumir alineado
            
            # 15M
            ohlcv_15m = self.connector.exchange.fetch_ohlcv(symbol, '15m', limit=100)
            if ohlcv_15m and len(ohlcv_15m) >= 8:
                closes_15m = [c[4] for c in ohlcv_15m]
                trend_15m = 'UP' if closes_15m[-1] > closes_15m[-4] else 'DOWN'
            else:
                trend_15m = trend_4h
            
            # 5M (ya lo tenemos del análisis)
            structure = analysis.get('structure', {})
            trend_5m = 'UP' if structure.get('trend') == 'BULLISH' else 'DOWN' if structure.get('trend') == 'BEARISH' else 'UNKNOWN'
            
            # ============================================================
            # PASO 4: VERIFICAR ALINEACIÓN COMPLETA
            # ============================================================
            
            aligned_1h = (trend_1h == trend_4h)
            aligned_15m = (trend_15m == trend_4h)
            aligned_5m = (trend_5m == 'UP') if trend_4h == 'UP' else (trend_5m == 'DOWN')
            
            all_aligned = aligned_1h and aligned_15m and aligned_5m
            
            # Mostrar resultado
            status_1d = '✓' if trend_1d == 'UP' and not is_correction else '✗'
            status_4h = '✓' if trend_4h == 'UP' else '✗'

    def _get_pattern_stats(self, pattern_id):
        """Consulta base de datos de observaciones."""
        try:
            import sqlite3
            from pathlib import Path
            db_path = Path('data/pattern_memory.db')
            
            if not db_path.exists():
                 return {'sample_size': 0, 'win_rate': 0, 'expectancy': 0}

            conn = sqlite3.connect(db_path)
            cursor = conn.cursor()
            
            # Check table exists
            cursor.execute("SELECT name FROM sqlite_master WHERE type='table' AND name='observations'")
            if not cursor.fetchone():
                 conn.close()
                 return {'sample_size': 0, 'win_rate': 0, 'expectancy': 0}
            
            outcomes = cursor.execute("""
                SELECT outcome, profit_pct 
                FROM observations 
                WHERE pattern_id = ?
            """, (pattern_id,)).fetchall()
            
            conn.close()
            
            if len(outcomes) < 20:
                return {'sample_size': len(outcomes), 'win_rate': 0, 'expectancy': 0}
            
            wins = [o for o in outcomes if o[0] == 'WIN']
            win_rate = len(wins) / len(outcomes)
            
            # Avg Profit/Loss
            wins_pct = [o[1] for o in wins]
            losses_pct = [o[1] for o in outcomes if o[0] == 'LOSS']
            
            avg_profit = sum(wins_pct) / len(wins_pct) if wins_pct else 0
            avg_loss = abs(sum(losses_pct) / len(losses_pct)) if losses_pct else 0
            
            # EV = (Win% * AvgWin) - (Loss% * AvgLoss)
            expectancy = ((win_rate * avg_profit) - ((1-win_rate) * avg_loss))
            
            return {
                'sample_size': len(outcomes),
                'win_rate': win_rate,
                'expectancy': expectancy
            }
        except Exception as e:
            # print(f"DB Error: {e}")
            return {'sample_size': 0, 'win_rate': 0, 'expectancy': 0}
