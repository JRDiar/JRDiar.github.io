"""
BRAIN TREND EXHAUSTION v1.0
============================
Joel: "Medir cuánto subió, cuánto le queda, dónde está el S/R"
IMPORTANTE: Este módulo NO aplica su propia penalty.
Devuelve MULTIPLICADORES para que brain_analyzer ajuste penalties existentes.
"""
import pandas as pd
from exchange_connection import fetch_ohlcv
from brain_calculator import calculate_adx_with_memory, calculate_momentum_with_memory
def analyze_trend_exhaustion(symbol, direction, df_5m=None, df_15m=None):
    """
    Analiza exhaustión y retorna MULTIPLICADORES para adjustar penalties.
    
    Returns:
        dict con:
        - exhaustion_pct (0-100)
        - room_pct (espacio hasta S/R)
        - phase (FRESH/DEVELOPING/MATURE/EXHAUSTED)
        - multipliers: {adx, momentum, late_entry}  <-- NUEVO
    """
    # Fetch si no se proporciona
    if df_5m is None:
        ohlcv = fetch_ohlcv(symbol, '5m', 100)
        if not ohlcv:
            return _default_result()
        df_5m = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    
    current_price = float(df_5m['close'].iloc[-1])
    
    # === CALCULAR S/R DINÁMICO ===
    high_50 = df_5m['high'].tail(50).max()
    low_50 = df_5m['low'].tail(50).min()
    
    exhaustion_score = 0
    details = {}
    
    # === 1. DISTANCIA A S/R (30 pts) ===
    if direction == 'BULLISH':
        dist_to_resistance = ((high_50 - current_price) / current_price) * 100
        dist_from_support = ((current_price - low_50) / low_50) * 100
        room_pct = dist_to_resistance
        moved_pct = dist_from_support
        details['target'] = high_50
    else:
        dist_to_support = ((current_price - low_50) / current_price) * 100
        dist_from_resistance = ((high_50 - current_price) / high_50) * 100
        room_pct = dist_to_support
        moved_pct = dist_from_resistance
        details['target'] = low_50
    
    details['room_pct'] = room_pct
    details['moved_pct'] = moved_pct
    
    # Penalty por poco espacio
    if room_pct < 0.3:
        exhaustion_score += 30
    elif room_pct < 0.8:
        exhaustion_score += 20
    elif room_pct < 1.5:
        exhaustion_score += 10
    elif room_pct < 3.0:
        exhaustion_score += 5
    
    # === 2. RATIO MOVED/REMAINING (25 pts) ===
    if room_pct > 0:
        ratio = moved_pct / room_pct
        details['ratio'] = ratio
        if ratio > 5:
            exhaustion_score += 25
        elif ratio > 3:
            exhaustion_score += 18
        elif ratio > 1.5:
            exhaustion_score += 10
        elif ratio > 0.5:
            exhaustion_score += 5
    
    # === 3. ADX LIFECYCLE (25 pts) ===
    try:
        adx_data = calculate_adx_with_memory(df_5m, period=14, lookback=10)
        adx_current = adx_data.get('adx_current', 25)
        adx_velocity = adx_data.get('adx_velocity', 0)
        details['adx'] = adx_current
        details['adx_vel'] = adx_velocity
        
        if adx_current > 50:
            exhaustion_score += 25
        elif adx_current > 45 and adx_velocity < 0:
            exhaustion_score += 20
        elif adx_current > 35:
            exhaustion_score += 10
        elif adx_current > 25:
            exhaustion_score += 5
    except:
        pass
    
    # === 4. MOMENTUM AGE (20 pts) ===
    try:
        mom_data = calculate_momentum_with_memory(df_5m, lookback=10)
        mom_age = mom_data.get('consecutive_count', 0)
        details['mom_age'] = mom_age
        
        if mom_age >= 10:
            exhaustion_score += 20
        elif mom_age >= 7:
            exhaustion_score += 15
        elif mom_age >= 4:
            exhaustion_score += 8
        elif mom_age >= 2:
            exhaustion_score += 3
    except:
        pass
    
    # === CALCULAR FASE Y MULTIPLICADORES ===
    exhaustion_pct = min(100, exhaustion_score)
    
    if exhaustion_pct >= 70:
        phase = 'EXHAUSTED'
    elif exhaustion_pct >= 50:
        phase = 'MATURE'
    elif exhaustion_pct >= 30:
        phase = 'DEVELOPING'
    else:
        phase = 'FRESH'
    
    # >>> MULTIPLICADORES según fase + room <<<
    multipliers = _calculate_multipliers(exhaustion_pct, room_pct, phase)
    
    return {
        'exhaustion_pct': exhaustion_pct,
        'room_pct': room_pct,
        'moved_pct': moved_pct,
        'phase': phase,
        'multipliers': multipliers,
        'details': details
    }
def _calculate_multipliers(exhaustion_pct, room_pct, phase):
    """
    Calcula multiplicadores para ajustar penalties existentes.
    
    FRESH + room alto = permite entrada temprana (reduce penalties)
    EXHAUSTED = agrava penalties
    """
    # Defaults
    adx_mult = 1.0
    momentum_mult = 1.0
    late_entry_mult = 1.0
    lateral_mult = 1.0
    
    # FRESH + room excelente = OPORTUNIDAD TEMPRANA
    if exhaustion_pct < 30 and room_pct > 3.0:
        adx_mult = 0.2        # ADX bajo ESPERADO en fase temprana
        momentum_mult = 0.1   # Momentum joven NORMAL
        late_entry_mult = 0.0 # No es late entry
        lateral_mult = 0.3    # Lateral puede ser acumulación
    
    # FRESH + room bueno = entrada con precaución
    elif exhaustion_pct < 30 and room_pct > 2.0:
        adx_mult = 0.4
        momentum_mult = 0.3
        late_entry_mult = 0.2
        lateral_mult = 0.5
    
    # FRESH pero poco room = cuidado
    elif exhaustion_pct < 30:
        adx_mult = 0.7
        momentum_mult = 0.6
        late_entry_mult = 0.5
        lateral_mult = 0.8
    
    # DEVELOPING = normal
    elif exhaustion_pct < 50:
        # Mantener en 1.0
        pass
    
    # MATURE = más estricto
    elif exhaustion_pct < 70:
        adx_mult = 1.3
        momentum_mult = 1.3
        late_entry_mult = 1.2
        lateral_mult = 1.2
    
    # EXHAUSTED = muy estricto
    else:
        adx_mult = 2.0
        momentum_mult = 2.0
        late_entry_mult = 1.5
        lateral_mult = 1.5
    
    return {
        'adx': adx_mult,
        'momentum': momentum_mult,
        'late_entry': late_entry_mult,
        'lateral': lateral_mult
    }
def _default_result():
    """Resultado por defecto si falla el análisis."""
    return {
        'exhaustion_pct': 50,
        'room_pct': 2.0,
        'moved_pct': 0,
        'phase': 'UNKNOWN',
        'multipliers': {'adx': 1.0, 'momentum': 1.0, 'late_entry': 1.0, 'lateral': 1.0},
        'details': {}
    }
def print_exhaustion_analysis(result, symbol, direction):
    """Imprime análisis visual."""
    m = result.get('multipliers', {})
    print(f"\n{'='*60}")
    print(f"📉 EXHAUSTION: {symbol} ({direction})")
    print(f"   Exhaustion: {result['exhaustion_pct']:.0f}% | Phase: {result['phase']}")
    print(f"   Room: {result['room_pct']:.2f}% | Moved: {result['moved_pct']:.2f}%")
    print(f"   Multipliers: ADX {m.get('adx', 1):.1f} | Mom {m.get('momentum', 1):.1f} | Late {m.get('late_entry', 1):.1f}")
    print(f"{'='*60}")
