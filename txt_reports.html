def generate_15m_warnings(data_15m: dict, direction_5m: str) -> dict:
    """
    15M revisa contexto macro y genera warnings LEVES para 5M.
    No decide, solo advierte.
    
    Args:
        data_15m: Return de analyze_15m()
        direction_5m: Direction decidida por 5M
        
    Returns:
        {
            'total_penalty': int (0-105 máximo teórico),
            'warnings': list[str],
            'should_hard_veto': bool (solo casos extremos)
        }
    """
    from brain_config import (
        ENABLE_15M_WARNINGS,
        WARNING_15M_ADX_DECAY,
        WARNING_15M_MOMENTUM_OLD,
        WARNING_15M_GRID_HIGH,
        WARNING_15M_DIRECTION,
        WARNING_15M_LATERAL,
        WARNING_15M_NO_STRUCTURE,
        WARNING_15M_ADX_WEAK,
        WARNING_15M_SPREAD_MIN
    )
    
    if not ENABLE_15M_WARNINGS:
        return {'total_penalty': 0, 'warnings': [], 'should_hard_veto': False}
    
    if not data_15m:
        return {'total_penalty': 0, 'warnings': [], 'should_hard_veto': False}
    
    penalties = []
    warnings = []
    
    # Extraer datos de 15M
    adx_15m = data_15m.get('adx', {})
    adx_current = adx_15m.get('adx_current', 0)
    adx_trend = adx_15m.get('adx_trend', 'estable')
    spread_15m = adx_15m.get('spread', 0)
    
    momentum_15m = data_15m.get('momentum', {})
    mom_age = momentum_15m.get('consecutive_count', 0)
    
    direction_15m = data_15m.get('direction', 'LATERAL')
    lateral_penalty_15m = data_15m.get('lateral_penalty', 0)
    
    # Grid position (si disponible)
    grid_15m = data_15m.get('grid', {})
    zigzag_15m = data_15m.get('zigzag_granular', {})
    
    # ================================================
    # HARD VETO: Solo para spread muy bajo
    # ================================================
    if abs(spread_15m) < WARNING_15M_SPREAD_MIN:
        return {
            'total_penalty': 0,
            'warnings': [f'Spread 15M {spread_15m:.1f} < {WARNING_15M_SPREAD_MIN} (inoperable)'],
            'should_hard_veto': True
        }
    
    # ================================================
    # WARNING 1: ADX decreciendo
    # ================================================
    if adx_trend == 'decreciendo':
        penalties.append(WARNING_15M_ADX_DECAY)
        warnings.append(f'ADX 15M decreciendo ({adx_current:.1f})')
    
    # ================================================
    # WARNING 2: Momentum agotado (>12 velas)
    # ================================================
    if mom_age > 12:
        penalties.append(WARNING_15M_MOMENTUM_OLD)
        warnings.append(f'Momentum 15M agotado ({mom_age}v)')
    
    # ================================================
    # WARNING 3: Grid alto (>70%)
    # Requiere que grid_position esté disponible
    # ================================================
    grid_diagonal = grid_15m.get('diagonal', {})
    if grid_diagonal.get('valid'):
        # Calcular posición aproximada (no siempre disponible)
        pass  # Se puede expandir si grid_position está disponible
    
    # ================================================
    # WARNING 4: Dirección contradice
    # ================================================
    if direction_15m != direction_5m and direction_15m != 'LATERAL':
        penalties.append(WARNING_15M_DIRECTION)
        warnings.append(f'15M dice {direction_15m}, 5M dice {direction_5m}')
    
    # ================================================
    # WARNING 5: LATERAL detectado (con penalty alta)
    # ================================================
    if lateral_penalty_15m > 30:
        penalties.append(WARNING_15M_LATERAL)
        warnings.append(f'15M lateral penalty -{lateral_penalty_15m}')
    
    # ================================================
    # WARNING 6: Sin estructura (choppy)
    # ================================================
    structure = zigzag_15m.get('current_trend', 'UNKNOWN')
    if structure == 'LATERAL':
        penalties.append(WARNING_15M_NO_STRUCTURE)
        warnings.append('15M sin estructura clara')
    
    # ================================================
    # WARNING 7: ADX muy débil (<20)
    # ================================================
    if adx_current < 20:
        penalties.append(WARNING_15M_ADX_WEAK)
        warnings.append(f'ADX 15M débil ({adx_current:.1f})')
    
    total_penalty = sum(penalties)
    
    if total_penalty > 0:
        print(f"   ⚠️ 15M WARNINGS: {', '.join(warnings)} → penalty total -{total_penalty}")
    else:
        print(f"   ✅ 15M: Sin warnings (contexto macro OK)")
    
    return {
        'total_penalty': total_penalty,
        'warnings': warnings,
        'should_hard_veto': False
    }

@staticmethod
    def is_perfect_setup(symbol: str, analyzer_score: float, decision: dict) -> bool:
        """
        v5.4.0: 5M DECIDE, 15M ADVIERTE
        
        - Score principal: 5M (con penalties robustas)
        - 15M: Solo genera warnings que reducen score
        - Threshold único: 60
        """
        from brain_config import (
            SCORE_WEIGHT_ANALYZER, SCORE_WEIGHT_VALIDATOR,
            SCORE_THRESHOLD_ENTER
        )
        from brain_frames import check_price_velocity
        from brain_entry_validator import entry_validator
        from brain_analyzer import generate_15m_warnings
        
        full_analysis = decision.get('full_analysis', {})
        ladder_data = full_analysis.get('ladder_data', {})
        
        data_5m = ladder_data.get('5m', {})
        data_15m = ladder_data.get('15m', {})
        
        # ===================================================================
        # PASO 1: OBTENER SCORE DE 5M (ahora es el score principal)
        # ===================================================================
        score_5m = data_5m.get('score_5m', 0)
        direction_5m = data_5m.get('direction', 'LATERAL')
        
        print(f"\n   📊 5M DOMINANCE SCORING v5.4.0:")
        print(f"      Base Score 5M: {score_5m}")
        print(f"      Direction 5M: {direction_5m}")
        
        # ===================================================================
        # PASO 2: PENALTY SI 5M ES LATERAL
        # ===================================================================
        if direction_5m == 'LATERAL':
            score_5m -= 50
            print(f"      ⚠️ 5M LATERAL: penalty -50 → score: {score_5m}")
            
            # Intentar inferir dirección desde DI
            adx_5m = data_5m.get('adx', {})
            if adx_5m.get('dominance') == 'DI+':
                direction_5m = 'BULLISH'
            elif adx_5m.get('dominance') == 'DI-':
                direction_5m = 'BEARISH'
            else:
                direction_5m = 'BULLISH'  # Default
            print(f"      → Usando {direction_5m} (DI inference)")
        
        # ===================================================================
        # PASO 3: WARNINGS DE 15M (solo advierten, no bloquean)
        # ===================================================================
        warnings_15m = generate_15m_warnings(data_15m, direction_5m)
        
        if warnings_15m['should_hard_veto']:
            print(f"\n   🚫 15M HARD VETO: {warnings_15m['warnings']}")
            return False
        
        if warnings_15m['total_penalty'] > 0:
            score_5m -= warnings_15m['total_penalty']
            print(f"      📉 15M Warnings: -{warnings_15m['total_penalty']} → score: {score_5m}")
        
        # ===================================================================
        # PASO 4: 5M VETO (mantener sistema existente)
        # ===================================================================
        veto_result = check_5m_veto(symbol, direction_5m, ladder_data)
        
        if veto_result['should_veto']:
            print(f"\n   🚫 5M VETO: {veto_result['reason']}")
            return False
        
        if veto_result['penalty'] > 0:
            score_5m -= veto_result['penalty']
            print(f"      📉 5M Veto Penalty: -{veto_result['penalty']} → score: {score_5m}")
        
        # ===================================================================
        # PASO 5: VALIDATOR SCORE (mantener lógica)
        # ===================================================================
        velocity_check = check_price_velocity(symbol, direction_5m, '5m')
        if 'insuficientes' in velocity_check.get('reason', '').lower():
            velocity_check = check_price_velocity(symbol, direction_5m, '15m')
        
        velocity_value = velocity_check.get('velocity', 0)
        print(f"      Velocity: {velocity_value:.3f}%")
        
        force_result = full_analysis.get('force_result', {}).copy()
        force_result['velocity'] = velocity_value
        
        try:
            validation = entry_validator._validate_explosion_entry(
                symbol, direction_5m, force_result, ladder_data
            )
            validator_score = validation.get('score', 50)
        except Exception as e:
            print(f"      ⚠️ Error validator: {e}")
            validator_score = 50
        
        print(f"      Validator Score: {validator_score:.0f}")
        
        # ===================================================================
        # PASO 6: COMBINAR SCORES
        # ===================================================================
        combined = (score_5m * SCORE_WEIGHT_ANALYZER) + \
                   (validator_score * SCORE_WEIGHT_VALIDATOR)
        
        print(f"      🎯 COMBINED: ({score_5m:.0f} × {SCORE_WEIGHT_ANALYZER}) + ({validator_score:.0f} × {SCORE_WEIGHT_VALIDATOR}) = {combined:.0f}")
        print(f"      Threshold: {SCORE_THRESHOLD_ENTER}")
        
        # ===================================================================
        # PASO 7: DECISIÓN FINAL
        # ===================================================================
        if combined >= SCORE_THRESHOLD_ENTER:
            print(f"   ✅ APROBADO: {combined:.0f} >= {SCORE_THRESHOLD_ENTER}")
            return True
        else:
            print(f"   ❌ RECHAZADO: {combined:.0f} < {SCORE_THRESHOLD_ENTER}")
            return False
