def build_time_ladder_context(symbol, watchlist_data):
    """
    Construye el contexto completo de Time Ladder
    Como subir una escalera viendo cada peldaño
    """
    
    context = {
        'symbol': symbol,
        'ladder': {},
        'evolution': {},
        'confluence': 0
    }
    
    # Fetch todos los timeframes necesarios
    timeframes = {
        '1m': 100,   # Explosiones
        '5m': 50,    # Velocidad
        '15m': 50,   # Timing
        '1h': 50,    # Tendencia
        '4h': 30     # Contexto
    }
    
    for tf, limit in timeframes.items():
        ohlcv = fetch_ohlcv(symbol, tf, limit)
        if not ohlcv:
            return None
            
        df = pd.DataFrame(ohlcv, columns=['timestamp','open','high','low','close','volume'])
        
        # Análisis específico por timeframe
        if tf == '1m':
            context['ladder']['1m'] = analyze_micro_structure(df)
        elif tf == '5m':
            context['ladder']['5m'] = analyze_execution_layer(df)
        elif tf == '15m':
            context['ladder']['15m'] = analyze_timing_layer(df)
        elif tf == '1h':
            context['ladder']['1h'] = analyze_trend_layer(df)
        elif tf == '4h':
            context['ladder']['4h'] = analyze_context_layer(df)
    
    # Calcular confluencia (alineación de timeframes)
    context['confluence'] = calculate_timeframe_confluence(context['ladder'])
    
    # Detectar evolución (cómo llegamos aquí)
    context['evolution'] = detect_trend_evolution(context['ladder'])
    
    return context

def analyze_micro_structure(df):
    """
    1M - Detecta microestructura y explosiones
    """
    analysis = {}
    
    # Velocidad y aceleración en tiempo real
    velocity = calculate_velocity(df, lookback=5)
    acceleration = calculate_acceleration(df, lookback=5)
    
    # Detectar explosión
    last_5_candles = df.tail(5)
    consecutive_ups = (last_5_candles['close'] > last_5_candles['open']).sum()
    
    analysis['is_explosive'] = (
        consecutive_ups >= 4 and 
        velocity > 0.001 and 
        acceleration > 0
    )
    
    # Microestructura
    analysis['micro_trend'] = 'UP' if velocity > 0 else 'DOWN'
    analysis['acceleration'] = acceleration
    analysis['velocity'] = velocity
    
    # Detección de trampa (pump fake)
    volume_spike = df['volume'].iloc[-1] / df['volume'].mean()
    analysis['pump_risk'] = volume_spike > 10  # Alerta si volumen >10x
    
    return analysis

def analyze_execution_layer(df):
    """
    5M - Capa de ejecución inmediata
    """
    analysis = {}
    
    # ADX y Momentum actuales
    adx = calculate_adx_with_memory(df, period=14, lookback=10)
    momentum = calculate_momentum_with_memory(df, lookback=10)
    
    analysis['adx_current'] = adx['adx_current']
    analysis['momentum_strength'] = momentum['consecutive_count']
    analysis['di_plus'] = adx['di_plus_current']
    analysis['di_minus'] = adx['di_minus_current']
    
    # Volume Pattern
    vol_current = df['volume'].iloc[-1]
    vol_avg = df['volume'].iloc[-20:].mean()
    analysis['volume_ratio'] = vol_current / vol_avg if vol_avg > 0 else 0
    
    # Presión compradora/vendedora
    buying_pressure = ((df['close'] - df['low']) / (df['high'] - df['low'])).mean()
    analysis['buying_pressure'] = buying_pressure
    
    return analysis

def analyze_timing_layer(df):
    """
    15M - Timing y punto de entrada
    """
    analysis = {}
    
    # ADX para fuerza de tendencia
    adx = calculate_adx_with_memory(df, period=14, lookback=10)
    analysis['adx'] = adx['adx_current']
    analysis['adx_trend'] = adx['adx_trend']
    
    # Distancia a EMAs (evitar comprar en techo)
    ema_20 = df['close'].ewm(span=20).mean().iloc[-1]
    current_price = df['close'].iloc[-1]
    analysis['distance_to_ema20'] = ((current_price - ema_20) / ema_20) * 100
    
    # Momentum quality
    momentum = calculate_momentum_with_memory(df, lookback=10)
    analysis['momentum_age'] = momentum['consecutive_count']
    analysis['momentum_quality'] = momentum['strength']
    
    return analysis

def analyze_trend_layer(df):
    """
    1H - Tendencia dominante
    """
    analysis = {}
    
    # Estructura de mercado
    highs = df['high'].rolling(5).max()
    lows = df['low'].rolling(5).min()
    
    # Higher Highs / Higher Lows
    analysis['structure'] = detect_market_structure(highs, lows)
    
    # ADX 1H (más confiable para tendencia)
    adx = calculate_adx_with_memory(df, period=14, lookback=10)
    analysis['trend_strength'] = adx['adx_current']
    
    # VWAP vs Price
    vwap = (df['close'] * df['volume']).cumsum() / df['volume'].cumsum()
    analysis['above_vwap'] = df['close'].iloc[-1] > vwap.iloc[-1]
    
    return analysis

def analyze_context_layer(df):
    """
    4H - Contexto macro
    """
    analysis = {}
    
    # Tendencia principal
    ema_50 = df['close'].ewm(span=50).mean().iloc[-1]
    ema_200 = df['close'].ewm(span=200).mean().iloc[-1] if len(df) > 200 else ema_50
    
    analysis['trend'] = 'BULLISH' if ema_50 > ema_200 else 'BEARISH'
    analysis['trend_strength'] = abs((ema_50 - ema_200) / ema_200) * 100
    
    # Volatilidad histórica
    analysis['volatility'] = df['close'].pct_change().std() * 100
    
    return analysis

def calculate_timeframe_confluence(ladder):
    """
    Calcula alineación entre timeframes (0-100)
    """
    bullish_count = 0
    bearish_count = 0
    
    # Check cada timeframe
    if ladder.get('1m', {}).get('micro_trend') == 'UP':
        bullish_count += 1
    else:
        bearish_count += 1
        
    if ladder.get('5m', {}).get('di_plus', 0) > ladder.get('5m', {}).get('di_minus', 0):
        bullish_count += 1
    else:
        bearish_count += 1
        
    if ladder.get('15m', {}).get('distance_to_ema20', 0) < 5:  # No sobreextendido
        bullish_count += 1
        
    if ladder.get('1h', {}).get('above_vwap', False):
        bullish_count += 1
        
    if ladder.get('4h', {}).get('trend') == 'BULLISH':
        bullish_count += 1
        
    total = bullish_count + bearish_count
    return (bullish_count / total * 100) if total > 0 else 50

def calculate_intelligent_score(time_ladder, market_context, watchlist_data):
    """
    Sistema de scoring no lineal con interacciones
    """
    score = 50  # Base neutral
    confidence_factors = []
    
    # ========================================
    # 1. TIME LADDER SCORING (40% peso)
    # ========================================
    
    # Confluencia de timeframes
    confluence = time_ladder['confluence']
    if confluence > 80:
        score += 20
        confidence_factors.append(('STRONG_CONFLUENCE', 20))
    elif confluence > 60:
        score += 10
        confidence_factors.append(('GOOD_CONFLUENCE', 10))
    elif confluence < 40:
        score -= 15
        confidence_factors.append(('POOR_CONFLUENCE', -15))
    
    # Evolución de tendencia
    evolution_stage = time_ladder['evolution']['stage']
    if evolution_stage == 'YOUNG':
        score += 25
        confidence_factors.append(('YOUNG_TREND', 25))
    elif evolution_stage == 'BIRTH':
        score += 15
        confidence_factors.append(('EMERGING_TREND', 15))
    elif evolution_stage == 'MATURE':
        score -= 10
        confidence_factors.append(('MATURE_TREND', -10))
    
    # ========================================
    # 2. EXECUTION LAYER SCORING (30% peso)
    # ========================================
    
    # ADX 5M (fuerza inmediata)
    adx_5m = time_ladder['ladder']['5m']['adx_current']
    if adx_5m > 35:
        score += 15
        confidence_factors.append(('STRONG_ADX', 15))
    elif adx_5m > 25:
        score += 8
        confidence_factors.append(('GOOD_ADX', 8))
    elif adx_5m < 20:
        score -= 10
        confidence_factors.append(('WEAK_ADX', -10))
    
    # Volumen
    vol_ratio = time_ladder['ladder']['5m']['volume_ratio']
    if vol_ratio > 2.0:
        score += 10
        confidence_factors.append(('HIGH_VOLUME', 10))
    elif vol_ratio > 1.5:
        score += 5
        confidence_factors.append(('GOOD_VOLUME', 5))
    elif vol_ratio < 0.5:
        score -= 15
        confidence_factors.append(('LOW_VOLUME', -15))
    
    # ========================================
    # 3. CONTEXT SCORING (20% peso)
    # ========================================
    
    # Sesión de trading
    session_quality = market_context['session_quality']
    score *= session_quality  # Multiplicador por calidad de sesión
    
    # BTC alignment
    if not market_context.get('btc_aligned', True):
        score *= 0.7  # Penalización 30% si va contra BTC
        confidence_factors.append(('AGAINST_BTC', -30))
    
    # ========================================
    # 4. RISK FACTORS (10% peso)
    # ========================================
    
    # Pump risk
    if time_ladder['ladder']['1m'].get('pump_risk', False):
        score *= 0.5  # Penalización 50% si hay riesgo de pump
        confidence_factors.append(('PUMP_RISK', -50))
    
    # Sobreextensión
    distance_ema = time_ladder['ladder']['15m'].get('distance_to_ema20', 0)
    if abs(distance_ema) > 5:
        score -= 10
        confidence_factors.append(('OVEREXTENDED', -10))
    
    # ========================================
    # 5. BONUS EXPLOSIVO
    # ========================================
    if time_ladder['ladder']['1m']['is_explosive']:
        score += 30
        confidence_factors.append(('EXPLOSIVE_1M', 30))
    
    return {
        'total_score': min(100, max(0, score)),
        'factors': confidence_factors,
        'confidence_level': get_confidence_level(score)
    }

def get_confidence_level(score):
    """
    Convierte score en nivel de confianza
    """
    if score >= 80:
        return 'VERY_HIGH'
    elif score >= 65:
        return 'HIGH'
    elif score >= 50:
        return 'MEDIUM'
    elif score >= 35:
        return 'LOW'
    else:
        return 'VERY_LOW'

def make_gradient_decision(symbol, confidence_score, time_ladder, market_context):
    """
    Decisión con gradiente - no binaria
    """
    
    score = confidence_score['total_score']
    confidence = confidence_score['confidence_level']
    
    # Umbrales adaptativos según contexto
    if market_context['session'] == 'US':
        threshold = 55  # Más agresivo en sesión US
    elif market_context['session'] == 'ASIA':
        threshold = 65  # Más conservador en Asia
    else:
        threshold = 60
    
    # Ajuste por evolución
    if time_ladder['evolution']['stage'] == 'YOUNG':
        threshold -= 5  # Más agresivo con tendencias jóvenes
    elif time_ladder['evolution']['stage'] == 'MATURE':
        threshold += 10  # Más conservador con tendencias maduras
    
    # DECISIÓN
    if score >= threshold:
        print(f"🎯 TRIGGER ACTIVADO: {symbol}")
        print(f"   Score: {score:.1f}/100 | Confianza: {confidence}")
        print(f"   Factores principales:")
        for factor, points in confidence_score['factors'][:5]:
            print(f"   • {factor}: {points:+d}")
        
        return {
            'activated': True,
            'score': score,
            'confidence': confidence,
            'threshold_used': threshold,
            'time_ladder': time_ladder['evolution']['stage'],
            'session': market_context['session'],
            'factors': confidence_score['factors']
        }
    else:
        print(f"⏳ Esperando mejor setup: {symbol}")
        print(f"   Score: {score:.1f}/{threshold} | Confianza: {confidence}")
        
        # Importante: devolver por qué no entró
        return {
            'activated': False,
            'score': score,
            'confidence': confidence,
            'threshold_used': threshold,
            'missing_points': threshold - score,
            'recommendation': get_improvement_recommendation(confidence_score['factors'])
        }

def get_improvement_recommendation(factors):
    """
    Sugiere qué necesita mejorar para activar el trigger
    """
    negative_factors = [f for f in factors if f[1] < 0]
    if negative_factors:
        worst_factor = min(negative_factors, key=lambda x: x[1])
        recommendations = {
            'POOR_CONFLUENCE': 'Esperando alineación de timeframes',
            'WEAK_ADX': 'Necesita más fuerza en la tendencia',
            'LOW_VOLUME': 'Volumen insuficiente',
            'MATURE_TREND': 'Tendencia muy madura, riesgo de reversión',
            'PUMP_RISK': 'Posible manipulación detectada',
            'AGAINST_BTC': 'Esperando alineación con BTC'
        }
        return recommendations.get(worst_factor[0], 'Mejorando condiciones generales')
    return 'Acumulando más señales positivas'

        return 'DOWNTREND_LH_LL'
    else:
        return 'RANGING'
