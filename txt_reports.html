Fase 1: Frame Buffer (brain_frames.py)

class FrameBuffer:
    def __init__(self, size=10):
        self.frames = deque(maxlen=size)  # Últimos 10 minutos
    
    def add_frame(self, adx, price, candle_size, momentum):
        self.frames.append({
            'timestamp': now(),
            'adx': adx,
            'price': price, 
            'candle_size': candle_size,
            'momentum': momentum
        })
    
    def detect_acceleration(self):
        """Ve la película: ¿acelera o desacelera?"""
        if len(self.frames) < 3:
            return None
            
        velocities = []
        for i in range(len(self.frames)-1):
            v = self.frames[i+1].price - self.frames[i].price
            velocities.append(v)
        
        # ¿Velocidad creciente? = aceleración
        return velocities[-1] > velocities[0]
    
    def detect_candle_growth(self):
        """Ve la película: ¿velas crecen?"""
        sizes = [f.candle_size for f in self.frames]
        return all(sizes[i] < sizes[i+1] for i in range(len(sizes)-1))

def analyze_market_space(symbol, direction):
    """Ve el mercado como canvas 2D."""
    
    # Posición actual
    current_price = get_price(symbol)
    
    # Obstáculos (resistencia/soporte)
    resistance = find_resistance(symbol)
    support = find_support(symbol)
    
    # Runway visual
    if direction == 'BULLISH':
        runway = ((resistance - current_price) / current_price) * 100
    else:
        runway = ((current_price - support) / current_price) * 100
    
    # Velocidad (frames históricos)
    velocity = calculate_velocity_from_frames(symbol)
    
    # Tiempo hasta colisión
    if velocity > 0:
        time_to_obstacle = runway / velocity  # minutos
    else:
        time_to_obstacle = 999
    
    return {
        'runway': runway,
        'velocity': velocity,
        'time_to_collision': time_to_obstacle,
        'can_enter': time_to_obstacle > 2  # Mínimo 2 min de runway
    }

Fase 3: Decisión Visual (brain_think.py)

def decide_entry(symbol, direction):
    """Decisión basada en VISIÓN, no scoring."""
    
    # 1. Ver la película (frames)
    frames = get_frame_buffer(symbol)
    accelerating = frames.detect_acceleration()
    candles_growing = frames.detect_candle_growth()
    
    # 2. Ver el espacio (canvas)
    space = analyze_market_space(symbol, direction)
    has_runway = space.time_to_collision > 2
    
    # 3. Decisión simple y visual
    if accelerating and candles_growing and has_runway:
        return ENTER  # Explosión iniciando + espacio libre
    
    return WAIT
