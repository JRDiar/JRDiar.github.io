"""
manual_analyzer.py - Análisis Manual de Moneda en Tiempo Real
==============================================================
Uso: python tools/manual_analyzer.py SYMBOL
Ejemplo: python tools/manual_analyzer.py BTC/USDT:USDT
Muestra TODOS los filtros por los que pasa la moneda.
"""
import sys
import os
sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))
from brain_analyzer import analyze_15m, analyze_1h, analyze_4h
from brain_calculator import (
    calculate_adx_with_memory,
    calculate_momentum_with_memory,
    calculate_zigzag
)
from brain_think import sniper_checklist
from brain_intelligent import decide_intelligently
from exchange_connection import fetch_ohlcv, fetch_current_price
import pandas as pd
def analyze_coin_manual(symbol):
    """
    Análisis completo paso a paso de una moneda
    """
    print("\n" + "=" * 70)
    print(f"🔬 ANÁLISIS MANUAL: {symbol}")
    print("=" * 70)
    
    # 1. PRECIO ACTUAL
    print("\n📊 PASO 1: DATOS BÁSICOS")
    print("-" * 50)
    current_price = fetch_current_price(symbol)
    print(f"  Precio actual: ${current_price}")
    
    # 2. ADX Y MOMENTUM (15M)
    print("\n📊 PASO 2: ADX + MOMENTUM (15M)")
    print("-" * 50)
    ohlcv = fetch_ohlcv(symbol, '15m', 50)
    if not ohlcv:
        print("  ❌ Error obteniendo datos 15M")
        return
    
    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    
    adx_data = calculate_adx_with_memory(df)
    mom_data = calculate_momentum_with_memory(df)
    
    print(f"  ADX: {adx_data['adx_current']:.1f}")
    print(f"  ADX Trend: {adx_data['adx_trend']}")
    print(f"  DI+: {adx_data['di_plus_current']:.1f}")
    print(f"  DI-: {adx_data['di_minus_current']:.1f}")
    print(f"  Spread: {abs(adx_data['di_plus_current'] - adx_data['di_minus_current']):.1f}")
    print(f"  Momentum Color: {mom_data['color_current']}")
    print(f"  Momentum Age: {mom_data['consecutive_count']}v")
    
    # 3. VALIDACIONES JOEL'S STRATEGY
    print("\n📊 PASO 3: JOEL'S STRATEGY VALIDATIONS")
    print("-" * 50)
    
    adx_current = adx_data['adx_current']
    adx_trend = adx_data['adx_trend']
    spread = abs(adx_data['di_plus_current'] - adx_data['di_minus_current'])
    mom_age = mom_data['consecutive_count']
    
    # Validación ADX
    if adx_current < 25:
        print(f"  ❌ ADX {adx_current:.1f} < 25 → PENALTY -60")
    elif adx_trend == 'decreciendo':
        print(f"  ❌ ADX {adx_current:.1f} DECRECIENDO → PENALTY -60")
    elif adx_trend == 'creciendo':
        print(f"  ✅ ADX {adx_current:.1f} CRECIENDO → OK")
    else:
        print(f"  ⚠️ ADX {adx_current:.1f} ESTABLE → NEUTRAL")
    
    # Validación Spread
    if spread < 5:
        print(f"  ❌ Spread {spread:.1f} < 5 → PENALTY -80")
    elif spread < 7:
        print(f"  ❌ Spread {spread:.1f} < 7 → PENALTY -50")
    else:
        print(f"  ✅ Spread {spread:.1f} >= 7 → OK")
    
    # Validación Momentum Age
    if mom_age > 5:
        print(f"  ❌ Momentum {mom_age}v > 5 → PENALTY -50")
    elif mom_age > 3:
        print(f"  ⚠️ Momentum {mom_age}v > 3 → PENALTY -25")
    else:
        print(f"  ✅ Momentum {mom_age}v <= 3 → OK")
    
    # 4. VELOCIDAD DEL PRECIO
    print("\n📊 PASO 4: VELOCIDAD DEL PRECIO")
    print("-" * 50)
    closes = df['close'].values
    price_changes = []
    for i in range(-10, -1):
        if closes[i-1] != 0:
            change = abs((closes[i] - closes[i-1]) / closes[i-1] * 100)
            price_changes.append(change)
    
    if price_changes:
        avg_change = sum(price_changes) / len(price_changes)
        if avg_change < 0.1:
            print(f"  ❌ Avg change: {avg_change:.3f}%/vela → ESTÁTICO (-40)")
        else:
            print(f"  ✅ Avg change: {avg_change:.3f}%/vela → MOVIMIENTO OK")
    
    # 5. GRID DIAGONAL
    print("\n📊 PASO 5: GRID DIAGONAL")
    print("-" * 50)
    try:
        from brain_grid import MarketGrid
        zigzag = calculate_zigzag(df)
        grid = MarketGrid()
        diagonal = grid._build_diagonal_trendlines(df, zigzag)
        if diagonal.get('valid'):
            pos = diagonal['price_position_pct']
            if pos > 80:
                print(f"  ❌ Posición: {pos:.0f}% → CERCA TECHO (-40)")
            elif pos < 20:
                print(f"  ✅ Posición: {pos:.0f}% → CERCA SOPORTE (+20)")
            else:
                print(f"  ⚪ Posición: {pos:.0f}% → ZONA NEUTRAL")
        else:
            print(f"  ⚠️ Grid diagonal no disponible")
    except Exception as e:
        print(f"  ⚠️ Error grid: {e}")
    
    # 6. LLAMAR A analyze_15m
    print("\n📊 PASO 6: RESULTADO analyze_15m()")
    print("-" * 50)
    
    # Determinar dirección
    if adx_data['di_plus_current'] > adx_data['di_minus_current']:
        direction = 'BULLISH'
    else:
        direction = 'BEARISH'
    
    result = analyze_15m(symbol, direction)
    print(f"  Score 15M: {result.get('score_15m', 'N/A')}")
    print(f"  Direction: {result.get('direction', 'N/A')}")
    print(f"  Strength: {result.get('strength', 'N/A')}")
    
    # 7. SNIPER CHECKLIST
    print("\n📊 PASO 7: SNIPER CHECKLIST")
    print("-" * 50)
    
    # Crear ladder básico
    ladder = {
        '15m': result,
        '1h': analyze_1h(symbol, direction) if 'analyze_1h' in dir() else {},
        '4h': analyze_4h(symbol, direction) if 'analyze_4h' in dir() else {},
    }
    
    sniper_result = sniper_checklist(symbol, direction, ladder=ladder)
    print(f"  Ready: {sniper_result.get('ready', 'N/A')}")
    print(f"  Total Score: {sniper_result.get('total_score', 'N/A')}")
    print(f"  Archetype: {sniper_result.get('archetype', 'N/A')}")
    print(f"  Summary: {sniper_result.get('summary', 'N/A')}")
    
    # 8. VEREDICTO FINAL
    print("\n" + "=" * 70)
    print("🎯 VEREDICTO FINAL")
    print("=" * 70)
    
    score_15m = result.get('score_15m', 0)
    if score_15m >= 65 and sniper_result.get('ready'):
        print(f"  ✅ APROBAR (Score: {score_15m})")
    else:
        print(f"  ❌ RECHAZAR (Score: {score_15m})")
        print(f"  Razones de rechazo:")
        if adx_current < 25:
            print(f"    - ADX {adx_current:.1f} < 25")
        if adx_trend == 'decreciendo':
            print(f"    - ADX DECRECIENDO")
        if spread < 7:
            print(f"    - Spread {spread:.1f} < 7")
        if mom_age > 5:
            print(f"    - Momentum {mom_age}v > 5")
if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("Uso: python tools/manual_analyzer.py SYMBOL")
        print("Ejemplo: python tools/manual_analyzer.py BTC/USDT:USDT")
        sys.exit(1)
    
    symbol = sys.argv[1]
    if not symbol.endswith(':USDT'):
        symbol = f"{symbol}:USDT"
    
    analyze_coin_manual(symbol)
