# Imports del Bot
from exchange.connection import MEXCConnection

# Colores para consola
class Colors:
    HEADER = '\033[95m'
    BLUE = '\033[94m'
    CYAN = '\033[96m'
    GREEN = '\033[92m'
    WARNING = '\033[93m'
    FAIL = '\033[91m'
    ENDC = '\033[0m'
    BOLD = '\033[1m'
    UNDERLINE = '\033[4m'
def print_header(symbol):
    print(f"\n{Colors.HEADER}╔{'═'*60}╗{Colors.ENDC}")
    print(f"{Colors.HEADER}║  🔍 BRAINTRADER INSPECTOR - {symbol:<31} ║{Colors.ENDC}")
    print(f"{Colors.HEADER}║  Timestamp: {datetime.now().strftime('%Y-%m-%d %H:%M:%S'):<37} ║{Colors.ENDC}")
    print(f"{Colors.HEADER}╚{'═'*60}╝{Colors.ENDC}\n")
def print_phase(phase_num, title, content_lines):
    print(f"{Colors.CYAN}┌─ 📊 FASE {phase_num}: {title} {'─'*(48-len(title))}{Colors.ENDC}┐")
    for line in content_lines:
        print(f"{Colors.CYAN}│{Colors.ENDC} {line}")
    print(f"{Colors.CYAN}└{'─'*58}{Colors.ENDC}")
def analyze_symbol(symbol_input):
    start_time = time.time()
    
    # Normalizar input
    symbol = symbol_input.upper()
    if not symbol.endswith(':USDT') and 'USDT' not in symbol:
        symbol += '/USDT:USDT'
    elif symbol.endswith('USDT') and ':' not in symbol:
        symbol += ':USDT'
    
    print_header(symbol)
    # ------------------------------------------------------------
    # FASE 0: INICIALIZACIÓN Y CONEXIÓN
    # ------------------------------------------------------------
    try:
        # Silenciar logs para no ensuciar output
        import logging
        logging.getLogger().setLevel(logging.CRITICAL)
        
        # Conexión Read-Only (sin keys)
        conn = MEXCConnection() 
        if not conn.connected:
            print(f"{Colors.FAIL}❌ Error conectando a MEXC{Colors.ENDC}")
            return
            
    except Exception as e:
        print(f"{Colors.FAIL}❌ Error fatal iniciando: {e}{Colors.ENDC}")
        return
    # ------------------------------------------------------------
    # FASE 1: VALIDACIÓN EXCHANGE
    # ------------------------------------------------------------
    phase1_output = []
    try:
        ticker = conn.fetch_ticker(symbol)
        if not ticker or ticker.get('last') == 0:
             phase1_output.append(f"{Colors.FAIL}❌ Símbolo no encontrado o sin datos{Colors.ENDC}")
             print_phase(1, "VALIDACIÓN EXCHANGE", phase1_output)
             return
        
        price = ticker['last']
        bid = ticker['bid']
        ask = ticker['ask']
        spread = ask - bid
        spread_pct = (spread / price) * 100
        
        phase1_output.append(f"✅ Símbolo válido: {symbol}")
        phase1_output.append(f"Precio Actual: ${price:,.5f}")
        phase1_output.append(f"Spread: ${spread:.2f} ({spread_pct:.4f}%)")
        
    except Exception as e:
        phase1_output.append(f"{Colors.FAIL}❌ Error validando: {e}{Colors.ENDC}")
        print_phase(1, "VALIDACIÓN EXCHANGE", phase1_output)
        return
    print_phase(1, "VALIDACIÓN EXCHANGE", phase1_output)
    # ------------------------------------------------------------
    # FASE 2: DATOS OHLCV
    # ------------------------------------------------------------
    phase2_output = []
    ohlcv = []
    try:
        ohlcv = conn.fetch_ohlcv(symbol, '5m', limit=100)
        count = len(ohlcv)
        phase2_output.append(f"Velas obtenidas (5m): {count}/100 requested")
        
        if count < 20:
             phase2_output.append(f"{Colors.FAIL}❌ Insuficientes datos (<20){Colors.ENDC}")
             print_phase(2, "DATOS OHLCV", phase2_output)
             return
             
        last_candle = ohlcv[-1]
        candle_range = (last_candle[2] - last_candle[3]) / last_candle[3] * 100
        
        phase2_output.append(f"Última Vela: O:{last_candle[1]} H:{last_candle[2]} L:{last_candle[3]} C:{last_candle[4]}")
        phase2_output.append(f"Rango Vela: {candle_range:.2f}%")
        
    except Exception as e:
        phase2_output.append(f"{Colors.FAIL}❌ Error fetching OHLCV: {e}{Colors.ENDC}")
        print_phase(2, "DATOS OHLCV", phase2_output)
        return
    print_phase(2, "DATOS OHLCV", phase2_output)
    # ------------------------------------------------------------
    # FASE 3: FÍSICA (MarketEntity)
    # ------------------------------------------------------------
    phase3_output = []
    entity = None
    try:
        entity = MarketEntity(symbol)
        # Poblar historia manualmente para simular el bot
        prices = [c[4] for c in ohlcv]
        # Necesitamos timestamps para calcular velocidad real
        # MarketEntity espera (price, timestamp)
        history_data = [(c[4], c[0]) for c in ohlcv[-20:]] # Simulamos updates recientes
        
        # update masivo simulado
        # En visual_sniper hace update() vela a vela o usa historia directa
        # MarketEntity tiene .update(price, timestamp)
        # Vamos a resetear y cargar
        entity.price_history = []
        for p, t in history_data:
            entity.update(p, t)
            
        # Métricas
        velocity = entity.velocity
        accel = entity.acceleration
        
        # Momentum check manual (reimaginando logica visual_sniper que usa su propio avg check a veces)
        # Pero market_entity.has_momentum() es la clave
        
        # Calculo de historical avg velocity para display
        hist_vels = []
        if len(entity.price_history) > 2:
            for i in range(1, len(entity.price_history)):
                 v = abs((entity.price_history[i]['price'] - entity.price_history[i-1]['price']) / entity.price_history[i-1]['price'] * 100)
                 hist_vels.append(v)
        avg_vel = np.mean(hist_vels) if hist_vels else 0.001
        
        ratio = velocity / avg_vel if avg_vel > 0 else 0
        threshold = avg_vel * 0.3
        has_momentum = abs(velocity) > threshold
        
        phase3_output.append(f"Velocidad Actual: {velocity:+.4f}%/min")
        phase3_output.append(f"Velocidad Promedio: {avg_vel:.4f}%/min")
        phase3_output.append(f"Ratio (Act/Avg): {ratio:.2f}x")
        phase3_output.append(f"Aceleración: {accel:+.6f}%/min²")
        phase3_output.append(f"Threshold Momentum: > {threshold:.4f}%/min")
        
        mom_icon = "✅" if has_momentum else "❌"
        phase3_output.append(f"Test Momentum: {mom_icon} (Es {abs(velocity):.4f} > {threshold:.4f}?)")
        
    except Exception as e:
        phase3_output.append(f"{Colors.FAIL}❌ Error cálculo física: {e}{Colors.ENDC}")
        import traceback
        traceback.print_exc()
    print_phase(3, "FÍSICA (MarketEntity)", phase3_output)
    # ------------------------------------------------------------
    # FASE 4: ESTRUCTURA (Walls)
    # ------------------------------------------------------------
    phase4_output = []
    walls = None
    try:
        walls = Walls(symbol)
        # Cargar OHLCV
        # Walls usa self.df internamente si se le pasa ohlcv
        # Ojo: Walls.update(ohlc) espera dict format o lista? 
        # Revisando codigo Walls... update toma single candle dict o lista de dicts?
        # Walls.update suele ser single candle. `detect_structure` usa self.closes
        # Simulamos carga:
        for c in ohlcv:
            walls.update({
                'timestamp': c[0],
                'open': c[1], 'high': c[2], 'low': c[3], 'close': c[4], 'volume': c[5]
            })
            
        structure = walls.detect_structure()
        trend = structure.get('trend', 'UNKNOWN')
        
        phase4_output.append(f"Trend Detectada: {Colors.BOLD}{trend}{Colors.ENDC}")
        phase4_output.append(f"Pivotes: HH={structure.get('hh_count')} HL={structure.get('hl_count')} LH={structure.get('lh_count')} LL={structure.get('ll_count')}")
        
        # Resistencias / Soportes
        nearest_res = walls.find_nearest_wall(price, 1) # 1 = arriba
        nearest_sup = walls.find_nearest_wall(price, -1) # -1 = abajo
        
        if nearest_res:
            dist_res_pct = (nearest_res[0] - price) / price * 100
            phase4_output.append(f"Resistencia Prox: ${nearest_res[0]:.2f} (+{dist_res_pct:.2f}%)")
        else:
            phase4_output.append("Resistencia Prox: Ninguna cercana (Cielo abierto)")
            
        if nearest_sup:
            dist_sup_pct = (price - nearest_sup[0]) / price * 100
            phase4_output.append(f"Soporte Prox: ${nearest_sup[0]:.2f} (-{dist_sup_pct:.2f}%)")
            
        # Runway
        runway_mins = 999
        if nearest_res and velocity > 0:
            runway_mins = walls.calculate_runway(price, nearest_res[0], velocity)
            phase4_output.append(f"Tiempo a Colisión: {runway_mins:.1f} min")
        elif nearest_sup and velocity < 0:
            runway_mins = walls.calculate_runway(price, nearest_sup[0], velocity) # velocity es neg
            phase4_output.append(f"Tiempo a Colisión: {runway_mins:.1f} min")
        else:
             phase4_output.append("Tiempo a Colisión: Infinito (scan/static)")
        # Clasificación Runway
        # Replicamos logica de visual_classifier para mostrar threshold
        # classify_runway calcula distancia %
        dist_pct = 0
        if velocity >= 0 and nearest_res:
            dist_pct = (nearest_res[0] - price) / price * 100
        elif velocity < 0 and nearest_sup:
            dist_pct = abs((nearest_sup[0] - price) / price * 100)
        else:
            dist_pct = 999
            
        runway_sprite = classify_runway(dist_pct)
        phase4_output.append(f"Runway Sprite: {runway_sprite} (Dist: {dist_pct:.2f}%)")
        
    except Exception as e:
         phase4_output.append(f"{Colors.FAIL}❌ Error estructura: {e}{Colors.ENDC}")
    print_phase(4, "ESTRUCTURA (Walls)", phase4_output)
    # ------------------------------------------------------------
    # FASE 5: CLASIFICACIÓN VISUAL
    # ------------------------------------------------------------
    phase5_output = []
    sprites = {}
    
    try:
        # 1. Price Direction
        p_dir = classify_price_direction_with_avg(velocity, accel, avg_vel)
        phase5_output.append(f"Price Direction: {p_dir}")
        sprites['price_direction'] = p_dir
        
        # 2. Candle Shape (simplificado sin acceso a historical body avg complejo aqui, usamos basic data)
        # Necesitamos 'body_size' relative to something.
        # classify_candle_shape usa (open, close, high, low, avg_body_size)
        bodies = [abs(c[4]-c[1]) for c in ohlcv[-20:]]
        avg_body = np.mean(bodies) if bodies else 0
        c_shape = classify_candle_shape(last_candle[1], last_candle[4], last_candle[2], last_candle[3], avg_body)
        phase5_output.append(f"Candle Shape: {c_shape}")
        sprites['candle_shape'] = c_shape
        
        # 3. Wind (EMAs)
        # Calculo simple de EMAs para pasar al clasificador
        import pandas as pd
        closes_s = pd.Series([c[4] for c in ohlcv])
        ema9 = closes_s.ewm(span=9).mean().iloc[-1]
        ema20 = closes_s.ewm(span=20).mean().iloc[-1]
        ema50 = closes_s.ewm(span=50).mean().iloc[-1]
        wind = classify_wind(price, ema9, ema20, ema50)
        phase5_output.append(f"Wind: {wind}")
        sprites['wind'] = wind
        
        # 4. Fuel (Momentum)
        # Usamos velocity > threshold como proxy de FUEL simple visual o logica de momentum LC
        # VisualSniper usa Indicators para 'fuel'. Aquí simplificamos con velocity ratio o re-calculamos.
        # Para ser fiel, usamos el estado de momentum calculado en F3
        fuel = 'FULL_TANK' if has_momentum and velocity > 0 else 'EMPTY_BEARISH' if has_momentum and velocity < 0 else 'FUMES'
        # O usar classify_fuel() si tuvieramos el indicador momentum especifico (ej. LazyBear)
        # Vamos a usar classify_fuel placeholder/simple
        # phase5_output.append(f"Fuel (Est): {fuel}")
        sprites['fuel'] = 'ANY' # Placeholder para evitar bloqueo por falta de lib externa compleja
        # Si fuel es critico en pattern, esto podría fallar. Revisemos patrón.
        # La mayoria acepta ANY o FULL_TANK.
        # Intentemos logica real:
        if has_momentum:
            sprites['fuel'] = 'FULL_TANK' if velocity > 0 else 'EMPTY_BEARISH'
        else:
            sprites['fuel'] = 'FUMES'
        phase5_output.append(f"Fuel: {sprites['fuel']}")
        # 5. Terrain
        terr = classify_terrain(structure)
        phase5_output.append(f"Terrain: {terr}")
        sprites['terrain'] = terr
        
        # 6. Runway
        sprites['runway'] = runway_sprite
        phase5_output.append(f"Runway: {runway_sprite}")
        
    except Exception as e:
        phase5_output.append(f"{Colors.FAIL}❌ Error clasificación: {e}{Colors.ENDC}")
    print_phase(5, "CLASIFICACIÓN VISUAL", phase5_output)
    # ------------------------------------------------------------
    # FASE 6: PATTERN MATCHING
    # ------------------------------------------------------------
    phase6_output = []
    best_pattern = None
    best_score = 0
    
    try:
        # Find pattern
        # find_pattern retorna (id, score, data)
        # Vamos a listar top matches
        matches = []
        for pid, pdata in PATTERNS.items():
            # Calc score manual para display
            score = 0
            total = 0
            details = []
            
            # Check Runway First logic (Critical)
            # Logica de visual_sniper es: filter by runway comp
            # Si runway no compatible, score = 0
            
            p_sprites = pdata['sprites']
            
            match_count = 0
            for k, v in p_sprites.items():
                if k == 'runway': continue
                total += 1
                if v == 'ANY' or v == sprites.get(k):
                    match_count += 1
                    details.append(f"✅ {k}")
                else:
                    details.append(f"❌ {k}({sprites.get(k)}!={v})")
            
            raw_score = match_count / total if total > 0 else 0
            
            matches.append({
                'id': pid,
                'name': pdata['name'],
                'score': raw_score,
                'action': pdata['action'],
                'details': details
            })
            
        # Sort by score
        matches.sort(key=lambda x: x['score'], reverse=True)
        
        phase6_output.append("Top 3 Patrones Candidatos:")
        for m in matches[:3]:
            phase6_output.append(f"  • {m['name']}: Score {m['score']:.2f} | Acc: {m['action']}")
            # phase6_output.append(f"    {' '.join(m['details'][:3])}...") # Detalle corto
            
        best = matches[0] if matches else None
        if best and best['score'] >= 0.70:
            best_pattern = best
            phase6_output.append(f"\n🏆 GANADOR: {best['name']} (Score {best['score']:.2f})")
            phase6_output.append(f"   Acción: {best['action']}")
        else:
            phase6_output.append(f"\n⚠️ Ningún patrón alcanzó score >= 0.70")
            phase6_output.append("   → Activando Fallback Físico")
            
    except Exception as e:
        phase6_output.append(f"{Colors.FAIL}❌ Error pattern matching: {e}{Colors.ENDC}")
    print_phase(6, "PATTERN MATCHING", phase6_output)
    # ------------------------------------------------------------
    # FASE 7: VALIDACIÓN FÍSICA (Fallback) & FASE 8 MTF
    # ------------------------------------------------------------
    # Unificamos logica de decision final
    phase7_output = []
    final_decision = "REJECTED"
    final_reason = ""
    is_priority = False
    
    # 1. Determinar candidato PRIORITY
    if best_pattern:
        if 'LONG' in best_pattern['action'] or 'SHORT' in best_pattern['action']:
             phase7_output.append(f"Ruta Patrón: {best_pattern['name']} valida entrada.")
             is_priority = True
             final_reason = f"Pattern {best_pattern['name']}"
    else:
        # Fallback Logic
        phase7_output.append("Evaluando 4 Condiciones Físicas:")
        c1 = has_momentum # Velocity > 30% avg
        c2 = accel > 0 if velocity > 0 else accel < 0 # Accelerating logic simple
        c3 = runway_sprite != 'ALLEY' # Espacio
        
        # Check EMAs alignment
        c4 = False
        if velocity > 0:
             c4 = price > ema9 > ema20
        else:
             c4 = price < ema9 < ema20
             
        phase7_output.append(f"  1. Momentum: {'✅' if c1 else '❌'}")
        phase7_output.append(f"  2. Aceleración: {'✅' if c2 else '❌'}")
        phase7_output.append(f"  3. Runway Safe: {'✅' if c3 else '❌'}")
        phase7_output.append(f"  4. EMAs Align: {'✅' if c4 else '❌'}")
        
        passed = sum([c1, c2, c3, c4])
        phase7_output.append(f"Condiciones: {passed}/4")
        
        if passed >= 3: # Seamos estrictos para demo, visual_sniper pide 2+ pero con caveats
             is_priority = True
             final_reason = "Physics Fallback (3+ Conds)"
             phase7_output.append("✅ Fallback APROBADO")
        else:
             phase7_output.append("❌ Fallback RECHAZADO")
    print_phase(7, "VALIDACIÓN FÍSICA", phase7_output)
    # FASE 8: MTF
    phase8_output = []
    diag_solution = []
    diag_cause = "Unknown"
    diag_problem = "Rejected"
    
    if is_priority:
        phase8_output.append("Simulando análisis MTF (1D, 4H, 1H, 15M)...")
        # Logica MTF real
        try:
             # Fetch 4H (Critico)
             ohlcv_4h = conn.fetch_ohlcv(symbol, '4h', limit=5)
             c4h = [c[4] for c in ohlcv_4h]
             trend_4h = 'UP' if c4h[-1] > c4h[-3] else 'DOWN'
             phase8_output.append(f"Trend 4H: {trend_4h} (vs Trade Dir: {'LONG' if velocity > 0 else 'SHORT'})")
             
             trade_dir = 'UP' if velocity > 0 else 'DOWN'
             
             if trend_4h == trade_dir:
                 phase8_output.append("✅ 4H Alineado con Trade")
                 
                 # 1H Check
                 ohlcv_1h = conn.fetch_ohlcv(symbol, '1h', limit=5)
                 c1h = [c[4] for c in ohlcv_1h]
                 trend_1h = 'UP' if c1h[-1] > c1h[-3] else 'DOWN'
                 
                 if trend_1h == trend_4h:
                     phase8_output.append("✅ 1H Alineado")
                     final_decision = "PRIORITY"
                 else:
                     phase8_output.append("❌ 1H Desalineado")
                     final_decision = "DEVELOPING" # Downgrade
                     final_reason += " (MTF 1H Fail)"
                     diag_problem = "Fallo MTF"
                     diag_cause = "Temporalidad 1H va en contra"
                     diag_solution = ["Esperar alineación 1H", "Buscar otro par"]
             else:
                 phase8_output.append("❌ 4H Desalineado (BLOCK)")
                 final_decision = "REJECTED"
                 final_reason += " (MTF 4H Fail)"
                 diag_problem = "Tendencia Mayor Opuesta"
                 diag_cause = f"4H es {trend_4h} pero setup es {trade_dir}"
                 diag_solution = ["No operar contra tendencia 4H", "Esperar reversión confirmada"]
                 
        except Exception as e:
             phase8_output.append(f"❌ Error MTF: {e}")
    else:
        phase8_output.append("⏹️ Saltado (No es Priority)")
        diag_problem = "Setup Débil"
        diag_cause = "No cumple patrón ni condiciones físicas mínimas"
        diag_solution = ["Buscar monedas con más volumen", "Esperar setup claro"]
    print_phase(8, "MTF CONFIRMATION", phase8_output)
    # DECISIÓN FINAL
    color = Colors.GREEN if final_decision == 'PRIORITY' else Colors.FAIL
    print(f"\n{color}╔{'═'*60}╗{Colors.ENDC}")
    print(f"{color}║  {'✅' if final_decision=='PRIORITY' else '❌'} DECISIÓN FINAL: {final_decision:<29} ║{Colors.ENDC}")
    print(f"{color}║  Razón: {final_reason:<49} ║{Colors.ENDC}")
    print(f"{color}╚{'═'*60}╝{Colors.ENDC}\n")
    # DIAGNOSTICO RAÍZ
    if final_decision != 'PRIORITY':
        print(f"🔧 {Colors.WARNING}DIAGNÓSTICO RAÍZ:{Colors.ENDC}")
        print(f"┌{'─'*60}┐")
        print(f"│ PROBLEMA: {diag_problem:<48} │")
        print(f"│ CAUSA: {diag_cause:<51} │")
        print(f"│ SOLUCIÓN SUGERIDA:                             │")
        for sol in diag_solution:
            print(f"│  • {sol:<50} │")
        print(f"└{'─'*60}┘")
    duration = time.time() - start_time
    print(f"\n⏱️ Análisis completado en {duration:.2f} segundos")
def main():
    while True:
        try:
            symbol = input(f"\n{Colors.BLUE}Ingrese símbolo (ej: BTC, HANA) o 'q' para salir: {Colors.ENDC}").strip()
            if symbol.lower() in ['q', 'quit', 'exit']:
                break
            if not symbol: continue
            
            analyze_symbol(symbol)
