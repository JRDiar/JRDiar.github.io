class BotBrainRecorder:
    """
    Records BOT's decision-making process at each historical frame.
    
    TIME LADDER COMPLIANT:
    - Windowed context (LOOKBACK frames)
    - Real velocity/acceleration calculations
    - Trajectory-based analysis
    """
    
    LOOKBACK = 5
    MIN_CONTEXT = 4  # Minimum frames for reliable calculation
    
    def record_decision_at_frame(self, symbol, frame_idx, all_frames, direction):
        """
        Simulates bot's decision with FULL CONTEXT.
        
        Args:
            symbol: str
            frame_idx: int - Position in all_frames
            all_frames: list - ALL frames for context
            direction: str - BULLISH/BEARISH
        
        Returns:
            dict with confidence_factors, decision, velocities
        """
        try:
            start_idx = max(0, frame_idx - self.LOOKBACK)
            context_window = all_frames[start_idx:frame_idx + 1]
            current_frame = all_frames[frame_idx]
            
            velocities = self._calculate_velocities(context_window)
            
            trajectory = self._build_trajectory(
                current_frame.get('indicators', {}),
                velocities,
                direction
            )
            
            ladder_analysis = self._build_ladder_analysis(
                symbol,
                current_frame.get('indicators', {}),
                velocities,
                direction
            )
            
            from brain_vision_unified import UnifiedVision
            vision = UnifiedVision()
            result = vision.evaluate_confidence(trajectory, direction, ladder_analysis)
            
            return self._extract_confidence_breakdown(result, velocities)
            
        except Exception as e:
            return {
                'confidence_factors': [],
                'total_confidence': 0,
                'decision': 'ERROR',
                'decision_reason': f'Simulation failed: {str(e)[:50]}',
                'velocities': {},
                'error': str(e)
            }
    
    def _calculate_velocities(self, context_window):
        """Calculates REAL velocities from context window."""
        if len(context_window) < self.MIN_CONTEXT:
            return {
                'adx': 0, 'adx_accel': 0,
                'exhaustion': 0, 'exhaustion_accel': 0,
                'room': 0, 'momentum': 0, 'spread': 0,
                'insufficient_context': True
            }
        
        def safe_get(frames, key):
            """KeyError-safe extraction."""
            values = []
            for f in frames:
                ind = f.get('indicators', {})
                val = ind.get(key)
                values.append(val if val is not None else 0)
            return values
        
        def velocity(series):
            if len(series) < 2:
                return 0
            return (series[-1] - series[0]) / (len(series) - 1)
        
        def acceleration(series):
            if len(series) < 4:
                return 0
            mid = len(series) // 2
            return velocity(series[mid:]) - velocity(series[:mid])
        
        adx = safe_get(context_window, 'adx')
        exh = safe_get(context_window, 'exhaustion_pct')
        room = safe_get(context_window, 'room_pct')
        mom = safe_get(context_window, 'momentum_age')
        spr = safe_get(context_window, 'spread')
        
        return {
            'adx': round(velocity(adx), 2),
            'adx_accel': round(acceleration(adx), 3),
            'exhaustion': round(velocity(exh), 2),
            'exhaustion_accel': round(acceleration(exh), 3),
            'room': round(velocity(room), 3),
            'momentum': round(velocity(mom), 2),
            'spread': round(velocity(spr), 2),
            'insufficient_context': False
        }
    
    def _build_trajectory(self, indicators, velocities, direction):
        """Builds trajectory with REAL velocities."""
        return {
            'direction': direction,
            'adx_end': indicators.get('adx', 0),
            'adx_delta': velocities['adx'],
            'candle_ratio': 1.0,
            'momentum_end': indicators.get('momentum_age', 0),
            'momentum_delta': velocities['momentum'],
            'exhaustion_pct': indicators.get('exhaustion_pct', 50),
            'room_pct': indicators.get('room_pct', 0),
            'exhaustion_velocity': velocities['exhaustion'],
            'pivot_types': [],
            'position_pct': indicators.get('grid_position', 50)
        }
    
    def _build_ladder_analysis(self, symbol, indicators, velocities, direction):
        """Constructs ladder_analysis with velocity-aware data."""
        adx = indicators.get('adx', 0)
        di_plus = indicators.get('di_plus', 0)
        di_minus = indicators.get('di_minus', 0)
        
        if velocities['adx'] > 0.5:
            adx_trend = 'creciendo'
        elif velocities['adx'] < -0.5:
            adx_trend = 'decreciendo'
        else:
            adx_trend = 'estable'
        
        phase = self._get_phase_time_ladder(
            indicators.get('exhaustion_pct', 50),
            velocities['exhaustion'],
            velocities['exhaustion_accel']
        )
        
        return {
            '15m': {
                'direction': direction,
                'adx': {
                    'adx_current': adx,
                    'di_plus': di_plus,
                    'di_minus': di_minus,
                    'spread': abs(di_plus - di_minus),
                    'trend': adx_trend,
                    'velocity': velocities['adx']
                },
                'exhaustion_pct': indicators.get('exhaustion_pct', 50),
                'room_pct': indicators.get('room_pct', 0),
                'exhaustion_phase': phase,
                'exhaustion_velocity': velocities['exhaustion'],
                'momentum': {
                    'consecutive_count': indicators.get('momentum_age', 0),
                    'current_color': 'green' if indicators.get('momentum_age', 0) > 0 else 'red',
                    'velocity': velocities['momentum']
                }
            },
            '5m': {}, '1h': {}, '4h': {}
        }
    
    def _get_phase_time_ladder(self, exhaustion_pct, velocity, acceleration):
        """Phase from TRAJECTORY, not snapshot."""
        if acceleration > 0.5:
            return 'ACCELERATING_EXHAUSTION'
        elif acceleration < -0.5:
            return 'RECOVERING_FAST'
        elif velocity > 5:
            return 'BURNING_BRUTAL'
        elif velocity > 2:
            return 'BURNING_FAST'
        elif velocity > 0:
            return 'DEVELOPING'
        elif velocity < -3:
            return 'RECOVERING'
        elif velocity < 0:
            return 'COOLING'
        else:
            return 'STABLE'
    
    def _extract_confidence_breakdown(self, result, velocities):
        """Extracts confidence factors with velocity context."""
        action = result.get('action', 'WAIT')
        confidence = result.get('confidence', 0) or 0
        details = result.get('details', {})
        
        factors = []
        if 'confidence_factors' in details:
            for ft in details['confidence_factors']:
                if len(ft) >= 2:
                    factors.append({
                        'name': ft[0],
                        'value': round(ft[1], 1) if ft[1] else 0,
                        'reason': str(ft[2]) if len(ft) > 2 else ''
                    })
        
        if not factors:
            factors.append({'name': 'Total', 'value': confidence, 'reason': 'Details unavailable'})
        
        threshold = 40
        decision = 'ENTER' if action == 'OPERAR' else 'WAIT'
        reason = f"Confidence {confidence:.0f}% {'>' if confidence >= threshold else '<'} {threshold}%"
        
        return {
            'confidence_factors': factors,
            'total_confidence': round(confidence, 1),
            'decision': decision,
            'decision_reason': reason,
            'velocities': velocities
        }

def _detect_bugs_time_ladder(self):
        """TIME LADDER bug detection with accelerations."""
        bugs = []
        brain_frames = [f.get('bot_brain', {}) for f in self.frames if 'bot_brain' in f]
        
        if not brain_frames:
            return bugs
        
        # BUG 1: Factor never used
        factor_usage = {}
        for brain in brain_frames:
            for factor in brain.get('confidence_factors', []):
                name = factor.get('name', 'Unknown')
                value = factor.get('value', 0) or 0
                if name not in factor_usage:
                    factor_usage[name] = []
                factor_usage[name].append(value)
        
        for name, values in factor_usage.items():
            if values and all(v == 0 for v in values):
                bugs.append({
                    'bug': f'{name.upper()}_NEVER_USED',
                    'evidence': f'{name} = 0 in ALL {len(values)} frames',
                    'severity': 'HIGH'
                })
        
        # BUG 2: Entered while decelerating
        confidences = [b.get('total_confidence', 0) for b in brain_frames]
        if len(confidences) >= 4:
            velocities = [confidences[i] - confidences[i-1] for i in range(1, len(confidences))]
            mid = len(velocities) // 2
            v1 = sum(velocities[:mid]) / mid if mid > 0 else 0
            v2 = sum(velocities[mid:]) / (len(velocities) - mid) if len(velocities) > mid else 0
            
            if v2 < v1 - 2:
                bugs.append({
                    'bug': 'ENTERED_WHILE_DECELERATING',
                    'evidence': f'Velocity: {v1:.1f} → {v2:.1f}',
                    'severity': 'HIGH'
                })
        
        # BUG 3: Exhaustion accelerating at entry (FIXED TIMESTAMP)
        entry_brain = None
        entry_time = self.trade_data['entry_time']
        
        for frame in self.frames:
            if 'bot_brain' not in frame:
                continue
            
            ts = frame['timestamp']
            if isinstance(ts, str):
                from datetime import datetime
                ts = datetime.fromisoformat(ts)
            
            rel_min = int((ts - entry_time).total_seconds() / 60)
            if abs(rel_min) <= 5:
                entry_brain = frame['bot_brain']
                break
        
        if entry_brain:
            vel = entry_brain.get('velocities', {})
            exh_accel = vel.get('exhaustion_accel', 0)
            if exh_accel and exh_accel > 0.3:
                bugs.append({
                    'bug': 'EXHAUSTION_ACCELERATING_AT_ENTRY',
                    'evidence': f'Accel: {exh_accel:.2f}',
                    'severity': 'CRITICAL'
                })
        
        return bugs
    
    def _analyze_confidence_trajectory(self):
        """Analyzes confidence velocity and acceleration."""
        brain_frames = [f.get('bot_brain', {}) for f in self.frames if 'bot_brain' in f]
        if not brain_frames:
            return {}
        
        confidences = [b.get('total_confidence', 0) for b in brain_frames]
        velocities = [confidences[i] - confidences[i-1] for i in range(1, len(confidences))]
        accelerations = [velocities[i] - velocities[i-1] for i in range(1, len(velocities))]
        
        avg_vel = sum(velocities) / len(velocities) if velocities else 0
        avg_acc = sum(accelerations) / len(accelerations) if accelerations else 0
        
        if avg_acc > 0.5:
            interp = f"ACCELERATING (+{avg_acc:.1f}/frame²)"
        elif avg_acc < -0.5:
            interp = f"DECELERATING ({avg_acc:.1f}/frame²)"
        elif avg_vel > 2:
            interp = f"GROWING (+{avg_vel:.1f}/frame)"
        elif avg_vel < -2:
            interp = f"FALLING ({avg_vel:.1f}/frame)"
        else:
            interp = "STABLE"
        
        return {
            'values': [round(c, 1) for c in confidences],
            'velocities': [round(v, 1) for v in velocities],
            'avg_velocity': round(avg_vel, 2),
            'avg_acceleration': round(avg_acc, 2),
            'interpretation': interp
        }
    
    def _analyze_factor_contributions(self):
        """Analyzes factor TRENDS, not just averages."""
        brain_frames = [f.get('bot_brain', {}) for f in self.frames if 'bot_brain' in f]
        if not brain_frames:
            return {}
        
        factor_series = {}
        for brain in brain_frames:
            for factor in brain.get('confidence_factors', []):
                name = factor.get('name', 'Unknown')
                value = factor.get('value', 0) or 0
                if name not in factor_series:
                    factor_series[name] = []
                factor_series[name].append(value)
        
        contributions = {}
        for name, values in factor_series.items():
            if not values:
                continue
            
            avg_val = sum(values) / len(values)
            trend = (values[-1] - values[0]) / (len(values) - 1) if len(values) >= 2 else 0
            
            # FIXED: Safe variance calculation
            variance = sum((v - avg_val) ** 2 for v in values) / len(values) if values else 0
            
            contributions[name] = {
                'range': [round(min(values), 1), round(max(values), 1)],
                'avg': round(avg_val, 1),
                'trend': round(trend, 2),
                'impact': self._classify_impact_time_ladder(values, trend, variance)
            }
        
        return contributions
    
    def _classify_impact_time_ladder(self, values, trend, variance):
        """Trend-based classification."""
        if not values:
            return 'NONE'
        
        if max(values) == 0:
            return 'NONE - BUG: never used'
        
        if abs(trend) > 2:
            return 'GROWING' if trend > 0 else 'SHRINKING'
        
        if variance > 25:
            return 'VOLATILE'
        
        avg = sum(values) / len(values)
        if avg > 10:
            return 'HIGH'
        elif avg > 5:
            return 'MEDIUM'
        else:
            return 'LOW'
