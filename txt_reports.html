"""
Market Entity - Representación física del mercado.

El precio como objeto en movimiento (pelota en Pong).
Solo física, cero penalties, cero scoring.
"""

import logging
from collections import deque
from datetime import datetime

# Use custom BotLogger if available, else fallback to standard logging

    from utils.logger import logger

    logger = logging.getLogger(__name__)

class MarketEntity:
    """
    Representa el mercado como entidad física en movimiento.
    
    Filosofía:
        - Precio = Posición en el canvas
        - Velocidad = Cambio de posición por tiempo
        - Aceleración = Cambio de velocidad
        - Fuerza = ADX (potencia del movimiento)
    
    Similar a física de videojuego:
        position += velocity * dt
        velocity += acceleration * dt
    """
    
    def __init__(self, symbol, history_size=10):
        """
        Args:
            symbol: Par de trading (ej: 'BTC/USDT:USDT')
            history_size: Frames históricos para calcular física
        """
        self.symbol = symbol
        self.history_size = history_size
        
        # Estado físico
        self.position = 0.0          # Precio actual
        self.velocity = 0.0          # %/min (cambio de precio relativo)
        self.acceleration = 0.0      # %/min² (cambio de velocidad relativo)
        self.force = 0.0             # ADX (0-100)
        self.direction = 0           # -1 (bear), 0 (neutral), 1 (bull)
        
        # Historia (para calcular derivadas)
        self.price_history = deque(maxlen=history_size)
        self.time_history = deque(maxlen=history_size)
        
        # Última actualización
        self.last_update = None
        
        logger.info(f"MarketEntity creado: {symbol}")
    
    def update(self, price, timestamp=None):
        """
        Actualiza estado físico con nuevo precio.
        
        Calcula:
            - Velocidad (primera derivada)
            - Aceleración (segunda derivada)
            - Dirección (signo de velocidad)
        
        Args:
            price: Precio actual
            timestamp: Datetime (opcional, usa now() si no provee)
        """
        if timestamp is None:
            timestamp = datetime.now()
        
        # Guardar en historia
        self.price_history.append(price)
        self.time_history.append(timestamp)
        
        # Actualizar posición
        old_position = self.position
        self.position = price
        
        # Calcular VELOCIDAD (primera derivada)
        if len(self.price_history) >= 2:
            self.velocity = self._calculate_velocity()
        
        # Calcular ACELERACIÓN (segunda derivada)
        if len(self.price_history) >= 3:
            self.acceleration = self._calculate_acceleration()
        
        # Determinar DIRECCIÓN
        if self.velocity > 0.1:
            self.direction = 1   # Bullish
        elif self.velocity < -0.1:
            self.direction = -1  # Bearish
        else:
            self.direction = 0   # Lateral
        
        self.last_update = timestamp
        
        logger.debug(f"{self.symbol} | "
                    f"Position: ${self.position:.2f} | "
                    f"Velocity: {self.velocity:+.3f}%/min | "
                    f"Accel: {self.acceleration:+.2f}")
    
    def _calculate_velocity(self):
        """
        Calcula velocidad (primera derivada del precio) en PORCENTAJE.
        
        Returns:
            float: %/min (positivo = subiendo, negativo = bajando)
        """
        if len(self.price_history) < 2:
            return 0.0
        
        # Tomar últimos 2 puntos
        price_now = self.price_history[-1]
        price_prev = self.price_history[-2]
        
        time_now = self.time_history[-1]
        time_prev = self.time_history[-2]
        
        # Calcular delta
        delta_price = price_now - price_prev
        delta_time = (time_now - time_prev).total_seconds() / 60.0  # Minutos
        
        if delta_time > 0 and price_prev > 0:
            # Convertir a PORCENTAJE por minuto
            velocity_pct = (delta_price / price_prev) * 100
            velocity = velocity_pct / delta_time # Normalizado por tiempo
        else:
            velocity = 0.0
        
        return velocity
    
    def _calculate_acceleration(self):
        """
        Calcula aceleración (segunda derivada del precio).
        
        Returns:
            float: %/min² (positivo = acelerando, negativo = frenando)
        """
        if len(self.price_history) < 3:
            return 0.0
        
        # Calcular velocidades de últimos 2 intervalos
        velocities = []
        
        for i in range(len(self.price_history) - 1, max(0, len(self.price_history) - 3), -1):
            price_now = self.price_history[i]
            price_prev = self.price_history[i-1]
            
            time_now = self.time_history[i]
            time_prev = self.time_history[i-1]
            
            delta_price = price_now - price_prev
            delta_time = (time_now - time_prev).total_seconds() / 60.0
            
            if delta_time > 0 and price_prev > 0:
                # Calcular velocidad porcentual para cada intervalo
                v_pct = ((delta_price / price_prev) * 100) / delta_time
                velocities.append(v_pct)
        
        if len(velocities) >= 2:
            # Aceleración = cambio de velocidad (%/min²)
            acceleration = velocities[0] - velocities[1]
        else:
            acceleration = 0.0
        
        return acceleration
    
    def set_force(self, adx_value):
        """
        Establece fuerza del movimiento (ADX).
        
        Args:
            adx_value: Valor ADX (0-100)
        """
        self.force = max(0.0, min(100.0, adx_value))  # Clamp 0-100
    
    def is_accelerating(self):
        """¿Está acelerando? (aceleración positiva)"""
        return self.acceleration > 0
    
    def is_decelerating(self):
        """¿Está frenando? (aceleración negativa)"""
        return self.acceleration < 0
    
    def has_momentum(self):
        """
        ¿Tiene momentum? Comparado con SU PROPIA velocidad promedio.
        
        NO usa threshold fijo (0.03%, 0.05%, etc).
        Cada moneda se compara consigo misma.
        """
        if len(self.price_history) < 5:
            # Sin historial suficiente, considerar cualquier movimiento positivo
            return abs(self.velocity) > 0
        
        # Calcular velocidades históricas (en %) para el promedio
        historical_vels = []
        for i in range(len(self.price_history) - 1):
            prev = self.price_history[i]
            curr = self.price_history[i+1]
            
            # Necesitamos timestamps para ser precisos, pero aproximamos con los índices si es uniforme
            # Mejor recalculamos similar al método _calculate_velocity
            if prev > 0:
                # Delta simple, asumiendo time delta similar
                v_pct = ((curr - prev) / prev) * 100
                historical_vels.append(abs(v_pct))
        
        if not historical_vels:
            return abs(self.velocity) > 0
        
        # Promedio histórico de ESTA moneda (magnitud absoluta)
        avg_velocity = sum(historical_vels) / len(historical_vels)
        
        # CAMBIO: 0.8 (80%) en vez de 1.0 (100%)
        threshold_velocity = avg_velocity * 0.8
        
        return abs(self.velocity) > threshold_velocity
    
    def get_state(self):
        """
        Retorna estado completo como dict.
        
        Returns:
            dict: Estado físico completo
        """
        return {
            'position': self.position,
            'velocity': self.velocity,
            'acceleration': self.acceleration,
            'force': self.force,
            'direction': self.direction,
            'is_accelerating': self.is_accelerating(),
            'has_momentum': self.has_momentum(),
            'last_update': self.last_update.isoformat() if self.last_update else None
        }
    
    def __repr__(self):
        return (f"MarketEntity({self.symbol} | "
                f"Pos: ${self.position:.2f} | "
                f"Vel: {self.velocity:+.2f} | "
                f"Acc: {self.acceleration:+.2f} | "
                f"Force: {self.force:.1f})")

"""
Walls - Detector de obstáculos (Resistencias/Soportes).

Las paredes del juego donde el precio rebota.

import logging

class Walls:
    """
    Detecta y gestiona resistencias y soportes.
    
    Como paredes en un videojuego:
        - Resistencia = Pared superior
        - Soporte = Pared inferior
        - Precio rebota en estas paredes
    """
    
    def __init__(self, symbol: str, lookback_periods: int = 50):
        """
        Args:
            symbol: Par de trading
            lookback_periods: Períodos históricos para detectar pivotes
        """
        self.symbol = symbol
        self.lookback_periods = lookback_periods
        
        # Obstáculos detectados
        self.resistances: List[float] = []
        self.supports: List[float] = []
        
        logger.info(f"Walls creado: {symbol}")
    
    def update(self, ohlc_data: List[dict]):
        """
        Actualiza detección de resistencias/soportes.
        
        Método simple: Encuentra pivotes (máximos/mínimos locales).
        
        Args:
            ohlc_data: Lista de velas [{'high': X, 'low': X}, ...]
        """
        if len(ohlc_data) < 5:
            logger.warning("Datos insuficientes para detectar walls")
            return
        
        # Detectar pivotes
        highs = [candle['high'] for candle in ohlc_data]
        lows = [candle['low'] for candle in ohlc_data]
        
        # Resistencias = máximos locales
        self.resistances = self._find_local_maxima(highs)
        
        # Soportes = mínimos locales
        self.supports = self._find_local_minima(lows)
        
        logger.debug(f"Walls actualizadas: {len(self.resistances)} R, {len(self.supports)} S")
    
    def _find_local_maxima(self, values: List[float], window: int = 3) -> List[float]:
        """
        Encuentra máximos locales (pivotes high).
        
        Un máximo local es un valor mayor que sus N vecinos.
        
        Args:
            values: Lista de valores
            window: Ventana de comparación (3 = compara con 3 a cada lado)
        
        Returns:
            Lista de valores que son máximos locales
        """
        maxima = []
        
        for i in range(window, len(values) - window):
            is_local_max = True
            
            # Comparar con vecinos
            for j in range(1, window + 1):
                if values[i] <= values[i - j] or values[i] <= values[i + j]:
                    is_local_max = False
                    break
            
            if is_local_max:
                maxima.append(values[i])
        
        # Ordenar y eliminar duplicados cercanos
        maxima = sorted(set(maxima), reverse=True)
        
        return maxima
    
    def _find_local_minima(self, values: List[float], window: int = 3) -> List[float]:
        """
        Encuentra mínimos locales (pivotes low).
        
        Similar a maxima pero busca menores.
        """
        minima = []
        
        for i in range(window, len(values) - window):
            is_local_min = True
            
            for j in range(1, window + 1):
                if values[i] >= values[i - j] or values[i] >= values[i + j]:
                    is_local_min = False
                    break
            
            if is_local_min:
                minima.append(values[i])
        
        minima = sorted(set(minima))
        
        return minima
    
    def detect_structure(self) -> dict:
        """
        Detecta estructura de pivotes (HH/HL vs LL/LH).
        
        Returns:
            dict: trend, counts, last pivots
        """
        if len(self.resistances) < 2 or len(self.supports) < 2:
            return {
                'trend': 'UNKNOWN',
                'last_hh': None,
                'last_hl': None,
                'hh_count': 0,
                'hl_count': 0,
                'll_count': 0
            }
        
        # Resistencias ordenadas (highs)
        highs = sorted(self.resistances)
        lows = sorted(self.supports)
        
        # Contar Higher Highs
        hh_count = sum(1 for i in range(1, len(highs)) if highs[i] > highs[i-1])
        
        # Contar Higher Lows
        hl_count = sum(1 for i in range(1, len(lows)) if lows[i] > lows[i-1])
        
        # Contar Lower Lows
        ll_count = sum(1 for i in range(1, len(lows)) if lows[i] < lows[i-1])
        
        # Contar Lower Highs
        lh_count = sum(1 for i in range(1, len(highs)) if highs[i] < highs[i-1])
        
        # Determinar tendencia
        if hh_count >= 2 and hl_count >= 2:
            trend = 'BULLISH'
        elif ll_count >= 2 and lh_count >= 1:
            trend = 'BEARISH'
        else:
            trend = 'LATERAL'
        
        return {
            'trend': trend,
            'last_hh': highs[-1] if highs else None,
            'last_hl': lows[-1] if lows else None,
            'last_ll': lows[0] if lows else None,
            'last_lh': highs[0] if highs else None,
            'hh_count': hh_count,
            'hl_count': hl_count,
            'll_count': ll_count,
            'lh_count': lh_count
        }
    
    def calculate_development_potential(
        self, 
        current_price: float, 
        structure: dict
    ) -> dict:
        """
        Calcula cuánto le queda por desarrollar al precio.
        
        Returns:
            dict: position_pct, room_to_develop_pct, stage
        """
        trend = structure.get('trend', 'UNKNOWN')
        
        if trend == 'BULLISH':
            hl = structure.get('last_hl')
            hh = structure.get('last_hh')
            
            if not hl or not hh or hh <= hl:
                return {'position_pct': 50, 'room_to_develop_pct': 50, 'stage': 'UNKNOWN'}
            
            total_range = hh - hl
            current_progress = max(0, current_price - hl)
            position_pct = min(100, (current_progress / total_range) * 100)
            room_to_develop_pct = 100 - position_pct
            
            if position_pct <= 30:
                stage = 'EARLY'
            elif position_pct <= 70:
                stage = 'MID'
            else:
                stage = 'LATE'
            
            return {
                'position_pct': position_pct,
                'room_to_develop_pct': room_to_develop_pct,
                'stage': stage,
                'range_low': hl,
                'range_high': hh
            }
        
        elif trend == 'BEARISH':
            lh = structure.get('last_lh') or structure.get('last_hh')
            ll = structure.get('last_ll')
            
            if not lh or not ll or lh <= ll:
                return {'position_pct': 50, 'room_to_develop_pct': 50, 'stage': 'UNKNOWN'}
            
            total_range = lh - ll
            current_progress = max(0, lh - current_price)
            position_pct = min(100, (current_progress / total_range) * 100)
            room_to_develop_pct = 100 - position_pct
            
            if position_pct <= 30:
                stage = 'EARLY'
            elif position_pct <= 70:
                stage = 'MID'
            else:
                stage = 'LATE'
            
            return {
                'position_pct': position_pct,
                'room_to_develop_pct': room_to_develop_pct,
                'stage': stage,
                'range_high': lh,
                'range_low': ll
            }
        
        else:
            return {
                'position_pct': 50,
                'room_to_develop_pct': 0,
                'stage': 'LATERAL'
            }
    
    def find_nearest_wall(
        self, 
        current_price: float, 
        direction: int
    ) -> Optional[Tuple[float, str, float]]:
        """
        Encuentra la pared más cercana en la dirección del movimiento.
        
        Args:
            current_price: Precio actual
            direction: 1 (bullish busca resistencia), -1 (bearish busca soporte)
        
        Returns:
            Tuple: (precio_wall, tipo, distancia) o None si no hay
            Ejemplo: (98000, 'resistance', 2000)
        """
        if direction > 0:
            # Buscar resistencia arriba
            walls_above = [r for r in self.resistances if r > current_price]
            
            if walls_above:
                nearest = min(walls_above)
                distance = nearest - current_price
                return (nearest, 'resistance', distance)
        
        elif direction < 0:
            # Buscar soporte abajo
            walls_below = [s for s in self.supports if s < current_price]
            
            if walls_below:
                nearest = max(walls_below)
                distance = current_price - nearest
                return (nearest, 'support', distance)
        
        return None
    
    def calculate_runway(
        self, 
        current_price: float, 
        target_wall: float, 
        velocity: float
    ) -> float:
        """
        Calcula tiempo hasta colisión con pared.
        
        velocity viene en %/min (desde MarketEntity).
        
        Args:
            current_price: Precio actual
            target_wall: Precio de la pared objetivo
            velocity: Velocidad (%/min)
        
        Returns:

            time_minutes = distance_pct / abs(velocity)

        direction: int,
