def detect_ignition(symbol, timeframe='5m'):
    """
    Detecta si una moneda está 'arrancando' antes de explotar.
    Busca:
    1. Volumen actual > 2x Volumen promedio (Anomalía)
    2. Vela actual verde y rompiendo el máximo de la anterior.
    """
    storage = get_frame_storage(symbol, timeframe)
    df = storage.df
    
    if len(df) < 20: return {'is_igniting': False}

    # Datos vela actual (sin cerrar) y anteriores
    current_vol = df['volume'].iloc[-1]
    avg_vol = df['volume'].iloc[-20:-1].mean() # Promedio últimas 20 velas
    
    current_close = df['close'].iloc[-1]
    current_open = df['open'].iloc[-1]
    prev_high = df['high'].iloc[-2]
    
    # 1. CÁLCULO DE FUERZA DE VOLUMEN
    vol_ratio = 0
    if avg_vol > 0:
        vol_ratio = current_vol / avg_vol
    
    # 2. CÁLCULO DE VELOCIDAD DE PRECIO (Vela actual)
    price_change_pct = ((current_close - current_open) / current_open) * 100
    
    # 3. LOGICA DE IGNICIÓN (El secreto de la anticipación)
    # Si hay mucho volumen (x1.5) Y el precio está subiendo > 0.3% en 5m
    is_volume_push = vol_ratio > 1.5
    is_price_push = price_change_pct > 0.35 
    is_breaking_structure = current_close > prev_high

    if is_volume_push and is_price_push and is_breaking_structure:
        return {
            'is_igniting': True,
            'reason': f'🔥 IGNICIÓN: Vol x{vol_ratio:.1f}, Price +{price_change_pct:.2f}%',
            'quality': 'HIGH' if vol_ratio > 3.0 else 'MEDIUM'

"""
brain_frames.py - DETECTOR DE IGNICIÓN (NUEVO)

import numpy as np

def detect_ignition(symbol, timeframe='5m'):
    """
    Busca la 'Chispa':
    1. Volumen anormalmente alto (entran las ballenas).
    2. Vela rompiendo máximos anteriores.
    3. Velocidad instantánea positiva.
    """
    storage = get_frame_storage(symbol, timeframe)
    df = storage.df
    
    if len(df) < 25: return {'is_igniting': False}

    # Datos de la vela ACTUAL (aún formándose) y anteriores
    curr_vol = df['volume'].iloc[-1]
    avg_vol = df['volume'].iloc[-21:-1].mean() # Promedio últimas 20 velas
    
    curr_close = df['close'].iloc[-1]
    curr_open = df['open'].iloc[-1]
    prev_high = df['high'].iloc[-2]
    
    # 1. FACTOR DE VOLUMEN (¿Hay dinero entrando?)
    # Si el volumen actual es 2 veces el promedio, es una anomalía seria.
    vol_factor = 0
    if avg_vol > 0:
        vol_factor = curr_vol / avg_vol
    
    # 2. VELOCIDAD DE LA VELA
    candle_body_pct = ((curr_close - curr_open) / curr_open) * 100
    
    # 3. RUPTURA DE ESTRUCTURA LOCAL
    breaking_high = curr_close > prev_high

    # === LA FÓRMULA DE IGNICIÓN ===
    # Caso A: Mucho volumen y vela verde creciendo (Inicio de explosión)
    igniting_strong = vol_factor > 2.0 and candle_body_pct > 0.25
    
    # Caso B: Volumen decente pero rompiendo estructura con fuerza
    igniting_breakout = vol_factor > 1.2 and candle_body_pct > 0.40 and breaking_high

    if igniting_strong or igniting_breakout:
        return {
            'is_igniting': True,
            'reason': f'🔥 IGNICIÓN 5m: Vol x{vol_factor:.1f}, Vela +{candle_body_pct:.2f}%',
            'quality': 'NUCLEAR' if vol_factor > 3.0 else 'HIGH'
        }
        
    return {'is_igniting': False, 'vol_factor': vol_factor}

def manage_zombie_trades(trade_data):
    """
    Si entramos esperando una explosión y no ocurre, SALIR.
    No te cases con la operación.
    """
    start_time = trade_data['entry_time']
    minutes_open = (time.time() - start_time) / 60
    pnl_pct = trade_data['current_pnl_pct']
    
    # REGLA DEL TIEMPO (Time Decay)
    
    # 1. A los 7 minutos: Si estamos perdiendo, CORTAR. 
    # (La ignición falló, es una falsa ruptura).
    if minutes_open > 7 and pnl_pct < -0.15:
        return {'close': True, 'reason': 'Ignición fallida (7m)'}
        
    # 2. A los 15 minutos: Si no ganamos al menos 0.3%, CORTAR.
    # (El precio se lateralizó, es un Zombie).
    if minutes_open > 15 and pnl_pct < 0.3:
        return {'close': True, 'reason': 'Zombie Kill (Lateral >15m)'}
        
    # 3. A los 30 minutos: Hard Close si no es rentable.
    if minutes_open > 30 and pnl_pct < 0.5:
         return {'close': True, 'reason': 'Tiempo máximo excedido'}
         
    return {'close': False}
