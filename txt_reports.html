if decision['decision'] == 'ENTER':
    
    # ===================================================================
    # SISTEMA DUAL: Position Size según Confianza
    # ===================================================================
    # Joel: "Más volumen pero gestión de riesgo inteligente"
    #
    # TIER ALTO    (60-100%): 15% capital - Setups perfectos
    # TIER MEDIO   (50-60%):  10% capital - Setups buenos (mayoría)
    # TIER BAJO    (40-50%):   5% capital - Setups mediocres (exploración)
    # ===================================================================
    
    confidence = decision['confidence']
    
    if confidence >= 60:
        position_size = 0.15  # Setups perfectos (como threshold 60 antiguo)
        risk_tier = 'ALTO'
    elif confidence >= 50:
        position_size = 0.10  # Setups buenos (nuevo rango)
        risk_tier = 'MEDIO'
    else:  # 40-50
        position_size = 0.05  # Setups mediocres (exploración conservadora)
        risk_tier = 'BAJO'
    
    print(f"   💰 Position Size: {position_size*100:.0f}% (Tier: {risk_tier}, Conf: {confidence:.0f}%)")
    
    return {
        'action': 'OPERAR',
        'trade_type': 'LONG' if ladder_analysis.get('15m', {}).get('direction') == 'BULLISH' else 'SHORT',
        'confidence': confidence,
        'pattern': 'UNIFIED_VISION',
        'reasons': [decision['reason']],
        'full_analysis': {
            'ladder_data': ladder_analysis, 
            'patterns': patterns_result,
            'force_result': force_result,
            'unified_vision': decision
        },
        'position_size': position_size,  # ← DINÁMICO
        'risk_tier': risk_tier,  # ← NUEVO campo para tracking
        'strategy': determine_strategy(patterns_result, ladder_analysis)
    }
Justificación:
60%+ (ALTO): Misma exposición que threshold 60 antiguo (15%)

Riesgo #1: Overtrading

Riesgo #2: Trades BAJO calidad
Síntoma: Win rate <45% en tier BAJO
Mitigación: Monitorear por 48h, si WR<45% → subir threshold a 45%

Riesgo #3: Capital insuficiente
Síntoma: No puede abrir trades (capital bloqueado)
Mitigación: Max 7.75% capital en riesgo simultáneo

Riesgo #4: Más pérdidas consecutivas
