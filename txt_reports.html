class FrameBuffer:
    def __init__(self, size=10):
        self.frames = deque(maxlen=size)  # Últimos 10 minutos
    
    def add_frame(self, adx, price, candle_size, momentum):
        self.frames.append({
            'timestamp': now(),
            'adx': adx,
            'price': price, 
            'candle_size': candle_size,
            'momentum': momentum
        })
    
    def detect_acceleration(self):
        """Ve la película: ¿acelera o desacelera?"""
        if len(self.frames) < 3:
            return None
            
        velocities = []
        for i in range(len(self.frames)-1):
            v = self.frames[i+1].price - self.frames[i].price
            velocities.append(v)
        
        # ¿Velocidad creciente? = aceleración
        return velocities[-1] > velocities[0]
    
    def detect_candle_growth(self):
        """Ve la película: ¿velas crecen?"""
        sizes = [f.candle_size for f in self.frames]
        return all(sizes[i] < sizes[i+1] for i in range(len(sizes)-1))

    def __init__(self, size=10):
        self.frames = deque(maxlen=size)  # Últimos 10 minutos
    
    def add_frame(self, adx, price, candle_size, momentum):
        self.frames.append({
            'timestamp': now(),
            'adx': adx,
            'price': price, 
            'candle_size': candle_size,
            'momentum': momentum
        })
    
    def detect_acceleration(self):
        """Ve la película: ¿acelera o desacelera?"""
        if len(self.frames) < 3:
            return None
            
        velocities = []
        for i in range(len(self.frames)-1):
            v = self.frames[i+1].price - self.frames[i].price
            velocities.append(v)
        
        # ¿Velocidad creciente? = aceleración
        return velocities[-1] > velocities[0]
    
    def detect_candle_growth(self):
        """Ve la película: ¿velas crecen?"""
        sizes = [f.candle_size for f in self.frames]
        return all(sizes[i] < sizes[i+1] for i in range(len(sizes)-1))
