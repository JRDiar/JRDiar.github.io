def calculate_profit_potential(current_price, target, stop):
    profit_pct = abs(target - current_price) / current_price * 100
    risk_pct = abs(current_price - stop) / current_price * 100
    return {
        'profit_potential': profit_pct,
        'risk': risk_pct,
        'rr_ratio': profit_pct / risk_pct
    }
