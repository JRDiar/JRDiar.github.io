Python
"""
================================================================================
TRADE DNA EXTRACTOR - Ingeniería Inversa de Setups Exitosos
================================================================================
Propósito:
    Permite al usuario seleccionar un rango de tiempo histórico donde ocurrió
    un movimiento ideal (la "Fotografía") y extraer sus métricas técnicas
    exactas (ADN) para usarlas como plantilla en el Sniper.

Uso:
    python tools/extract_trade_dna.py --symbol LUNC/USDT --start "2023-10-27 06:00" --end "2023-10-27 06:15"

Autor: BrainTrader System
"""

import sys
import os
import json
import pandas as pd
import numpy as np
from datetime import datetime, timedelta
import argparse

# Añadir directorio raíz al path para importar módulos del cerebro
sys.path.append(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

from exchange_connection import fetch_ohlcv_history
from brain_grid import grid
from brain_calculator import (
    calculate_adx_with_memory,
    calculate_momentum_with_memory,
    calculate_zigzag
)

def extract_dna(symbol, start_str, end_str):
    """
    Extrae el ADN técnico de un movimiento histórico.
    """
    print(f"\n🧬 INICIANDO EXTRACCIÓN DE ADN PARA {symbol}")
    print(f"   📅 Rango: {start_str} a {end_str}")
    print("=" * 60)

    # 1. CONVERTIR FECHAS
    try:
        start_dt = datetime.strptime(start_str, "%Y-%m-%d %H:%M")
        end_dt = datetime.strptime(end_str, "%Y-%m-%d %H:%M")
    except ValueError:
        print("❌ Error de formato de fecha. Usa: 'YYYY-MM-DD HH:MM'")
        return

    # 2. OBTENER DATOS HISTÓRICOS (BUFFER GRANDE PARA CÁLCULOS)
    # Necesitamos datos ANTES del inicio para que los indicadores (EMA, RSI) estén calientes.
    # Bajamos 200 velas de 5m antes del start_dt.
    fetch_start = start_dt - timedelta(minutes=5*200)
    
    # Timeframes a analizar (Time Ladder)
    tfs = ['1h', '15m', '5m']
    dna_data = {}

    for tf in tfs:
        print(f"   📡 Descargando datos {tf}...")
        ohlcv = fetch_ohlcv_history(symbol, tf, start_dt=fetch_start, end_dt=end_dt)
        
        if not ohlcv or len(ohlcv) < 50:
            print(f"❌ Datos insuficientes para {tf}")
            return

        df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
        df['timestamp'] = pd.to_datetime(df['timestamp'], unit='ms')
        
        # Filtrar solo el rango "Golden" seleccionado por el usuario para la foto final
        # Pero calculamos indicadores sobre TODO el df primero
        
        # --- CÁLCULOS TÉCNICOS ---
        adx_data = calculate_adx_with_memory(df)
        mom_data = calculate_momentum_with_memory(df)
        zz_data = calculate_zigzag(df, threshold_pct=2.0) # ZigZag sensible
        
        # Recortar al momento exacto del final del movimiento (La "Foto Finish")
        mask = (df['timestamp'] >= start_dt) & (df['timestamp'] <= end_dt)
        df_window = df.loc[mask]
        
        if df_window.empty:
            print(f"⚠️ No hay velas en el rango exacto para {tf}")
            continue

        # Tomamos el PUNTO MÁXIMO del movimiento para ver a qué llegó
        # O el punto de inicio para ver cómo empezó.
        # Estrategia: Tomar el promedio de las métricas durante la subida.
        
        avg_adx = np.mean(adx_data['adx_history'][-len(df_window):])
        max_adx = np.max(adx_data['adx_history'][-len(df_window):])
        start_adx = adx_data['adx_history'][-len(df_window)]
        
        mom_colors = mom_data['color_history'][-len(df_window):]
        dominant_color = max(set(mom_colors), key=mom_colors.count)
        
        # ZigZag del movimiento
        # Buscamos el último swing dentro de la ventana
        zz_pivots = [v for i, v in enumerate(zz_data['values']) if zz_data['pivots'][i] >= df_window.index[0]]
        if len(zz_pivots) >= 2:
            swing_pct = abs((zz_pivots[-1] - zz_pivots[0]) / zz_pivots[0]) * 100
        else:
            # Si no hay pivotes, usamos Open vs Close de la ventana
            swing_pct = abs((df_window['close'].iloc[-1] - df_window['open'].iloc[0]) / df_window['open'].iloc[0]) * 100

        # --- GUARDAR ADN DEL TIMEFRAME ---
        dna_data[tf] = {
            'adx_avg': round(avg_adx, 2),
            'adx_max': round(max_adx, 2),
            'adx_start': round(start_adx, 2),
            'momentum_color': dominant_color,
            'swing_size_pct': round(swing_pct, 2),
            'volume_trend': 'RISING' if df_window['volume'].iloc[-1] > df_window['volume'].iloc[0] else 'FALLING'
        }

    # 3. EXTRAER GRID 5M (La Estructura Física)
    print("   🌐 Analizando Estructura Grid 5M...")
    # Usamos el DF de 5m recortado al final para ver cómo estaba el Grid
    # Reconstruimos el Grid object simulado
    # Nota: Grid necesita un 'ladder' simulado, usaremos los datos crudos
    
    # Calculamos dimensiones del Grid en el momento pico
    grid_snapshot = {}
    
    # Simulamos lo que grid._build_dimensions haría
    df_5m_full = pd.DataFrame(fetch_ohlcv_history(symbol, '5m', start_dt=fetch_start, end_dt=end_dt), columns=['ts','o','h','l','c','v'])
    highs = df_5m_full['h'].rolling(20).max()
    lows = df_5m_full['l'].rolling(20).min()
    techo = highs.iloc[-1]
    piso = lows.iloc[-1]
    close = df_5m_full['c'].iloc[-1]
    
    # Posición relativa (0 a 1)
    posicion_grid = (close - piso) / (techo - piso) if (techo - piso) > 0 else 0.5
    
    dna_data['grid_structure'] = {
        'posicion_relativa': round(posicion_grid, 2), # 0.1=Piso, 0.9=Techo
        'ancho_grid_pct': round((techo - piso)/piso * 100, 2)
    }

    # 4. GENERAR ARCHIVO JSON
    dna_filename = f"dna_{symbol.replace('/','')}_{start_dt.strftime('%Y%m%d_%H%M')}.json"
    dna_path = os.path.join(os.path.dirname(__file__), dna_filename)
    
    final_dna = {
        'metadata': {
            'symbol': symbol,
            'start': start_str,
            'end': end_str,
            'description': 'User defined successful setup'
        },
        'dna_metrics': dna_data,
        'trigger_rules': {
            'min_adx_15m': dna_data['15m']['adx_start'], # El ADX que tenía al empezar
            'required_momentum': dna_data['15m']['momentum_color'],
            'min_grid_width': dna_data['grid_structure']['ancho_grid_pct'] * 0.8, # 80% del ancho original
            'allow_buying_top': dna_data['grid_structure']['posicion_relativa'] > 0.8 # ¿Compró en techo?
        }
    }

    with open(dna_path, 'w') as f:
        json.dump(final_dna, f, indent=4)

    print("=" * 60)
    print(f"✅ ADN EXTRAÍDO EXITOSAMENTE: {dna_filename}")
    print(f"📄 Contenido del ADN:")
    print(json.dumps(final_dna['trigger_rules'], indent=2))
    print("\n👉 Copia este archivo a la carpeta 'strategies/dna/' para que el Sniper lo use.")

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description='Extract Trade DNA')
    parser.add_argument('--symbol', type=str, required=True, help='Ej: LUNC/USDT')
    parser.add_argument('--start', type=str, required=True, help='YYYY-MM-DD HH:MM')
    parser.add_argument('--end', type=str, required=True, help='YYYY-MM-DD HH:MM')
    
    args = parser.parse_args()
    
    extract_dna(args.symbol, args.start, args.end)
