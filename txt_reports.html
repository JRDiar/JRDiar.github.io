def validate_5m_entry_conditions(symbol, direction='BULLISH'):
    """
    Checklist quirúrgico de 5M para entrada precisa.
    NO rechaza - solo informa qué falta.
    
    Returns:
        dict: {
            'ready': bool (True si TODAS las condiciones OK),
            'conditions': {
                'momentum_green': {'ok': bool, 'value': str, 'reason': str},
                'adx_accompanied': {'ok': bool, 'value': str, 'reason': str},
                'not_in_correction': {'ok': bool, 'value': str, 'reason': str},
                'above_ema_support': {'ok': bool, 'value': str, 'reason': str}
            },
            'score': int (0-100, cuántas condiciones OK),
            'missing': [lista de condiciones faltantes]
        }
    """
    # 1. Fetch Data Fresca (5M)
    ohlcv = fetch_ohlcv(symbol, '5m', 50)
    if not ohlcv or len(ohlcv) < 20:
        return {'ready': False, 'conditions': {}, 'score': 0, 'missing': ['No Data']}
        
    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    current_price = float(df['close'].iloc[-1])
    
    # 2. Calcular Indicadores
    mom_data = calculate_momentum_with_memory(df, lookback=5)
    adx_data = calculate_adx_with_memory(df, period=14, lookback=5)
    emas_data = calculate_emas_with_memory(df, periods=[21], lookback=5)
    
    # 3. Validar Condiciones
    conditions = {}
    missing = []
    score_count = 0
    
    # A. Momentum Green
    mom_color = mom_data['color_current']
    if direction == 'BULLISH':
        mom_ok = mom_color in ['green', 'lime']
    else:
        mom_ok = mom_color in ['red', 'maroon']
        
    conditions['momentum_green'] = {
        'ok': mom_ok,
        'value': mom_color,
        'reason': f"Momentum is {mom_color}"
    }
    if mom_ok: score_count += 1
    else: missing.append(f"Momentum {mom_color} (Need {direction} colors)")

    # B. ADX Accompanied (Fuerza + Dirección)
    adx_trend = adx_data['adx_trend']
    di_trend = adx_data['di_plus_trend'] if direction == 'BULLISH' else adx_data['di_minus_trend']
    
    adx_ok = (adx_trend == 'creciendo' and di_trend == 'creciendo')
    
    conditions['adx_accompanied'] = {
        'ok': adx_ok,
        'value': f"ADX:{adx_trend} DI:{di_trend}",
        'reason': "ADX and DI must both be growing"
    }
    if adx_ok: score_count += 1
    else: missing.append(f"ADX/DI not synced (ADX:{adx_trend}, DI:{di_trend})")

    # C. Not in Correction (Velas)
    closes = df['close'].values[-3:]
    opens = df['open'].values[-3:]
    
    is_correction = True
    for c, o in zip(closes, opens):
        if direction == 'BULLISH':
            if c > o: # Vela verde
                is_correction = False
                break
        else:
            if c < o: # Vela roja
                is_correction = False
                break
                
    conditions['not_in_correction'] = {
        'ok': not is_correction,
        'value': 'Correction' if is_correction else 'Clean',
        'reason': "Last 3 candles cannot be all against trend"
    }
    if not is_correction: score_count += 1
    else: missing.append("Price in active correction (3 candles against trend)")

    # D. Above EMA Support (Soporte)
    ema21 = emas_data['emas'].get(21, 0)
    
    if ema21 > 0:
        dist_pct = (current_price - ema21) / ema21 * 100
        if direction == 'BULLISH':
            ema_ok = dist_pct > -0.5 # Puede estar apenas abajo (retest)
        else:
            ema_ok = dist_pct < 0.5
    else:
        dist_pct = 0
        ema_ok = True # Sin EMA, no penalizar
        
    conditions['above_ema_support'] = {
        'ok': ema_ok,
        'value': f"{dist_pct:.2f}% vs EMA21",
        'reason': "Price must hold EMA21 support"
    }
    if ema_ok: score_count += 1
    else: missing.append(f"Lost EMA21 support (Dist: {dist_pct:.2f}%)")

    # Resultado Final
    final_score = score_count * 25
    ready = (score_count == 4)
    
    return {
        'ready': ready,
        'conditions': conditions,
        'score': final_score,
        'missing': missing
    }
