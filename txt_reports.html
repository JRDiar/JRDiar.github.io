    def __init__(self):
        self.timeframe_weights = {
            '1M': 0.10,   # Micro - Entrada inmediata

    def analyze_complete_ladder(self, symbol):
        """
        Análisis completo del Time Ladder con inteligencia real
        """
        # 1. Recolectar datos de todos los timeframes
        ladder_data = self.collect_ladder_data(symbol)

        confluence = self.calculate_smart_confluence(ladder_data)

            'ladder_data': ladder_data,
            'patterns': patterns,
            'confluence': confluence,
            'projection': projection,
            'quality_score': quality_score,
            'decision': self.make_intelligent_decision(quality_score)
        }
    
    def collect_ladder_data(self, symbol):
        """
        Recolecta y enriquece datos de cada timeframe
        """
        data = {}
        
        # Análisis base de cada timeframe
        raw_analysis = {
            '1D': analyze_1d(symbol) if ENABLE_ANALYSIS_1D else None,
            '4H': analyze_4h(symbol),
            '1H': analyze_1h(symbol),
            '15M': analyze_15m(symbol, strict=False),
            '5M': analyze_5m(symbol),
            '1M': analyze_trend_evolution(symbol)
        }
        
        # Enriquecer cada análisis con contexto
        for tf, analysis in raw_analysis.items():
            if analysis is None:
                continue
                
            enriched = self.enrich_timeframe_analysis(tf, analysis, raw_analysis)
            data[tf] = enriched
            
        return data
    
    def enrich_timeframe_analysis(self, timeframe, analysis, all_analyses):
        """
        Enriquece el análisis de un timeframe con contexto
        """
        enriched = analysis.copy()
        
        # Agregar fuerza relativa
        adx = self._extract_adx(analysis)
        enriched['strength_category'] = self.categorize_strength(adx)
        
        # Agregar madurez del movimiento
        momentum_age = analysis.get('momentum_age', 0)
        enriched['maturity'] = self.categorize_maturity(momentum_age)
        
        # Agregar alineación con timeframe superior
        enriched['alignment'] = self.check_alignment_with_higher_tf(
            timeframe, analysis, all_analyses
        )
        
        # Detectar características especiales
        enriched['special_features'] = self.detect_special_features(
            timeframe, analysis
        )
        
        return enriched
    
    def detect_ladder_patterns(self, ladder_data):

        Detecta patrones específicos en el Time Ladder

        patterns = []

                'impact': 'VERY_HIGH'

        # 2. PATRÓN: Compresión Multi-Timeframe
        if self.check_compression_pattern(ladder_data):
            patterns.append({
                'name': 'MULTI_TF_COMPRESSION',
                'confidence': 85,
                'description': 'Volatilidad comprimida, explosión inminente',
                'impact': 'HIGH'
            })
        
        # 3. PATRÓN: Divergencia Oculta
        if self.check_hidden_divergence(ladder_data):
            patterns.append({
                'name': 'HIDDEN_DIVERGENCE',
                'confidence': 75,
                'description': 'TF menores fuertes mientras mayores consolidan',
                'impact': 'MEDIUM'
            })

        if self.check_nascent_momentum(ladder_data):

                'name': 'NASCENT_MOMENTUM',
                'confidence': 80,
                'description': 'Momentum iniciando en TF menores',
                'impact': 'HIGH'
            })
        
        # 5. PATRÓN: Confirmación Institucional
        if self.check_institutional_pattern(ladder_data):
            patterns.append({
                'name': 'INSTITUTIONAL_ENTRY',
                'confidence': 90,
                'description': '4H y 1H con volumen y ADX excepcional',
                'impact': 'VERY_HIGH'
            })
        
        return patterns
    
    def calculate_smart_confluence(self, ladder_data):
        """
        Calcula confluencia ponderada inteligente
        """
        confluence_score = 0
        confluence_details = {}
        
        # 1. Confluencia Direccional Ponderada
        direction_score = 0
        for tf, data in ladder_data.items():
            if data.get('direction') == 'BULLISH':
                weight = self.timeframe_weights.get(tf, 0.1)
                direction_score += weight * 100
                
        confluence_details['direction'] = direction_score
        
        # 2. Confluencia de Fuerza (ADX)
        strength_score = 0
        strong_timeframes = []
        for tf, data in ladder_data.items():
            adx = self._extract_adx(data)
            if adx > 25:
                weight = self.timeframe_weights.get(tf, 0.1)
                strength_score += weight * (adx / 50) * 100
                strong_timeframes.append(tf)
                
        confluence_details['strength'] = strength_score
        confluence_details['strong_tfs'] = strong_timeframes
        
        # 3. Confluencia de Momentum
        momentum_score = 0
        for tf, data in ladder_data.items():
            if data.get('momentum_direction') == 'UP':
                age = data.get('momentum_age', 0)
                if age < 10:  # Momentum joven
                    weight = self.timeframe_weights.get(tf, 0.1)
                    momentum_score += weight * (10 - age) * 10
                    
        confluence_details['momentum'] = momentum_score
        
        # 4. Confluencia de Volumen
        volume_score = self.calculate_volume_confluence(ladder_data)
        confluence_details['volume'] = volume_score
        
        # Score total ponderado
        total_score = (
            direction_score * 0.3 +
            strength_score * 0.3 +
            momentum_score * 0.25 +
            volume_score * 0.15
        )
        
        return {
            'total_score': total_score,
            'details': confluence_details,
            'quality': self.rate_confluence_quality(total_score)
        }
    
    def project_future_movement(self, ladder_data, patterns):
        """
        Proyecta el movimiento futuro basado en Time Ladder
        """
        projection = {
            'direction': None,
            'strength': None,
            'duration': None,
            'confidence': 0,
            'targets': [],
            'risks': []
        }
        
        # Analizar velocidad de propagación entre timeframes
        propagation_speed = self.analyze_propagation_speed(ladder_data)
        
        # Si el momentum está subiendo por la escalera (1M → 5M → 15M)
        if propagation_speed > 0:
            projection['direction'] = 'UP'
            projection['strength'] = 'STRONG' if propagation_speed > 0.7 else 'MODERATE'
            projection['duration'] = self.estimate_duration(ladder_data)
            projection['confidence'] = min(95, propagation_speed * 100)
            
            # Calcular targets basados en ATR y estructura
            projection['targets'] = self.calculate_smart_targets(ladder_data)
            
        # Identificar riesgos
        projection['risks'] = self.identify_risks(ladder_data, patterns)
        
        return projection
    
    def evaluate_opportunity_quality(self, ladder_data, patterns, confluence, projection):
        """
        Evalúa la calidad total de la oportunidad
        """
        base_score = 50
        
        # 1. Bonus por patrones detectados
        for pattern in patterns:
            if pattern['impact'] == 'VERY_HIGH':
                base_score += 15
            elif pattern['impact'] == 'HIGH':
                base_score += 10
            elif pattern['impact'] == 'MEDIUM':
                base_score += 5
                
        # 2. Bonus por confluencia
        confluence_score = confluence['total_score']
        if confluence_score > 80:
            base_score += 20
        elif confluence_score > 60:
            base_score += 10
        elif confluence_score < 30:
            base_score -= 10
            
        # 3. Ajuste por proyección
        if projection['confidence'] > 80:
            base_score += 15
        elif projection['confidence'] < 40:
            base_score -= 15
            
        # 4. Penalizaciones por riesgos
        for risk in projection['risks']:
            if risk['severity'] == 'HIGH':
                base_score -= 10
            elif risk['severity'] == 'MEDIUM':
                base_score -= 5
                
        # 5. Bonus por condiciones especiales
        base_score = self.apply_special_conditions(base_score, ladder_data)
        
        return {
            'total_score': min(100, max(0, base_score)),
            'grade': self.score_to_grade(base_score),
            'tradeable': base_score >= 60,
            'optimal_strategy': self.determine_strategy(base_score, ladder_data)
        }
    
    def make_intelligent_decision(self, quality_score):
        """
        Toma decisión final basada en calidad
        """
        score = quality_score['total_score']
        grade = quality_score['grade']
        strategy = quality_score['optimal_strategy']
        
        if score >= 80:

        elif score >= 65:

            }

    def check_bullish_cascade(self, ladder_data):
        """Verifica cascada alcista perfecta"""
        bullish_count = sum(1 for d in ladder_data.values() 
                           if d.get('direction') == 'BULLISH')
        return bullish_count >= 5
    
    def check_compression_pattern(self, ladder_data):
        """Detecta compresión de volatilidad multi-TF"""
        low_volatility_count = 0
        for tf, data in ladder_data.items():
            if tf in ['15M', '1H', '4H']:
                atr = data.get('atr', 0)
                if atr < data.get('atr_average', atr) * 0.7:
                    low_volatility_count += 1
        return low_volatility_count >= 2
    
    def check_hidden_divergence(self, ladder_data):
        """Detecta divergencias ocultas entre TFs"""
        # TF menores fuertes mientras mayores consolidan
        minor_strong = (
            self._extract_adx(ladder_data.get('5M', {})) > 25 and
            self._extract_adx(ladder_data.get('15M', {})) > 25
        )
        major_weak = (
            self._extract_adx(ladder_data.get('4H', {})) < 20
        )
        return minor_strong and major_weak
    
    def check_nascent_momentum(self, ladder_data):
        """Detecta momentum naciente"""
        data_1m = ladder_data.get('1M', {})
        data_5m = ladder_data.get('5M', {})
        
        return (
            data_1m.get('is_explosive', False) and
            data_5m.get('momentum_age', 10) < 3
        )
    
    def check_institutional_pattern(self, ladder_data):
        """Detecta patrones institucionales"""
        data_1h = ladder_data.get('1H', {})
        data_4h = ladder_data.get('4H', {})
        
        return (
            self._extract_adx(data_1h) > 35 and
            self._extract_adx(data_4h) > 30 and
            data_1h.get('volume_ratio', 0) > 2
        )

    def _extract_adx(self, data):
        """Extrae ADX de manera segura"""
        adx = data.get('adx', 0)
        if isinstance(adx, dict):
            return adx.get('adx_current', 0)
        return float(adx) if adx else 0
    
    def categorize_strength(self, adx):
        """Categoriza fuerza basada en ADX"""
        if adx >= 40:
            return 'VERY_STRONG'
        elif adx >= 30:
            return 'STRONG'
        elif adx >= 20:
            return 'MODERATE'
        else:
            return 'WEAK'
    
    def categorize_maturity(self, age):
        """Categoriza madurez del momentum"""
        if age <= 3:
            return 'NASCENT'
        elif age <= 7:
            return 'YOUNG'
        elif age <= 15:
            return 'MATURE'
        else:
            return 'OLD'
    
    def check_alignment_with_higher_tf(self, tf, analysis, all_analyses):
        """Verifica alineación con TF superior"""
        tf_hierarchy = ['1M', '5M', '15M', '1H', '4H', '1D']
        current_idx = tf_hierarchy.index(tf) if tf in tf_hierarchy else -1
        
        if current_idx < len(tf_hierarchy) - 1:
            higher_tf = tf_hierarchy[current_idx + 1]
            higher_analysis = all_analyses.get(higher_tf, {})
            
            if higher_analysis:
                return (
                    analysis.get('direction') == 
                    higher_analysis.get('direction')
                )
        return True
    
    def detect_special_features(self, tf, analysis):
        """Detecta características especiales del TF"""
        features = []
        
        adx = self._extract_adx(analysis)
        if adx > 40:
            features.append('EXTREME_STRENGTH')
            
        if analysis.get('volume_ratio', 0) > 3:
            features.append('VOLUME_EXPLOSION')
            
        if analysis.get('is_explosive'):
            features.append('EXPLOSIVE_MOVE')
            
        return features
    
    def calculate_volume_confluence(self, ladder_data):
        """Calcula confluencia de volumen"""
        volume_score = 0
        for tf, data in ladder_data.items():
            vol_ratio = data.get('volume_ratio', 1)
            if vol_ratio > 1.5:
                weight = self.timeframe_weights.get(tf, 0.1)
                volume_score += weight * min(100, vol_ratio * 20)
        return volume_score
    
    def rate_confluence_quality(self, score):
        """Califica la calidad de confluencia"""
        if score >= 80:
            return 'EXCELLENT'
        elif score >= 60:
            return 'GOOD'
        elif score >= 40:
            return 'FAIR'
        else:
            return 'POOR'
    
    def analyze_propagation_speed(self, ladder_data):
        """Analiza velocidad de propagación entre TFs"""
        propagation = 0
        
        # Verificar si el momentum se propaga hacia arriba
        if ladder_data.get('1M', {}).get('is_explosive'):
            propagation += 0.3
        if ladder_data.get('5M', {}).get('direction') == 'BULLISH':
            propagation += 0.3
        if ladder_data.get('15M', {}).get('direction') == 'BULLISH':
            propagation += 0.4
            
        return propagation
    
    def estimate_duration(self, ladder_data):
        """Estima duración del movimiento"""
        # Basado en qué TFs están alineados
        if self._extract_adx(ladder_data.get('4H', {})) > 25:
            return 'MULTI_DAY'
        elif self._extract_adx(ladder_data.get('1H', {})) > 25:
            return 'INTRADAY'
        else:
            return 'SCALPING'

        targets = []

        return targets
    
    def identify_risks(self, ladder_data, patterns):
        """Identifica riesgos potenciales"""
        risks = []
        
        # Riesgo de sobreextensión
        if ladder_data.get('15M', {}).get('distance_to_ema20', 0) > 3:
            risks.append({
                'type': 'OVEREXTENSION',
                'severity': 'MEDIUM',
                'description': 'Precio alejado de EMAs'
            })
            
        # Riesgo de madurez
        if any(d.get('maturity') == 'OLD' for d in ladder_data.values()):
            risks.append({
                'type': 'TREND_EXHAUSTION',
                'severity': 'HIGH',
                'description': 'Tendencia madura, posible reversión'
            })
            
        return risks
    
    def apply_special_conditions(self, score, ladder_data):
        """Aplica condiciones especiales al score"""
        # Bonus por sesión de trading
        hour = datetime.now().hour
        if 14 <= hour <= 20:  # Sesión US
            score *= 1.1
        elif 2 <= hour <= 8:  # Sesión Asia
            score *= 0.9
            
        return score
    
    def score_to_grade(self, score):
        """Convierte score a grado"""
        if score >= 90:
            return 'A+'
        elif score >= 80:
            return 'A'
        elif score >= 70:
            return 'B'
        elif score >= 60:
            return 'C'
        else:
            return 'D'

            else:
                return 'SWING_TRADING'
        elif score >= 60:
            return 'SCALPING'
        else:
            return 'NO_TRADE'
