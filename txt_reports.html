class BotBrainRecorder:
    """
    Records BOT's decision-making process at each historical frame.
    
    Simulates what the bot would have thought/decided at each moment
    by reconstructing the analysis with historical data.
    
    This is the "first-person camera" inside the bot's brain.
    """
    
    def __init__(self):
        self.cache = {}
    
    def record_decision_at_frame(self, symbol, frame_data, direction):
        """
        Simulates bot's decision process at this historical frame.
        
        Args:
            symbol: str - Trading pair
            frame_data: dict - Market indicators at this frame
            direction: str - BULLISH/BEARISH
        
        Returns:
            dict - {
                'confidence_factors': [...],
                'total_confidence': 55,
                'decision': 'ENTER',
                'decision_reason': 'Confidence crossed threshold'
            }
        """
        try:
            # Build synthetic ladder_analysis from frame data
            ladder_analysis = self._build_ladder_analysis(symbol, frame_data, direction)
            
            # Call UnifiedVision to get decision (lightweight)
            from brain_vision_unified import UnifiedVision
            vision = UnifiedVision()
            
            # Get trajectory data (simplified)
            traj = self._build_trajectory(frame_data, direction)
            
            # Evaluate confidence (this is what we want to see!)
            result = vision.evaluate_confidence(traj, direction, ladder_analysis)
            
            # Extract confidence breakdown
            confidence_breakdown = self._extract_confidence_breakdown(result)
            
            return confidence_breakdown
            
        except Exception as e:
            # If simulation fails, return error state
            return {
                'confidence_factors': [],
                'total_confidence': 0,
                'decision': 'ERROR',
                'decision_reason': f'Simulation failed: {str(e)[:50]}',
                'error': str(e)
            }
    
    def _build_ladder_analysis(self, symbol, frame_data, direction):
        """
        Constructs synthetic ladder_analysis for this historical frame.
        
        UnifiedVision needs ladder_analysis, so we build one from
        the indicator data we have in this frame.
        """
        # Extract key indicators from frame
        adx = frame_data.get('adx', 0)
        di_plus = frame_data.get('di_plus', 0)
        di_minus = frame_data.get('di_minus', 0)
        exhaustion_pct = frame_data.get('exhaustion_pct', 50)
        room_pct = frame_data.get('room_pct', 0)
        momentum_age = frame_data.get('momentum_age', 0)
        
        # Build minimal 15m analysis (UnifiedVision mainly uses this)
        ladder_15m = {
            'direction': direction,
            'adx': {
                'adx_current': adx,
                'di_plus': di_plus,
                'di_minus': di_minus,
                'spread': abs(di_plus - di_minus),
                'trend': 'creciendo' if adx > 20 else 'estable'
            },
            'exhaustion_pct': exhaustion_pct,
            'room_pct': room_pct,
            'exhaustion_phase': self._get_phase(exhaustion_pct),
            'momentum': {
                'consecutive_count': momentum_age,
                'current_color': 'green' if momentum_age > 0 else 'red'
            }
        }
        
        # Return ladder with 15m (minimum required)
        return {
            '15m': ladder_15m,
            '5m': {},  # Placeholder
            '1h': {},  # Placeholder
            '4h': {}   # Placeholder
        }
    
    def _build_trajectory(self, frame_data, direction):
        """
        Builds trajectory dict for UnifiedVision.evaluate_confidence().
        """
        # Extract data from frame
        adx_end = frame_data.get('adx', 0)
        exhaustion_pct = frame_data.get('exhaustion_pct', 50)
        room_pct = frame_data.get('room_pct', 0)
        momentum_age = frame_data.get('momentum_age', 0)
        
        # Build trajectory (simplified - no velocity for now)
        return {
            'direction': direction,
            'adx_end': adx_end,
            'adx_delta': 0,  # We don't have velocity in single frame
            'candle_ratio': 1.0,  # Neutral
            'momentum_end': momentum_age,
            'momentum_delta': 0,
            'exhaustion_pct': exhaustion_pct,
            'room_pct': room_pct,
            'exhaustion_velocity': 0,  # No velocity data
            'pivot_types': [],  # No structure data
            'position_pct': 50  # Neutral grid position
        }
    
    def _get_phase(self, exhaustion_pct):
        """Maps exhaustion percentage to phase."""
        if exhaustion_pct < 30:
            return 'FRESH'
        elif exhaustion_pct < 50:
            return 'DEVELOPING'
        elif exhaustion_pct < 70:
            return 'MATURE'
        else:
            return 'EXHAUSTED'
    
    def _extract_confidence_breakdown(self, result):
        """
        Extracts confidence factors from UnifiedVision result.
        
        Args:
            result: dict from evaluate_confidence()
        
        Returns:
            dict with confidence_factors, total, decision, reason
        """
        # Result format from UnifiedVision:
        # {
        #   'action': 'OPERAR' or 'WAIT',
        #   'confidence': 55,
        #   'reason': 'Confidence sufficient',
        #   'details': {...}
        # }
        
        action = result.get('action', 'WAIT')
        confidence = result.get('confidence', 0)
        reason = result.get('reason', 'Unknown')
        details = result.get('details', {})
        
        # Try to extract factors from details
        # UnifiedVision in evaluate_confidence builds confidence_factors list
        factors = []
        
        # If details has confidence_factors (from our implementation)
        if 'confidence_factors' in details:
            # Format: [('ADX', 12, 'reason'), ...]
            for factor_tuple in details['confidence_factors']:
                if len(factor_tuple) >= 2:
                    name = factor_tuple[0]
                    value = factor_tuple[1]
                    factor_reason = factor_tuple[2] if len(factor_tuple) > 2 else ''
                    
                    factors.append({
                        'name': name,
                        'value': round(value, 1),
                        'reason': str(factor_reason)
                    })
        
        # If no factors extracted, create generic one
        if not factors:
            factors.append({
                'name': 'Total',
                'value': confidence,
                'reason': 'Details not available'
            })
        
        # Determine decision
        decision = 'ENTER' if action == 'OPERAR' else 'WAIT'
        
        # Build decision reason
        threshold = 40  # Known threshold
        if confidence >= threshold:
            decision_reason = f"Confidence {confidence}% >= {threshold}% threshold"
        else:
            decision_reason = f"Confidence {confidence}% < {threshold}% threshold"
        
        return {
            'confidence_factors': factors,
            'total_confidence': round(confidence, 1),
            'decision': decision,
            'decision_reason': decision_reason
        }

def _detect_bugs(self):
        """
        Detects bugs automatically from bot_brain frames.
        
        Returns:
            list of detected bugs
        """
        bugs = []
        
        # Collect all bot_brain data
        brain_frames = [f.get('bot_brain', {}) for f in self.frames if 'bot_brain' in f]
        
        if not brain_frames:
            return bugs
        
        # BUG 1: Factor never used
        factor_usage = {}
        for brain in brain_frames:
            for factor in brain.get('confidence_factors', []):
                name = factor['name']
                value = factor['value']
                
                if name not in factor_usage:
                    factor_usage[name] = []
                factor_usage[name].append(value)
        
        for factor_name, values in factor_usage.items():
            if all(v == 0 for v in values):
                bugs.append({
                    'bug': f'{factor_name.upper()}_NEVER_USED',
                    'evidence': f'{factor_name} factor = 0 pts in ALL {len(values)} frames',
                    'impact': f'Bot cannot evaluate {factor_name} in decisions',
                    'fix': f'Implement {factor_name} factor in evaluate_confidence()',
                    'severity': 'HIGH'
                })
        
        # BUG 2: Multiple threshold crossings (indecisive)
        confidences = [b.get('total_confidence', 0) for b in brain_frames]
        threshold = 40
        
        crossings = []
        for i in range(1, len(confidences)):
            if confidences[i-1] < threshold <= confidences[i]:
                crossings.append(i)
            elif confidences[i-1] >= threshold > confidences[i]:
                crossings.append(i)
        
        if len(crossings) > 2:
            bugs.append({
                'bug': 'MULTIPLE_THRESHOLD_CROSSINGS',
                'evidence': f'Confidence crossed {threshold}% threshold {len(crossings)} times',
                'impact': 'Indecisive bot, threshold too close to noise',
                'fix': f'Adjust threshold or increase signal-to-noise ratio',
                'severity': 'MEDIUM'
            })
        
        # BUG 3: Contradictory factor values at entry
        entry_brain = next((f.get('bot_brain', {}) for f in self.frames 
                           if f.get('relative_min') == 0), None)
        
        if entry_brain:
            for factor in entry_brain.get('confidence_factors', []):
                reason_lower = factor.get('reason', '').lower()
                value = factor.get('value', 0)
                
                # Check for contradictions
                negative_signals = ['closing', 'cayendo', 'débil', 'contradice', 'bajando']
                if any(sig in reason_lower for sig in negative_signals) and value > 0:
                    bugs.append({
                        'bug': f'{factor["name"].upper()}_BONUS_ON_NEGATIVE_SIGNAL',
                        'evidence': f"{factor['name']} gave +{value} pts despite '{factor['reason']}'",
                        'impact': 'Bot rewards negative market signals',
                        'fix': f'Check {factor["name"]} calculation logic',
                        'severity': 'HIGH'
                    })
        
        return bugs
    
    def _analyze_confidence_trajectory(self):
        """
        Analyzes how confidence evolved over time.
        
        Returns:
            dict with frames, values, interpretation
        """
        brain_frames = [f.get('bot_brain', {}) for f in self.frames if 'bot_brain' in f]
        
        if not brain_frames:
            return {}
        
        # Extract relative times and confidences
        times = [f.get('relative_min', 0) for f in self.frames if 'bot_brain' in f]
        confidences = [b.get('total_confidence', 0) for b in brain_frames]
        
        # Find threshold crossing
        threshold = 40
        crossing_time = None
        for i, (t, c) in enumerate(zip(times, confidences)):
            if i > 0 and confidences[i-1] < threshold <= c:
                crossing_time = t
                break
        
        # Interpretation
        if crossing_time is not None:
            if crossing_time < -10:
                interpretation = f"Confidence crossed {threshold}% at {crossing_time}min (EARLY), entered later at 0min"
            elif crossing_time < 0:
                interpretation = f"Confidence crossed {threshold}% at {crossing_time}min (JUST before entry)"
            elif crossing_time == 0:
                interpretation = f"Confidence crossed {threshold}% exactly at entry (0min)"
            else:
                interpretation = f"Entered BEFORE confidence crossed {threshold}% (at {crossing_time}min)"
        else:
            avg_conf = sum(confidences) / len(confidences) if confidences else 0
            if avg_conf < threshold:
                interpretation = f"Confidence NEVER crossed {threshold}% (avg: {avg_conf:.0f}%)"
            else:
                interpretation = f"Confidence always > {threshold}% (avg: {avg_conf:.0f}%)"
        
        return {
            'frames': times,
            'values': [round(c, 1) for c in confidences],
            'threshold': threshold,
            'crossing_time': crossing_time,
            'interpretation': interpretation
        }
    
    def _analyze_factor_contributions(self):
        """
        Analyzes contribution of each factor across all frames.
        
        Returns:
            dict with factor stats
        """
        brain_frames = [f.get('bot_brain', {}) for f in self.frames if 'bot_brain' in f]
        
        if not brain_frames:
            return {}
        
        factor_stats = {}
        
        # Collect all factor values
        for brain in brain_frames:
            for factor in brain.get('confidence_factors', []):
                name = factor['name']
                value = factor['value']
                
                if name not in factor_stats:
                    factor_stats[name] = []
                factor_stats[name].append(value)
        
        # Calculate stats
        contributions = {}
        for name, values in factor_stats.items():
            contributions[name] = {
                'range': [round(min(values), 1), round(max(values), 1)],
                'avg': round(sum(values) / len(values), 1),
                'impact': self._classify_impact(values)
            }
        
        return contributions
    
    def _classify_impact(self, values):
        """Classifies factor impact based on value distribution."""
        if not values:
            return 'NONE'
        
        avg = sum(values) / len(values)
        max_val = max(values)
        min_val = min(values)
        range_val = max_val - min_val
        
        # All zeros = not used
        if max_val == 0:
            return 'NONE - BUG: never used'
        
        # High average = strong contributor
        if avg > 10:
            return 'HIGH - strong contributor'
        
        # Large range = dynamic factor
        if range_val > 15:
            return 'DYNAMIC - high variance'
        
        # Medium average
        if avg > 5:
            return 'MEDIUM - moderate contributor'
        
        return 'LOW - minor contributor'
