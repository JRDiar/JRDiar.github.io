def check_market_force(ladder):
    """
    Verifica FUERZA DE MERCADO en 15M (Setup preciso según Joel).
    Reglas: ADX > 25, Spread > 5, Momentum Alineado.
    """
    # Usar 15M para setup preciso
    a15m = ladder.get('15m') or {}
    if not a15m: 
        return {'detected': False, 'reason': "No data 15M"}
    
    adx_data = a15m.get('adx', {})
    mom_data = a15m.get('momentum', {})
    
    adx_val = adx_data.get('adx_current', 0)
    di_plus = adx_data.get('di_plus_current', 0)
    di_minus = adx_data.get('di_minus_current', 0)
    mom_color = mom_data.get('color_current', 'gray')
    
    # Determinar dirección por DI spread
    if di_plus > di_minus:
        force_direction = 'BULLISH'
        spread = di_plus - di_minus
    else:
        force_direction = 'BEARISH'
        spread = di_minus - di_plus
        
    # Validaciones estrictas
    if adx_val < 25: 
        return {'detected': False, 'reason': f"ADX débil 15M ({adx_val:.1f})"}
    if spread < 5.0: 
        return {'detected': False, 'reason': f"Spread sucio 15M ({spread:.1f})"}
    
    # Momentum DEBE alinear (CRÍTICO)
    aligned = False
    if force_direction == 'BULLISH' and mom_color in ['green', 'lime']: 
        aligned = True
    elif force_direction == 'BEARISH' and mom_color in ['red', 'maroon']: 
        aligned = True
    
    if not aligned: 
        return {
            'detected': False, 
            'reason': f"⛔ Momentum 15M ({mom_color}) NO alinea con {force_direction}"
        }
    
    return {
        'detected': True,
        'direction': force_direction,
        'adx': adx_val,
        'spread': spread,
        'momentum': mom_color,
        'reason': f"💪 FUERZA 15M: ADX {adx_val:.1f} | {force_direction} | Momentum {mom_color}"
    }

"""
================================================================================
BRAIN GRID v1.0 - SISTEMA DE NAVEGACIÓN ESPACIAL
================================================================================
Filosofía: Ver el mercado como un MAPA, no como números sueltos

Responsabilidades:
1. Construir la MALLA (dimensiones del campo de juego)
2. Calcular el PUNTERO (dónde está el precio + hacia dónde va)
3. Detectar CONTEXTOS (lateralización, tendencia, explosión)
4. Validar ZONAS OPERABLES (setup según la imagen de Joel)

Creado por: Claude Sonnet (Anthropic)
Autor del concepto: Joel Rodríguez Diarte
Fecha: 2025-12-13
"""

import pandas as pd
import numpy as np
from datetime import datetime

# ============================================================================
# CONFIGURACIÓN (Hardcoded - sin dependencias externas)
# ============================================================================

GRID_CONFIG = {
    'lookback_velas': 20,           # Velas para definir techo/piso
    'lateral_threshold_atr': 2.0,   # Ancho < 2 ATRs = lateral
    'tendencia_threshold_atr': 3.5, # Ancho > 3.5 ATRs = tendencia fuerte
    'puntero_sensibilidad': 3,      # Velas para calcular vector velocidad
    'fuerza_min_adx': 22,           # ADX mínimo para considerar fuerza
    'fuerza_fuerte_adx': 30,        # ADX para fuerza brutal
}

# ============================================================================
# CLASE PRINCIPAL
# ============================================================================

class MarketGrid:
    """
    La Malla del Mercado - Navegación Espacial
    """
    
    def __init__(self):
        self.history = {}  # Cache de estados anteriores
    
    def analyze(self, symbol, df_5m, ladder=None):
        """
        Análisis completo de la malla
        
        Args:
            symbol: Símbolo (ej: 'BTC/USDT')
            df_5m: DataFrame 5M con al menos 50 velas
            ladder: Dict con análisis multi-timeframe (opcional)
        
        Returns:
            dict: {
                'status': str,           # 'OPERABLE', 'LATERAL', 'WAIT'
                'contexto': str,         # Descripción del contexto
                'score': int,            # -100 a +100
                'razon': str,            # Explicación
                'grid_data': dict,       # Datos técnicos
                'puntero_data': dict,    # Datos del puntero
                'zona_ideal': bool       # True si es zona perfecta
            }
        """
        
        if df_5m is None or len(df_5m) < 30:
            return self._return_error("Sin datos suficientes")
        
        # 1. CONSTRUIR DIMENSIONES DE LA MALLA
        grid_dims = self._build_dimensions(df_5m)
        
        # 2. CALCULAR PUNTERO (Vector de precio)
        puntero = self._calculate_pointer(df_5m, ladder)
        
        # 3. DETECTAR CONTEXTO
        contexto = self._detect_context(grid_dims, puntero, df_5m)
        
        # 4. VALIDAR FUERZA (si hay ladder)
        fuerza = self._validate_force(ladder) if ladder else {'score': 0}
        
        # 5. SCORING FINAL
        resultado = self._calculate_final_score(
            grid_dims, puntero, contexto, fuerza, df_5m
        )
        
        # 6. Guardar estado
        self.history[symbol] = {
            'timestamp': datetime.now(),
            'grid': grid_dims,
            'puntero': puntero,
            'contexto': contexto
        }
        
        return resultado
    
    # ========================================================================
    # CONSTRUCCIÓN DE DIMENSIONES
    # ========================================================================
    
    def _build_dimensions(self, df):
        """
        Construye las paredes de la malla (Techo, Piso, Ancho)
        """
        lookback = GRID_CONFIG['lookback_velas']
        
        # Highs y Lows rolling
        highs = df['high'].rolling(lookback).max()
        lows = df['low'].rolling(lookback).min()
        
        techo = highs.iloc[-1]
        piso = lows.iloc[-1]
        medio = (techo + piso) / 2
        
        # Ancho en % (método simple)
        ancho_pct = ((techo - piso) / medio) * 100 if medio > 0 else 0
        
        # Ancho en ATRs (método robusto)
        atr_val = self._calculate_atr_simple(df)
        ancho_atr = (techo - piso) / atr_val if atr_val > 0 else 0
        
        # Slope (pendiente de la línea media)
        medios = (df['high'].rolling(lookback).max() + df['low'].rolling(lookback).min()) / 2
        if len(medios) >= 5:
            slope = medios.iloc[-1] - medios.iloc[-5]
        else:
            slope = 0
        
        return {
            'techo': techo,
            'piso': piso,
            'medio': medio,
            'ancho_pct': ancho_pct,
            'ancho_atr': ancho_atr,
            'atr': atr_val,
            'slope': slope,
            'lookback': lookback
        }
    
    def _calculate_atr_simple(self, df, period=14):
        """ATR simplificado (no necesita brain_calculator)"""
        high = df['high']
        low = df['low']
        close = df['close']
        
        tr1 = high - low
        tr2 = abs(high - close.shift(1))
        tr3 = abs(low - close.shift(1))
        
        tr = pd.concat([tr1, tr2, tr3], axis=1).max(axis=1)
        atr = tr.rolling(period).mean().iloc[-1]
        
        return atr if not pd.isna(atr) else 0.01

    # CÁLCULO DEL PUNTERO

    def _calculate_pointer(self, df, ladder):
        """
        Calcula el puntero (vector de velocidad + dirección)
        """
        sens = GRID_CONFIG['puntero_sensibilidad']
        closes = df['close'].values
        
        # Vector precio (cambio en últimas N velas)
        if len(closes) > sens:
            delta = closes[-1] - closes[-sens]
            velocidad = delta / sens
        else:
            delta = 0
            velocidad = 0
        
        # Dirección básica
        if delta > 0:
            direccion_precio = 'UP'
        elif delta < 0:
            direccion_precio = 'DOWN'
        else:
            direccion_precio = 'NEUTRAL'
        
        # Si hay ladder, usar momentum color
        if ladder and '15m' in ladder:
            mom_data = ladder['15m'].get('momentum', {})
            mom_color = mom_data.get('color_current', 'neutral')
            
            if mom_color in ['green', 'lime']:
                direccion_final = 'UP'
                color = 'VERDE'
            elif mom_color in ['red', 'maroon']:
                direccion_final = 'DOWN'
                color = 'ROJO'
            else:
                direccion_final = direccion_precio
                color = 'GRIS'
        else:
            direccion_final = direccion_precio
            color = 'GRIS'
        
        # Fuerza del movimiento (% de cambio)
        fuerza_pct = abs(delta / closes[-1] * 100) if closes[-1] > 0 else 0
        
        return {
            'direccion': direccion_final,
            'color': color,
            'velocidad': velocidad,
            'fuerza_pct': fuerza_pct,
            'delta': delta
        }
    
    # ========================================================================
    # DETECCIÓN DE CONTEXTO
    # ========================================================================
    
    def _detect_context(self, grid_dims, puntero, df):
        """
        Detecta el contexto del mercado:
        - LATERAL: Caja cerrada
        - TENDENCIA_ALCISTA: Malla subiendo
        - TENDENCIA_BAJISTA: Malla bajendo
        - EXPLOSION: Breakout violento
        """
        ancho_atr = grid_dims['ancho_atr']
        slope = grid_dims['slope']
        techo = grid_dims['techo']
        piso = grid_dims['piso']
        
        current_price = df['close'].iloc[-1]
        
        # Posición relativa en la malla (0 = piso, 1 = techo)
        denominator = techo - piso
        if denominator < 0.0001:
            posicion = 0.5
        else:
            posicion = (current_price - piso) / denominator
        
        # CLASIFICACIÓN
        contexto_tipo = 'UNKNOWN'
        zona_operacion = 'NEUTRAL'
        
        # 1. LATERAL (Caja cerrada)
        if ancho_atr < GRID_CONFIG['lateral_threshold_atr']:
            contexto_tipo = 'LATERAL'
            zona_operacion = 'PROHIBIDA'
        
        # 2. TENDENCIA ALCISTA
        elif slope > 0 and puntero['direccion'] == 'UP':
            contexto_tipo = 'TENDENCIA_ALCISTA'
            
            # Zona ideal: Precio en zona baja/media (0.2-0.6)
            if 0.15 < posicion < 0.65:
                zona_operacion = 'IDEAL_LONG'
            elif posicion > 0.85:
                zona_operacion = 'TECHO'  # Peligro de comprar techo
            else:
                zona_operacion = 'ACEPTABLE_LONG'
        
        # 3. TENDENCIA BAJISTA
        elif slope < 0 and puntero['direccion'] == 'DOWN':
            contexto_tipo = 'TENDENCIA_BAJISTA'
            
            # Zona ideal: Precio en zona alta/media (0.4-0.8)
            if 0.35 < posicion < 0.85:
                zona_operacion = 'IDEAL_SHORT'
            elif posicion < 0.15:
                zona_operacion = 'PISO'  # Peligro de vender piso
            else:
                zona_operacion = 'ACEPTABLE_SHORT'
        
        # 4. CONFLICTO (Malla dice una cosa, puntero otra)
        elif slope > 0 and puntero['direccion'] == 'DOWN':
            contexto_tipo = 'CORRECCION_ALCISTA'
            zona_operacion = 'ESPERAR_REBOTE'
        
        elif slope < 0 and puntero['direccion'] == 'UP':
            contexto_tipo = 'REBOTE_BAJISTA'
            zona_operacion = 'ESPERAR_CONTINUACION'
        
        else:
            contexto_tipo = 'INDEFINIDO'
            zona_operacion = 'NEUTRAL'
        
        return {
            'tipo': contexto_tipo,
            'zona': zona_operacion,
            'posicion': posicion,
            'ancho_atr': ancho_atr
        }
