"""
brain_frames.py - Sistema de Frames Temporales (PELÍCULA)
=========================================================
Guarda historia de indicadores como "frames" de una película.

Concepto de Joel:
"Es como un dibujante que apila sus dibujos con secuencia y los 
mueve rápido para generar el dibujo animado."

Cada frame contiene:
- ADX, DI+, DI-, Spread
- Momentum color, age
- Precio, volumen
- Grid posición
- ZigZag estado
- Timestamp

Los frames se apilan secuencialmente para ver MOVIMIENTO.

Autor: Joel Rodríguez Diarte (concepto)
Implementación: Claude (Anthropic)
Fecha: 2025-12-21
"""

from datetime import datetime
from collections import deque
import json
import os

# ============================================================================
# CLASE PRINCIPAL: FrameStorage
# ============================================================================

class FrameStorage:
    """
    Almacena frames temporales de indicadores.
    
    Permite ver la PELÍCULA del mercado, no solo fotos estáticas.
    
    Ejemplo:
        Frame -5: ADX = 45
        Frame -4: ADX = 48  (+3)
        Frame -3: ADX = 52  (+4)
        Frame -2: ADX = 56  (+4)
        Frame -1: ADX = 58  (+2)
        
        → ADX ACELERANDO → EXPLOSIÓN EN CURSO
    """
    
    def __init__(self, max_frames=20):
        """
        Args:
            max_frames: Máximo de frames a guardar (20 = últimas 20 velas)
        """
        self.max_frames = max_frames
        self.frames = deque(maxlen=max_frames)
    
    def add_frame(self, symbol, timeframe, indicators):
        """
        Agrega un frame nuevo (un "dibujo" del momento actual)
        
        Args:
            symbol: Par (ej: RAVE/USDT:USDT)
            timeframe: TF (5m, 15m, 1h, 4h)
            indicators: Dict con todos los indicadores actuales
        """
        frame = {
            'timestamp': datetime.now().isoformat(),
            'symbol': symbol,
            'timeframe': timeframe,
            
            # ADX (líneas del indicador)
            'adx': {
                'value': indicators.get('adx', {}).get('adx_current', 0),
                'di_plus': indicators.get('adx', {}).get('di_plus_current', 0),
                'di_minus': indicators.get('adx', {}).get('di_minus_current', 0),
                'spread': abs(
                    indicators.get('adx', {}).get('di_plus_current', 0) -
                    indicators.get('adx', {}).get('di_minus_current', 0)
                ),
                'trend': indicators.get('adx', {}).get('adx_trend', 'unknown')
            },
            
            # Momentum (barras)
            'momentum': {
                'color': indicators.get('momentum', {}).get('color_current', 'gray'),
                'age': indicators.get('momentum', {}).get('consecutive_count', 0),
                'intensity': indicators.get('momentum', {}).get('intensity', 0)
            },
            
            # >>> NUEVO: OHLC Completo para ver "dibujo" del precio
            'ohlc': {
                'open': indicators.get('open', 0),
                'high': indicators.get('high', 0),
                'low': indicators.get('low', 0),
                'close': indicators.get('close', indicators.get('price', 0))
            },
            
            # Precio (legacy - mantener compatibilidad)
            'price': indicators.get('price', indicators.get('close', 0)),
            
            # Volumen
            'volume': indicators.get('volume', 0),
            'volume_ratio': indicators.get('volume_ratio', 100),
            
            # Grid
            'grid': {
                'position_pct': indicators.get('position_in_range', 50),
                'rompiendo': indicators.get('position_in_range', 50) > 85
            },
            
            # Score
            'score': indicators.get('score_15m', 0)
        }
        
        self.frames.append(frame)
    
    def get_frames(self, n=None):
        """
        Obtiene últimos N frames
        
        Args:
            n: Número de frames (None = todos)
            
        Returns:
            list: Lista de frames [más viejo ... más nuevo]
        """
        if n is None:
            return list(self.frames)
        else:
            return list(self.frames)[-n:]
    
    def get_trajectory(self, indicator_path, n=5):
        """
        Obtiene trayectoria de un indicador (la "animación")
        
        Args:
            indicator_path: Ruta al indicador (ej: 'adx.value')
            n: Frames a analizar
            
        Returns:
            list: Valores históricos [más viejo ... más nuevo]
            
        Example:
            trajectory = storage.get_trajectory('adx.value', 5)
            # [45, 48, 52, 56, 58] → subiendo
        """
        frames = self.get_frames(n)
        values = []
        
        for frame in frames:
            # Navegar por el path (ej: 'adx.value')
            parts = indicator_path.split('.')
            value = frame
            
            try:
                for part in parts:
                    if isinstance(value, dict):
                        value = value.get(part, 0)
                    else:
                        value = 0
                        break
                values.append(value)
            except:
                values.append(0)
        
        return values
    
    def get_frame_count(self):
        """Retorna cantidad de frames almacenados"""
        return len(self.frames)
    
    def clear(self):
        """Limpia todos los frames"""
        self.frames.clear()
    
    def get_latest_frame(self):
        """Obtiene el frame más reciente"""
        if self.frames:
            return self.frames[-1]
        return None
    
    def has_enough_frames(self, min_frames=3):
        """Verifica si hay suficientes frames para análisis"""
        return len(self.frames) >= min_frames

def load_frames_from_ohlcv(symbol: str, timeframe: str, ohlcv_data: list, num_frames: int = 10):
    """
    CARGA INSTANTÁNEA DE FRAMES DESDE OHLCV HISTÓRICO.
    
    CONCEPTO DE JOEL:
    "El Time Ladder debe ser rápido, no en slow motion"
    
    Esta función carga los últimos N frames inmediatamente desde los datos
    OHLCV históricos del exchange, en lugar de esperar múltiples ciclos.
    
    Args:
        symbol: Par (ej: RAVE/USDT:USDT)
        timeframe: TF (5m, 15m, 1h, 4h)
        ohlcv_data: Lista de OHLCV [[timestamp, open, high, low, close, volume], ...]
        num_frames: Cantidad de frames a cargar (default 10)
        
    Returns:
        int: Cantidad de frames cargados
    """
    if not ohlcv_data or len(ohlcv_data) < 5:
        return 0
    
    storage = get_frame_storage(symbol, timeframe)
    
    # Si ya tiene suficientes frames, no recargar
    if storage.has_enough_frames(num_frames):
        return storage.get_frame_count()
    
    # Limpiar frames antiguos si los hay
    storage.clear()
    
    # Cargar últimos N candles como frames
    candles_to_load = ohlcv_data[-num_frames:]
    
    for candle in candles_to_load:
        if len(candle) >= 6:
            timestamp, open_p, high_p, low_p, close_p, volume = candle[:6]
            
            # Crear indicadores simulados para el frame
            indicators = {
                'open': open_p,
                'high': high_p,
                'low': low_p,
                'close': close_p,
                'price': close_p,
                'volume': volume,
                'volume_ratio': 100,  # Se calculará después si es necesario
                'position_in_range': 50,  # Se calculará después
                
                # ADX y momentum se llenarán con datos reales cuando se analice
                'adx': {'adx_current': 0, 'di_plus_current': 0, 'di_minus_current': 0},
                'momentum': {'color_current': 'gray', 'consecutive_count': 0}
            }
            
            storage.add_frame(symbol, timeframe, indicators)
    
    loaded = storage.get_frame_count()
    if loaded >= 5:
        print(f"   ⚡ Time Ladder INSTANTÁNEO: {symbol} - {loaded} frames cargados")
    
    return loaded

def detect_price_pattern(symbol, direction='BULLISH', timeframe='15m'):
    """
    Detecta patrón de precio usando Time Ladder.
    
    CONCEPTO DE JOEL:
    "El bot no esperaba el piso y luego el impulso, entraba sin ver ese piso."
    
    Esta función ve la PELÍCULA del precio frame a frame para detectar:
    - LONG: Corrección → Piso → Rebote confirmado → ENTRAR
    - SHORT: Rally → Techo → Caída confirmada → ENTRAR
    
    Args:
        symbol: Par a analizar
        direction: 'BULLISH' o 'BEARISH'
        timeframe: TF para análisis
        
    Returns:
        dict: {
            'ready': bool - Si el patrón está listo para entrar
            'pattern': str - Nombre del patrón detectado
            'phase': str - Fase actual (CORRECCION, PISO, IMPULSO, etc)
            'reason': str - Razón descriptiva
            'wait_for': str - Qué esperar si no está listo
        }
    """
    storage = get_frame_storage(symbol, timeframe)
    
    # Necesitamos mínimo 5 frames para ver la película
    if not storage.has_enough_frames(5):
        return {
            'ready': False,
            'pattern': 'INSUFFICIENT_DATA',
            'phase': 'UNKNOWN',
            'reason': f'Solo {storage.get_frame_count()} frames, necesita 5',
            'wait_for': 'Más datos de precio'
        }
    
    # Obtener trayectorias
    closes = storage.get_trajectory('ohlc.close', 5)
    lows = storage.get_trajectory('ohlc.low', 5)
    highs = storage.get_trajectory('ohlc.high', 5)
    
    # Fallback si OHLC no está disponible
    if not closes or closes[-1] == 0:
        closes = storage.get_trajectory('price', 5)
        lows = closes.copy()
        highs = closes.copy()
    
    if not closes or len(closes) < 5:
        return {
            'ready': False,
            'pattern': 'NO_PRICE_DATA',
            'phase': 'UNKNOWN',
            'reason': 'Sin datos de precio',
            'wait_for': 'Datos de precio',
            'structure': None
        }
    
    # ===================================================================
    # >>> NUEVO: DETECCIÓN DE HH/HL/LL/LH (ESTRUCTURA DE TENDENCIA)
    # ===================================================================
    # Esto es como ver el gráfico completo de velas pero con el precio
    
    # Identificar swing highs y swing lows (pivots locales)
    swing_highs = []  # [(idx, price)]
    swing_lows = []   # [(idx, price)]
    
    for i in range(1, len(highs) - 1):
        # Swing High: precio mayor que vecinos
        if highs[i] > highs[i-1] and highs[i] >= highs[i+1]:
            swing_highs.append((i, highs[i]))
        # Swing Low: precio menor que vecinos  
        if lows[i] < lows[i-1] and lows[i] <= lows[i+1]:
            swing_lows.append((i, lows[i]))
    
    # Analizar estructura HH/HL/LL/LH
    structure = {
        'swing_highs': swing_highs,
        'swing_lows': swing_lows,
        'trend': 'LATERAL',
        'hh_count': 0,
        'hl_count': 0,
        'll_count': 0,
        'lh_count': 0,
        'last_pivot': 'NONE'
    }
    
    # Contar HH (Higher High) y LH (Lower High)
    for i in range(1, len(swing_highs)):
        if swing_highs[i][1] > swing_highs[i-1][1]:
            structure['hh_count'] += 1
        else:
            structure['lh_count'] += 1
    
    # Contar HL (Higher Low) y LL (Lower Low)
    for i in range(1, len(swing_lows)):
        if swing_lows[i][1] > swing_lows[i-1][1]:
            structure['hl_count'] += 1
        else:
            structure['ll_count'] += 1
    
    # Determinar tendencia por estructura
    alcista_score = structure['hh_count'] + structure['hl_count']
    bajista_score = structure['ll_count'] + structure['lh_count']
    
    if alcista_score > bajista_score + 1:
        structure['trend'] = 'UPTREND'
        structure['last_pivot'] = 'HL' if swing_lows and swing_lows[-1][0] > (swing_highs[-1][0] if swing_highs else 0) else 'HH'
    elif bajista_score > alcista_score + 1:
        structure['trend'] = 'DOWNTREND'
        structure['last_pivot'] = 'LL' if swing_lows and swing_lows[-1][0] > (swing_highs[-1][0] if swing_highs else 0) else 'LH'
    else:
        structure['trend'] = 'LATERAL'
    
    print(f"         📊 Estructura: {structure['trend']} | HH:{structure['hh_count']} HL:{structure['hl_count']} LL:{structure['ll_count']} LH:{structure['lh_count']}")
    
    # === ANÁLISIS PARA LONG (BULLISH) ===
    if direction == 'BULLISH':
        # Calcular movimientos
        lowest_idx = lows.index(min(lows)) if lows else -1
        current_close = closes[-1]
        previous_close = closes[-2]
        lowest_low = min(lows) if lows else 0
        highest_high = max(highs) if highs else 0
        
        # >>> NUEVO: BLOQUEAR LONG EN ZONA DE RESISTENCIA
        # El bot no debe entrar LONG justo en la resistencia (probable rechazo)
        if highest_high > 0:
            dist_to_resistance = abs(highest_high - current_close) / highest_high * 100
            if dist_to_resistance < 1.0:  # Menos de 1% del máximo
                return {
                    'ready': False,
                    'pattern': 'AT_RESISTANCE',
                    'phase': 'RESISTENCIA',
                    'reason': f'LONG en zona de RESISTENCIA (a {dist_to_resistance:.2f}% del máximo)',
                    'wait_for': 'Esperar rechazo o ruptura de resistencia',
                    'structure': structure
                }
        
        # Detectar fases
        # FASE 1: Corrección (precio bajando)
        if closes[-3] > closes[-2] > closes[-1]:
            return {
                'ready': False,
                'pattern': 'CORRECTION_IN_PROGRESS',
                'phase': 'CORRECCION',
                'reason': 'Precio bajando - esperar piso',
                'wait_for': 'Rebote desde piso',
                'structure': structure
            }
        
        # FASE 2: Piso detectado (mínimo tocado, precio rebotando)
        if lowest_idx >= 2 and current_close > lowest_low * 1.005:
            # Verificar que hay rebote confirmado
            if current_close > previous_close:
                return {
                    'ready': True,
                    'pattern': 'FLOOR_BOUNCE',
                    'phase': 'IMPULSO',
                    'reason': f'Rebote desde piso ${lowest_low:.4f} confirmado',
                    'wait_for': None,
                    'structure': structure
                }
            else:
                return {
                    'ready': False,
                    'pattern': 'FLOOR_TOUCHED',
                    'phase': 'PISO',
                    'reason': 'Piso tocado, esperando confirmación de rebote',
                    'wait_for': 'Vela verde desde piso',
                    'structure': structure
                }
        
        # FASE 3: Impulso en curso (subiendo) - Requiere 4 velas
        # Joel: "El sniper debe esperar un poco para que se desarrolle la tendencia"
        if len(closes) >= 5 and closes[-2] < closes[-1] and closes[-3] < closes[-2] and closes[-4] < closes[-3]:
            # >>> NUEVO: Verificar que el movimiento es FUERTE (no solo subiendo)
            price_change_pct = ((closes[-1] - closes[-3]) / closes[-3]) * 100
            
            if price_change_pct >= 0.5:  # Subida fuerte (>=0.5% en 3 velas)
                return {
                    'ready': True,
                    'pattern': 'IMPULSE_STRONG',
                    'phase': 'EXPLOSIÓN',
                    'reason': f'Impulso FUERTE confirmado (+{price_change_pct:.2f}% en 3 velas)',
                    'wait_for': None,
                    'structure': structure,
                    'price_momentum': price_change_pct
                }
            else:
                return {
                    'ready': True,
                    'pattern': 'IMPULSE_CONFIRMED',
                    'phase': 'IMPULSO',
                    'reason': f'Impulso alcista (+{price_change_pct:.2f}%)',
                    'wait_for': None,
                    'structure': structure,
                    'price_momentum': price_change_pct
                }
        
        # FASE 4: Lateral (sin dirección clara)
        return {
            'ready': False,
            'pattern': 'LATERAL',
            'phase': 'CONSOLIDACION',
            'reason': 'Sin patrón claro - lateralizando',
            'wait_for': 'Dirección definida',
            'structure': structure
        }
    
    # === ANÁLISIS PARA SHORT (BEARISH) ===
    else:
        highest_idx = highs.index(max(highs)) if highs else -1
        current_close = closes[-1]
        previous_close = closes[-2]
        highest_high = max(highs) if highs else 0
        lowest_low = min(lows) if lows else 0
        
        # >>> NUEVO: BLOQUEAR SHORT EN ZONA DE SOPORTE
        # El bot no debe entrar SHORT justo en el soporte (probable rebote)
        if lowest_low > 0:
            dist_to_support = abs(current_close - lowest_low) / lowest_low * 100
            if dist_to_support < 1.0:  # Menos de 1% del mínimo
                return {
                    'ready': False,
                    'pattern': 'AT_SUPPORT',
                    'phase': 'SOPORTE',
                    'reason': f'SHORT en zona de SOPORTE (a {dist_to_support:.2f}% del mínimo)',
                    'wait_for': 'Esperar rebote o ruptura del soporte',
                    'structure': structure
                }
        
        # FASE 1: Rally (precio subiendo - contra-tendencia)
        if closes[-3] < closes[-2] < closes[-1]:
            return {
                'ready': False,
                'pattern': 'RALLY_IN_PROGRESS',
                'phase': 'RALLY',
                'reason': 'Precio subiendo - esperar techo',
                'wait_for': 'Rechazo desde techo',
                'structure': structure
            }
        
        # FASE 2: Techo detectado (máximo tocado, precio cayendo)
        if highest_idx >= 2 and current_close < highest_high * 0.995:
            if current_close < previous_close:
                return {
                    'ready': True,
                    'pattern': 'CEILING_REJECTION',
                    'phase': 'CAIDA',
                    'reason': f'Rechazo desde techo ${highest_high:.4f} confirmado',
                    'wait_for': None,
                    'structure': structure
                }
            else:
                return {
                    'ready': False,
                    'pattern': 'CEILING_TOUCHED',
                    'phase': 'TECHO',
                    'reason': 'Techo tocado, esperando confirmación de caída',
                    'wait_for': 'Vela roja desde techo',
                    'structure': structure
                }
        
        # FASE 3: Caída en curso - Requiere 4 velas (igual que LONG)
        if len(closes) >= 5 and closes[-2] > closes[-1] and closes[-3] > closes[-2] and closes[-4] > closes[-3]:
            # >>> NUEVO: Verificar que el movimiento es FUERTE (no solo cayendo)
            price_change_pct = ((closes[-3] - closes[-1]) / closes[-3]) * 100
            
            if price_change_pct >= 0.5:  # Caída fuerte (>=0.5% en 3 velas)
                return {
                    'ready': True,
                    'pattern': 'DROP_STRONG',
                    'phase': 'EXPLOSIÓN',
                    'reason': f'Caída FUERTE confirmada (-{price_change_pct:.2f}% en 3 velas)',
                    'wait_for': None,
                    'structure': structure,
                    'price_momentum': -price_change_pct
                }
            else:
                return {
                    'ready': True,
                    'pattern': 'DROP_CONFIRMED',
                    'phase': 'CAIDA',
                    'reason': f'Caída confirmada (-{price_change_pct:.2f}%)',
                    'wait_for': None,
                    'structure': structure,
                    'price_momentum': -price_change_pct
                }
        
        # FASE 4: Lateral
        return {
            'ready': False,
            'pattern': 'LATERAL',
            'phase': 'CONSOLIDACION',
            'reason': 'Sin patrón claro - lateralizando',
            'wait_for': 'Dirección definida',
            'structure': structure
        }

def check_price_velocity(symbol, direction, timeframe='5m'):
    """
    v5.2: Mide velocidad del precio en TIEMPO REAL para scalping.
    
    Joel: "Si BTC pasa de 80.300 a 80.900 en 1 min, ENTRAR RÁPIDO"
    
    Correcciones Sonnet:
    - Usa fetch_current_price() para precio actual (no vela cerrada)
    - Default timeframe 5m (más estable que 1m)
    - Mode VELOCITY_SCALP para explosiones
    
    Returns:
        dict: {'valid': bool, 'velocity': float, 'mode': str, 'reason': str}
    """
    from brain_config import VELOCITY_TRIGGER_PCT, ENABLE_MOTOR_VELOCITY
    from exchange_connection import fetch_current_price
    
    # v5.4.1 FIX: Velocity es MÉTRICA, Motor es ESTRATEGIA
    # Velocity SIEMPRE se calcula. Motor decide si ACTÚA en base a velocity.
    # Antes: Si motor apagado, velocity = 0 (MAL - perdíamos la métrica)
    # Ahora: Velocity se calcula siempre, motor solo controla si scalping activo
    
    storage = get_frame_storage(symbol, timeframe)
    
    if not storage.has_enough_frames(2):
        return {
            'valid': False,
            'velocity': 0,
            'mode': 'NORMAL',
            'reason': f'Datos insuficientes ({storage.get_frame_count()} frames)',
            'frames_analyzed': storage.get_frame_count()
        }
    
    # === OBTENER PRECIO ACTUAL EN TIEMPO REAL (Sonnet fix) ===
    try:
        current_price = fetch_current_price(symbol)
        if not current_price or current_price == 0:
            raise ValueError("Precio 0")
    except Exception as e:
        # Fallback a último close si falla API
        closes = storage.get_trajectory('ohlc.close', 3)
        current_price = closes[-1] if closes and len(closes) > 0 else 0
        if current_price == 0:
            return {'valid': False, 'mode': 'ERROR', 'velocity': 0, 'reason': f'Error precio: {e}'}
    
    # Obtener cierres históricos
    closes = storage.get_trajectory('ohlc.close', 3)
    if not closes or len(closes) < 2:
        closes = storage.get_trajectory('price', 3)
    
    if not closes or len(closes) < 2:
        return {'valid': False, 'mode': 'NORMAL', 'velocity': 0, 'reason': 'Sin historial de cierres'}
    
    last_close = closes[-1]  # Último cierre confirmado
    prev_close = closes[-2]  # Cierre anterior
    
    # === CALCULAR VELOCIDAD ACTUAL (precio ahora vs último cierre) ===
    velocity_pct = ((current_price - last_close) / last_close) * 100 if last_close > 0 else 0
    
    # Calcular aceleración
    prev_velocity = ((last_close - prev_close) / prev_close) * 100 if prev_close > 0 else 0
    acceleration = velocity_pct - prev_velocity
    
    print(f"   🏎️ VELOCIDAD {symbol}: {velocity_pct:.3f}%/vela (Accel: {acceleration:.3f})")
    print(f"      Precio actual: {current_price:.6f} | Último cierre: {last_close:.6f}")
    
    # === DETECCIÓN DE EXPLOSIÓN PARA SCALPING ===
    if direction == 'BULLISH':
        if velocity_pct > VELOCITY_TRIGGER_PCT:
            return {
                'valid': True,
                'velocity': velocity_pct,
                'acceleration': acceleration,
                'mode': 'VELOCITY_SCALP',
                'reason': f'🚀 EXPLOSIÓN: +{velocity_pct:.2f}% (>{VELOCITY_TRIGGER_PCT}%)',
                'current_price': current_price,
                'frames_analyzed': len(closes)
            }
        elif velocity_pct > 0.10:
            return {
                'valid': True,
                'velocity': velocity_pct,
                'mode': 'MODERATE_UP',
                'reason': f'Precio subiendo +{velocity_pct:.2f}%',
                'frames_analyzed': len(closes)
            }
        elif velocity_pct < -0.25:  # v5.4.3d: Subido de -0.1 a -0.25
            return {
                'valid': False,
                'velocity': velocity_pct,
                'mode': 'OPPOSITE',
                'reason': f'Precio BAJANDO {velocity_pct:.2f}% (contradice LONG)',
                'frames_analyzed': len(closes)
            }
        else:
            return {
                'valid': False,
                'velocity': velocity_pct,
                'mode': 'FLAT',
                'reason': f'Velocidad plana ({velocity_pct:.2f}%)',
                'frames_analyzed': len(closes)
            }
    
    elif direction == 'BEARISH':
        if velocity_pct < -VELOCITY_TRIGGER_PCT:
            return {
                'valid': True,
                'velocity': velocity_pct,
                'acceleration': acceleration,
                'mode': 'VELOCITY_SCALP',
                'reason': f'🔻 DESPLOME: {velocity_pct:.2f}%',
                'current_price': current_price,
                'frames_analyzed': len(closes)
            }
        elif velocity_pct < -0.10:
            return {
                'valid': True,
                'velocity': velocity_pct,
                'mode': 'MODERATE_DOWN',
                'reason': f'Precio bajando {velocity_pct:.2f}%',
                'frames_analyzed': len(closes)
            }
        elif velocity_pct > 0.25:  # v5.4.3d: Subido de 0.1 a 0.25
            return {
                'valid': False,
                'velocity': velocity_pct,
                'mode': 'OPPOSITE',
                'reason': f'Precio SUBIENDO +{velocity_pct:.2f}% (contradice SHORT)',
                'frames_analyzed': len(closes)
            }
        else:
            return {
                'valid': False,
                'velocity': velocity_pct,
                'mode': 'FLAT',
                'reason': f'Velocidad plana ({velocity_pct:.2f}%)',
                'frames_analyzed': len(closes)
            }
    
    return {'valid': False, 'velocity': velocity_pct, 'mode': 'NORMAL', 'reason': 'Dirección desconocida'}

def unified_time_ladder_vision(symbol, direction, zigzag_data, grid_data, timeframe='15m'):

           las tres cosas unirlas para sacar una conclusión, una sola visión."

        dict: {'approved': bool, 'score': int, 'issues': list, 'validations': dict}

    issues = []
    score = 100

    price_pattern = detect_price_pattern(symbol, direction, timeframe)
    validations['price_pattern'] = price_pattern
    
    if not price_pattern.get('ready'):
        score -= 40
        issues.append(f"Precio: {price_pattern.get('reason', 'Sin patrón')}")

    zigzag_trend = zigzag_data.get('current_trend', 'LATERAL')
    validations['zigzag'] = {'trend': zigzag_trend, 'aligned': False}

        if zigzag_trend == 'BEARISH':
            score -= 50
            issues.append('ZigZag en ROJO (corrección bajista activa)')
        elif zigzag_trend == 'BULLISH':
            validations['zigzag']['aligned'] = True
        else:
            score -= 15
            issues.append('ZigZag LATERAL')
    else:  # BEARISH
        if zigzag_trend == 'BULLISH':
            score -= 50
            issues.append('ZigZag en VERDE (rally activo)')
        elif zigzag_trend == 'BEARISH':
            validations['zigzag']['aligned'] = True
        else:
            score -= 15
            issues.append('ZigZag LATERAL')
    
    # === VALIDACIÓN 3: GRID/MALLA ===
    grid_position = grid_data.get('position_pct', 50)
    validations['grid'] = {'position_pct': grid_position, 'safe_zone': True}
    
    if direction == 'BULLISH':
        if grid_position > 90:
            score -= 50  # v5.4.3b: Reducido de 80
            issues.append(f'BULLISH en RESISTENCIA EXTREMA (grid {grid_position:.0f}%)')
            validations['grid']['safe_zone'] = False
        elif grid_position > 80:
            score -= 30  # v5.4.3b: Reducido de 50
            issues.append(f'BULLISH cerca resistencia (grid {grid_position:.0f}%)')
            validations['grid']['safe_zone'] = False
        elif grid_position > 70:
            score -= 15  # v5.4.3b: Reducido de 30
            issues.append(f'Advertencia resistencia ({grid_position:.0f}%)')
    else:  # BEARISH
        if grid_position < 10:
            score -= 50  # v5.4.3b: Reducido de 80
            issues.append(f'BEARISH en SOPORTE EXTREMO (grid {grid_position:.0f}%)')
            validations['grid']['safe_zone'] = False
        elif grid_position < 20:
            score -= 30  # v5.4.3b: Reducido de 50
            issues.append(f'BEARISH cerca soporte (grid {grid_position:.0f}%)')
            validations['grid']['safe_zone'] = False
        elif grid_position < 30:
            score -= 15  # v5.4.3b: Reducido de 30
            issues.append(f'Advertencia soporte ({grid_position:.0f}%)')
    
    # === VALIDACIÓN 4: VELOCIDAD DE PRECIO ===
    # Usar mismo timeframe que analysis (no 5m que no tiene frames)
    velocity_check = check_price_velocity(symbol, direction, timeframe)
    validations['velocity'] = velocity_check

def detect_early_ignition(symbol, direction, timeframe='5m'):
    """
    Detecta explosión EN FORMACIÓN (antes de confirmación completa).
    
    Joel: "El bot entra tarde cuando ya explotó. Necesita anticipar."
    
    Señales tempranas:
    1. Volume spike >200% promedio (capital entrando)
    2. Vela actual >0.15% sin cerrar (movimiento iniciando)
    3. Rompe máximo/mínimo anterior (estructura rompiendo)
    4. ADX velocity >2 pts/vela (fuerza acelerando)
    
    Returns:
        {
            'is_igniting': bool,
            'confidence': str ('HIGH', 'MEDIUM', 'LOW'),
            'reason': str,
            'signals': list
        }
    """
    from exchange_connection import fetch_current_price
    
    storage = get_frame_storage(symbol, timeframe)
    
    if not storage.has_enough_frames(25):
        return {'is_igniting': False, 'confidence': 'NONE', 'reason': 'Datos insuficientes'}
    
    try:
        df = storage.df
    except:
        return {'is_igniting': False, 'confidence': 'NONE', 'reason': 'Sin DataFrame'}
    
    if len(df) < 25:
        return {'is_igniting': False, 'confidence': 'NONE', 'reason': 'Datos insuficientes'}
    
    signals = []
    confidence_score = 0
    
    # v5.4.3 Performance: Fetch precio UNA sola vez
    try:
        current_price = fetch_current_price(symbol)
    except:
        current_price = None
    
    # ===== SEÑAL 1: VOLUME SPIKE =====
    try:
        volume_current = float(df['volume'].iloc[-1])
        volume_avg = df['volume'].iloc[-20:-1].mean()
        volume_ratio = volume_current / volume_avg if volume_avg > 0 else 0
        
        if volume_ratio > 2.5:
            signals.append(f'Volume spike x{volume_ratio:.1f}')
            confidence_score += 40
        elif volume_ratio > 1.8:
            signals.append(f'Volume elevado x{volume_ratio:.1f}')
            confidence_score += 25
    except:
        pass
    
    # ===== SEÑAL 2: VELA EN FORMACIÓN =====
    try:
        if current_price:
            current_open = float(df['open'].iloc[-1])
            candle_change_pct = ((current_price - current_open) / current_open) * 100
            
            if direction == 'BULLISH':
                if candle_change_pct > 0.20:
                    signals.append(f'Vela verde +{candle_change_pct:.2f}%')
                    confidence_score += 30
                elif candle_change_pct > 0.10:
                    signals.append(f'Vela subiendo +{candle_change_pct:.2f}%')
                    confidence_score += 15
            else:  # BEARISH
                if candle_change_pct < -0.20:
                    signals.append(f'Vela roja {candle_change_pct:.2f}%')
                    confidence_score += 30
                elif candle_change_pct < -0.10:
                    signals.append(f'Vela bajando {candle_change_pct:.2f}%')
                    confidence_score += 15
    except:
        pass
    
    # ===== SEÑAL 3: RUPTURA ESTRUCTURA =====
    try:
        if current_price:
            prev_high = float(df['high'].iloc[-2])
            prev_low = float(df['low'].iloc[-2])
            
            if direction == 'BULLISH' and current_price > prev_high:
                signals.append('Rompió máximo anterior')
                confidence_score += 20
            elif direction == 'BEARISH' and current_price < prev_low:
                signals.append('Rompió mínimo anterior')
                confidence_score += 20
    except:
        pass
    
    # ===== SEÑAL 4: ADX ACELERANDO =====
    try:
        adx_data = storage.get_trajectory('adx.adx_current', 5)
        if adx_data and len(adx_data) >= 3:
            adx_velocity = adx_data[-1] - adx_data[-2]
            if adx_velocity > 2:
                signals.append(f'ADX acelerando +{adx_velocity:.1f}')
                confidence_score += 10
    except:
        pass
    
    # ===== EVALUACIÓN FINAL =====
    if confidence_score >= 60:
        confidence = 'HIGH'
        is_igniting = True
    elif confidence_score >= 40:
        confidence = 'MEDIUM'
        is_igniting = True
    else:
        confidence = 'LOW'
        is_igniting = False
    
    return {
        'is_igniting': is_igniting,
        'confidence': confidence,
        'confidence_score': confidence_score,
        'reason': ', '.join(signals) if signals else 'Sin señales',
        'signals': signals
    }

def check_5m_veto(symbol: str, direction: str, ladder_data: dict) -> dict:
    """
    5M puede vetar entrada si detecta condiciones críticas.
    
    Time Ladder Puro:
    - No hay hard blocks individuales
    - Penalties se acumulan
    - Veto solo si total >= 100
    
    Args:
        symbol: Par de trading
        direction: 'BULLISH' o 'BEARISH' del trade propuesto
        ladder_data: Dict con análisis de todos los timeframes
        
    Returns:
        {
            'should_veto': bool,
            'reason': str,
            'penalty': int,
            'details': list
        }
    """
    from brain_config import (
        ENABLE_5M_VETO,
        VETO_5M_ADX_MIN,
        VETO_5M_ADX_VELOCITY_CRITICAL,
        VETO_5M_MOMENTUM_PENALTY,
        VETO_5M_CONFLICT_PENALTY,
        VETO_5M_LATERAL_PENALTY,
        VETO_5M_ADX_WEAK_PENALTY,
        VETO_5M_ADX_COLLAPSE_PENALTY,
        VETO_5M_UNKNOWN_PENALTY,
        VETO_5M_THRESHOLD
    )
    
    # Kill switch
    if not ENABLE_5M_VETO:
        return {'should_veto': False, 'reason': 'Veto disabled', 'penalty': 0, 'details': []}
    
    # Obtener data 5M
    data_5m = ladder_data.get('5m', {})
    
    # Edge case: Sin datos de 5M
    if not data_5m:
        print(f"   ⚠️ 5M VETO: Sin datos 5M para {symbol}")
        return {'should_veto': False, 'reason': 'No 5M data', 'penalty': 0, 'details': []}
    
    # Extraer campos con validación
    adx_5m = data_5m.get('adx', {})
    adx_current = adx_5m.get('adx_current', None)
    adx_trend = adx_5m.get('adx_trend', 'insuficiente')
    adx_velocity = adx_5m.get('adx_velocity', 0)
    
    mom_5m = data_5m.get('momentum', {})
    mom_color = mom_5m.get('color_current', 'gray')
    
    direction_5m = data_5m.get('direction', 'UNKNOWN')
    
    # Edge case: ADX current es None (sin datos válidos)
    if adx_current is None:
        print(f"   ⚠️ 5M VETO: ADX 5M sin valor para {symbol}")
        return {'should_veto': False, 'reason': 'No ADX data', 'penalty': 0, 'details': []}
    
    # Acumuladores Time Ladder
    penalties = []
    vetos = []
    
    # ===============================================
    # CONDICIÓN 1: ADX débil + decreciendo (COMBO)
    # No es veto individual, es penalty
    # ===============================================
    if adx_current < VETO_5M_ADX_MIN and adx_trend == 'decreciendo':
        penalty = VETO_5M_ADX_WEAK_PENALTY  # 50
        penalties.append(penalty)
        vetos.append(f'ADX 5M {adx_current:.1f} < {VETO_5M_ADX_MIN} + decreciendo')
        print(f"   ⚠️ 5M: ADX {adx_current:.1f} débil + decreciendo → penalty -{penalty}")
    
    # ===============================================
    # CONDICIÓN 2: ADX colapsando rápido
    # Velocity < -2 pts/frame = colapso crítico
    # ===============================================
    if adx_velocity < VETO_5M_ADX_VELOCITY_CRITICAL:
        penalty = VETO_5M_ADX_COLLAPSE_PENALTY  # 40
        penalties.append(penalty)
        vetos.append(f'ADX 5M colapsando ({adx_velocity:.1f} pts/frame)')
        print(f"   🚨 5M: ADX colapsando rápido ({adx_velocity:.1f} pts/frame) → penalty -{penalty}")
    
    # ===============================================
    # CONDICIÓN 3: Direction LATERAL en 5M
    # Penalty gradual, no veto absoluto
    # ===============================================
    if direction_5m == 'LATERAL':
        penalty = VETO_5M_LATERAL_PENALTY  # 40
        penalties.append(penalty)
        vetos.append('5M Direction LATERAL')
        print(f"   ⚠️ 5M: Direction LATERAL → penalty -{penalty}")
    
    # ===============================================
    # CONDICIÓN 4: Direction UNKNOWN (sin claridad)
    # ===============================================
    if direction_5m == 'UNKNOWN':
        penalty = VETO_5M_UNKNOWN_PENALTY  # 30
        penalties.append(penalty)
        vetos.append('5M Direction UNKNOWN')
        print(f"   ⚠️ 5M: Direction UNKNOWN → penalty -{penalty}")
    
    # ===============================================
    # CONDICIÓN 5: Momentum 5M contradice trade (CRÍTICO)
    # ===============================================
    if direction == 'BULLISH' and mom_color in ['red', 'maroon']:
        penalty = VETO_5M_MOMENTUM_PENALTY  # 60
        penalties.append(penalty)
        vetos.append(f'Momentum 5M {mom_color} contradice LONG')
        print(f"   🚨 5M: Momentum {mom_color} contradice LONG → penalty -{penalty}")
    elif direction == 'BEARISH' and mom_color in ['green', 'lime']:
        penalty = VETO_5M_MOMENTUM_PENALTY  # 60
        penalties.append(penalty)
        vetos.append(f'Momentum 5M {mom_color} contradice SHORT')
        print(f"   🚨 5M: Momentum {mom_color} contradice SHORT → penalty -{penalty}")
    
    # ===============================================
    # CONDICIÓN 6: Conflicto de dirección 5M vs trade
    # ===============================================
    if direction_5m == 'BEARISH' and direction == 'BULLISH':
        penalty = VETO_5M_CONFLICT_PENALTY  # 50
        penalties.append(penalty)
        vetos.append('5M BEARISH pero trade es BULLISH')
        print(f"   🚨 5M: Direction {direction_5m} contradice trade {direction} → penalty -{penalty}")
    elif direction_5m == 'BULLISH' and direction == 'BEARISH':
        penalty = VETO_5M_CONFLICT_PENALTY  # 50
        penalties.append(penalty)
        vetos.append('5M BULLISH pero trade es BEARISH')
        print(f"   🚨 5M: Direction {direction_5m} contradice trade {direction} → penalty -{penalty}")
    
    # ===============================================
    # DECISIÓN FINAL: Veto si penalty total >= threshold
    # ===============================================
    total_penalty = sum(penalties)
    should_veto = total_penalty >= VETO_5M_THRESHOLD  # 100
    
    if should_veto:
        print(f"\n   🚫 5M VETO ACTIVADO: Penalty total {total_penalty} >= {VETO_5M_THRESHOLD}")
        print(f"      Razones: {' | '.join(vetos)}")
    elif total_penalty > 0:
        print(f"   ⚠️ 5M Penalty: {total_penalty} (< {VETO_5M_THRESHOLD}, no veto)")
    else:
        print(f"   ✅ 5M: Sin problemas detectados")
    
    return {
        'should_veto': should_veto,
        'reason': ' | '.join(vetos) if vetos else '',
        'penalty': total_penalty,
        'details': vetos
    }

        Criterios EXPANDIDOS para no rechazar buenos setups:
        - Score >= 40 (antes era 50-65)
        - O WAIT/HOLD con buena razón
        - O tiene buena fuerza pero espera datos
        """
        # Cualquier score >= 40 merece seguimiento
        if score >= 40:
            return True
        
        # WAIT o HOLD van a waiting list
        if decision['action'] in ['WAIT', 'HOLD']:
            return True
        
        # Si tiene force_result con buena fuerza, esperar
        full_analysis = decision.get('full_analysis', {})
        force_result = full_analysis.get('force_result', {})
        if force_result.get('has_force', False):
            return True
        
        return False


class SniperQueue:
    """
    Gestor de colas dinámicas:
    - Priority Queue: Monedas con mejor setup
    - Waiting List: Monedas cerca del setup
    """
    
    def __init__(self, max_priority_size: int = 10):
        self.priority_queue: List[CoinSetup] = []
        self.waiting_list: List[CoinSetup] = []
        self.rejected_cache: Dict[str, datetime] = {}  # symbol: last_check_time
        self.max_priority_size = max_priority_size
    
    def add_to_priority(self, coin_setup: CoinSetup):
        """Agrega o actualiza moneda en Priority Queue"""
        # Remover si ya existe en cualquier lista
        self.remove_symbol(coin_setup.symbol)
        
        # Agregar a priority
        self.priority_queue.append(coin_setup)
        
        # Ordenar por score descendente
        self.priority_queue.sort(key=lambda x: x.score, reverse=True)
        
        # Limitar tamaño
        if len(self.priority_queue) > self.max_priority_size:
            # Degradar la peor a waiting list
            demoted = self.priority_queue.pop()
            demoted.classification = 'WAITING'
            self.waiting_list.append(demoted)
        
        print(f"  🎯 PRIORITY: {coin_setup.symbol} (Score: {coin_setup.score:.1f})")
    
    def add_to_waiting(self, coin_setup: CoinSetup):
        """Agrega o actualiza moneda en Waiting List"""
        # Remover si ya existe
        self.remove_symbol(coin_setup.symbol)
        
        # Agregar a waiting
        self.waiting_list.append(coin_setup)
        
        # Ordenar por score descendente
        self.waiting_list.sort(key=lambda x: x.score, reverse=True)
        
        print(f"  ⏳ WAITING: {coin_setup.symbol} (Score: {coin_setup.score:.1f})")
    
    def reject_symbol(self, symbol: str):
        """Marca símbolo como rechazado (cache temporal)"""
        self.remove_symbol(symbol)
        self.rejected_cache[symbol] = datetime.now(ARGENTINA_TZ)
        print(f"  ❌ REJECTED: {symbol}")
    
    def remove_symbol(self, symbol: str):
        """Remueve símbolo de todas las listas"""
        self.priority_queue = [c for c in self.priority_queue if c.symbol != symbol]
        self.waiting_list = [c for c in self.waiting_list if c.symbol != symbol]
    
    def is_recently_rejected(self, symbol: str, minutes: int = 10) -> bool:
        """Verifica si el símbolo fue rechazado recientemente"""
        if symbol not in self.rejected_cache:
            return False
        
        elapsed = (datetime.now(ARGENTINA_TZ) - self.rejected_cache[symbol]).total_seconds() / 60
        return elapsed < minutes
    
    def get_priority_coins(self) -> List[CoinSetup]:
        """Retorna lista de monedas en priority queue"""
        return self.priority_queue.copy()
    
    def get_waiting_coins(self) -> List[CoinSetup]:
        """Retorna lista de monedas en waiting list"""
        return self.waiting_list.copy()
    
    def get_best_setup(self) -> Optional[CoinSetup]:
        """Retorna el mejor setup disponible"""
        if self.priority_queue:
            return self.priority_queue[0]
        return None
    
    def cleanup_stale(self, max_age_minutes: int = 15):
        """Limpia setups antiguos"""
        now = datetime.now(ARGENTINA_TZ)
        
        # Limpiar priority
        self.priority_queue = [
            c for c in self.priority_queue 
            if (now - c.last_update).total_seconds() < max_age_minutes * 60
        ]
        
        # Limpiar waiting
        self.waiting_list = [
            c for c in self.waiting_list 
            if (now - c.last_update).total_seconds() < max_age_minutes * 60
        ]
        
        # Limpiar cache de rechazados
        self.rejected_cache = {
            symbol: timestamp 
            for symbol, timestamp in self.rejected_cache.items()
            if (now - timestamp).total_seconds() < max_age_minutes * 60
        }

        """
        Analiza siguiente lote de monedas.

            int: Cantidad de monedas analizadas
        """
        if not self.symbols_to_scan:
            return 0
        
        analyzed_count = 0
        
        for _ in range(batch_size):
            if self.scan_index >= len(self.symbols_to_scan):
                self.scan_index = 0  # Reiniciar desde el principio
                self.cycle_count += 1
                print(f"\n🔄 SNIPER Ciclo #{self.cycle_count} completado")
            
            symbol = self.symbols_to_scan[self.scan_index]
            self.scan_index += 1
            
            # Skip si fue rechazado recientemente
            if self.queue.is_recently_rejected(symbol, minutes=10):
                continue
            
            # Analizar y clasificar
            self.classify_coin(symbol)
            analyzed_count += 1
        
        return analyzed_count
    
    def classify_coin(self, symbol: str):
        """
        Analiza y clasifica una moneda en Priority/Waiting/Rejected.
        """
        try:
            # Filtro rápido: Stablecoins
            base_currency = symbol.split('/')[0]
            if base_currency in ['USDC', 'USDT', 'DAI', 'BUSD', 'TUSD']:
                return
            
            # Análisis inteligente
            decision = decide_intelligently(symbol)

            score = self.scorer.score_setup(symbol, decision)
            
            # Crear objeto setup
            coin_setup = CoinSetup(
                symbol=symbol,

            full_analysis = decision.get('full_analysis', {})
            force_result = full_analysis.get('force_result', {})

            force_level = force_result.get('force_level', 'NONE')
            
            # FILOSOFÍA DE JOEL: "Primero fuerza, luego setup"
            # Si tiene fuerza, debe ir a waiting para esperar el setup perfecto
            
            # 1. Setup perfecto → Priority (operar ahora)
            if self.scorer.is_perfect_setup(symbol, score, decision):

        """Convierte decisión inteligente a formato compatible con trade_manager"""

        strategy = decision.get('strategy', {})
        
        # Obtener precio actual de 5m
        current_price = ladder_data.get('5m', {}).get('current_price', 0)
        direction = ladder_data.get('5m', {}).get('direction', 'UNKNOWN')
        
        # Calcular Stop Loss y Target basado en strategy
        stop_mult = strategy.get('stop_multiplier', 2.0)
        target_mult = strategy.get('target_multiplier', 2.0)
        
        # Calcular Stop Loss (2% por defecto)
        from brain_config import STOP_LOSS_PERCENT
        sl_percent = STOP_LOSS_PERCENT / 100  # 2% = 0.02
        
        if direction == 'BULLISH':
            stop_loss = current_price * (1 - sl_percent * stop_mult)
            target_price = current_price * (1 + sl_percent * target_mult)
        elif direction == 'BEARISH':
            stop_loss = current_price * (1 + sl_percent * stop_mult)
            target_price = current_price * (1 - sl_percent * target_mult)
        else:
            # Fallback si direction es UNKNOWN
            stop_loss = current_price * 0.98
            target_price = current_price * 1.04
        
        # Calcular R:R ratio
        rr_ratio = abs(target_price - current_price) / abs(current_price - stop_loss) if stop_loss != current_price else 0
        
        return {
            'viable': True,
            'score': decision.get('confidence', 60),
            'clasificacion': decision.get('trade_type', 'PATTERN'),
            'direction': direction,
            'analysis_snapshot': ladder_data,
            'intelligent_decision': decision,
            'triggers': {
                'target_price': target_price,
                'stop_loss': stop_loss,
                'rr_ratio': rr_ratio,
                'tiempo_proyectado_min': 30  # Estimado default
            }
        }
