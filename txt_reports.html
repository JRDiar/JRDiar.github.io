Lógica:
pythondef check_5m_veto(symbol, direction, ladder_data):
    """
    5M puede vetar entrada si detecta condiciones peligrosas.
    
    Returns:
        {
            'should_veto': bool,
            'reason': str,
            'penalty': int
        }
    """
    data_5m = ladder_data.get('5m', {})
    
    # Verificar condiciones de veto
    vetos = []
    
    # Veto 1: ADX muriendo en 5M
    adx_5m = data_5m.get('adx', {}).get('adx_current', 0)
    if adx_5m < 18:
        vetos.append('ADX 5M débil')
    
    # Veto 2: Direction LATERAL en 5M
    direction_5m = data_5m.get('direction', 'UNKNOWN')
    if direction_5m == 'LATERAL':
        vetos.append('5M LATERAL')
    
    # Veto 3: Momentum opuesto en 5M
    momentum_5m = data_5m.get('momentum', {})
    mom_color = momentum_5m.get('color_current', 'gray')
    if direction == 'BULLISH' and mom_color in ['red', 'maroon']:
        vetos.append('Momentum 5M contradice LONG')
    elif direction == 'BEARISH' and mom_color in ['green', 'lime']:
        vetos.append('Momentum 5M contradice SHORT')
    
    # Veto 4: ADX colapsando rápido en 5M
    adx_velocity_5m = data_5m.get('adx', {}).get('velocity', 0)
    if adx_velocity_5m < -1.5:
        vetos.append('ADX 5M colapsando')
    
    if vetos:
        return {
            'should_veto': True,
            'reason': ', '.join(vetos),
            'penalty': 100  # Forzar bloqueo
        }
    
    return {'should_veto': False, 'reason': '', 'penalty': 0}
