# Estructura para almacenar la historia reciente en memoria RAM (más rápido que disco para scalping)
class PriceMemory:
    def __init__(self):
        self.history = {}  # {symbol: [{'time': timestamp, 'price': float}]}

    def update(self, symbol, current_price, current_time):
        if symbol not in self.history:
            self.history[symbol] = []
        
        # Agregar nuevo punto
        self.history[symbol].append({'time': current_time, 'price': current_price})
        
        # Mantener solo los últimos 5 minutos (limpieza)
        # Asumiendo que update se llama cada ~10-30 segundos
        if len(self.history[symbol]) > 30: 
            self.history[symbol].pop(0)

    def get_velocity(self, symbol, minutes=3):
        """Calcula cambio porcentual en los últimos X minutos"""
        if symbol not in self.history or len(self.history[symbol]) < 2:
            return 0.0
            
        data = self.history[symbol]
        current = data[-1]
        
        # Buscar el punto más cercano a X minutos atrás
        target_time = current['time'] - (minutes * 60) # timestamp en segundos
        
        # Encontrar el registro más antiguo dentro de la ventana de tiempo
        past_record = data[0] 
        
        velocity = ((current['price'] - past_record['price']) / past_record['price']) * 100
        return velocity
