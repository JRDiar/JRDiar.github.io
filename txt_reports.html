def analyze_1d(symbol):
    ohlcv = fetch_ohlcv(symbol, '1d', 100)
    if not ohlcv or len(ohlcv) < 50:
        return None

    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    adx_data = calculate_adx_with_memory(df, period=14, lookback=10)
    momentum_data = calculate_momentum_with_memory(df, lookback=10)
    rsi_data = calculate_rsi_with_memory(df, period=14, lookback=10)
    emas_data = calculate_emas_with_memory(df, periods=[9, 21, 50, 200], lookback=10)

    adx_current = adx_data['adx_current']
    dominance = adx_data['dominance']
    momentum_color = momentum_data['color_current']
    momentum_count = momentum_data['consecutive_count']

    if adx_current >= 25 and dominance == 'DI+' and momentum_color in ['green', 'lime'] and momentum_count >= 5:
        phase = 'BULLISH'
        strength = 'STRONG'
    elif adx_current >= 25 and dominance == 'DI-' and momentum_color in ['red', 'maroon'] and momentum_count >= 5:
        phase = 'BEARISH'
        strength = 'STRONG'
    elif adx_current < 25 and momentum_color in ['green', 'lime']:
        phase = 'BULLISH'
        strength = 'WEAK'
    elif adx_current < 25 and momentum_color in ['red', 'maroon']:
        phase = 'BEARISH'
        strength = 'WEAK'
    else:
        phase = 'LATERAL'
        strength = 'WEAK'

    print(f"📅 1D | {symbol} | {phase} {strength} | ADX:{adx_current:.1f}")

    return {
        'phase': phase,
        'strength': strength,
        'adx': adx_data,
        'momentum': momentum_data,
        'rsi': rsi_data,
        'emas': emas_data,
        'context': f"1D {phase} - ADX {adx_current:.1f}",
        'timestamp': datetime.now().isoformat()
    }

def analyze_temporal_context(symbol):
    """
    Analiza el contexto temporal reciente (últimos 60 min) usando la DB.
    Busca estabilidad y consistencia.
    """
    try:
        history = db_manager.get_recent_context(symbol, limit_minutes=60)
        if not history or len(history) < 10:
            return {'stability': 'DESCONOCIDO', 'bonus': 0}
        
        # 1. Verificar cambios de dirección (inestabilidad)
        directions = [row['direction'] for row in history]
        flips = 0
        for i in range(1, len(directions)):
            if directions[i] != directions[i-1]:
                flips += 1
        
        # 2. Verificar consistencia de ADX
        adx_values = [row['adx_15m'] for row in history]
        adx_rising = 0
        for i in range(1, len(adx_values)):
            if adx_values[i] > adx_values[i-1]:
                adx_rising += 1
        
        # Evaluación
        bonus = 0
        stability = 'ESTABLE'
        
        if flips > 3:
            stability = 'INESTABLE'
            bonus -= 10  # Penalización por inestabilidad
        elif flips == 0:
            stability = 'MUY_ESTABLE'
            bonus += 10  # Bonus por tendencia sostenida
            
        if adx_rising > len(adx_values) * 0.7:
            bonus += 5  # Bonus si ADX sube la mayoría del tiempo
            
        return {'stability': stability, 'bonus': bonus, 'flips': flips}
        
    except Exception as e:
        print(f"⚠️ Error analizando contexto temporal: {e}")
        return {'stability': 'ERROR', 'bonus': 0}

def analyze_trend_evolution(symbol):
    """
    ANÁLISIS DE EVOLUCIÓN (TIME LADDER / ROADMAP)
    Analiza velas de 1M para detectar tendencias explosivas en tiempo real.
    Permite 'bypassear' filtros estrictos si la moneda está explotando ahora mismo.
    """
    try:
        # Buscar 30 velas de 1M (media hora de evolución)
        ohlcv = fetch_ohlcv(symbol, '1m', 30)
        if not ohlcv or len(ohlcv) < 15:
            return {'is_explosive': False, 'score': 0, 'reason': 'Sin datos 1M'}

        df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
        
        # 1. Análisis de ADX en 1M (Velocidad de tendencia)
        adx_data = calculate_adx_with_memory(df, period=14, lookback=5)
        adx_current = adx_data['adx_current']
        adx_history = adx_data['adx_history']
        
        # 2. Análisis de Precio (Escalera de precios)
        closes = df['close'].values
        price_velocity = calculate_velocity(df, lookback=5)
        
        # DETECCIÓN DE EXPLOSIÓN
        is_explosive = False
        score = 0
        reason = []

        # Criterio A: ADX Disparándose (Roadmap de fuerza)
        # Si ADX subió más de 5 puntos en las últimas 5 velas 1M
        if len(adx_history) >= 5:
            adx_change = adx_history[-1] - adx_history[0]
            if adx_change > 3.0 and adx_current > 20:
                score += 40
                reason.append(f"ADX 1M disparándose (+{adx_change:.1f})")
                if adx_current > 25:
                    is_explosive = True

        # Criterio B: Escalera de Precios (ZigZag alcista agresivo)
        # Verificar si las últimas 3 velas son verdes y con cuerpos crecientes o fuertes
        last_3_closes = closes[-3:]
        last_3_opens = df['open'].values[-3:]
        is_green_ladder = all(c > o for c, o in zip(last_3_closes, last_3_opens))
        
        if is_green_ladder and price_velocity['velocity_current'] > 0:
            score += 30
            reason.append("Escalera de precios 1M alcista")
            if price_velocity['trend'] == 'acelerando':
                is_explosive = True
                score += 20

        # Criterio C: Volumen Climático
        volumes = df['volume'].values
        vol_avg = volumes[:-1].mean()
        if volumes[-1] > vol_avg * 2:
            score += 20
            reason.append("Volumen 1M climático")

        # Decisión final de explosividad
        if score >= 70:
            is_explosive = True
        
        return {
            'is_explosive': is_explosive,
            'score': score,
            'adx_1m': adx_current,
            'reason': ", ".join(reason),
            'timestamp': datetime.now().isoformat()
        }

    except Exception as e:
        print(f"⚠️ Error en analyze_trend_evolution: {e}")
        return {'is_explosive': False, 'score': 0, 'reason': 'Error'}

def _extract_score(score_value, default=0):
    """
    Recursively extract integer score from potentially nested dict.
    
    Args:
        score_value: The value to extract (can be int, float, dict, or None)
        default: Default value if extraction fails
    
    Returns:
        int: Extracted integer score
    """
    if score_value is None:
        return default
    if isinstance(score_value, (int, float)):
        return int(score_value)
    if isinstance(score_value, dict):
        # Recursively extract from nested dict
        return _extract_score(score_value.get('score', default), default)
    # Fallback for any other type
    return default


def calculate_score(analysis_4h, analysis_1h, analysis_15m, analysis_5m):
    """Calcula score usando los datos reales de cada timeframe y CONTEXTO."""
    score = 0  # Initialize as integer
    
    # Usar los scores calculados en cada timeframe - ULTRA DEFENSIVE
    if analysis_4h and 'score_4h' in analysis_4h:
        score_4h_value = _extract_score(analysis_4h['score_4h'], 0)
        try:
            score_4h_value = int(score_4h_value) if score_4h_value is not None else 0
        except (ValueError, TypeError):
            score_4h_value = 0
        score = int(score + min(int(score_4h_value), 30))
    elif analysis_4h and analysis_4h.get('strength') == 'STRONG':
        score = int(score + 25)
    
    if analysis_1h and 'score_1h' in analysis_1h:
        score_1h_value = _extract_score(analysis_1h['score_1h'], 0)
        try:
            score_1h_value = int(score_1h_value) if score_1h_value is not None else 0
        except (ValueError, TypeError):
            score_1h_value = 0
        score = int(score + min(int(score_1h_value), 25))
    elif analysis_1h and analysis_1h.get('strength') == 'STRONG':
        score = int(score + 15)
    
    if analysis_15m and 'score_15m' in analysis_15m:
        score_15m_value = _extract_score(analysis_15m['score_15m'], 0)
        try:
            score_15m_value = int(score_15m_value) if score_15m_value is not None else 0
        except (ValueError, TypeError):
            score_15m_value = 0
        score = int(score + min(int(score_15m_value), 25))
    elif analysis_15m and analysis_15m.get('strength') == 'STRONG':
        score = int(score + 15)
    
    if analysis_5m and 'score_5m' in analysis_5m:
        score_5m_value = _extract_score(analysis_5m['score_5m'], 0)
        try:
            score_5m_value = int(score_5m_value) if score_5m_value is not None else 0
        except (ValueError, TypeError):
            score_5m_value = 0
        score = int(score + min(int(score_5m_value), 20))
    elif analysis_5m and analysis_5m.get('strength') == 'STRONG':
        score = int(score + 10)
    
    # Bonus por alineación de direcciones
    directions = []
    if analysis_4h and 'direction' in analysis_4h:
        directions.append(analysis_4h['direction'])
    if analysis_1h and 'direction' in analysis_1h:
        directions.append(analysis_1h['direction'])
    if analysis_15m and 'direction' in analysis_15m:
        directions.append(analysis_15m['direction'])
    if analysis_5m and 'direction' in analysis_5m:
        directions.append(analysis_5m['direction'])
    
    # Si 3+ direcciones coinciden (excluyendo LATERAL), suma bonus
    non_lateral = [d for d in directions if d != 'LATERAL']
    if len(non_lateral) >= 3 and len(set(non_lateral)) == 1:
        score = int(score + 10)
    
    # --- CONTEXTUAL MULTIPLIERS (Phase 4.1) ---
    multiplier = 1.0
    
    # 1. Volatility (ATR)
    if analysis_15m and 'atr' in analysis_15m and 'current_price' in analysis_15m:
        atr = analysis_15m['atr']
        price = analysis_15m['current_price']
        if price > 0 and (atr / price) > 0.05:
            multiplier *= 0.9
            # print(f"⚠️ High Volatility Penalty (ATR > 5%): x0.9")

    # 2. Time/Liquidity (00-06 UTC assumed low volume)
    try:
        current_utc_hour = datetime.utcnow().hour
        if 0 <= current_utc_hour < 6:
            multiplier *= 0.98  # Reducir penalización
            # print(f"⚠️ Low Liquidity Hours: x0.95")
    except Exception:
        pass # Ignore if time check fails

    # 3. Volume Profile (Price > POC)
    if analysis_15m and 'volume_profile' in analysis_15m and 'current_price' in analysis_15m:
        vp = analysis_15m['volume_profile']
        price = analysis_15m['current_price']
        poc = vp.get('poc_current', 0)
        if poc > 0 and price > poc:
            multiplier *= 1.1
            # print(f"✅ Price above POC Bonus: x1.1")

    # 4. Session Context (Asia Penalty - REDUCIDA)
    hour = datetime.now().hour
    if 2 <= hour <= 8: # Asia Session (aprox)
        multiplier *= 0.95  # Reducir penalización Asia
        # print(f"⚠️ Asia Session Penalty: x0.92")

    score = int(score * multiplier)

    # Ensure score is definitely an integer before comparisons
    score = int(score)
    
    # Clasificación
    clasificacion = 'REGULAR'
    if score >= 90:
        clasificacion = 'ÓPTIMA'
    elif score >= 80:
        clasificacion = 'ALTA'
    elif score >= 70:
        clasificacion = 'BUENA'
    elif score >= 50:
        clasificacion = 'BUENA'
    else:
        clasificacion = 'POBRE'

    return {
        'score': int(score),  # Absolutely ensure it's int
        'clasificacion': clasificacion
    }

def validate_5m_entry_conditions(symbol, direction='BULLISH'):
    """
    Checklist quirúrgico de 5M para entrada precisa.
    NO rechaza - solo informa qué falta.
    
    Returns:
        dict: {
            'ready': bool (True si TODAS las condiciones OK),
            'conditions': {
                'momentum_green': {'ok': bool, 'value': str, 'reason': str},
                'adx_accompanied': {'ok': bool, 'value': str, 'reason': str},
                'not_in_correction': {'ok': bool, 'value': str, 'reason': str},
                'above_ema_support': {'ok': bool, 'value': str, 'reason': str}
            },
            'score': int (0-100, cuántas condiciones OK),
            'missing': [lista de condiciones faltantes]
        }
    """
    try:
        # 1. Fetch Data Fresca (5M)
        ohlcv = fetch_ohlcv(symbol, '5m', 50)
        if not ohlcv or len(ohlcv) < 20:
            return {'ready': False, 'conditions': {}, 'score': 0, 'missing': ['No Data']}
            
        df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
        current_price = float(df['close'].iloc[-1])
        
        # 2. Calcular Indicadores
        mom_data = calculate_momentum_with_memory(df, lookback=5)
        adx_data = calculate_adx_with_memory(df, period=14, lookback=5)
        emas_data = calculate_emas_with_memory(df, periods=[21], lookback=5)
        
        # 3. Validar Condiciones
        conditions = {}
        missing = []
        score_count = 0
        
        # A. Momentum Green
        mom_color = mom_data['color_current']
        if direction == 'BULLISH':
            mom_ok = mom_color in ['green', 'lime']
        else:
            mom_ok = mom_color in ['red', 'maroon']
            
        conditions['momentum_green'] = {
            'ok': mom_ok,
            'value': mom_color,
            'reason': f"Momentum is {mom_color}"
        }
        if mom_ok: score_count += 1
        else: missing.append(f"Momentum {mom_color} (Need {direction} colors)")

        # B. ADX Accompanied (Fuerza + Dirección) - FLEXIBLE
        adx_trend = adx_data['adx_trend']
        adx_current = adx_data['adx_current']
        di_trend = adx_data['di_plus_trend'] if direction == 'BULLISH' else adx_data['di_minus_trend']
        
        # TIME LADDER: Permitir si ADX alto aunque decrezca (post-impulso normal)
        adx_ok = (
            (adx_trend == 'creciendo' and di_trend == 'creciendo') or  # Ideal
            (adx_current > 25 and di_trend == 'creciendo') or  # ADX alto + DI subiendo
            (adx_current > 35)  # ADX muy alto = fuerza brutal (bypass)
        )
        
        conditions['adx_accompanied'] = {
            'ok': adx_ok,
            'value': f"ADX:{adx_trend} DI:{di_trend}",
            'reason': "ADX and DI must both be growing"
        }
        if adx_ok: score_count += 1
        else: missing.append(f"ADX/DI not synced (ADX:{adx_trend}, DI:{di_trend})")

        # C. Not in Correction (Velas)
        closes = df['close'].values[-3:]
        opens = df['open'].values[-3:]
        
        is_correction = True
        for c, o in zip(closes, opens):
            if direction == 'BULLISH':
                if c > o: # Vela verde
                    is_correction = False
                    break
            else:
                if c < o: # Vela roja
                    is_correction = False
                    break
                    
        conditions['not_in_correction'] = {
            'ok': not is_correction,
            'value': 'Correction' if is_correction else 'Clean',
            'reason': "Last 3 candles cannot be all against trend"
        }
        if not is_correction: score_count += 1
        else: missing.append("Price in active correction (3 candles against trend)")

        # D. Above EMA Support (Soporte)
        ema21 = emas_data['emas'].get(21, 0)
        
        if ema21 > 0:
            dist_pct = (current_price - ema21) / ema21 * 100
            if direction == 'BULLISH':
                ema_ok = dist_pct > -0.5 # Puede estar apenas abajo (retest)
            else:
                ema_ok = dist_pct < 0.5
        else:
            dist_pct = 0
            ema_ok = True # Sin EMA, no penalizar
            
        conditions['above_ema_support'] = {
            'ok': ema_ok,
            'value': f"{dist_pct:.2f}% vs EMA21",
            'reason': "Price must hold EMA21 support"
        }
        if ema_ok: score_count += 1
        else: missing.append(f"Lost EMA21 support (Dist: {dist_pct:.2f}%)")

        # Resultado Final
        final_score = score_count * 25
        
        # LÓGICA GRADIENTE (No exigir perfección)
        # 4/4 = Perfecto (100%)
        # 3/4 = Suficiente para entrar (75%)
        # 2/4 = Esperar (50%)
        # 1/4 o menos = Rechazar

        if score_count >= 3:
            ready = True
            status = 'READY'
        elif score_count == 2:
            ready = False
            status = 'WAITING'
        else:
            ready = False
            status = 'REJECT'
        
        return {
            'ready': ready,
            'status': status,
            'conditions': conditions,
            'score': final_score,
            'missing': missing
        }
    except Exception as e:
        print(f"⚠️ Error en validate_5m_entry_conditions: {e}")
        return {'ready': False, 'conditions': {}, 'score': 0, 'missing': ['Error de cálculo']}
