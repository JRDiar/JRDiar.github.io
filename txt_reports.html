pythonclass MLOptimizer:
    def __init__(self):
        self.trade_history = []
        self.pattern_success_rate = {}
        
    def record_trade(self, entry_conditions, result):
        """Registra cada trade con sus condiciones"""
        pattern = self.extract_pattern(entry_conditions)
        self.trade_history.append({
            'pattern': pattern,
            'profit': result['profit'],
            'duration': result['duration']
        })
        
    def extract_pattern(self, conditions):
        """Extrae patrón identificable"""
        return f"{conditions['adx_range']}_{conditions['momentum_color']}_{conditions['volume_level']}"
    
    def update_success_rates(self):
        """Actualiza tasas de éxito después de 100 trades"""
        if len(self.trade_history) < 100:
            return
            
        for pattern in set([t['pattern'] for t in self.trade_history]):
            trades = [t for t in self.trade_history if t['pattern'] == pattern]
            wins = [t for t in trades if t['profit'] > 0]
            self.pattern_success_rate[pattern] = len(wins) / len(trades)
    
    def should_trade(self, current_conditions):
        """Decide si operar basado en historial"""
        pattern = self.extract_pattern(current_conditions)
        
        if pattern in self.pattern_success_rate:
            if self.pattern_success_rate[pattern] < 0.4:  # <40% win rate
                return False, "Pattern histórico negativo"
        
        return True, "Pattern OK"

pythondef calculate_position_size_kelly(self, win_rate, avg_win, avg_loss):
    """Calcula tamaño óptimo con Kelly Criterion simplificado"""
    
    if avg_loss == 0:
        return 0.01  # 1% default
    
    # Kelly formula: f = (p * b - q) / b
    # p = probabilidad de ganar, b = ratio win/loss, q = probabilidad de perder
    p = win_rate
    q = 1 - win_rate
    b = avg_win / abs(avg_loss)
    
    kelly_fraction = (p * b - q) / b
    
    # Aplicar factor de seguridad (25% de Kelly)
    safe_fraction = kelly_fraction * 0.25
    
    # Límites
    return max(0.005, min(0.05, safe_fraction))  # Entre 0.5% y 5%

    base_score *= (1 + aligned_timeframes * 0.1)
