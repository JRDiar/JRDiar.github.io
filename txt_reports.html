def _classify_pivots(self, pivots: List[float]) -> List[str]:
    """
    Clasifica pivots en HH, HL, LH, LL.
    
    Asume que pivots vienen alternados: [low, high, low, high, ...] o [high, low, high, low, ...]
    """
    if len(pivots) < 3:
        return []
    
    pivot_types = []
    
    # Determinar si empezamos con high o low
    # Si primer pivot < segundo pivot → empezamos con low
    # Si primer pivot > segundo pivot → empezamos con high
    starts_with_low = pivots[0] < pivots[1]
    
    for i in range(len(pivots)):
        # Determinar si este índice es high o low
        is_low_index = (i % 2 == 0) if starts_with_low else (i % 2 == 1)
        
        if i == 0:
            # Primer pivot, no hay comparación
            pivot_types.append('L' if is_low_index else 'H')
            continue
        
        current = pivots[i]
        
        if is_low_index:
            # Es un LOW, comparar con el LOW anterior (i-2)
            if i >= 2:
                prev_low = pivots[i-2]
                if current > prev_low:
                    pivot_types.append('HL')  # Higher Low
                else:
                    pivot_types.append('LL')  # Lower Low
            else:
                pivot_types.append('L')
        else:
            # Es un HIGH, comparar con el HIGH anterior (i-2)
            if i >= 2:
                prev_high = pivots[i-2]
                if current > prev_high:
                    pivot_types.append('HH')  # Higher High
                else:
                    pivot_types.append('LH')  # Lower High
            else:
                pivot_types.append('H')
    
    return pivot_types
