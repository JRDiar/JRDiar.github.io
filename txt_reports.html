def make_pattern_based_decision(patterns, ladder, signals):
    """Decisión basada en confluencia REAL"""
    score = 0
    reasons = []
    
    # 1. Patrón principal detectado (Base sólida)
    if patterns['best_pattern']:
        pattern = patterns['best_pattern']
        score += pattern['confidence'] * 0.4
        reasons.append(f"Pattern: {pattern['type']} ({pattern['tf']})")
    
    # 2. Estructura de mercado (Alineación)
    # Dirección del patrón vs Dirección de los TFs
    pattern_dir = patterns['confidence']['direction'] # BULLISH/BEARISH
    
    structure_aligned = 0
    for tf, analysis in ladder.items():
        if analysis and analysis.get('direction') == pattern_dir:
            structure_aligned += 1
            
    if structure_aligned >= 3:
        score += 30
        reasons.append(f"Trend aligned {structure_aligned}/4")
    elif structure_aligned >= 2:
        score += 15
        reasons.append(f"Trend aligned {structure_aligned}/4")

    # 3. Indicadores matemáticos (Confirmación)
    if signals['adx_aligned']:
        score += 15
        reasons.append("ADX aligned")
    
    if signals['momentum_propagating']:
        score += 15
        reasons.append("Momentum propagating")
        
    if signals['volume_confirming']:
        score += 10
        reasons.append("Volume confirming")

    # 4. SMART CONTEXT LOGIC (La "Inteligencia" real)
    a5m = ladder.get('5m', {})
    if a5m:
        current_price = a5m.get('current_price', 0)
        
        # A. DYNAMIC OVEREXTENSION (ATR-based)
        # En vez de % fijo, usamos ATRs de distancia
        dist_ema = abs(a5m.get('dist_ema_pct', 0))
        atr_pct = a5m.get('atr_pct', 1.0) # Default 1% si falla
        
        # Ratio de extensión: Cuántos ATRs estamos lejos de la media
        extension_ratio = dist_ema / atr_pct if atr_pct > 0 else 0
        
        if extension_ratio > 3.0: # >3 ATRs lejos (Extremo)
            score -= 25
            reasons.append(f"⚠️ Overextended ({extension_ratio:.1f} ATRs)")
        elif extension_ratio > 2.0: # >2 ATRs lejos
            score -= 10
            reasons.append(f"⚠️ Extended ({extension_ratio:.1f} ATRs)")
        elif extension_ratio < 0.5:
            score += 5
            reasons.append("In Value Zone")

        # B. ZONE AWARENESS (Soportes y Resistencias)
        # Evitar comprar contra una pared
        sr_data = a5m.get('sr', {})
        if pattern_dir == 'BULLISH':
            resistances = sr_data.get('resistances', [])
            if resistances:
                nearest_res = resistances[0]
                dist_to_res = (nearest_res - current_price) / current_price * 100
                if 0 < dist_to_res < 0.5: # Menos de 0.5% hasta la resistencia
                    score -= 20
                    reasons.append(f"⚠️ Resistance ahead (+{dist_to_res:.2f}%)")
        elif pattern_dir == 'BEARISH':
            supports = sr_data.get('supports', [])
            if supports:
                nearest_sup = supports[0]
                dist_to_sup = (current_price - nearest_sup) / current_price * 100
                if 0 < dist_to_sup < 0.5: # Menos de 0.5% hasta el soporte
                    score -= 20
                    reasons.append(f"⚠️ Support below (-{dist_to_sup:.2f}%)")

        # C. RSI TIMING (Evitar extremos sin fuerza)
        rsi_val = a5m.get('rsi', {}).get('rsi_current', 50)
        mom_strength = a5m.get('momentum', {}).get('strength', 'débil')
        
        if pattern_dir == 'BULLISH':
            if rsi_val > 75 and mom_strength != 'fuerte':
                score -= 15
                reasons.append(f"⚠️ RSI Overbought ({rsi_val:.0f})")
        elif pattern_dir == 'BEARISH':
            if rsi_val < 25 and mom_strength != 'fuerte':
                score -= 15
                reasons.append(f"⚠️ RSI Oversold ({rsi_val:.0f})")

    # DECISIÓN FINAL
    # Umbral 60 para ser "Inteligente pero no restrictivo"
    if score >= 60:
        return {
            'action': 'ENTER',
            'confidence': score,
            'pattern': patterns['best_pattern']['type'] if patterns['best_pattern'] else "STRUCTURE_PLAY",
            'reasons': reasons,
            'strategy': determine_strategy(patterns, ladder)
        }
    else:
        return {
            'action': 'WAIT',
            'confidence': score,
            'missing': 60 - score,
            'reasons': reasons
        }

def determine_strategy(patterns, ladder):
    """Define estrategia según patrón"""
    pattern_type = "STANDARD"
    if patterns['best_pattern']:
        pattern_type = patterns['best_pattern']['type']
    
    # Reversión = Stop ajustado, target conservador
    if 'DOUBLE' in pattern_type or 'BOS' in pattern_type:
        return {
            'type': 'REVERSAL',
            'stop_multiplier': 1.5, 
            'target_multiplier': 2.0,
            'position_size': 'HALF' # Mitad del tamaño
        }
    # Continuación = Stop normal, target agresivo
    elif 'FLAG' in pattern_type or 'TRIANGLE' in pattern_type or 'TREND' in pattern_type:
        return {
            'type': 'CONTINUATION',
            'stop_multiplier': 2.0,
            'target_multiplier': 3.0,
            'position_size': 'FULL' # Tamaño completo
        }
    
    # Default
    return {
        'type': 'STANDARD',
        'stop_multiplier': 2.0,
        'target_multiplier': 2.0,
        'position_size': 'STANDARD'
    }

def check_adx_alignment(ladder):
    """Verifica si ADX apoya la tendencia en TFs clave"""
    # Ideal: ADX subiendo en 1H y 15M
    a1h = ladder.get('1h')
    a15m = ladder.get('15m')
    
    if not a1h or not a15m: return False
    
    # ADX > 20 y subiendo es fuerte
    # Extraer valor numérico de ADX (puede ser dict o int/float)
    adx_1h_val = a1h.get('adx', 0)
    if isinstance(adx_1h_val, dict):
        adx_1h_val = adx_1h_val.get('adx_current', adx_1h_val.get('current', 0))
        
    adx_15m_val = a15m.get('adx', 0)
    if isinstance(adx_15m_val, dict):
        adx_15m_val = adx_15m_val.get('adx_current', adx_15m_val.get('current', 0))

    strong_1h = float(adx_1h_val) > 20
    strong_15m = float(adx_15m_val) > 20
    
    return strong_1h or strong_15m
