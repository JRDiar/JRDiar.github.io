import ccxt
from brain_config import (
    MEXC_API_KEY, MEXC_SECRET_KEY, EXCHANGE_NAME,
    DEFAULT_TYPE, RATE_LIMIT, ENABLE_RATE_LIMIT
)

# ============================================================================
# INICIALIZAR EXCHANGE
# ============================================================================

exchange = ccxt.mexc({
    'apiKey': MEXC_API_KEY,
    'secret': MEXC_SECRET_KEY,
    'options': {'defaultType': DEFAULT_TYPE},
    'enableRateLimit': ENABLE_RATE_LIMIT,
    'rateLimit': RATE_LIMIT
})

# ============================================================================
# FUNCIONES FETCH
# ============================================================================

def fetch_ohlcv(symbol, timeframe='5m', limit=100):
    """
    Obtiene datos OHLCV para símbolo y timeframe
    
    Args:
        symbol: Símbolo (ej: 'BTC/USDT:USDT')
        timeframe: Timeframe (1d, 4h, 1h, 15m, 5m)
        limit: Número de velas
    
    Returns:
        list: OHLCV [[timestamp, open, high, low, close, volume], ...]
        None: Si error
    """
    try:
        ohlcv = exchange.fetch_ohlcv(symbol, timeframe, limit=limit)
        
        if not ohlcv or len(ohlcv) < 10:
            return None
            
        return ohlcv
        
    except Exception as e:
        print(f"❌ Error fetch_ohlcv {symbol} {timeframe}: {e}")
        return None


def fetch_markets():
    """
    Obtiene lista de mercados disponibles
    
    Returns:
        dict: Markets {symbol: market_info}
        None: Si error
    """
    try:
        markets = exchange.load_markets()
        return markets
    except Exception as e:
        print(f"❌ Error fetch_markets: {e}")
        return None


def get_futures_symbols():
    """
    Obtiene lista de símbolos futuros USDT
    
    Returns:
        list: ['BTC/USDT:USDT', 'ETH/USDT:USDT', ...]
        None: Si error
    """
    try:
        markets = fetch_markets()
        if not markets:
            return None
        
        # Filtrar solo futuros USDT
        symbols = [
            symbol for symbol, market in markets.items()
            if market.get('type') == 'swap' and symbol.endswith(':USDT')
        ]
        
        return sorted(symbols)
        
    except Exception as e:
        print(f"❌ Error get_futures_symbols: {e}")
        return None


def get_top_symbols_by_volume(limit=100):
    """
    Obtiene símbolos ordenados por volumen 24h (top movers)
    
    Args:
        limit: Cantidad de símbolos a retornar
    
    Returns:
        list: Símbolos ordenados por volumen descendente
    """
    try:
        tickers = exchange.fetch_tickers()
        
        # Filtrar y ordenar por volumen
        volume_list = []
        for symbol, ticker in tickers.items():
            if ':USDT' in symbol and ticker.get('quoteVolume'):
                volume_list.append((symbol, ticker['quoteVolume']))
        
        # Ordenar por volumen descendente
        volume_list.sort(key=lambda x: x[1], reverse=True)
        
        # Retornar top N
        return [symbol for symbol, _ in volume_list[:limit]]
        
    except Exception as e:
        print(f"❌ Error obteniendo top symbols: {e}")
        return []
