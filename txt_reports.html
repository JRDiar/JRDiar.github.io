"""
BRAIN TREND EXHAUSTION v1.0
============================
Joel: "Medir cuánto subió, cuánto le queda, dónde está el S/R"
Componentes de exhaustión:
1. Distancia a S/R (30 pts) - ¿Cuánto room queda?
2. Ratio moved/remaining (25 pts) - ¿Cuánto ya recorrió vs lo que queda?
3. ADX lifecycle (25 pts) - ¿En qué fase está la fuerza?
4. Momentum age (20 pts) - ¿Cuántas velas lleva el impulso?
Fases:
- FRESH (0-30%): ENTER
- DEVELOPING (30-50%): CAREFUL  
- MATURE (50-70%): WAIT
- EXHAUSTED (70-100%): AVOID
"""
import pandas as pd
from exchange_connection import fetch_ohlcv
from brain_calculator import calculate_adx_with_memory, calculate_momentum_with_memory
def analyze_trend_exhaustion(symbol, direction, df_5m=None, df_15m=None):
    """
    Analiza exhaustión combinando 5M (crítico) + 15M (contexto).
    
    Args:
        symbol: Par de trading
        direction: 'BULLISH' o 'BEARISH'
        df_5m: DataFrame 5M opcional
        df_15m: DataFrame 15M opcional
    
    Returns:
        dict con exhaustion_pct, room_pct, phase, penalty, recommendation
    """
    # Fetch si no se proporciona
    if df_5m is None:
        ohlcv = fetch_ohlcv(symbol, '5m', 100)
        if not ohlcv:
            return {'exhaustion_pct': 50, 'room_pct': 2, 'phase': 'UNKNOWN', 'penalty': 0}
        df_5m = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    
    current_price = float(df_5m['close'].iloc[-1])
    
    # === CALCULAR S/R DINÁMICO ===
    high_50 = df_5m['high'].tail(50).max()  # Resistencia
    low_50 = df_5m['low'].tail(50).min()    # Soporte
    high_20 = df_5m['high'].tail(20).max()  # Resistencia reciente
    low_20 = df_5m['low'].tail(20).min()    # Soporte reciente
    
    exhaustion_score = 0
    details = {}
    
    # === 1. DISTANCIA A S/R (30 pts máx) ===
    if direction == 'BULLISH':
        dist_to_resistance = ((high_50 - current_price) / current_price) * 100
        dist_from_support = ((current_price - low_50) / low_50) * 100
        room_pct = dist_to_resistance
        moved_pct = dist_from_support
        details['target'] = 'RESISTANCE'
        details['target_price'] = high_50
    else:  # BEARISH
        dist_to_support = ((current_price - low_50) / current_price) * 100
        dist_from_resistance = ((high_50 - current_price) / high_50) * 100
        room_pct = dist_to_support
        moved_pct = dist_from_resistance
        details['target'] = 'SUPPORT'
        details['target_price'] = low_50
    
    details['room_pct'] = room_pct
    details['moved_pct'] = moved_pct
    
    # Penalty por poco espacio
    if room_pct < 0.3:
        exhaustion_score += 30
        details['room_phase'] = 'BLOCKED'
    elif room_pct < 0.8:
        exhaustion_score += 20
        details['room_phase'] = 'TIGHT'
    elif room_pct < 1.5:
        exhaustion_score += 10
        details['room_phase'] = 'MODERATE'
    elif room_pct < 3.0:
        exhaustion_score += 5
        details['room_phase'] = 'GOOD'
    else:
        exhaustion_score += 0
        details['room_phase'] = 'EXCELLENT'
    
    # === 2. RATIO MOVED/REMAINING (25 pts máx) ===
    if room_pct > 0:
        ratio = moved_pct / room_pct
        details['move_ratio'] = ratio
        
        if ratio > 5:
            exhaustion_score += 25
            details['ratio_phase'] = 'VERY_EXHAUSTED'
        elif ratio > 3:
            exhaustion_score += 18
            details['ratio_phase'] = 'EXHAUSTED'
        elif ratio > 1.5:
            exhaustion_score += 10
            details['ratio_phase'] = 'BALANCED'
        elif ratio > 0.5:
            exhaustion_score += 5
            details['ratio_phase'] = 'FRESH'
        else:
            exhaustion_score += 0
            details['ratio_phase'] = 'VERY_FRESH'
    
    # === 3. ADX LIFECYCLE (25 pts máx) ===
    try:
        adx_data = calculate_adx_with_memory(df_5m, period=14, lookback=10)
        adx_current = adx_data.get('adx_current', 25)
        adx_velocity = adx_data.get('adx_velocity', 0)
        
        details['adx_value'] = adx_current
        details['adx_velocity'] = adx_velocity
        
        if adx_current > 50:
            exhaustion_score += 25
            details['adx_phase'] = 'OVEREXTENDED'
        elif adx_current > 45 and adx_velocity < 0:
            exhaustion_score += 20
            details['adx_phase'] = 'PEAKED'
        elif adx_current > 35:
            exhaustion_score += 10
            details['adx_phase'] = 'STRONG'
        elif adx_current > 25:
            exhaustion_score += 5
            details['adx_phase'] = 'DEVELOPING'
        else:
            exhaustion_score += 0
            details['adx_phase'] = 'NASCENT'
    except:
        details['adx_phase'] = 'UNKNOWN'
    
    # === 4. MOMENTUM AGE (20 pts máx) ===
    try:
        mom_data = calculate_momentum_with_memory(df_5m, lookback=10)
        mom_age = mom_data.get('consecutive_count', 0)
        details['momentum_age'] = mom_age
        
        if mom_age >= 10:
            exhaustion_score += 20
            details['momentum_phase'] = 'EXHAUSTED'
        elif mom_age >= 7:
            exhaustion_score += 15
            details['momentum_phase'] = 'OLD'
        elif mom_age >= 4:
            exhaustion_score += 8
            details['momentum_phase'] = 'DEVELOPING'
        elif mom_age >= 2:
            exhaustion_score += 3
            details['momentum_phase'] = 'FRESH'
        else:
            exhaustion_score += 0
            details['momentum_phase'] = 'NASCENT'
    except:
        details['momentum_phase'] = 'UNKNOWN'
    
    # === RESULTADO FINAL ===
    exhaustion_pct = min(100, exhaustion_score)
    
    if exhaustion_pct >= 70:
        phase = 'EXHAUSTED'
        penalty = -50
        recommendation = 'AVOID'
    elif exhaustion_pct >= 50:
        phase = 'MATURE'
        penalty = -30
        recommendation = 'WAIT'
    elif exhaustion_pct >= 30:
        phase = 'DEVELOPING'
        penalty = -15
        recommendation = 'CAREFUL'
    else:
        phase = 'FRESH'
        penalty = 0
        recommendation = 'ENTER'
    
    # >>> BONUS: Si hay mucho room, reducir penalty
    if room_pct > 3.0 and penalty < 0:
        penalty = int(penalty * 0.5)  # Reducir 50% la penalty si hay buen room
        details['room_bonus'] = True
    
    return {
        'exhaustion_pct': exhaustion_pct,
        'room_pct': room_pct,
        'moved_pct': moved_pct,
        'phase': phase,
        'penalty': penalty,
        'recommendation': recommendation,
        'details': details
    }
def get_dynamic_sl_from_exhaustion(exhaustion_result, base_sl_pct=2.0):
    """
    Calcula SL dinámico basado en room disponible.
    Más room = SL más amplio (dejar desarrollar)
    Menos room = SL más ajustado (proteger)
    """
    room = exhaustion_result.get('room_pct', 2.0)
    
    if room > 4.0:
        return min(base_sl_pct * 1.5, 3.5)  # MAX 3.5%
    elif room > 2.0:
        return base_sl_pct * 1.2  # +20%
    elif room > 1.0:
        return base_sl_pct
    elif room > 0.5:
        return base_sl_pct * 0.8  # -20%
    else:
        return base_sl_pct * 0.6  # -40%
def get_smart_trailing_distance(exhaustion_result, pnl_pct, base_trail=2.0):
    """
    Trailing dinámico: se aprieta cerca de S/R.
    """
    room = exhaustion_result.get('room_pct', 2.0)
    
    # Base según profit
    if pnl_pct > 3.0:
        trail = 1.0
    elif pnl_pct > 1.5:
        trail = 1.5
    elif pnl_pct > 0.8:
        trail = 2.0
    else:
        trail = 2.5
    
    # Ajustar según cercanía a S/R (aprieta cuando se acerca)
    if room < 0.5:
        trail = min(trail, 0.5)  # Muy apretado
    elif room < 1.0:
        trail = min(trail, 1.0)
    elif room < 1.5:
        trail = min(trail, 1.5)
    
    return trail
