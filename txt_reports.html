def is_perfect_setup(symbol, analyzer_score, decision):
    # 1. Obtener datos
    ladder_data = decision.get('full_analysis', {}).get('ladder_data', {})
    direction = decision.get('direction', 'LATERAL')
    
    # 2. LATERAL penalty
    if direction == 'LATERAL':
        analyzer_score -= 50
        # Intenta determinar direction desde DI dominance
    
    # 3. Obtener velocity
    velocity_check = check_price_velocity(symbol, direction, '5m')
    
    # 4. Validator score
    validation = entry_validator._validate_explosion_entry(...)
    validator_score = validation.get('score', 50)
    
    # 5. 5M VETO CHECK (v5.3.0)
    veto_result = check_5m_veto(symbol, direction, ladder_data)
    if veto_result['should_veto']:
        return False
    if veto_result['penalty'] > 0:
        analyzer_score -= veto_result['penalty']
    
    # 6. Combinar scores
    combined = (analyzer_score * 0.60) + (validator_score * 0.40)
    
    # 7. Decisión
    if combined >= 60:
        return True
    return False
