def detect_current_movement(structure_data, current_price):
    """
    Detecta movimiento actual del precio (últimas 5 velas + current_price)
    
    Args:
        structure_data: Dict con datos ZigZag incluyendo 'prices'
        current_price: Precio actual para comparar con última vela
    
    Returns:
        str: 'RISING', 'FALLING', or 'SIDEWAYS'
    """
    prices = structure_data.get('prices', [])
    
    if len(prices) < 5:
        return 'UNKNOWN'
    
    # Últimas 5 velas
    recent = prices[-5:]
    last_price = prices[-1]
    
    # Comparar current_price vs último precio cerrado
    if current_price < last_price * 0.998:  # -0.2% threshold
        movement_now = 'FALLING'
    elif current_price > last_price * 1.002:  # +0.2% threshold
        movement_now = 'RISING'
    else:
        movement_now = 'SIDEWAYS'
    
    # Confirmar con tendencia de 5 velas
    rising_count = sum(1 for i in range(4) if recent[i+1] > recent[i])
    
    if rising_count >= 4 and movement_now in ['RISING', 'SIDEWAYS']:
        return 'RISING'
    elif rising_count <= 1 and movement_now in ['FALLING', 'SIDEWAYS']:
        return 'FALLING'
    
    return 'SIDEWAYS'
Impact: Detecta si precio está cayendo AHORA (no solo estructura pasada)

"""
Momentum Validator - Detecta reversiones de momentum
Crítico para identificar fin de corrección y comienzo de rebote
"""
def validate_momentum_reversal(ladder, direction, timeframe='5m'):
    """
    Detecta reversiones de momentum (señal de fin de corrección)
    
    Args:
        ladder: Dict con análisis multi-timeframe
        direction: 'BULLISH' o 'BEARISH'
        timeframe: Timeframe a analizar
    
    Returns:
        dict: {
            'reversal_detected': bool,
            'score_adjustment': int,
            'reason': str,
            'momentum_color': str,
            'consecutive': int
        }
    """
    result = {
        'reversal_detected': False,
        'score_adjustment': 0,
        'reason': '',
        'momentum_color': 'neutral',
        'consecutive': 0
    }
    
    # Obtener datos de momentum
    tf_data = ladder.get(timeframe, {})
    momentum = tf_data.get('momentum', {})
    
    # VALIDACIÓN: Verificar que momentum existe y tiene campos necesarios
    if not momentum or 'color_current' not in momentum:
        result['score_adjustment'] = -10
        result['reason'] = "⚠️ Momentum no disponible"
        return result
    
    color = momentum['color_current']
    color_prev = momentum.get('color_prev', 'neutral')
    consecutive = momentum.get('consecutive_count', 1)  # Default 1, no 0
    
    result['momentum_color'] = color
    result['consecutive'] = consecutive
    
    if direction == 'BULLISH':
        # Caso 1: Reversión confirmada (rojo → verde, 3+ velas) - MÁXIMO BONUS
        if color_prev == 'red' and color in ['green', 'lime']:
            if consecutive >= 3:
                result['reversal_detected'] = True
                result['score_adjustment'] = 30  # Aumentado
                result['reason'] = f"✅ Reversión MADURA (3v+)"
            
            elif consecutive >= 2:
                result['reversal_detected'] = True
                result['score_adjustment'] = 25
                result['reason'] = f"✅ Reversión confirmada (2v)"
            
            elif consecutive == 1:
                result['reversal_detected'] = True
                result['score_adjustment'] = 15  # Aumentado de 10 → 15
                result['reason'] = f"⚠️ Reversión iniciando (1v)"
        
        # Caso 2: Momentum verde maduro (3+ velas)
        elif color in ['green', 'lime'] and consecutive >= 3:
            result['score_adjustment'] = 15
            result['reason'] = f"✅ Momentum desarrollado ({consecutive}v)"
        
        # Caso 3: Momentum verde temprano (2 velas)
        elif color in ['green', 'lime'] and consecutive == 2:
            result['score_adjustment'] = 10
            result['reason'] = f"✅ Momentum confirmado ({consecutive}v)"
        
        # Caso 4: Corrección en progreso (rojo)
        elif color in ['red', 'maroon']:
            result['score_adjustment'] = -25
            result['reason'] = f"❌ Corrección activa ({consecutive}v rojas)"
        
        # Caso 5: Neutral/débil
        else:
            result['score_adjustment'] = -10
            result['reason'] = f"⚠️ Momentum neutral"
    
    elif direction == 'BEARISH':
        # Invertir lógica para SHORT
        if color_prev == 'green' and color in ['red', 'maroon']:
            if consecutive >= 3:
                result['reversal_detected'] = True
                result['score_adjustment'] = 30
                result['reason'] = f"✅ Reversión bajista MADURA (3v+)"
            elif consecutive >= 2:
                result['reversal_detected'] = True
                result['score_adjustment'] = 25
                result['reason'] = f"✅ Reversión bajista confirmada (2v)"
            elif consecutive == 1:
                result['reversal_detected'] = True
                result['score_adjustment'] = 15
                result['reason'] = f"⚠️ Reversión bajista iniciando (1v)"
        elif color in ['red', 'maroon'] and consecutive >= 3:
            result['score_adjustment'] = 15
            result['reason'] = f"✅ Momentum bajista desarrollado ({consecutive}v)"
        elif color in ['red', 'maroon'] and consecutive == 2:
            result['score_adjustment'] = 10
            result['reason'] = f"✅ Momentum bajista confirmado ({consecutive}v)"
        elif color in ['green', 'lime']:
            result['score_adjustment'] = -25
            result['reason'] = f"❌ Recuperación alcista ({consecutive}v verdes)"
        else:
            result['score_adjustment'] = -10
            result['reason'] = f"⚠️ Momentum neutral"
    
    return result
Impact: +30 pts para reversión madura (3v+), +25 para confirmada (2v), +15 para iniciando (1v)
