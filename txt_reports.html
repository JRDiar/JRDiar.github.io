def get_frame_storage(symbol, timeframe='15m'):
    """
    Obtiene o crea storage para símbolo/timeframe
    
    Args:
        symbol: Par (ej: RAVE/USDT:USDT)
        timeframe: TF (default '15m')
        
    Returns:
        FrameStorage: Instancia para ese par/tf
    """
    key = f"{symbol}_{timeframe}"
    if key not in _frame_storages:
        _frame_storages[key] = FrameStorage(max_frames=20)
    return _frame_storages[key]
