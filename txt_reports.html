        has_mature_force = adx_val > 25 and mom_confirms and consecutive >= 3  # Maduro

            'has_mature': has_mature_force,
            'has_confirmed': has_confirmed_force,

            'consecutive': consecutive,
            'mom_color': mom_color
        }
    
    # Detectar propagación (MATURE/CONFIRMED)
    consecutive_strong = 0
    max_consecutive_strong = 0
