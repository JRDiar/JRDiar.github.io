"""
brain_adx_zones.py - ADX Cortometraje con Zonas
================================================
Analiza ADX como movimiento entre ZONAS, no valores estáticos.
"""
class ADXZoneAnalyzer:
    """
    Analiza ADX como cortometraje con zonas
    """
    
    # Definición de zonas
    ZONES = {
        'LATERAL': (0, 20),
        'INICIANDO': (20, 25),
        'TENDENCIA': (25, 35),
        'FUERTE': (35, 50),
        'BRUTAL': (50, 70),
        'EXTREMO': (70, 100)
    }
    
    # Bonus por zona actual
    ZONE_BONUS = {
        'LATERAL': -40,
        'INICIANDO': 0,
        'TENDENCIA': 10,
        'FUERTE': 20,
        'BRUTAL': 30,
        'EXTREMO': -10
    }
    
    def get_zone(self, adx_value):
        """Determina zona del ADX"""
        for zone_name, (min_val, max_val) in self.ZONES.items():
            if min_val <= adx_value < max_val:
                return zone_name
        return 'EXTREMO'
    
    def analyze_zone_trajectory(self, adx_frames):
        """
        Analiza trayectoria del ADX entre zonas
        
        Args:
            adx_frames: Lista [más viejo ... más nuevo]
                       Ej: [28, 30, 33, 35, 38, 41, 45]
        
        Returns:
            dict con zonas, cruces, dirección, historia, bonus
        """
        if len(adx_frames) < 3:
            return self._default_result()
        
        # Mapear zonas
        zones = [self.get_zone(adx) for adx in adx_frames]
        current_zone = zones[-1]
        
        # Detectar cruces de zona
        zone_changes = []
        for i in range(1, len(zones)):
            if zones[i] != zones[i-1]:
                zone_changes.append({
                    'from': zones[i-1],
                    'to': zones[i],
                    'adx': adx_frames[i]
                })
        
        # Analizar dirección
        delta = adx_frames[-1] - adx_frames[0]
        if delta > 5:
            direction = 'ASCENDING'
        elif delta < -5:
            direction = 'DESCENDING'
        else:
            direction = 'STABLE'
        
        # Calcular fuerza
        velocity = delta / len(adx_frames)
        if abs(velocity) > 3:
            strength = 'EXPLOSIVE'
        elif abs(velocity) > 1.5:
            strength = 'STRONG'
        elif abs(velocity) > 0.5:
            strength = 'MODERATE'
        else:
            strength = 'WEAK'
        
        # Calcular bonus
        bonus = self._calculate_zone_bonus(
            current_zone, zone_changes, direction, strength
        )
        
        # Crear historia
        story = self._create_story(
            adx_frames, zones, zone_changes, direction, strength
        )
        
        return {
            'zones': zones,
            'current_zone': current_zone,
            'zone_changes': zone_changes,
            'direction': direction,
            'strength': strength,
            'story': story,
            'bonus': bonus,
            'is_aligned': direction == 'ASCENDING' and current_zone not in ['LATERAL'],
            'valid': True
        }
    
    def _calculate_zone_bonus(self, current_zone, zone_changes, direction, strength):
        """Calcula bonus por zonas y cruces"""
        bonus = self.ZONE_BONUS.get(current_zone, 0)
        
        # Bonus por cruces ascendentes
        if direction == 'ASCENDING':
            for change in zone_changes:
                if change['from'] == 'INICIANDO' and change['to'] == 'TENDENCIA':
                    bonus += 20  # Confirmación
                elif change['from'] == 'TENDENCIA' and change['to'] == 'FUERTE':
                    bonus += 30  # Fortalecimiento
                elif change['from'] == 'FUERTE' and change['to'] == 'BRUTAL':
                    bonus += 40  # Explosión
            
            # Bonus por fuerza
            if strength == 'EXPLOSIVE':
                bonus += 25
            elif strength == 'STRONG':
                bonus += 15
        
        # Penalty por cruces descendentes
        elif direction == 'DESCENDING':
            for change in zone_changes:
                if change['from'] == 'BRUTAL' and change['to'] == 'FUERTE':
                    bonus -= 20
                elif change['from'] == 'FUERTE' and change['to'] == 'TENDENCIA':
                    bonus -= 30
        
        return bonus
    
    def _create_story(self, adx_frames, zones, zone_changes, direction, strength):
        """Crea historia del cortometraje"""
        start_adx = adx_frames[0]
        end_adx = adx_frames[-1]
        
        if len(zone_changes) == 0:
            return f"ADX {start_adx:.0f}→{end_adx:.0f} ({direction}) en zona {zones[-1]}"
        elif len(zone_changes) == 1:
            c = zone_changes[0]
            return f"ADX cruzó {c['from']}→{c['to']} ({start_adx:.0f}→{end_adx:.0f})"
        else:
            return f"ADX cruzó {len(zone_changes)} zonas ({start_adx:.0f}→{end_adx:.0f}) | {strength}"
    
    def _default_result(self):
        return {
            'zones': [],
            'current_zone': 'UNKNOWN',
            'zone_changes': [],
            'direction': 'UNKNOWN',
            'strength': 'UNKNOWN',
            'story': 'Datos insuficientes',
            'bonus': 0,
            'is_aligned': False,
            'valid': False
        }

"""
brain_alignment.py - Detector de Alineación Global
==================================================
Detecta cuando TODOS los indicadores están alineados.
NO busca perfección individual.
Busca COHERENCIA direccional del conjunto.
"""
from brain_adx_zones import adx_zone_analyzer
from brain_frames import get_frame_storage
from brain_trajectory import TrajectoryAnalyzer
class AlignmentDetector:
    """
    Detecta alineación global de todos los indicadores
    """
    
    def __init__(self):
        self.trajectory_analyzer = TrajectoryAnalyzer()
    
    def analyze_alignment(self, symbol, direction, ladder_data):
        """
        Analiza alineación COMPLETA de todos los componentes
        
        Args:
            symbol: Par a analizar
            direction: 'BULLISH' o 'BEARISH'
            ladder_data: Datos de todos los TFs
        
        Returns:
            dict con alignment_score, signals, story, decision
        """
        
        signals = {}
        
        # Obtener datos del 15m
        data_15m = ladder_data.get('15m', {})
        adx_data = data_15m.get('adx', {})
        momentum_data = data_15m.get('momentum', {})
        zigzag_data = data_15m.get('zigzag', {})
        
        # Obtener frames para análisis dinámico
        storage = get_frame_storage(symbol, '15m')
        
        # ===============================================
        # SEÑAL 1: ADX DIRECTION (usando zonas)
        # ===============================================
        adx_values = storage.get_trajectory('adx.value', 5)
        
        if len(adx_values) >= 3:
            zone_analysis = adx_zone_analyzer.analyze_zone_trajectory(adx_values)
            
            if zone_analysis['is_aligned']:
                signals['adx'] = 'ALIGNED'
            elif zone_analysis['direction'] == 'DESCENDING':
                signals['adx'] = 'CONFLICT'
            else:
                signals['adx'] = 'NEUTRAL'
        else:
            # Sin frames: usar ADX trend como proxy
            adx_trend = adx_data.get('adx_trend', 'estable')
            if adx_trend == 'creciendo':
                signals['adx'] = 'ALIGNED'
            elif adx_trend == 'decreciendo':
                signals['adx'] = 'CONFLICT'
            else:
                signals['adx'] = 'NEUTRAL'
        
        # ===============================================
        # SEÑAL 2: PRICE DIRECTION
        # ===============================================
        price_values = storage.get_trajectory('price', 5)
        
        if len(price_values) >= 3:
            price_delta = price_values[-1] - price_values[0]
            
            if direction == 'BULLISH':
                signals['price'] = 'ALIGNED' if price_delta > 0 else 'CONFLICT'
            else:
                signals['price'] = 'ALIGNED' if price_delta < 0 else 'CONFLICT'
        else:
            # Sin frames: asumir neutral
            signals['price'] = 'NEUTRAL'
        
        # ===============================================
        # SEÑAL 3: ZIGZAG STRUCTURE
        # ===============================================
        structure = zigzag_data.get('structure', '')
        current_trend = zigzag_data.get('current_trend', 'LATERAL')
        
        if direction == 'BULLISH':
            if current_trend == 'BULLISH' or 'HH' in structure or 'HL' in structure:
                signals['zigzag'] = 'ALIGNED'
            elif current_trend == 'BEARISH' or 'LL' in structure or 'LH' in structure:
                signals['zigzag'] = 'CONFLICT'
            else:
                signals['zigzag'] = 'NEUTRAL'
        else:
            if current_trend == 'BEARISH' or 'LL' in structure or 'LH' in structure:
                signals['zigzag'] = 'ALIGNED'
            elif current_trend == 'BULLISH' or 'HH' in structure or 'HL' in structure:
                signals['zigzag'] = 'CONFLICT'
            else:
                signals['zigzag'] = 'NEUTRAL'
        
        # ===============================================
        # SEÑAL 4: MOMENTUM COLOR
        # ===============================================
        mom_color = momentum_data.get('color_current', 'gray')
        
        if direction == 'BULLISH':
            if mom_color in ['lime', 'green']:
                signals['momentum'] = 'ALIGNED'
            elif mom_color in ['red', 'maroon']:
                signals['momentum'] = 'CONFLICT'
            else:
                signals['momentum'] = 'NEUTRAL'
        else:
            if mom_color in ['red', 'maroon']:
                signals['momentum'] = 'ALIGNED'
            elif mom_color in ['lime', 'green']:
                signals['momentum'] = 'CONFLICT'
            else:
                signals['momentum'] = 'NEUTRAL'
        
        # ===============================================
        # SEÑAL 5: DI DOMINANCE
        # ===============================================
        di_plus = adx_data.get('di_plus_current', 0)
        di_minus = adx_data.get('di_minus_current', 0)
        spread = abs(di_plus - di_minus)
        
        if spread < 5:
            signals['di'] = 'NEUTRAL'  # DIs muy juntos
        elif direction == 'BULLISH':
            signals['di'] = 'ALIGNED' if di_plus > di_minus else 'CONFLICT'
        else:
            signals['di'] = 'ALIGNED' if di_minus > di_plus else 'CONFLICT'
        
        # ===============================================
        # SEÑAL 6: VOLUME TREND
        # ===============================================
        volume_values = storage.get_trajectory('volume', 5)
        
        if len(volume_values) >= 3:
            vol_delta = volume_values[-1] - volume_values[0]
            if vol_delta > 0:
                signals['volume'] = 'ALIGNED'  # Volumen aumentando siempre es bueno
            else:
                signals['volume'] = 'NEUTRAL'  # No penalizar vol bajando
        else:
            signals['volume'] = 'NEUTRAL'
        
        # ===============================================
        # CALCULAR ALINEACIÓN
        # ===============================================
        aligned_count = sum(1 for s in signals.values() if s == 'ALIGNED')
        conflict_count = sum(1 for s in signals.values() if s == 'CONFLICT')
        total_signals = len(signals)
        
        alignment_score = (aligned_count / total_signals) * 100
        
        # Listas para reporte
        aligned_list = [k for k, v in signals.items() if v == 'ALIGNED']
        conflict_list = [k for k, v in signals.items() if v == 'CONFLICT']
        neutral_list = [k for k, v in signals.items() if v == 'NEUTRAL']
        
        # ===============================================
        # DECISIÓN BASADA EN ALINEACIÓN
        # ===============================================
        if alignment_score >= 80:
            decision = True
            story = f"ALTA COHERENCIA ({alignment_score:.0f}%): {', '.join(aligned_list)}"
        elif alignment_score >= 60 and conflict_count <= 1:
            decision = True
            story = f"COHERENCIA MODERADA ({alignment_score:.0f}%): {aligned_count}/{total_signals}"
        elif alignment_score >= 50 and conflict_count == 0:
            decision = True
            story = f"COHERENCIA ACEPTABLE ({alignment_score:.0f}%): Sin conflictos"
        else:
            decision = False
            if conflict_list:
                story = f"CONFLICTOS ({conflict_count}): {', '.join(conflict_list)}"
            else:
                story = f"ALINEACIÓN BAJA ({alignment_score:.0f}%)"
        
        return {
            'alignment_score': alignment_score,
            'aligned_count': aligned_count,
            'conflict_count': conflict_count,
            'total_signals': total_signals,
            'signals': signals,
            'aligned_list': aligned_list,
            'conflict_list': conflict_list,
            'neutral_list': neutral_list,
            'story': story,
            'decision': decision
        }
# ===============================================
# FUNCIÓN HELPER SIMPLIFICADA
# ===============================================
def check_alignment(symbol, direction, ladder_data):
    """
    Verificación rápida de alineación
    
    Returns:
        tuple: (decision, alignment_score, story)
    """
    detector = AlignmentDetector()
    result = detector.analyze_alignment(symbol, direction, ladder_data)
    
    return (
        result['decision'],
        result['alignment_score'],
        result['story']
    )

def sniper_checklist(symbol, direction, ladder, triggers=None, current_price=None):
    """
    Checklist del Sniper basado en ALINEACIÓN (versión 6.0)
    
    NO más penalties acumulativos.
    SOLO: ¿Están todos alineados?
    """
    
    from brain_alignment import alignment_detector
    from brain_adx_zones import adx_zone_analyzer
    from brain_frames import get_frame_storage
    
    print(f"\n🧠 SNIPER COGNITIVO v6.0: {symbol} ({direction})")
    print("=" * 50)
    
    # ===================================================
    # PASO 1: ANÁLISIS DE ALINEACIÓN (NUEVO)
    # ===================================================
    
    alignment = alignment_detector.analyze_alignment(symbol, direction, ladder)
    
    print(f"\n🎯 ALINEACIÓN GLOBAL:")
    print(f"   Score: {alignment['alignment_score']:.0f}%")
    print(f"   Alineados: {', '.join(alignment['aligned_list'])}")
    if alignment['conflict_list']:
        print(f"   Conflictos: {', '.join(alignment['conflict_list'])}")
    print(f"   Historia: {alignment['story']}")
    
    # Si alineación < 50% o > 2 conflictos = RECHAZAR
    if alignment['alignment_score'] < 50 or alignment['conflict_count'] > 2:
        return {
            'ready': False,
            'approved': False,
            'reason': alignment['story'],
            'archetype': 'LOW_ALIGNMENT',
            'total_score': int(alignment['alignment_score']),
            'alignment': alignment
        }
    
    # ===================================================
    # PASO 2: ANÁLISIS DE ZONAS ADX (NUEVO)
    # ===================================================
    
    storage = get_frame_storage(symbol, '15m')
    adx_values = storage.get_trajectory('adx.value', 5)
    
    if len(adx_values) >= 3:
        zone_analysis = adx_zone_analyzer.analyze_zone_trajectory(adx_values)
        
        print(f"\n📊 ADX ZONAS:")
        print(f"   Zona actual: {zone_analysis['current_zone']}")
        print(f"   Dirección: {zone_analysis['direction']}")
        print(f"   Fuerza: {zone_analysis['strength']}")
        print(f"   Historia: {zone_analysis['story']}")
        print(f"   Bonus: {zone_analysis['bonus']:+d}")
        
        # Si ADX en zona LATERAL y NO subiendo = RECHAZAR
        if zone_analysis['current_zone'] == 'LATERAL' and zone_analysis['direction'] != 'ASCENDING':

                'ready': False,
                'approved': False,
                'reason': 'ADX en zona LATERAL sin movimiento',
                'archetype': 'ADX_LATERAL',
                'total_score': 0,
                'zone_analysis': zone_analysis
            }
    else:
        zone_analysis = None
    
    # ===================================================
    # PASO 3: VALIDACIONES MÍNIMAS (SIMPLIFICADAS)

    data_15m = ladder.get('15m', {})

    # ÚNICO bloqueo duro: Spread < 5 (DIs convergiendo)
    if spread < 5:
        return {
            'ready': False,
            'approved': False,
            'reason': f'Spread {spread:.1f} < 5 (sin claridad)',
            'archetype': 'NO_CLARITY',
            'total_score': 0
        }
    
    # ===================================================
    # PASO 4: CALCULAR SCORE FINAL
    # ===================================================

    total_score = int(alignment['alignment_score'])

        total_score += zone_analysis['bonus']

        total_score += 10

    # ===================================================

    elif alignment['alignment_score'] >= 60:

        archetype = 'MODERATE_ALIGNMENT'
    
    # ===================================================

    # ===================================================
    
    # ENTRAR si:
    # - Alineación >= 50% Y
    # - No más de 2 conflictos Y
    # - Spread >= 5
    
    ready = alignment['decision']
    
    print(f"\n🎯 VEREDICTO:")

    print(f"   Archetype: {archetype}")
    print(f"   Ready: {'✅ SÍ' if ready else '❌ NO'}")
    
    return {

        'approved': ready,
        'total_score': total_score,
        'archetype': archetype,
