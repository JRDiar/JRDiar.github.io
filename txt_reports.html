def check_optimal_entry_timing(current_price, patterns, ladder, score):
    """
    Verifica si ES EL MOMENTO ÓPTIMO para entrar
    Aunque tenga fuerza, puede esperar mejor punto
    """
    a5m = ladder.get('5m', {})
    
    # 1. ¿Está overextended? → Esperar pullback
    dist_ema = abs(a5m.get('dist_ema_pct', 0))
    atr_pct = a5m.get('atr_pct', 1.0)
    extension_ratio = dist_ema / atr_pct if atr_pct > 0 else 0
    
    if extension_ratio > 3.0:  # Más de 3 ATRs lejos
        # Calcular zona pullback óptima (POC o EMA20)
        vp_poc = a5m.get('volume_profile', {}).get('poc_current', 0)
        target_pullback = vp_poc if vp_poc > 0 else current_price * 0.98
        
        return {
            'status': 'WAIT_PULLBACK',
            'reason': f'Overextended {extension_ratio:.1f} ATRs. Esperando pullback',
            'target_price': target_pullback
        }
    
    # 2. ¿Resistencia muy cerca? → Esperar breakout
    sr_data = a5m.get('sr', {})
    resistances = sr_data.get('resistances', [])
    
    if resistances:
        nearest_res = resistances[0]
        dist_pct = (nearest_res - current_price) / current_price * 100
        
        if 0 < dist_pct < 0.8:  # Menos de 0.8% a resistencia
            return {
                'status': 'WAIT_BREAKOUT',
                'reason': f'Resistencia a {dist_pct:.1f}%. Esperando breakout',
                'breakout_level': nearest_res * 1.002  # +0.2% sobre resistencia
            }
    
    # 3. ¿En medio de subida con espacio? → PUEDE ENTRAR
    # Verificar HH próximo (ZigZag)
    zigzag = a5m.get('zigzag', {})
    pivot_highs = zigzag.get('pivot_highs', [])
    
    if pivot_highs and len(pivot_highs) >= 2:
        last_hh = pivot_highs[-1]
        dist_to_hh = (last_hh - current_price) / current_price * 100
        
        if dist_to_hh > 2.0:  # Más de 2% hasta HH
            return {
                'status': 'OPTIMAL_NOW',
                'reason': f'Espacio hasta HH ({dist_to_hh:.1f}%). Entrada válida',
                'target_hh': last_hh
            }
    
    # 4. Default: timing OK
    return {'status': 'OPTIMAL_NOW', 'reason': 'Setup óptimo'}
