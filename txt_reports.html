# FACTOR 3.5: EXHAUSTION ABSOLUTO + ACELERACIÓN (Time Ladder)
exhaustion_pct = traj.get('exhaustion_pct', 50)
exhaustion_velocity = traj.get('exhaustion_velocity', 0)
exhaustion_accel = traj.get('exhaustion_accel', 0)
# VER TRAYECTORIA: Exhaustion acelerando es CRÍTICO
if exhaustion_accel > 0.3:
    # Acelerando hacia exhaustion = bloqueo casi garantizado
    exh_penalty = -50
    confidence_factors.append(('Exhaustion acelerando', exh_penalty, 
                             f'{exhaustion_pct:.0f}% (accel: +{exhaustion_accel:.2f})'))
elif exhaustion_pct >= 85:
    # Exhaustion crítico (más severo)
    exh_penalty = -50  # Era -40
    confidence_factors.append(('Exhaustion crítico', exh_penalty, f'{exhaustion_pct:.0f}%'))
elif exhaustion_pct >= 70:
    # Exhausted (MÁS SEVERO - el fix clave)
    if exhaustion_velocity > 5:
        # Subiendo rápido (peor)
        exh_penalty = -40  # Era -25
    else:
        # Subiendo lento o estable
        exh_penalty = -35  # Era -25
    confidence_factors.append(('Exhausted', exh_penalty, 
                             f'{exhaustion_pct:.0f}% (vel: +{exhaustion_velocity:.1f})'))
elif exhaustion_pct >= 50:
    # Mature (gradual)
    exh_penalty = int(-15 * (exhaustion_pct - 50) / 20)
    if exh_penalty < 0:
        confidence_factors.append(('Mature', exh_penalty, f'{exhaustion_pct:.0f}%'))
elif exhaustion_velocity < -5:
    # BONUS: Recovering fast (buena señal)
    exh_bonus = 12
    confidence_factors.append(('Recovering fast', exh_bonus, 
                             f'{exhaustion_pct:.0f}% (vel: {exhaustion_velocity:.1f})'))
elif exhaustion_pct < 30:
    # Fresh
    exh_bonus = int(10 * (30 - exhaustion_pct) / 30)
    if exh_bonus > 0:
        confidence_factors.append(('Fresh', exh_bonus, f'{exhaustion_pct:.0f}%'))
