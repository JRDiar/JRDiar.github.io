    VOLUME_RATIO_MIN, ENABLE_BREAKEVEN_PROTECTION, ENABLE_TRAILING_STOP,
    ENABLE_SMART_EXIT, BREAKEVEN_PROFIT_TRIGGER_PCT,
    DECELERATION_THRESHOLD, VOLUME_DECLINE_THRESHOLD,
    CRITICAL_ZONE_DISTANCE_PCT, BUFFETT_EARLY_EXIT_TIME_MIN,
    BUFFETT_EARLY_EXIT_LOSS_PCT, ARGENTINA_TZ
)

class TradeManager:
    def __init__(self):
        self.active_trades = {}
        self.capital_available = CAPITAL_INICIAL
        self.capital_per_trade = CAPITAL_INICIAL / MAX_TRADES_SIMULTANEOUS
        self.trade_history = []
        self.breakeven_exited_symbols = {}  # Monedas que salieron en breakeven para re-analizar
    
    def can_open_trade(self):
        """Verifica si puede abrir nuevo trade"""
        return len(self.active_trades) < MAX_TRADES_SIMULTANEOUS
    
    def check_trigger_activated(self, symbol, watchlist_data):
        """
        Verifica si trigger está activado para entrada
        
        Args:
            symbol: Símbolo
            watchlist_data: Dict con datos de watchlist
        
        Returns:
            bool: True si trigger activado
        """
        # Fetch datos actuales 5M
        ohlcv = fetch_ohlcv(symbol, '5m', 50)
        if not ohlcv or len(ohlcv) < 20:
            return False
        
        df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
        
        # Verificar ADX
        adx, di_plus, di_minus = calculate_adx(df)
        if adx is None or adx < ADX_THRESHOLD:
            return False
        
        # Verificar momentum color consecutivo
        momentum_color = calculate_momentum_color(df)
        expected_direction = watchlist_data['direction']
        
        if (expected_direction == 'BULLISH' and momentum_color != 'green') or \
           (expected_direction == 'BEARISH' and momentum_color != 'red'):
            return False
        
        # Verificar volumen ratio
        volumes = df['volume'].values
        volume_current = volumes[-1]
        volume_avg = volumes[-20:].mean()
        volume_ratio = volume_current / volume_avg if volume_avg > 0 else 0
        
        if volume_ratio < VOLUME_RATIO_MIN:
            return False
        
        # Verificar aceleración ADX (opcional pero recomendado)
        if len(ohlcv) >= 60:
            df_past = df.iloc[:-5]
            adx_past, _, _ = calculate_adx(df_past)
            if adx_past:
                adx_acceleration = (adx - adx_past) / 5  # Cambio por minuto
                if adx_acceleration < ADX_MIN_ACCELERATION:
                    return False
        
        print(f"🎯 TRIGGER ACTIVADO: {symbol}")
        print(f"   ADX: {adx:.1f} | Momentum: {momentum_color} | Vol ratio: {volume_ratio:.2f}x")
        
        return True
    
    def enter_trade(self, symbol, watchlist_data):
        """
        Entra a trade
        
        Args:
            symbol: Símbolo
            watchlist_data: Dict con datos de watchlist
        """
        if not self.can_open_trade():
            print(f"⚠️ Máximo trades alcanzado ({MAX_TRADES_SIMULTANEOUS})")
            return
        
        # Fetch precio actual
        ohlcv = fetch_ohlcv(symbol, '5m', 10)
        if not ohlcv:
            return
        
        entry_price = ohlcv[-1][4]  # Close
        direction = watchlist_data['direction']
        triggers = watchlist_data['triggers']
        stop_loss = triggers['stop_loss']
        
        # Crear trade
        now = datetime.now(ARGENTINA_TZ)
        
        trade_data = {
            'symbol': symbol,
            'direction': direction,
            'entry_price': entry_price,
            'entry_time': now,
            'stop_loss': stop_loss,
            'target_price': triggers['target_price'],
            'score': watchlist_data['score'],
            'clasificacion': watchlist_data['clasificacion'],
            'capital': self.capital_per_trade,
            'leverage': SIMULATE_LEVERAGE,
            'breakeven_activated': False,
            'trailing_activated': False,
            'max_pnl_reached': 0.0,
            'min_pnl_reached': 0.0
        }
        
        self.active_trades[symbol] = trade_data
        
        # Remover de watchlist
        watchlist_manager.remove_from_watchlist(symbol, "trade abierto")
        
        # Notificar
        print(f"\n{'='*80}")
        print(f"✅ ENTRADA: {symbol} {direction}")
        print(f"   Precio: ${entry_price:.8f}")
        print(f"   Stop: ${stop_loss:.8f}")
        print(f"   Target: ${triggers['target_price']:.8f}")
        print(f"   Score: {watchlist_data['score']} ({watchlist_data['clasificacion']})")
        print(f"{'='*80}\n")
        
        # Sheets
        sheets_manager.write_trade_entry(
            symbol, direction, entry_price,
            stop_loss, triggers['target_price'], watchlist_data['score'],
            triggers['rr_ratio']
        )
        
        # Telegram
        notify_entry(symbol, direction, entry_price, stop_loss, watchlist_data['score'])
    
    def monitor_active_trades(self):
        """Monitorea trades activos y aplica gestión inteligente"""
        if not self.active_trades:
            return
        
        now = datetime.now(ARGENTINA_TZ)
        symbols_to_close = []
        
        for symbol, trade_data in self.active_trades.items():
            # Fetch precio actual
            ohlcv = fetch_ohlcv(symbol, '5m', 50)
            if not ohlcv:
                continue
            
            current_price = ohlcv[-1][4]
            df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
            
            # Calcular PnL
            entry_price = trade_data['entry_price']
            direction = trade_data['direction']
            
            if direction == 'BULLISH':
                pnl_pct = ((current_price - entry_price) / entry_price) * 100
            else:
                pnl_pct = ((entry_price - current_price) / entry_price) * 100
            
            # Actualizar max/min PnL
            if pnl_pct > trade_data['max_pnl_reached']:
                trade_data['max_pnl_reached'] = pnl_pct
            if pnl_pct < trade_data['min_pnl_reached']:
                trade_data['min_pnl_reached'] = pnl_pct
            
            duration_min = (now - trade_data['entry_time']).total_seconds() / 60
            
            # GESTIÓN 1: Breakeven
            if ENABLE_BREAKEVEN_PROTECTION and not trade_data['breakeven_activated']:
                if pnl_pct >= BREAKEVEN_PROFIT_TRIGGER_PCT:
                    trade_data['stop_loss'] = entry_price
                    trade_data['breakeven_activated'] = True
                    print(f"🛡️ BREAKEVEN: {symbol} (PnL: +{pnl_pct:.2f}%)")
                    notify_breakeven(symbol, pnl_pct)
            
            # GESTIÓN 2: Verificar stop loss
            if (direction == 'BULLISH' and current_price <= trade_data['stop_loss']) or \
               (direction == 'BEARISH' and current_price >= trade_data['stop_loss']):
                
                # Si es breakeven, guardar para re-análisis
                if trade_data['breakeven_activated'] and abs(pnl_pct) < 0.5:
                    self.breakeven_exited_symbols[symbol] = now
                    reason = "Stop loss (breakeven) - re-analizar después"
                else:
                    reason = "Stop loss"
                
                symbols_to_close.append((symbol, reason))
                continue
            
            # GESTIÓN 3: Salida inteligente (5 señales)
            if ENABLE_SMART_EXIT:
                exit_signals = self.check_exit_signals(df, trade_data, duration_min)
                
                if exit_signals['count'] >= 2:
                    reason = f"Salida inteligente: {', '.join(exit_signals['active'])}"
                    symbols_to_close.append((symbol, reason))
                    continue
            
            # Monitoreo log
            print(f"📊 {symbol}: ${current_price:.8f} | PnL: {pnl_pct:+.2f}% | {duration_min:.1f}min")
            
            # Sheets
            adx, _, _ = calculate_adx(df)
            
            sheets_manager.write_monitoreo(
                symbol, current_price, pnl_pct, duration_min,
                trade_data['breakeven_activated'], current_price
            )
        
        # Cerrar trades marcados
        for symbol, reason in symbols_to_close:
            self.close_trade(symbol, reason)
    
    def check_exit_signals(self, df, trade_data, duration_min):
        """
        Verifica 5 señales de salida inteligente
        
        Returns:
            dict: {'count': int, 'active': [list]}
        """
        signals_active = []
        
        # SEÑAL 1: Desaceleración (velas más chicas)
        try:
            ranges = (df['high'] - df['low']).values
            avg_range_10 = ranges[-10:].mean()
            avg_range_3 = ranges[-3:].mean()
            
            if avg_range_3 < avg_range_10 * DECELERATION_THRESHOLD:
                signals_active.append("Desaceleración")
        except:
            pass
        
        # SEÑAL 2: Volumen bajando
        try:
            volumes = df['volume'].values
            avg_vol_10 = volumes[-10:].mean()
            avg_vol_3 = volumes[-3:].mean()
            
            if avg_vol_3 < avg_vol_10 * VOLUME_DECLINE_THRESHOLD:
                signals_active.append("Volumen bajo")
        except:
            pass
        
        # SEÑAL 3: Divergencia oculta (precio sube pero momentum flat)
        try:
            prices = df['close'].values
            price_slope = (prices[-1] - prices[-5]) / 5
            
            # Calcular momentum simplificado
            ema = df['close'].ewm(span=10, adjust=False).mean().values
            momentum_slope = (ema[-1] - ema[-5]) / 5
            
            direction = trade_data['direction']
            if direction == 'BULLISH' and price_slope > 0 and abs(momentum_slope) < 0.001:
                signals_active.append("Divergencia")
            elif direction == 'BEARISH' and price_slope < 0 and abs(momentum_slope) < 0.001:
                signals_active.append("Divergencia")
        except:
            pass
        
        # SEÑAL 4: Llegada zona crítica (cerca de target)
        try:
            current_price = df['close'].iloc[-1]
            target = trade_data['target_price']
            distance_pct = abs((target - current_price) / current_price * 100)
            
            if distance_pct < CRITICAL_ZONE_DISTANCE_PCT:
                signals_active.append("Zona crítica")
        except:
            pass
        
        # SEÑAL 5: Protección Buffett (pérdida temprana)
        if duration_min < BUFFETT_EARLY_EXIT_TIME_MIN:
            current_price = df['close'].iloc[-1]
            entry_price = trade_data['entry_price']
            direction = trade_data['direction']
            
            if direction == 'BULLISH':
                pnl_pct = ((current_price - entry_price) / entry_price) * 100
            else:
                pnl_pct = ((entry_price - current_price) / entry_price) * 100
            
            if pnl_pct < -BUFFETT_EARLY_EXIT_LOSS_PCT:
                signals_active.append("Buffett (setup falló)")
        
        return {
            'count': len(signals_active),
            'active': signals_active
        }
    
    def close_trade(self, symbol, reason):
        """Cierra trade y registra resultado"""
        if symbol not in self.active_trades:
            return
        
        trade_data = self.active_trades[symbol]
        
        # Fetch precio actual
        ohlcv = fetch_ohlcv(symbol, '5m', 10)
        exit_price = ohlcv[-1][4] if ohlcv else trade_data['entry_price']
        
        # Calcular PnL final
        entry_price = trade_data['entry_price']
        direction = trade_data['direction']
        
        if direction == 'BULLISH':
            pnl_pct = ((exit_price - entry_price) / entry_price) * 100
        else:
            pnl_pct = ((entry_price - exit_price) / entry_price) * 100
        
        pnl_usd = (self.capital_per_trade * pnl_pct / 100) * SIMULATE_LEVERAGE
        
        # Duración
        now = datetime.now(ARGENTINA_TZ)
        duration_min = (now - trade_data['entry_time']).total_seconds() / 60
        
        # Log
        emoji = "✅" if pnl_pct > 0 else "❌"
        print(f"\n{'='*80}")
        print(f"{emoji} SALIDA: {symbol}")
        print(f"   PnL: {pnl_pct:+.2f}% | ${pnl_usd:+.2f}")
        print(f"   Duración: {duration_min:.1f} min")
        print(f"   Razón: {reason}")
        print(f"{'='*80}\n")
        
        # Sheets
        sheets_manager.write_trade_exit(
            symbol, exit_price, pnl_pct, pnl_usd,
            duration_min, reason
        )
        
        # Telegram
        notify_exit(symbol, pnl_pct, pnl_usd, duration_min, reason)
        
        # Remover de activos
        del self.active_trades[symbol]
        
        # Guardar en historial
        trade_data['exit_price'] = exit_price
        trade_data['exit_time'] = now
        trade_data['pnl_pct'] = pnl_pct
        trade_data['pnl_usd'] = pnl_usd
        trade_data['duration_min'] = duration_min
        trade_data['exit_reason'] = reason
        self.trade_history.append(trade_data)
    
    def reanalyze_breakeven_exits(self):
        """Re-analiza monedas que salieron en breakeven"""
        if not self.breakeven_exited_symbols:
            return
        
        now = datetime.now(ARGENTINA_TZ)
        symbols_to_remove = []
        
        for symbol, exit_time in self.breakeven_exited_symbols.items():
            # Solo re-analizar después de 5 minutos
            if (now - exit_time).total_seconds() < 300:
                continue
            
            print(f"🔄 Re-analizando {symbol} (salió en breakeven)...")
            
            from brain_analyzer import analyze_coin_complete
            analysis = analyze_coin_complete(symbol)
            
            if analysis and analysis['viable']:
                print(f"✅ {symbol} sigue viable - agregando a watchlist")
                watchlist_manager.add_to_watchlist(symbol, analysis)
            else:
                print(f"❌ {symbol} ya no viable")
            
            symbols_to_remove.append(symbol)
        
        # Limpiar procesados
        for symbol in symbols_to_remove:
            del self.breakeven_exited_symbols[symbol]

    def pre_filter_weekly_candles(self):
        """
        PRE-FILTRO: Rechaza monedas con <10 velas semanales
        
        Returns:
            list: Símbolos que pasan filtro
        """
        print(f"\n{'='*80}")
        print("🔍 FASE 0: PRE-FILTRO VELAS SEMANALES")
        print(f"{'='*80}")
        
        all_symbols = get_futures_symbols()
        if not all_symbols:
            print("❌ Error obteniendo símbolos")
            return []
        
        total_symbols = len(all_symbols)
        print(f"📊 Total símbolos: {total_symbols}")
        
        filtered = []
        rejected = []
        
        print("Analizando velas semanales...")
        
        for i, symbol in enumerate(all_symbols):
            if i % 100 == 0:
                print(f"  Progreso: {i}/{total_symbols} ({i/total_symbols*100:.1f}%)")
            
            try:
                ohlcv = fetch_ohlcv(symbol, '1w', 15)
                if not ohlcv:
                    rejected.append((symbol, 0))
                    continue
                
                # Contar velas reales
                real_candles = 0
                for candle in ohlcv:
                    volume = candle[5]
                    high = candle[2]
                    low = candle[3]
                    
                    if volume > 0 and high != low:
                        real_candles += 1
                
                if real_candles >= MIN_WEEKLY_CANDLES:
                    filtered.append(symbol)
                    print(f"  ✅ {symbol}: {real_candles} velas semanales - APROBADO")
                else:
                    rejected.append((symbol, real_candles))
                    print(f"  ❌ {symbol}: {real_candles} velas semanales - RECHAZADO")
            
            except Exception as e:
                rejected.append((symbol, 0))
        
        print(f"\n✅ FILTRADAS: {len(filtered)}/{total_symbols} ({len(filtered)/total_symbols*100:.1f}%)")
        print(f"❌ RECHAZADAS: {len(rejected)} (<{MIN_WEEKLY_CANDLES} velas semanales)")
        
        return filtered

    def monitor_and_trade(self):
        """
        FASE 3: Monitorea watchlist y trades activos
        """

        # Monitorear trades activos

        # Verificar triggers en watchlist

import json

    ENABLE_WATCHLIST_JSON, ARGENTINA_TZ,

    MOMENTUM_CONSECUTIVE_CANDLES, VOLUME_RATIO_MIN,

class WatchlistManager:
    def __init__(self):
        self.watchlist = {}
        self.load_from_file()
    
    def load_from_file(self):
        """Carga watchlist desde JSON"""
        if not ENABLE_WATCHLIST_JSON:
            return
        
        try:
            with open(WATCHLIST_JSON, 'r') as f:
                data = json.load(f)
                self.watchlist = data.get('watchlist', {})
            print(f"✅ Watchlist cargada: {len(self.watchlist)} monedas")
        except FileNotFoundError:
            print("ℹ️ Watchlist vacía (archivo no existe)")
            self.watchlist = {}
        except Exception as e:
            print(f"❌ Error cargando watchlist: {e}")
            self.watchlist = {}
    
    def save_to_file(self):
        """Guarda watchlist a JSON"""
        if not ENABLE_WATCHLIST_JSON:
            return
        
        try:
            data = {
                'version': '1.0',
                'last_updated': datetime.now(ARGENTINA_TZ).isoformat(),
                'watchlist': self.watchlist
            }
            
            with open(WATCHLIST_JSON, 'w') as f:
                json.dump(data, f, indent=2)
            
            print(f"✅ Watchlist guardada: {len(self.watchlist)} monedas")
        except Exception as e:
            print(f"❌ Error guardando watchlist: {e}")
    
    def calculate_triggers(self, analysis_data):
        """
        Calcula triggers de entrada para una moneda
        
        Args:
            analysis_data: Dict con análisis completo
        
        Returns:
            dict: Triggers calculados o None
        """
        analysis_5m = analysis_data['analysis_5m']
        analysis_4h = analysis_data['analysis_4h']
        sr_zones = analysis_data['sr_zones']
        
        if not analysis_5m or not analysis_4h:
            return None
        
        direction = analysis_5m['direction']
        current_price = analysis_5m['current_price']
        velocity = analysis_5m['velocity']
        acceleration = analysis_5m['acceleration']
        
        # Target (resistencia/soporte más cercano)
        target_price = None
        target_distance_pct = None
        
        if sr_zones:
            if direction == 'BULLISH':
                nearest = find_nearest_resistance(current_price, sr_zones['resistances'])
                if nearest:
                    target_price = nearest[0]
                    target_distance_pct = nearest[1]
            else:  # BEARISH
                nearest = find_nearest_support(current_price, sr_zones['supports'])
                if nearest:
                    target_price = nearest[0]
                    target_distance_pct = nearest[1]
        
        # Si no hay target, usar proyección simple (3% para bulls, -3% para bears)
        if not target_price:
            if direction == 'BULLISH':
                target_price = current_price * 1.03
                target_distance_pct = 3.0
            else:
                target_price = current_price * 0.97
                target_distance_pct = 3.0
        
        # Tiempo proyectado a target
        tiempo_proyectado = calculate_time_to_target(
            current_price, target_price, velocity, acceleration
        )
        
        # Risk/Reward (stop loss simple 2%)
        stop_loss = current_price * 0.98 if direction == 'BULLISH' else current_price * 1.02
        reward = abs(target_price - current_price)
        risk = abs(current_price - stop_loss)
        rr_ratio = reward / risk if risk > 0 else 0
        
        # Validar viabilidad
        if rr_ratio < MIN_RR_RATIO:
            return None  # R:R insuficiente
        
        if tiempo_proyectado and (tiempo_proyectado < MIN_TIEMPO_PROYECTADO_MIN or 
                                  tiempo_proyectado > MAX_TIEMPO_PROYECTADO_MIN):
            return None  # Timing fuera de rango
        
        return {
            'trigger_adx': ADX_THRESHOLD,
            'trigger_adx_acceleration': ADX_MIN_ACCELERATION,
            'trigger_momentum_candles': MOMENTUM_CONSECUTIVE_CANDLES,
            'trigger_volume_ratio': VOLUME_RATIO_MIN,
            'target_price': target_price,
            'target_distance_pct': target_distance_pct,
            'stop_loss': stop_loss,
            'tiempo_proyectado_min': tiempo_proyectado,
            'rr_ratio': rr_ratio
        }
    
    def add_to_watchlist(self, symbol, analysis_data):
        """
        Agrega moneda a watchlist
        
        Args:
            symbol: Símbolo
            analysis_data: Dict con análisis completo
        
        Returns:
            bool: True si agregado, False si rechazado
        """
        # Verificar límite
        if len(self.watchlist) >= MAX_WATCHLIST_SIZE:
            # Remover moneda con score más bajo
            lowest_symbol = min(self.watchlist.keys(), 
                               key=lambda s: self.watchlist[s]['score'])
            lowest_score = self.watchlist[lowest_symbol]['score']
            
            if analysis_data['score_data']['score'] > lowest_score:
                del self.watchlist[lowest_symbol]
                print(f"⚠️ Watchlist llena, removido {lowest_symbol} (score {lowest_score})")
            else:
                return False  # Score muy bajo
        
        # Calcular triggers
        triggers = self.calculate_triggers(analysis_data)
        if not triggers:
            return False  # Triggers inviables
        
        # Agregar a watchlist
        self.watchlist[symbol] = {
            'added_timestamp': datetime.now(ARGENTINA_TZ).isoformat(),
            'score': analysis_data['score_data']['score'],
            'clasificacion': analysis_data['score_data']['clasificacion'],
            'direction': analysis_data['analysis_5m']['direction'],
            'triggers': triggers,
            'analysis_snapshot': {
                'adx_5m': analysis_data['analysis_5m']['adx'],
                'adx_4h': analysis_data['analysis_4h']['adx'],
                'momentum_age_5m': analysis_data['analysis_5m']['momentum_age'],
                'emas_classification_15m': analysis_data['analysis_15m']['emas_classification'] if analysis_data['analysis_15m'] else None
            },
            'status': 'MONITORING'
        }
        
        print(f"✅ Agregado a watchlist: {symbol} (Score: {analysis_data['score_data']['score']})")
        return True
    
    def remove_from_watchlist(self, symbol, reason=""):
        """Remueve moneda de watchlist"""
        if symbol in self.watchlist:
            del self.watchlist[symbol]
            print(f"❌ Removido de watchlist: {symbol} ({reason})")
    
    def get_watchlist(self):
        """Retorna watchlist ordenada por score"""
        sorted_list = sorted(
            self.watchlist.items(),
            key=lambda x: x[1]['score'],
            reverse=True
        )
        return sorted_list
    
    def update_watchlist_from_analysis(self, symbols_analyzed):
        """
        Actualiza watchlist con nuevos análisis
        
        Args:
            symbols_analyzed: List de (symbol, analysis_data) tuples
        """
        added_count = 0
        
        for symbol, analysis_data in symbols_analyzed:
            if not analysis_data or not analysis_data.get('viable'):
                continue
            
            # Si ya está en watchlist, actualizar score
            if symbol in self.watchlist:
                old_score = self.watchlist[symbol]['score']
                new_score = analysis_data['score_data']['score']
                
                # Si score bajó mucho (>15 puntos), remover
                if new_score < old_score - 15:
                    self.remove_from_watchlist(symbol, "score bajó significativamente")
                else:
                    # Actualizar score
                    self.watchlist[symbol]['score'] = new_score
            
            # Si no está, agregar
            else:
                if self.add_to_watchlist(symbol, analysis_data):
                    added_count += 1
        
        print(f"📊 Watchlist actualizada: {len(self.watchlist)} monedas ({added_count} nuevas)")
        self.save_to_file()
    
    def clear_watchlist(self):
        """Limpia watchlist completa"""
        self.watchlist = {}
        self.save_to_file()
        print("🗑️ Watchlist limpiada")

def calculate_adx(df, period=14):
    """
    Calcula ADX, DI+, DI-
    
    Returns:
        tuple: (adx, di_plus, di_minus)
    """
    try:
        high = df['high']
        low = df['low']
        close = df['close']
        
        # True Range
        tr1 = high - low
        tr2 = abs(high - close.shift(1))
        tr3 = abs(low - close.shift(1))
        tr = pd.concat([tr1, tr2, tr3], axis=1).max(axis=1)
        atr = tr.rolling(window=period).mean()
        
        # Directional Movement
        up_move = high - high.shift(1)
        down_move = low.shift(1) - low
        
        plus_dm = pd.Series(0.0, index=df.index)
        minus_dm = pd.Series(0.0, index=df.index)
        
        plus_dm[(up_move > down_move) & (up_move > 0)] = up_move
        minus_dm[(down_move > up_move) & (down_move > 0)] = down_move
        
        # Directional Indicators
        plus_di = 100 * (plus_dm.rolling(window=period).mean() / atr)
        minus_di = 100 * (minus_dm.rolling(window=period).mean() / atr)
        
        # ADX
        dx = 100 * abs(plus_di - minus_di) / (plus_di + minus_di)
        adx = dx.rolling(window=period).mean()
        
        return adx.iloc[-1], plus_di.iloc[-1], minus_di.iloc[-1]
    
    except:
        return None, None, None


def calculate_momentum_color(df, period=10):
    """
    Calcula momentum color (verde/rojo según EMA)
    
    Returns:
        str: 'green', 'red', 'neutral'
    """
    try:
        ema = df['close'].ewm(span=period, adjust=False).mean()
        current_close = df['close'].iloc[-1]
        current_ema = ema.iloc[-1]
        
        if current_close > current_ema * 1.002:  # 0.2% margen
            return 'green'
        elif current_close < current_ema * 0.998:
            return 'red'
        else:
            return 'neutral'
    except:
        return 'neutral'


def count_momentum_age(df, period=10):
    """
    Cuenta velas consecutivas con mismo color momentum
    
    Returns:
        int: Número de velas
    """
    try:
        ema = df['close'].ewm(span=period, adjust=False).mean()
        
        # Determinar color cada vela
        colors = []
        for i in range(len(df)):
            if df['close'].iloc[i] > ema.iloc[i] * 1.002:
                colors.append('green')
            elif df['close'].iloc[i] < ema.iloc[i] * 0.998:
                colors.append('red')
            else:
                colors.append('neutral')
        
        # Contar desde el final
        current_color = colors[-1]
        age = 0
        
        for i in range(len(colors) - 1, -1, -1):
            if colors[i] == current_color:
                age += 1
            else:
                break
        
        return age
    except:
        return 0


def calculate_emas_separation(df, fast=10, slow=50):
    """
    Calcula separación entre EMAs rápida/lenta
    
    Returns:
        tuple: (clasificacion, separacion_pct)
    """
    try:
        if len(df) < slow + 10:
            return 'CERCANAS', 0.0
        
        ema_fast = df['close'].ewm(span=fast, adjust=False).mean().iloc[-1]
        ema_slow = df['close'].ewm(span=slow, adjust=False).mean().iloc[-1]
        
        separation_pct = abs((ema_fast - ema_slow) / ema_slow * 100)
        
        if separation_pct < 2.0:
            return 'CERCANAS', separation_pct
        elif separation_pct <= 5.0:
            return 'MODERADAS', separation_pct
        else:
            return 'SEPARADAS', separation_pct
    except:
        return 'CERCANAS', 0.0

        'emas_separation': emas_sep,
        'velocity': velocity,
        'acceleration': acceleration,
        'current_price': prices[-1]
    }

def calculate_score(analysis_1d, analysis_4h, analysis_15m, analysis_5m, 
                   sr_zones=None, vp_data=None):
    """
    Calcula score 0-100 basado en análisis multi-timeframe
    
    Args:
        analysis_1d: Dict con análisis 1D
        analysis_4h: Dict con análisis 4H
        analysis_15m: Dict con análisis 15M
        analysis_5m: Dict con análisis 5M
        sr_zones: Dict con soportes/resistencias (opcional)
        vp_data: Dict con volume profile (opcional)
    
    Returns:
        dict: {
            'score': int (0-100),
            'clasificacion': str ('ÓPTIMA', 'BUENA', 'REGULAR', 'MALA'),
            'detalles': dict con breakdown
        }
    """
    score = 50  # Base
    detalles = {}
    
    # BONUS 1: Confluencia 4H (crítico)
    if analysis_4h and analysis_5m:
        if analysis_4h['direction'] == analysis_5m['direction']:
            score += BONUS_CONFLUENCE_4H
            detalles['confluence_4h'] = f"+{BONUS_CONFLUENCE_4H} (alineado)"
        else:
            detalles['confluence_4h'] = "0 (contradicción)"
    
    # BONUS 2: Confluencia 1D (informativo)
    if analysis_1d and analysis_5m:
        tendencia_1d = analysis_1d['tendencia']
        direction_5m = analysis_5m['direction']
        
        if (tendencia_1d == 'ALCISTA' and direction_5m == 'BULLISH') or \
           (tendencia_1d == 'BAJISTA' and direction_5m == 'BEARISH'):
            bonus = BONUS_CONFLUENCE_1D
            if analysis_1d['intensidad'] == 'FUERTE':
                bonus += 5  # Bonus extra
            score += bonus
            detalles['confluence_1d'] = f"+{bonus} (alineado {analysis_1d['intensidad']})"
        elif tendencia_1d == 'LATERAL':
            detalles['confluence_1d'] = "0 (1D lateral)"
        else:
            detalles['confluence_1d'] = "0 (no alineado)"
    
    # BONUS 3: EMAs cercanas (precio arrancando)
    if analysis_15m:
        if analysis_15m['emas_classification'] == 'CERCANAS':
            score += BONUS_EMAS_CERCANAS
            detalles['emas_15m'] = f"+{BONUS_EMAS_CERCANAS} (cercanas)"
        elif analysis_15m['emas_classification'] == 'MODERADAS':
            score += 5
            detalles['emas_15m'] = "+5 (moderadas)"
        else:
            score -= PENALTY_EMAS_SEPARADAS
            detalles['emas_15m'] = f"-{PENALTY_EMAS_SEPARADAS} (separadas)"
    
    if analysis_5m:
        if analysis_5m['emas_classification'] == 'CERCANAS':
            score += BONUS_EMAS_CERCANAS
            detalles['emas_5m'] = f"+{BONUS_EMAS_CERCANAS} (cercanas)"
        elif analysis_5m['emas_classification'] == 'MODERADAS':
            score += 5
            detalles['emas_5m'] = "+5 (moderadas)"
        else:
            score -= PENALTY_EMAS_SEPARADAS
            detalles['emas_5m'] = f"-{PENALTY_EMAS_SEPARADAS} (separadas)"
    
    # BONUS 4: Edad momentum joven
    if analysis_15m:
        edad_15m = analysis_15m['momentum_age']
        if edad_15m <= 5:
            score += BONUS_EDAD_JOVEN
            detalles['edad_15m'] = f"+{BONUS_EDAD_JOVEN} (joven: {edad_15m}v)"
        elif edad_15m <= 10:
            score += 10
            detalles['edad_15m'] = f"+10 (desarrollo: {edad_15m}v)"
        elif edad_15m <= 15:
            detalles['edad_15m'] = f"0 (avanzado: {edad_15m}v)"
        else:
            score -= PENALTY_EDAD_MADURA
            detalles['edad_15m'] = f"-{PENALTY_EDAD_MADURA} (maduro: {edad_15m}v)"
    
    if analysis_5m:
        edad_5m = analysis_5m['momentum_age']
        if edad_5m <= 3:
            score += 15
            detalles['edad_5m'] = f"+15 (muy joven: {edad_5m}v)"
        elif edad_5m <= 6:
            score += 10
            detalles['edad_5m'] = f"+10 (joven: {edad_5m}v)"
        elif edad_5m <= 10:
            score += 5
            detalles['edad_5m'] = f"+5 (desarrollo: {edad_5m}v)"
        else:
            score -= PENALTY_EDAD_MADURA
            detalles['edad_5m'] = f"-{PENALTY_EDAD_MADURA} (maduro: {edad_5m}v)"
    
    # BONUS 5: Sin obstáculos cercanos (soportes/resistencias)
    if sr_zones and analysis_5m:
        current_price = analysis_5m['current_price']
        direction = analysis_5m['direction']
        
        if direction == 'BULLISH':
            nearest_r = find_nearest_resistance(current_price, sr_zones['resistances'])
            if nearest_r:
                distance_pct = nearest_r[1]
                if distance_pct > 5.0:
                    score += BONUS_NO_OBSTACULOS
                    detalles['obstaculos'] = f"+{BONUS_NO_OBSTACULOS} (sin resistencia cercana)"
                elif distance_pct < 2.0:
                    score -= 10
                    detalles['obstaculos'] = f"-10 (resistencia muy cerca: {distance_pct:.1f}%)"
                else:
                    detalles['obstaculos'] = f"0 (resistencia moderada: {distance_pct:.1f}%)"
            else:
                detalles['obstaculos'] = "0 (no detectado)"
        
        elif direction == 'BEARISH':
            nearest_s = find_nearest_support(current_price, sr_zones['supports'])
            if nearest_s:
                distance_pct = nearest_s[1]
                if distance_pct > 5.0:
                    score += BONUS_NO_OBSTACULOS
                    detalles['obstaculos'] = f"+{BONUS_NO_OBSTACULOS} (sin soporte cercano)"
                elif distance_pct < 2.0:
                    score -= 10
                    detalles['obstaculos'] = f"-10 (soporte muy cerca: {distance_pct:.1f}%)"
                else:
                    detalles['obstaculos'] = f"0 (soporte moderado: {distance_pct:.1f}%)"
            else:
                detalles['obstaculos'] = "0 (no detectado)"
    
    # Clamp score entre 0-100
    score = max(0, min(100, score))
    
    # Clasificación
    if score >= 80:
        clasificacion = 'ÓPTIMA'
    elif score >= 70:
        clasificacion = 'BUENA'
    elif score >= 60:
        clasificacion = 'REGULAR'
    else:
        clasificacion = 'MALA'
    
    return {
        'score': score,
        'clasificacion': clasificacion,
        'detalles': detalles
    }


def analyze_coin_complete(symbol):
    """
    Análisis completo de una moneda (todas las temporalidades + scoring)
    
    Args:
        symbol: Símbolo (ej: 'BTC/USDT:USDT')
    
    Returns:
        dict: {
            'symbol': str,
            'analysis_1d': dict,
            'analysis_4h': dict,
            'analysis_15m': dict,
            'analysis_5m': dict,
            'sr_zones': dict,
            'vp_data': dict,
            'score_data': dict,
            'viable': bool
        }
        None si no pasa filtros básicos
    """
    
    # Análisis 4H (obligatorio, determina si vale seguir)
    analysis_4h = analyze_4h(symbol)
    if not analysis_4h:
        return None
    
    # Si 4H es LATERAL o WEAK, skip
    if analysis_4h['direction'] == 'LATERAL' or analysis_4h['strength'] == 'WEAK':
        return None
    
    # Análisis resto timeframes
    analysis_1d = analyze_1d(symbol)
    analysis_15m = analyze_15m(symbol)
    analysis_5m = analyze_5m(symbol)
    
    if not analysis_5m:
        return None
    
    # Soportes/Resistencias (si habilitado)
    sr_zones = None
    if ENABLE_CLUSTERING_SR:
        ohlcv_4h = fetch_ohlcv(symbol, '4h', 200)
        if ohlcv_4h:
            sr_zones = detect_support_resistance(ohlcv_4h)
    
    # Volume Profile (si habilitado)
    vp_data = None
    if ENABLE_VOLUME_PROFILE:
        ohlcv_4h = fetch_ohlcv(symbol, '4h', 200)
        if ohlcv_4h:
            vp_data = calculate_volume_profile(ohlcv_4h)
    
    # Scoring
    score_data = calculate_score(
        analysis_1d, analysis_4h, analysis_15m, analysis_5m,
        sr_zones, vp_data
    )
    
    # Determinar viabilidad
    viable = (
        score_data['score'] >= MIN_SCORE_WATCHLIST and
        analysis_4h['direction'] == analysis_5m['direction']  # Confluencia obligatoria
    )
    
    return {
        'symbol': symbol,
        'analysis_1d': analysis_1d,
        'analysis_4h': analysis_4h,
        'analysis_15m': analysis_15m,
        'analysis_5m': analysis_5m,
        'sr_zones': sr_zones,
        'vp_data': vp_data,
        'score_data': score_data,
        'viable': viable
    }
