def _build_diagonal_trendlines(self, df, zigzag_data):
    """
    Construye trendlines diagonales usando regresión lineal de pivots.
    
    Returns:
        dict: {
            'upper_trendline': (slope, intercept),
            'lower_trendline': (slope, intercept),
            'upper_r2': float,  # Calidad de la línea superior
            'lower_r2': float,  # Calidad de la línea inferior
            'channel_width': float,  # Ancho actual del canal
            'price_position_pct': float,  # 0-100% posición en canal
        }
    """
    high_pivots = zigzag_data.get('high_pivots', [])
    low_pivots = zigzag_data.get('low_pivots', [])
    
    # Necesitamos mínimo 3 pivots de cada tipo
    if len(high_pivots) < 2 or len(low_pivots) < 2:
        return self._default_trendlines()
    
    # Regresión lineal para HIGHs
    x_high = np.array([p[0] for p in high_pivots])
    y_high = np.array([p[1] for p in high_pivots])
    upper_coeffs = np.polyfit(x_high, y_high, 1)
    upper_predicted = np.polyval(upper_coeffs, x_high)
    upper_r2 = self._calculate_r2(y_high, upper_predicted)
    
    # Regresión lineal para LOWs
    x_low = np.array([p[0] for p in low_pivots])
    y_low = np.array([p[1] for p in low_pivots])
    lower_coeffs = np.polyfit(x_low, y_low, 1)
    lower_predicted = np.polyval(lower_coeffs, x_low)
    lower_r2 = self._calculate_r2(y_low, lower_predicted)
    
    # Calcular posición actual del precio en canal
    current_idx = len(df) - 1
    current_price = df['close'].iloc[-1]
    
    upper_at_current = np.polyval(upper_coeffs, current_idx)
    lower_at_current = np.polyval(lower_coeffs, current_idx)
    channel_width = upper_at_current - lower_at_current
    
    if channel_width > 0:
        price_position_pct = ((current_price - lower_at_current) / channel_width) * 100
    else:
        price_position_pct = 50  # Default si canal colapsado
    
    return {
        'upper_trendline': tuple(upper_coeffs),
        'lower_trendline': tuple(lower_coeffs),
        'upper_r2': upper_r2,
        'lower_r2': lower_r2,
        'avg_r2': (upper_r2 + lower_r2) / 2,
        'channel_width': channel_width,
        'channel_width_pct': (channel_width / current_price * 100) if current_price > 0 else 0,
        'price_position_pct': max(0, min(100, price_position_pct)),  # Clamp 0-100
        'upper_at_current': upper_at_current,
        'lower_at_current': lower_at_current,
    }
def _calculate_r2(self, y_actual, y_predicted):
    """Calcula R² (coeficiente de determinación)"""
    ss_res = np.sum((y_actual - y_predicted) ** 2)
    ss_tot = np.sum((y_actual - np.mean(y_actual)) ** 2)
    if ss_tot == 0:
        return 0.0
    return max(0.0, 1 - (ss_res / ss_tot))
def _default_trendlines(self):
    """Retorno por defecto si no hay suficientes datos"""
    return {
        'upper_trendline': (0, 0),
        'lower_trendline': (0, 0),
        'upper_r2': 0.0,
        'lower_r2': 0.0,
        'avg_r2': 0.0,
        'channel_width': 0.0,
        'channel_width_pct': 0.0,
        'price_position_pct': 50.0,
        'upper_at_current': 0.0,
        'lower_at_current': 0.0,
        'valid': False
    }
