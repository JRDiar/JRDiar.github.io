def make_pattern_based_decision(patterns, ladder, signals, symbol, current_price, recent_history=None):
    """
    Núcleo de Decisión v5.0 (Corregido)
    Integra: Física (Wrecking Ball), Estructura (ZigZag), Zonas (S/R), Memoria.
    """
    score = 0
    reasons = []
    block_trade = False

    # --- 1. DATOS DE ENTRADA ---
    pattern_dir = patterns['confidence'].get('direction', 'NEUTRAL')
    force_dir = signals.get('market_force', {}).get('direction', 'NEUTRAL')
    
    # Prioridad: Patrón > Fuerza
    trade_dir = pattern_dir if pattern_dir in ['BULLISH', 'BEARISH'] else force_dir
    
    a5m = ladder.get('5m', {})
    if not a5m: return {'action': 'WAIT', 'confidence': 0, 'reasons': ['No data 5M']}
    
    # --- 2. FÍSICA (WRECKING BALL - Lógica de Compensación) ---
    adx_val = signals.get('adx_val', 0)
    velocity_info = a5m.get('velocity', {})
    
    # 1. Analizar la calidad del movimiento
    trend_state = velocity_info.get('trend', 'estable')
    is_accelerating = trend_state in ['acelerando', 'aumentando']
    
    # 2. Definir el Umbral de Resistencia (La "Vara" a saltar)
    # Base: Se requiere Fuerza Extrema (45) para romper paredes por pura masa.
    required_force = 45 
    
    if is_accelerating:
        # BONUS DINÁMICO: La aceleración reduce el requisito de fuerza base.
        # Filosofía: "Si vienes con impulso (aceleración), es más fácil romper la pared."
        required_force -= 15  # La vara baja a 30
        
    # 3. Evaluación Final (Gradiente)
    is_wrecking_ball = adx_val > required_force
    
    if is_wrecking_ball:
        reasons.append(f"☄️ FÍSICA: Inercia suficiente (ADX {adx_val:.1f} > Req {required_force})")

    # --- 3. ESTRUCTURA (ZIGZAG) ---
    zz_data = a5m.get('zigzag', {})
    pivot_types = zz_data.get('pivot_types', [])
    last_pivot = pivot_types[-1] if pivot_types else 'UNKNOWN'
    
    if trade_dir == 'BEARISH':
        if last_pivot == 'HL': 
            score -= 50
            reasons.append("⛔ ESTRUCTURA: ZigZag Higher Low (Posible reversión alcista)")
            if not is_wrecking_ball: block_trade = True
        elif last_pivot == 'LH': score += 20; reasons.append("✅ ESTRUCTURA: ZigZag Lower High")
        elif last_pivot == 'LL': score += 10; reasons.append("✅ ESTRUCTURA: ZigZag Lower Low")
            
    elif trade_dir == 'BULLISH':
        if last_pivot == 'LH': 
            score -= 50
            reasons.append("⛔ ESTRUCTURA: ZigZag Lower High (Posible reversión bajista)")
            if not is_wrecking_ball: block_trade = True
        elif last_pivot == 'HL': score += 20; reasons.append("✅ ESTRUCTURA: ZigZag Higher Low")
        elif last_pivot == 'HH': score += 10; reasons.append("✅ ESTRUCTURA: ZigZag Higher High")

    # --- 4. GEOGRAFÍA (SOPORTES/RESISTENCIAS) ---
    sr_data = a5m.get('sr', {})
    
    if trade_dir == 'BEARISH':
        supports = sr_data.get('supports', [])
        near = [s for s in supports if s < current_price]
        if near:
            nearest = max(near)
            dist = (current_price - nearest) / current_price * 100
            if dist < 1.5:
                if is_wrecking_ball: score += 15; reasons.append(f"🔨 ROMPIENDO SOPORTE ({dist:.2f}%)")
                else: score -= 100; reasons.append(f"🧱 MURO: Vendiendo en Soporte ({dist:.2f}%)"); block_trade = True
    
    elif trade_dir == 'BULLISH':
        resistances = sr_data.get('resistances', [])
        near = [r for r in resistances if r > current_price]
        if near:
            nearest = min(near)
            dist = (nearest - current_price) / current_price * 100
            if dist < 1.5:
                if is_wrecking_ball: score += 15; reasons.append(f"🔨 ROMPIENDO RESISTENCIA ({dist:.2f}%)")
                else: score -= 100; reasons.append(f"🧱 MURO: Comprando en Resistencia ({dist:.2f}%)"); block_trade = True

    # --- 5. VALIDACIÓN DE FATIGA (GOMA ELÁSTICA) ---
    # Evita el FOMO: No perseguir precios que ya corrieron demasiado lejos de la media.
    dist_ema_pct = a5m.get('dist_ema_pct', 0)
    rsi_val = a5m.get('rsi', {}).get('rsi_current', 50)
    
    # UMBRALES DE FATIGA (Límites físicos de estiramiento)
    MAX_EXTENSION = 3.5  # Si el precio se aleja >3.5% de la EMA 21, el rebote es inminente.
    EXTREME_RSI_HIGH = 80
    EXTREME_RSI_LOW = 20
    
    if trade_dir == 'BULLISH':
        # Caso Long: Precio muy arriba de la media o RSI hirviendo
        if dist_ema_pct > MAX_EXTENSION:
            score -= 100 # Bloqueo duro: Es muy tarde para entrar
            reasons.append(f"🛑 FATIGA: Precio sobre-extendido (+{dist_ema_pct:.2f}% vs EMA21)")
            block_trade = True
        elif rsi_val > EXTREME_RSI_HIGH:
            score -= 50
            reasons.append(f"⚠️ FATIGA: RSI Sobrecomprado ({rsi_val:.1f})")
            # INTELIGENCIA: Solo una 'Wrecking Ball' puede ignorar un RSI extremo
            if not is_wrecking_ball: 
                block_trade = True 
            
    elif trade_dir == 'BEARISH':
        # Caso Short: Precio muy abajo de la media o RSI congelado
        if dist_ema_pct < -MAX_EXTENSION:
            score -= 100
            reasons.append(f"🛑 FATIGA: Precio sobre-extendido ({dist_ema_pct:.2f}% vs EMA21)")
            block_trade = True
        elif rsi_val < EXTREME_RSI_LOW:
            score -= 50
            reasons.append(f"⚠️ FATIGA: RSI Sobrevendido ({rsi_val:.1f})")
            if not is_wrecking_ball: 
                block_trade = True

    # --- 6. MEMORIA (PERSISTENCIA) ---
    if recent_history and not block_trade:
        attempts = 0
        for row in recent_history:
            # Detectar si intentamos romper este muro antes
            rec_reasons = str(row[3])
            if "MURO" in rec_reasons or "Soporte" in rec_reasons or "Resistencia" in rec_reasons:
                attempts += 1
        
        if attempts > 0:
            if is_wrecking_ball or signals.get('market_force', {}).get('detected'):
                score += min(attempts * 5, 15)
                reasons.append(f"🧠 MEMORIA: Persistencia ({attempts} intentos previos)")

    # --- 7. SCORE FINAL ---
    if signals.get('market_force', {}).get('detected'):
        score += 40
        reasons.append(signals['market_force']['reason'])
    
    if patterns.get('best_pattern'):
        score += patterns['best_pattern']['confidence'] * 0.3
        reasons.append(f"Patrón: {patterns['best_pattern']['type']}")

    final_score = max(0, min(100, score))
    
    # Proyección de Escenarios (si vale la pena)
    scenarios = None
    if final_score >= 50 and not block_trade:
        try:
            from brain_scenarios import project_scenarios
            scenarios = project_scenarios(symbol, current_price, ladder)
        except: pass

    if block_trade:
        return {'action': 'WAIT', 'confidence': 0, 'reasons': [f"🚫 BLOQUEO: {reasons[-1]}"] + reasons}

    MIN_THRESHOLD = 70
    
    if final_score >= MIN_THRESHOLD:
        return {
            'action': 'ENTER',
            'confidence': final_score,
            'reasons': reasons,
            'strategy': determine_strategy(patterns, ladder, scenarios),
            'pattern': patterns.get('best_pattern', {}).get('type', 'STRUCTURE')
        }
    else:
        return {
            'action': 'WAIT',
            'confidence': final_score,
            'reasons': reasons,
            'missing': MIN_THRESHOLD - final_score
        }

def determine_strategy(patterns, ladder, scenarios=None):
    """
    Define Take Profit y Stop Loss.
    Usa Proyección Dinámica (Escenarios) si están disponibles.
    """
    pattern_type = patterns.get('best_pattern', {}).get('type', 'STANDARD')
    
    # Valores Base
    stop_mult = 2.0
    target_mult = 2.0
    position = 0.75
    
    # Ajuste por Patrón
    if 'DOUBLE' in pattern_type or 'BOS' in pattern_type:
        target_mult = 2.5
        position = 0.5
    elif 'FLAG' in pattern_type:
        target_mult = 3.0
        position = 1.0

    # Ajuste por Escenario (Proyección)
    if scenarios:
        primary = scenarios.get('primary', {})
        prob = primary.get('probability', 0)
        rr = primary.get('rr_ratio', 0)
        
        if prob > 60 and rr > 2.0:
            target_mult = rr # Usar target real del gráfico
            position = min(position + 0.25, 1.0)
        elif prob < 50:
            position *= 0.8 # Reducir mano

    return {
        'type': 'SCENARIO' if scenarios else 'PATTERN',
        'stop_multiplier': stop_mult,
        'target_multiplier': target_mult,
        'position_size': position
    }

def check_market_force(ladder):
    """
    Verifica FUERZA DE MERCADO (Blindada v5.0).
    Reglas: ADX > 25, Spread > 5, Momentum Alineado.
    """
    a1h = ladder.get('1h') or {}
    if not a1h: return {'detected': False, 'reason': "No data 1H"}

    adx_data = a1h.get('adx', {})
    mom_data = a1h.get('momentum', {})
    
    adx_val = adx_data.get('adx_current', 0)
    di_plus = adx_data.get('di_plus_current', 0)
    di_minus = adx_data.get('di_minus_current', 0)
    mom_color = mom_data.get('color_current', 'gray')
    
    force_direction = 'NEUTRAL'
    if di_plus > di_minus: force_direction = 'BULLISH'; spread = di_plus - di_minus
    else: force_direction = 'BEARISH'; spread = di_minus - di_plus
        
    if adx_val < 25: return {'detected': False, 'reason': f"ADX débil ({adx_val:.1f})"}
    if spread < 5.0: return {'detected': False, 'reason': f"Spread sucio ({spread:.1f})"}
    
    aligned = False
    if force_direction == 'BULLISH' and mom_color in ['green', 'lime']: aligned = True
    elif force_direction == 'BEARISH' and mom_color in ['red', 'maroon']: aligned = True
    
    if not aligned: return {'detected': False, 'reason': f"Divergencia Mom/ADX"}

    return {
        'detected': True,
        'direction': force_direction,
        'reason': f"💪 FUERZA: ADX {adx_val:.1f} {force_direction}"
    }

def check_adx_alignment(ladder):
    """Verifica si ADX apoya la tendencia en TFs clave"""
    a1h = ladder.get('1h') or {}
    a15m = ladder.get('15m') or {}
    
    def get_val(data):
        if isinstance(data, dict):
            return data.get('adx_current', 0)
        return float(data) if data else 0

    adx_1h = get_val(a1h.get('adx'))
    adx_15m = get_val(a15m.get('adx'))

    # Alineación: Al menos uno debe mostrar fuerza real
    return adx_1h > 20 or adx_15m > 20

def check_momentum_propagation(ladder):
    """Verifica si el momentum se propaga de 4H a 15M"""
    dirs = []
    for tf in ['4h', '1h', '15m']:
        analysis = ladder.get(tf)
        if analysis and isinstance(analysis, dict):
            dirs.append(analysis.get('direction', 'NEUTRAL'))
            
    # Si tenemos 3 direcciones y son iguales (y no LATERAL)
    if len(dirs) >= 3 and len(set(dirs)) == 1 and dirs[0] in ['BULLISH', 'BEARISH']:
        return True
    return False

def check_volume_confirmation(ladder):
    """Verifica si hay volumen confirmando en 5M"""
    a5m = ladder.get('5m') or {}
    vp_data = a5m.get('volume_profile', {})
    
    # Si no hay perfil, no bloqueamos (neutral)
    if not vp_data: return True
    
    # Lógica simple: Si estamos sobre el POC en subida, o bajo el POC en bajada
    return True

def validate_adx_layer(symbol, tf='15m'):
    """
    CAPA 1: Validación ADX
    Requisitos:
    - ADX >= 20 (fuerza mínima)
    - Spread DI > 5 (separación clara)
    - ADX o DI+ creciendo (no muriendo)
    
    Returns:
        dict: {valid: bool, score: 0-25, reason: str, data: dict}
    """
    try:
        ohlcv = fetch_ohlcv(symbol, tf, 50)
        if not ohlcv:
            return {'valid': False, 'score': 0, 'reason': 'No data', 'data': {}}
        
        df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
        adx_data = calculate_adx_with_memory(df, period=14, lookback=10)
        
        adx_val = adx_data['adx_current']
        spread = adx_data['spread']
        adx_trend = adx_data['adx_trend']
        di_plus_trend = adx_data['di_plus_trend']
        dominance = adx_data['dominance']
        
        # Validaciones (STRICTAS - ADX debe estar creciendo)
        checks = {
            'adx_min': adx_val >= 20,
            'spread_min': spread >= 8.0,
            'adx_growing': adx_trend == 'creciendo',  # ADX DEBE crecer (no OR)
            'dominance_ok': dominance == 'DI+'
        }
        
        passed = sum(checks.values())
        score = passed * 6  # Max 24 puntos
        
        # Requiere al menos 3/4 checks
        valid = passed >= 3 and checks['adx_min']  # ADX mínimo es obligatorio
        
        reason = f"ADX {adx_val:.1f} | Spread {spread:.1f} | {adx_trend}"
        if not valid:
            failed = [k for k, v in checks.items() if not v]
            reason = f"❌ ADX Layer FAILED: {', '.join(failed)}"
        
        return {
            'valid': valid,
            'score': score,
            'reason': reason,
            'data': adx_data
        }
        
    except Exception as e:
        return {'valid': False, 'score': 0, 'reason': f'Error: {e}', 'data': {}}


def validate_momentum_layer(symbol, tf='15m', direction='BULLISH'):
    """
    CAPA 2: Validación Momentum (SNIPER TIMING)
    Requisitos:
    - Color alineado (Verde/Lime para BULLISH)
    - EN DESARROLLO: Las mejores entradas son en las velas 1-6.
    - Si tiene > 8 velas, es momentum "viejo" (riesgo de agotamiento).
    
    Returns:
        dict: {valid: bool, score: 0-25, reason: str, data: dict}
    """
    try:
        ohlcv = fetch_ohlcv(symbol, tf, 50)
        if not ohlcv:
            return {'valid': False, 'score': 0, 'reason': 'No data', 'data': {}}
        
        df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
        mom_data = calculate_momentum_with_memory(df, lookback=10)
        
        color = mom_data['color_current']
        consecutive = mom_data['consecutive_count']
        strength = mom_data['strength']
        squeeze = mom_data.get('squeeze_active', False)
        
        # Colores válidos según dirección
        if direction == 'BULLISH':
            valid_colors = ['green', 'lime']
        else:
            valid_colors = ['red', 'maroon']
        
        # Validaciones
        is_aligned = color in valid_colors
        not_neutral = color not in ['neutral', 'gray']
        
        score = 0
        reason = ""
        valid = False
        
        if is_aligned and not_neutral:
            # SCORING BASADO EN EDAD DEL MOMENTUM (TIMING)
            if 1 <= consecutive <= 3:
                score = 25
                reason = f"🚀 MOMENTUM STARTING ({consecutive} bars) - PERFECT TIMING"
                valid = True
            elif 4 <= consecutive <= 7:
                score = 20
                reason = f"✅ MOMENTUM DEVELOPING ({consecutive} bars) - GOOD"
                valid = True
            elif consecutive >= 8:
                score = 10
                reason = f"⚠️ MOMENTUM MATURE ({consecutive} bars) - CAUTION"
                # Aún válido, pero con menos score. No bloqueamos, pero avisamos.
                valid = True 
            else:
                score = 0
                reason = "❌ Momentum Error"
        else:
            reason = f"❌ Momentum Misaligned: {color} (need {valid_colors})"
            valid = False
            
        if squeeze:
            reason += " | ⚡SQUEEZE (Potential Explosion)"
            score += 5 # Bonus por squeeze
            
        return {
            'valid': valid,
            'score': min(25, score),
            'reason': reason,
            'data': mom_data
        }
        
    except Exception as e:
        return {'valid': False, 'score': 0, 'reason': f'Error: {e}', 'data': {}}


def validate_structure_layer(symbol, tf='15m', direction='BULLISH', pattern_type=None, ladder=None):
    """
    CAPA 3: Validación Estructura ZigZag (Pattern-Aware)
    Requisitos:
    - CONTINUACIÓN: Último movimiento HH o HL (estructura sana)
    - REVERSIÓN: Permite LH/LL (necesario para detectar agotamiento)
    
    Args:
        symbol: Símbolo a analizar
        tf: Timeframe (default '15m')
        direction: Dirección esperada ('BULLISH' o 'BEARISH')
        pattern_type: Tipo de patrón detectado ('DOUBLE_TOP', 'DOUBLE_BOTTOM', etc.)
    
    Returns:
        dict: {valid: bool, score: 0-25, reason: str, data: dict}
    """
    try:
        ohlcv = fetch_ohlcv(symbol, tf, 100)
        if not ohlcv:
            return {'valid': False, 'score': 0, 'reason': 'No data', 'data': {}}
        
        df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
        zz_data = calculate_zigzag(df, threshold_pct=1.5)
        
        pivot_types = zz_data.get('pivot_types', [])
        last_structure = zz_data.get('last_structure', 'N/A')
        
        # Último tipo de pivot
        last_pivot = pivot_types[-1] if pivot_types else 'N/A'
        
        # NUEVO: Detectar si es un patrón de REVERSIÓN
        reversal_patterns = ['DOUBLE_TOP', 'DOUBLE_BOTTOM', 'HCH', 'WEDGE', 'BOS_BULLISH', 'BOS_BEARISH']
        is_reversal = pattern_type in reversal_patterns if pattern_type else False
        
        # BYPASS: Si es patrón de reversión + ADX brutal (>45), confiar en pattern detector
        # No exigir ver LH/LL en ZigZag si la fuerza es extrema
        if is_reversal and ladder:
            a1h = ladder.get('1h', {}) or {}
            adx_1h = a1h.get('adx', {}).get('adx_current', 0)
            
            if adx_1h > 45:
                # Con fuerza tan brutal, el pattern detector es suficiente
                return {
                    'valid': True,
                    'score': 25,
                    'reason': f"✅ REVERSAL ({pattern_type}) + BRUTAL FORCE (ADX {adx_1h:.1f}) - Trust pattern detector",
                    'data': zz_data
                }
        
        # Validaciones para BULLISH
        if direction == 'BULLISH':

def validate_zone_layer(symbol, tf='5m'):
    """
    CAPA 4: Validación Zona y Estructura Dinámica (EMAs)
    Requisitos:
    - EMAs como "Carretera": Precio debe estar apoyado en EMAs (o cerca), no lejísimos.
    - VP: Permitir acercarse al High si hay fuerza, pero no comprar SI YA ESTAMOS en el High sin fuerza.
    
    Returns:
        dict: {valid: bool, score: 0-25, reason: str, data: dict}
    """
    try:
        ohlcv = fetch_ohlcv(symbol, tf, 50)
        if not ohlcv:
            return {'valid': False, 'score': 0, 'reason': 'No data', 'data': {}}
        
        df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
        current_price = df['close'].iloc[-1]
        
        # Volume Profile
        vp_data = calculate_volume_profile_with_memory(df, num_levels=10, lookback=50)
        profile_high = vp_data.get('profile_high', 0)
        poc = vp_data.get('poc_current', 0)
        
        # EMAs (La Carretera)
        ema9 = df['close'].ewm(span=9, adjust=False).mean().iloc[-1]
        ema21 = df['close'].ewm(span=21, adjust=False).mean().iloc[-1]
        
        # Distancia a EMAs (Overextension Check)
        # Si el precio está > 2% de la EMA21 en 5M, es peligroso (elastico estirado)
        dist_ema21_pct = ((current_price - ema21) / ema21) * 100
        
        # Distancia a VP High
        dist_to_high = ((profile_high - current_price) / current_price) * 100 if current_price > 0 else 0
        
        score = 0
        reasons = []
        valid = True
        
        # 1. CHECK CARRETERA (EMAs)
        if abs(dist_ema21_pct) < 1.5:
            score += 15
            reasons.append(f"✅ Supported by EMAs (Dist {dist_ema21_pct:.2f}%)")
        else:
            score += 5
            reasons.append(f"⚠️ Extended from EMAs (Dist {dist_ema21_pct:.2f}%)")
            # No invalidamos, pero bajamos score. El usuario quiere flexibilidad.
            
        # 2. CHECK VP (Techo)
        # Si estamos MUY cerca del techo (<0.5%) y NO hay breakout confirmado, cuidado.
        if 0 < dist_to_high < 0.5:
            reasons.append(f"⚠️ Near VP High ({dist_to_high:.2f}%)")
            # Aquí confiamos en que la capa de FUERZA (ADX) ya validó si tenemos potencia para romper.
            # Si llegamos aquí, asumimos que hay fuerza, así que no bloqueamos, solo advertimos.
            score += 5
        elif dist_to_high < 0:
            reasons.append("🚀 BREAKOUT MODE (Above VP High)")
            score += 10
        else:
            reasons.append("✅ Room to move")
            score += 10
            
        return {
            'valid': valid,
            'score': min(25, score),
            'reason': " | ".join(reasons),
            'data': {'dist_ema': dist_ema21_pct, 'dist_high': dist_to_high}
        }
        
    except Exception as e:
        return {'valid': False, 'score': 0, 'reason': f'Error: {e}', 'data': {}}



def validate_context_layer(ladder, direction):

    if dir_4h == direction:

        'reason': ' | '.join(reasons),

def validate_force_propagation(ladder, direction):
    """
    Detecta fuerza PROPAGÁNDOSE en CUALQUIER combinación de TFs.
    """
    score = 0
    reasons = []
    
    # Extraer datos
    tfs = ['4h', '1h', '15m', '5m']
    force_map = {}
    
    for tf in tfs:
        analysis = ladder.get(tf, {}) or {}
        adx_val = analysis.get('adx', {}).get('adx_current', 0)
        mom_data = analysis.get('momentum', {})
        mom_color = mom_data.get('color_current', 'neutral')
        consecutive = mom_data.get('consecutive_count', 0)
        
        # Validar momentum confirma dirección (filosofía FUERZA)
        if direction == 'BULLISH':
            mom_confirms = mom_color in ['green', 'lime']
        else:
            mom_confirms = mom_color in ['red', 'maroon']
        
        # Graduar fuerza por madurez
        has_mature = adx_val > 25 and mom_confirms and consecutive >= 3  # Maduro
        has_confirmed = adx_val > 25 and mom_confirms and consecutive >= 2  # Confirmado
        has_emerging = adx_val > 25 and mom_confirms and consecutive >= 1  # Emergente
        
        force_map[tf] = {
            'has_mature': has_mature,
            'has_confirmed': has_confirmed,
            'has_emerging': has_emerging,
            'adx': adx_val,
            'consecutive': consecutive,
            'mom_color': mom_color
        }
    
    # Detectar propagación STRONG (mature/confirmed)
    consecutive_strong = 0
    max_consecutive_strong = 0
    
    # Detectar propagación ANY (mature/confirmed/emerging)
    consecutive_any = 0
    max_consecutive_any = 0
    
    strongest_tf = None
    max_adx = 0
    
    for tf in tfs:
        # Track STRONG
        if force_map[tf]['has_mature'] or force_map[tf]['has_confirmed']:
            consecutive_strong += 1
        else:
            max_consecutive_strong = max(max_consecutive_strong, consecutive_strong)
            consecutive_strong = 0
        
        # Track ANY
        if force_map[tf]['has_emerging']:
            consecutive_any += 1
            if force_map[tf]['adx'] > max_adx:
                max_adx = force_map[tf]['adx']
                strongest_tf = tf
        else:
            max_consecutive_any = max(max_consecutive_any, consecutive_any)
            consecutive_any = 0
    
    max_consecutive_strong = max(max_consecutive_strong, consecutive_strong)
    max_consecutive_any = max(max_consecutive_any, consecutive_any)
    
    # Scoring GRADUADO (Time Ladder)
    if max_consecutive_strong >= 4:
        score = 40  # Full cascade STRONG
        reasons.append(f"🔥 Full cascade STRONG (4H→5M, ADX {max_adx:.1f})")
    elif max_consecutive_strong == 3:
        score = 35  # 3 TFs STRONG
        reasons.append(f"⚡ Strong cascade (3 TFs, ADX {max_adx:.1f})")
    elif max_consecutive_strong == 2:
        score = 30  # 2 TFs STRONG
        reasons.append(f"✅ Force confirmed (2 TFs strong, ADX {max_adx:.1f})")
    elif max_consecutive_any >= 3:
        # 3 TFs con fuerza (incluyendo emergente 1v)
        score = 25
        reasons.append(f"⚡ Emerging force (3 TFs, ADX {max_adx:.1f})")
    elif max_consecutive_any == 2:
        # 2 TFs con fuerza
        score = 20
        reasons.append(f"✅ Force propagating (2 TFs, ADX {max_adx:.1f})")
    elif max_consecutive_any == 1 and max_adx > 35:
        # 1 TF pero ADX muy alto
        score = 15
        reasons.append(f"💪 Single TF strong force (ADX {max_adx:.1f})")
    else:
        score = 0
        reasons.append("❌ No force propagation")
    
    # Bonus ADX extremo
    if max_adx > 40:
        score += 10
        reasons.append(f"💥 BRUTAL FORCE (ADX {max_adx:.1f})")
    elif max_adx > 30:
        score += 5
        reasons.append(f"💪 Strong ADX ({max_adx:.1f})")
    
    # Válido si: 2+ TFs con fuerza O 1 TF con ADX > 35
    valid = max_consecutive_any >= 2 or (max_adx > 35 and max_consecutive_any >= 1)
    
    return {
        'valid': valid,
        'score': score,
        'reason': ' | '.join(reasons),
        'data': {
            'max_consecutive_strong': max_consecutive_strong,
            'max_consecutive_any': max_consecutive_any,
            'strongest_tf': strongest_tf,
            'max_adx': max_adx
        }
    }

def validate_timing_layer(symbol, ladder, current_price):
    """
    Valida timing de entrada:
    - Si precio MUY cerca resistencia (< 0.5%): -30 (WAIT pullback)
    - Detecta REBOTE ACTIVO (velas contrarias)
    """
    score = 0
    reasons = []
    
    a5m = ladder.get('5m', {}) or {}
    sr_data = a5m.get('sr', {})
    vp_data = a5m.get('volume_profile', {})
    
    resistances = sr_data.get('resistances', [])
    poc = vp_data.get('poc_current', current_price)
    
    # --- NUEVO: DETECTAR REBOTE ACTIVO (SCORING) ---
    try:
        from exchange_connection import fetch_ohlcv
        import pandas as pd
        
        ohlcv = fetch_ohlcv(symbol, '5m', 5)
        
        if ohlcv and len(ohlcv) >= 3:
            df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
            closes = df['close'].values
            opens = df['open'].values
            
            last_3_green = sum(1 for i in range(-3, 0) if closes[i] > opens[i])
            
            # RESISTENCIA
            if resistances and current_price > 0:
                nearest_res = resistances[0]
                dist_pct = ((nearest_res - current_price) / current_price) * 100
                
                if 0 < dist_pct < 2.0 and last_3_green >= 2:
                    # Rebote activo contra resistencia: Penalización
                    score -= 30
                    reasons.append(f"⚠️ Rebote activo vs Res ({last_3_green}v verdes)")
                elif 0 < dist_pct < 2.0 and last_3_green == 1:
                     score -= 15
                     reasons.append(f"⚠️ Inicio rebote vs Res (1v verde)")

    except Exception as e:
        pass

    # Distancia a resistencia más cercana
    valid = True
    if resistances:
        nearest_res = resistances[0]
        if current_price > 0:
            dist_pct = ((nearest_res - current_price) / current_price) * 100
            
            if 0 < dist_pct < 0.5:
                # Detectar fuerza extrema para relajar bloqueo
                a1h = ladder.get('1h', {}) or {}
                adx_1h = a1h.get('adx', {}).get('adx_current', 0)
                
                if adx_1h > 40:
                    score -= 10
                    reasons.append(f"⚠️ BREAKOUT SETUP (Res {dist_pct:.2f}% away, FORCE overrides)")
                else:
                    score -= 30
                    reasons.append(f"🚫 BUYING TOP (Res {dist_pct:.2f}% away)")
                    valid = False
            elif 0.5 <= dist_pct < 2.0:
                score -= 15
                reasons.append(f"⚠️ Near resistance ({dist_pct:.2f}%)")
            elif dist_pct > 3.0:
                score += 15
                reasons.append(f"✅ Clean space ({dist_pct:.2f}%)")
        else:
            reasons.append("⚠️ Invalid price")
    else:
        score += 10
        reasons.append("✅ No immediate resistance")
    
    # Posición vs POC
    if poc > 0 and current_price > poc * 0.98 and current_price < poc * 1.02:
        score += 10
        reasons.append("✅ At POC (value zone)")
    
    return {
        'valid': valid,
        'score': max(0, score),
        'reason': ' | '.join(reasons),
        'data': {'dist_to_res': 0} # Placeholder
    }

        'total_score': total_score,

    }

def calculate_zigzag(df, percent=None, threshold_pct=None):
    """
    Calcula los puntos ZigZag y la ESTRUCTURA DE MERCADO (HH, HL, LH, LL).
    
    Args:
        percent/threshold_pct: Porcentaje mínimo de reversión.
        
    Returns:
        dict: {
            'pivots': list of indices,
            'values': list of prices,
            'pivot_types': list of strings (HH, HL, LH, LL, N/A),
            'last_structure': str (Resumen, e.g., "HH -> HL")
        }
    """
    # Aceptar ambos nombres de parámetro
    pct = threshold_pct if threshold_pct is not None else (percent if percent is not None else 5)
    
    if len(df) < 2:
        return {'pivots': [], 'values': [], 'pivot_types': [], 'last_structure': "N/A"}
        
    close = df['close'].values
    pivots = [0]
    values = [close[0]]
    last_pivot = close[0]
    direction = 0  # 0: indefinido, 1: arriba, -1: abajo
    
    # 1. Identificar Pivotes
    for i in range(1, len(close)):
        change = (close[i] - last_pivot) / last_pivot * 100
        if direction == 0:
            if abs(change) >= pct:
                direction = 1 if change > 0 else -1
                pivots.append(i)
                values.append(close[i])
                last_pivot = close[i]
        else:
            if (direction == 1 and close[i] < last_pivot * (1 - pct/100)) or \
               (direction == -1 and close[i] > last_pivot * (1 + pct/100)):
                direction *= -1
                pivots.append(i)
                values.append(close[i])
                last_pivot = close[i]

    # 2. Clasificar Estructura (HH, HL, LH, LL)
    pivot_types = ["Start"] * len(values)
    if len(values) >= 2:
        last_high = None
        last_low = None
        
        # Determinar si el primer movimiento fue High o Low
        # Si values[1] > values[0], el 1 es un High
        
        for i in range(1, len(values)):
            curr_val = values[i]
            prev_val = values[i-1]
            
            is_high = False
            is_low = False
            
            # En ZigZag, los puntos alternan.
            # Si values[i] > values[i-1], entonces 'i' es un pico (High)
            # Si values[i] < values[i-1], entonces 'i' es un valle (Low)
            
            if curr_val > prev_val:
                is_high = True
            else:
                is_low = True
                
            type_str = ""
            if is_high:
                if last_high is None: type_str = "H"
                elif curr_val > last_high: type_str = "HH" # Higher High
                elif curr_val < last_high: type_str = "LH" # Lower High
                else: type_str = "DH" # Double High
                last_high = curr_val
            elif is_low:
                if last_low is None: type_str = "L"
                elif curr_val > last_low: type_str = "HL" # Higher Low
                elif curr_val < last_low: type_str = "LL" # Lower Low
                else: type_str = "DL" # Double Low
                last_low = curr_val
            
            pivot_types[i] = type_str

    # 3. Resumen de última estructura
    last_structure = "Indefinido"
    if len(pivot_types) >= 2:
        # Tomar los últimos 2 tipos significativos
        last_structure = f"{pivot_types[-2]} -> {pivot_types[-1]}"

    return {
        'pivots': pivots, 
        'values': values, 
        'pivot_types': pivot_types, 
        'last_structure': last_structure
    }

def calculate_linear_regression_slope(values):
    """
    Calcula pendiente de regresión lineal simple
    
    Args:
        values: Lista de valores (últimas N velas)
        
    Returns:
        float: Pendiente (slope)
    """
    n = len(values)
    if n < 2:
        return 0.0
    
    x = np.arange(n)
    y = np.array(values)
    
    # y = mx + b → m = (n*Σxy - Σx*Σy) / (n*Σx² - (Σx)²)
    x_mean = np.mean(x)
    y_mean = np.mean(y)
    
    numerator = np.sum((x - x_mean) * (y - y_mean))
    denominator = np.sum((x - x_mean) ** 2)
    
    if denominator == 0:
        return 0.0
    
    slope = numerator / denominator
    return slope


def classify_trend(slope, threshold=0.3):
    """
    Clasifica tendencia según pendiente
    
    Args:
        slope: Pendiente calculada
        threshold: Umbral para considerar creciendo/decreciendo
        
    Returns:
        str: 'creciendo', 'decreciendo', 'estable'
    """
    if slope > threshold:
        return 'creciendo'
    elif slope < -threshold:
        return 'decreciendo'
    else:
        return 'estable'


def calculate_velocity(df, lookback=10):
    """
    Calcula velocidad de precio con memoria temporal
    
    Args:
        df: DataFrame con columna 'close'
        lookback: Velas a analizar
        
    Returns:
        dict: {
            'velocity_current': float (cambio precio por vela),
            'velocity_history': [list últimos lookback],
            'trend': 'acelerando'/'desacelerando'/'constante'
        }
    """
    if len(df) < lookback + 1:
        return {
            'velocity_current': 0.0,
            'velocity_history': [],
            'trend': 'constante'
        }
    
    prices = df['close'].values
    velocities = []
    
    # Calcular velocidad para cada par de velas
    for i in range(len(prices) - lookback, len(prices)):
        if i > 0:
            vel = prices[i] - prices[i-1]
            velocities.append(vel)
    
    velocity_current = velocities[-1] if velocities else 0.0
    
    # Calcular tendencia (velocidad creciendo/decreciendo)
    if len(velocities) >= 3:
        # Pendiente de las velocidades
        x = list(range(len(velocities)))
        y = velocities
        n = len(x)
        
        sum_x = sum(x)
        sum_y = sum(y)
        sum_xy = sum(x[i] * y[i] for i in range(n))
        sum_x2 = sum(x[i] ** 2 for i in range(n))
        
        slope = (n * sum_xy - sum_x * sum_y) / (n * sum_x2 - sum_x ** 2) if (n * sum_x2 - sum_x ** 2) != 0 else 0
        
        if slope > 0.0001:
            trend = 'acelerando'
        elif slope < -0.0001:
            trend = 'desacelerando'
        else:
            trend = 'constante'
    else:
        trend = 'constante'
    
    return {
        'velocity_current': velocity_current,
        'velocity_history': velocities,
        'trend': trend
    }

def calculate_adx_with_memory(df, period=14, lookback=10):
    """
    ADX con Wilder's Smoothing EXACTO del Bot 3.5.
    Retorna dict con valores actuales + memoria temporal.
    """
    df = df.copy()
    n = len(df)
    
    if n < period + 1:
        return {
            'adx_current': 0.0,
            'adx_history': [],
            'adx_trend': 'insuficiente',
            'di_plus_current': 0.0,
            'di_plus_history': [],
            'di_plus_trend': 'insuficiente',
            'di_minus_current': 0.0,
            'di_minus_history': [],
            'di_minus_trend': 'insuficiente',
            'dominance': 'neutral',
            'spread': 0.0
        }
    
    high = df['high'].values
    low = df['low'].values
    close = df['close'].values
    
    # 1. Calcular True Range y Directional Movements
    tr = np.zeros(n)
    dm_plus = np.zeros(n)
    dm_minus = np.zeros(n)
    
    for i in range(1, n):
        prev_close = close[i-1]
        prev_high = high[i-1]
        prev_low = low[i-1]
        
        # True Range (máximo de 3 valores)
        tr1 = high[i] - low[i]
        tr2 = abs(high[i] - prev_close)
        tr3 = abs(low[i] - prev_close)
        tr[i] = max(tr1, tr2, tr3)
        
        # Directional Movement
        move_up = high[i] - prev_high
        move_down = prev_low - low[i]
        
        if move_up > move_down and move_up > 0:
            dm_plus[i] = move_up
        else:
            dm_plus[i] = 0
            
        if move_down > move_up and move_down > 0:
            dm_minus[i] = move_down
        else:
            dm_minus[i] = 0
    
    # 2. Wilder's Smoothing MANUAL
    smoothed_tr = np.zeros(n)
    smoothed_dm_plus = np.zeros(n)
    smoothed_dm_minus = np.zeros(n)
    
    for i in range(n):
        if i < period:
            if i == period - 1:
                # Primera suavización: suma de los primeros 'period' valores
                smoothed_tr[i] = tr[:i+1].sum()
                smoothed_dm_plus[i] = dm_plus[:i+1].sum()
                smoothed_dm_minus[i] = dm_minus[:i+1].sum()
            else:
                smoothed_tr[i] = np.nan
                smoothed_dm_plus[i] = np.nan
                smoothed_dm_minus[i] = np.nan
        else:
            # Wilder's smoothing: prev - (prev/period) + current
            smoothed_tr[i] = smoothed_tr[i-1] - (smoothed_tr[i-1] / period) + tr[i]
            smoothed_dm_plus[i] = smoothed_dm_plus[i-1] - (smoothed_dm_plus[i-1] / period) + dm_plus[i]
            smoothed_dm_minus[i] = smoothed_dm_minus[i-1] - (smoothed_dm_minus[i-1] / period) + dm_minus[i]
    
    # 3. Calcular DI+ y DI- (con protección contra división por cero)
    di_plus = np.zeros(n)
    di_minus = np.zeros(n)
    for i in range(n):
        if smoothed_tr[i] != 0 and not np.isnan(smoothed_tr[i]):
            di_plus[i] = (smoothed_dm_plus[i] / smoothed_tr[i]) * 100
            di_minus[i] = (smoothed_dm_minus[i] / smoothed_tr[i]) * 100
        else:
            di_plus[i] = np.nan
            di_minus[i] = np.nan
    
    # 4. Calcular DX y ADX
    dx = np.zeros(n)
    for i in range(n):
        denominator = di_plus[i] + di_minus[i]
        if denominator != 0 and not np.isnan(denominator):
            dx[i] = np.abs(di_plus[i] - di_minus[i]) / denominator * 100
        else:
            dx[i] = np.nan
    
    # ADX es la media móvil simple de DX
    adx = np.zeros(n)
    for i in range(n):
        if i < period * 2 - 1:
            adx[i] = np.nan
        elif i == period * 2 - 1:
            # Calcular media ignorando NaN
            valid_dx = dx[period:i+1]
            valid_dx = valid_dx[~np.isnan(valid_dx)]
            if len(valid_dx) > 0:
                adx[i] = np.nanmean(valid_dx)
            else:
                adx[i] = np.nan
        else:
            # Continuar con Wilder's smoothing para ADX también
            if not np.isnan(adx[i-1]) and not np.isnan(dx[i]):
                adx[i] = (adx[i-1] * (period - 1) + dx[i]) / period
            else:
                adx[i] = np.nan
    
    # 5. Extraer historia y calcular tendencias
    start_idx = max(0, n - lookback)
    
    adx_history = [float(x) for x in adx[start_idx:n] if not np.isnan(x)]
    di_plus_history = [float(x) for x in di_plus[start_idx:n] if not np.isnan(x)]
    di_minus_history = [float(x) for x in di_minus[start_idx:n] if not np.isnan(x)]
    
    adx_current = float(adx[-1]) if not np.isnan(adx[-1]) else 0.0
    di_plus_current = float(di_plus[-1]) if not np.isnan(di_plus[-1]) else 0.0
    di_minus_current = float(di_minus[-1]) if not np.isnan(di_minus[-1]) else 0.0
    
    # 6. Determinar tendencias
    def get_trend(history):
        if len(history) < 3:
            return 'insuficiente'
        if history[-1] > history[-2] > history[-3]:
            return 'creciendo'
        elif history[-1] < history[-2] < history[-3]:
            return 'decreciendo'
        else:
            return 'estable'
    
    adx_trend = get_trend(adx_history)
    di_plus_trend = get_trend(di_plus_history)

def calculate_rsi_with_memory(df, period=14, lookback=10):
    """
    Calcula RSI con memoria temporal (Wilder's Smoothing exacto).
    Devuelve dict con valor actual, historia y tendencia.
    """
    close = df['close'].values
    n = len(close)
    if n < period + 1:
        return {
            'rsi_current': 0.0,
            'rsi_history': [],
            'rsi_trend': 'insuficiente'
        }
    deltas = np.diff(close)
    seed = deltas[:period]
    up = seed[seed >= 0].sum() / period
    down = -seed[seed < 0].sum() / period
    rs = up / down if down != 0 else 0
    rsi = np.zeros(n)
    rsi[:period] = np.nan
    rsi[period] = 100. - 100. / (1. + rs)
    up_avg = up
    down_avg = down
    for i in range(period + 1, n):
        delta = deltas[i - 1]
        upval = max(delta, 0)
        downval = -min(delta, 0)
        up_avg = (up_avg * (period - 1) + upval) / period
        down_avg = (down_avg * (period - 1) + downval) / period
        rs = up_avg / down_avg if down_avg != 0 else 0
        rsi[i] = 100. - 100. / (1. + rs)
    start_idx = max(0, n - lookback)
    rsi_history = [float(x) for x in rsi[start_idx:n] if not np.isnan(x)]
    rsi_current = float(rsi[-1]) if not np.isnan(rsi[-1]) else 0.0
    # Tendencia simple
    if len(rsi_history) >= 3:
        if rsi_history[-1] > rsi_history[-2] > rsi_history[-3]:
            trend = 'creciendo'
        elif rsi_history[-1] < rsi_history[-2] < rsi_history[-3]:
            trend = 'decreciendo'
        else:
            trend = 'estable'
    else:
        trend = 'insuficiente'
    return {
        'rsi_current': rsi_current,
        'rsi_history': rsi_history,
        'rsi_trend': trend
    }

def detect_support_resistance_with_memory(df, lookback=50, num_clusters=5):
    """
    Detección de soportes y resistencias por clustering de precios.
    Retorna dict con listas de resistencias y soportes ordenadas por cercanía.
    """
    closes = df['close'].values[-lookback:]
    if len(closes) < 10:
        return {'resistances': [], 'supports': []}

    # Clustering simple por bins de precio
    min_price = min(closes)
    max_price = max(closes)
    bin_size = (max_price - min_price) / num_clusters if num_clusters > 0 else 1
    bins = [min_price + i * bin_size for i in range(num_clusters + 1)]
    counts = [0] * num_clusters
    levels = [[] for _ in range(num_clusters)]

    for price in closes:
        for i in range(num_clusters):
            if bins[i] <= price < bins[i+1]:
                counts[i] += 1
                levels[i].append(price)
                break

    # Soportes: bins con más toques en la parte baja
    # Resistencias: bins con más toques en la parte alta
    supports = []
    resistances = []
    for i in range(num_clusters):
        if counts[i] >= 2:  # Al menos 2 toques para considerar nivel
            avg_level = sum(levels[i]) / len(levels[i])
            if avg_level < (min_price + max_price) / 2:
                supports.append(avg_level)
            else:
                resistances.append(avg_level)

    # Ordenar por cercanía al precio actual
    current_price = closes[-1]
    supports = sorted(supports, key=lambda x: abs(current_price - x))
    resistances = sorted(resistances, key=lambda x: abs(current_price - x))

    return {
        'resistances': resistances,
        'supports': supports
    }


def calculate_time_to_target(current_price, target_price, velocity, 
                             acceleration=0.0, candle_duration_min=5):
    """
    Proyecta tiempo estimado para alcanzar target
    
    Args:
        current_price: Precio actual
        target_price: Precio objetivo
        velocity: Velocidad actual (precio/vela)
        acceleration: Aceleración (cambio velocidad)
        candle_duration_min: Duración vela (default 5min para 5M)
    
    Returns:
        int: Minutos estimados para alcanzar target (0-120 max)
    """
    try:
        if velocity == 0 or velocity is None:
            return None
        
        distance = abs(target_price - current_price)
        
        # Si aceleración positiva, tiempo menor
        if acceleration > 0:
            effective_velocity = velocity * (1 + acceleration / 100)
        else:
            effective_velocity = velocity
        
        if effective_velocity <= 0:
            return None
        
        # Velas estimadas
        estimated_candles = distance / abs(effective_velocity) if effective_velocity != 0 else float('inf')
        
        # Convertir a minutos
        time_minutes = estimated_candles * candle_duration_min
        
        # Limitar a máximo 120 minutos (2 horas)
        return min(int(time_minutes), 120) if time_minutes < float('inf') else None
    
    except Exception as e:
        print(f"❌ Error calculate_time_to_target: {e}")
        return None


def calculate_volume_profile_with_memory(df, num_levels=10, lookback=50):
    """
    Calcula Volume Profile (Perfil de Volumen)
    
    Args:
        df: DataFrame con columnas ['close', 'volume']
        num_levels: Cantidad de niveles de precio
        lookback: Velas a analizar
    
    Returns:
        dict: {
            'poc_current': float (price of control - máximo volumen),
            'poc_history': [últimos values],
            'high_volume_nodes': [precios con alto volumen],
            'low_volume_nodes': [precios con bajo volumen]
        }
    """
    try:
        if len(df) < lookback:
            lookback = len(df)
        
        # Últimas N velas
        subset = df.tail(lookback).copy()
        
        # Crear bins de precio
        min_price = subset['close'].min()
        max_price = subset['close'].max()
        
        if min_price == max_price:
            return {
                'poc_current': float(min_price),
                'poc_history': [float(min_price)],
                'high_volume_nodes': [float(min_price)],
                'low_volume_nodes': []
            }
        
        # Crear niveles
        bins = np.linspace(min_price, max_price, num_levels)
        
        # Contar volumen en cada nivel
        volume_per_level = {}
        for i, price_level in enumerate(bins):
            tolerance = (max_price - min_price) / num_levels / 2
            # Velas que cerraron cerca de este nivel
            mask = (subset['close'] >= price_level - tolerance) & \
                   (subset['close'] <= price_level + tolerance)
            vol = subset.loc[mask, 'volume'].sum()
            volume_per_level[price_level] = vol
        
        # Point of Control (mayor volumen)
        poc = max(volume_per_level, key=volume_per_level.get)
        max_volume = volume_per_level[poc]
        min_volume = min(volume_per_level.values())
        
        # Nodos de alto/bajo volumen
        high_volume_threshold = min_volume + (max_volume - min_volume) * 0.7
        low_volume_threshold = min_volume + (max_volume - min_volume) * 0.3
        
        high_volume_nodes = [price for price, vol in volume_per_level.items() if vol > high_volume_threshold]
        low_volume_nodes = [price for price, vol in volume_per_level.items() if vol < low_volume_threshold]
        
        return {
            'poc_current': float(poc),
            'poc_history': [float(poc)],
            'high_volume_nodes': [float(p) for p in high_volume_nodes],
            'low_volume_nodes': [float(p) for p in low_volume_nodes],
            'profile_high': float(max(high_volume_nodes)) if high_volume_nodes else float(max_price),
            'profile_low': float(min(high_volume_nodes)) if high_volume_nodes else float(min_price)
        }
    
    except Exception as e:
        print(f"❌ Error calculate_volume_profile_with_memory: {e}")
        return {
            'poc_current': 0.0,
            'poc_history': [],
            'high_volume_nodes': [],
            'low_volume_nodes': [],
            'profile_high': 0.0,
            'profile_low': 0.0
        }

def calculate_bollinger_bands(df, period=20, std_dev=2):

def calculate_atr(df, period=14):
    """
    Calcula el Average True Range (ATR).
    
    Args:
        df: DataFrame con columnas 'high', 'low', 'close'
        period: Período para el ATR (default 14)
        
    Returns:
        float: Valor actual del ATR
    """
    try:
        high = df['high']
        low = df['low']
        close = df['close']
        
        # Calculate True Range
        tr1 = high - low
        tr2 = abs(high - close.shift(1))
        tr3 = abs(low - close.shift(1))
        
        tr = pd.concat([tr1, tr2, tr3], axis=1).max(axis=1)
        
        # Calculate ATR using rolling mean (Simple Moving Average of TR)
        # Note: Wilder's smoothing is more standard for ATR, but SMA is often used too.
        # Using Wilder's Smoothing to match standard TradingView ATR
        atr = tr.ewm(alpha=1/period, adjust=False).mean()
        
        return {
            'atr_current': float(atr.iloc[-1]) if not pd.isna(atr.iloc[-1]) else 0.0,
            'atr_history': atr.tail(10).tolist()
        }
        
    except Exception as e:
        print(f"❌ Error calculate_atr: {e}")
        return {
            'atr_current': 0.0,
            'atr_history': []
        }
