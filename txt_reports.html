def _validate_explosion_entry(self, symbol, direction, force_result, ladder_data):
    """
    v5.2.1 FINAL: Retorna SOLO score, no can_enter.
    El sniper decide con score combinado.
    """
    from brain_config import (
        MIN_DIST_BLOCK, MIN_DIST_PENALTY,
        SPACE_BASE_PENALTY, VELOCITY_PENALTY_REDUCTION_MAX
    )
    
    data_15m = ladder_data.get('15m', {})
    current_price = data_15m.get('current_price', 0)
    
    if current_price == 0:
        return {'score': 0, 'reason': 'No price data'}
    
    score = 100
    penalties = []
    
    # Velocity factor
    velocity_pct = abs(force_result.get('velocity', 0))
    velocity_factor = min(velocity_pct / 0.30, VELOCITY_PENALTY_REDUCTION_MAX)
    
    # === ESPACIO A RESISTENCIAS ===
    vp_data = data_15m.get('volume_profile', {})
    vah = vp_data.get('value_area_high', 0)
    
    if direction == 'BULLISH' and vah > 0 and current_price < vah:
        dist_pct = ((vah - current_price) / current_price) * 100
        
        if dist_pct < MIN_DIST_BLOCK:  # < 0.15% = Muy grave
            penalty = 60  # Penalty máxima pero no block
            penalties.append(f'EN resistencia: -{penalty}')
        elif dist_pct < MIN_DIST_PENALTY:  # < 0.50%
            base_penalty = SPACE_BASE_PENALTY * (1 - velocity_factor)
            penalties.append(f'Cerca VAH: -{base_penalty:.0f}')
            penalty = base_penalty
        else:
            penalty = 0
        
        score -= penalty
    
    # === DIAGONAL (si existe) ===
    grid_data = data_15m.get('grid', {})
    diagonal = grid_data.get('diagonal', {})
    
    if diagonal.get('valid') and direction == 'BULLISH':
        res_diag = diagonal.get('resistencia_diagonal', 0)
        if res_diag > current_price:
            dist_diag = ((res_diag - current_price) / current_price) * 100
            if dist_diag < MIN_DIST_PENALTY:
                diag_penalty = 25 * (1 - velocity_factor)
                score -= diag_penalty
                penalties.append(f'Diagonal cerca: -{diag_penalty:.0f}')
    
    score = max(0, min(100, score))
    
    print(f"   📊 Validator: {score:.0f} | Penalties: {penalties}")
    
    # SOLO retorna score, no can_enter
    return {
        'score': score,
        'reason': f'Penalties: {penalties}' if penalties else 'Clean',
        'velocity_factor': velocity_factor
    }

@staticmethod
def is_perfect_setup(symbol: str, analyzer_score: float, decision: dict) -> bool:
    """
    v5.2.1 FINAL: UN SOLO punto de decisión, TODO es scoring.
    """
    from brain_config import (
        ENABLE_MOTOR_VELOCITY, ENABLE_MOTOR_CHANNEL,
        SCORE_WEIGHT_ANALYZER, SCORE_WEIGHT_VALIDATOR,
        SCORE_THRESHOLD_ENTER, MOMENTUM_CONTRA_PENALTY,
        GRID_HIGH_PENALTY_PER_PCT
    )
    from brain_frames import check_price_velocity
    from brain_entry_validator import entry_validator
    
    full_analysis = decision.get('full_analysis', {})
    ladder_data = full_analysis.get('ladder_data', {})
    data_15m = ladder_data.get('15m', {})
    direction = decision.get('direction', 'LATERAL')
    
    if direction == 'LATERAL':
        return False
    
    # === AJUSTAR ANALYZER SCORE CON PENALTIES ADICIONALES ===
    adjusted_analyzer = analyzer_score
    
    # PENALTY: Momentum contradictorio (NO hard block)
    mom_color = data_15m.get('momentum_color', 'gray')
    if direction == 'BULLISH' and mom_color in ['red', 'maroon']:
        adjusted_analyzer -= MOMENTUM_CONTRA_PENALTY
        print(f"   📉 Momentum {mom_color}: -{MOMENTUM_CONTRA_PENALTY}")
    elif direction == 'BEARISH' and mom_color in ['lime', 'green']:
        adjusted_analyzer -= MOMENTUM_CONTRA_PENALTY
        print(f"   📉 Momentum {mom_color}: -{MOMENTUM_CONTRA_PENALTY}")
    
    # PENALTY: Grid position gradual (NO cliff)
    grid_data = data_15m.get('grid', {})
    diagonal = grid_data.get('diagonal', {})
    
    if diagonal.get('valid'):
        grid_position = diagonal.get('price_position_pct', 50)
    else:
        grid_position = data_15m.get('position_in_range', 50)
    
    if direction == 'BULLISH' and grid_position > 60:
        grid_penalty = (grid_position - 60) * GRID_HIGH_PENALTY_PER_PCT
        adjusted_analyzer -= grid_penalty
        print(f"   📉 Grid {grid_position:.0f}%: -{grid_penalty:.0f}")
    elif direction == 'BEARISH' and grid_position < 40:
        grid_penalty = (40 - grid_position) * GRID_HIGH_PENALTY_PER_PCT
        adjusted_analyzer -= grid_penalty
        print(f"   📉 Grid {grid_position:.0f}%: -{grid_penalty:.0f}")
    
    adjusted_analyzer = max(0, adjusted_analyzer)
    
    # === OBTENER VELOCITY ===
    velocity_check = check_price_velocity(symbol, direction, '5m')
    if 'insuficientes' in velocity_check.get('reason', '').lower():
        velocity_check = check_price_velocity(symbol, direction, '15m')
    
    velocity_value = velocity_check.get('velocity', 0)
    print(f"   🏎️ Velocity: {velocity_value:.3f}%")
    
    # === VALIDATOR SCORE ===
    force_result = full_analysis.get('force_result', {}).copy()
    force_result['velocity'] = velocity_value
    
    validation = entry_validator._validate_explosion_entry(
        symbol, direction, force_result, ladder_data
    )
    
    validator_score = validation.get('score', 50)
    
    # === SCORE COMBINADO - UN SOLO THRESHOLD ===
    combined = (adjusted_analyzer * SCORE_WEIGHT_ANALYZER) + \
               (validator_score * SCORE_WEIGHT_VALIDATOR)
    
    print(f"   📊 Analyzer ajustado: {adjusted_analyzer:.0f}")
    print(f"   📊 Validator: {validator_score:.0f}")
    print(f"   🎯 COMBINED: {combined:.0f} (threshold: {SCORE_THRESHOLD_ENTER})")
    
    # === UN SOLO PUNTO DE DECISIÓN ===
    if combined >= SCORE_THRESHOLD_ENTER:
        print(f"   ✅ APROBADO: {combined:.0f} >= {SCORE_THRESHOLD_ENTER}")
        return True
    else:
        print(f"   ❌ Score insuficiente: {combined:.0f} < {SCORE_THRESHOLD_ENTER}")
        return False
