"""
BRAIN GRID v6.1 - Sistema Navegación Espacial
Detecta el "espacio físico" disponible para el trade (Malla Donchian Normalizada).
"""
import pandas as pd
import numpy as np
from brain_config import GRID_LOOKBACK, GRID_COMPRESSION_TH
from brain_calculator import calculate_atr
class MarketGridNavigator:
    def navigate(self, symbol, df_5m, ladder):
        """
        Navegación espacial usando DF de 5m y datos del Ladder.
        Retorna decisión basada en espacio disponible y dirección del puntero.
        """
        if df_5m is None or len(df_5m) < 30:
            return {
                'decision': 'BLOCK', 
                'reason': 'Sin datos suficientes para Grid', 
                'score': -100,
                'grid_data': {}
            }
        
        # 1. CONSTRUIR MALLA (Donchian + ATR normalizado)
        grid = self._build_grid(df_5m)
        
        # 2. PUNTERO (Velocity + Momentum 15m)
        pointer = self._calc_pointer(df_5m, ladder)
        
        # 3. FUERZA (ADX real del ladder)
        force = self._get_force(ladder)
        
        # 4. DECISIÓN ESPACIAL (Scoring)
        return self._decide(grid, pointer, force, df_5m)
    
    def _build_grid(self, df):
        """Construye la malla espacial normalizada por volatilidad (ATR)"""
        lookback = GRID_LOOKBACK
        
        # Donchian Channel (Techo y Piso)
        ceiling = df['high'].rolling(lookback).max().iloc[-1]
        floor = df['low'].rolling(lookback).min().iloc[-1]
        mid = (ceiling + floor) / 2
        
        # ATR para normalización espacial
        atr_data = calculate_atr(df, period=14)
        atr = atr_data.get('atr_current', 0)
        
        # Ancho normalizado (Cuántos ATRs caben en el canal)
        width_atr = (ceiling - floor) / atr if atr > 0 else 0
        
        return {
            'ceiling': ceiling,
            'floor': floor,
            'mid': mid,
            'width_atr': width_atr,  # Métrica clave de compresión
            'atr': atr
        }
    
    def _calc_pointer(self, df, ladder):
        """Calcula la dirección del puntero (Intención inmediata)"""
        # Velocity (precio)
        closes = df['close'].values
        velocity = closes[-1] - closes[-3] if len(closes) > 3 else 0
        
        # Momentum LazyBear (15m - más confiable y estable)
        mom_15m = ladder.get('15m', {}).get('momentum', {})
        mom_color = mom_15m.get('color_current', 'neutral')
        
        direction = "NEUTRAL"
        if mom_color in ['green', 'lime']:
            direction = "UP"
        elif mom_color in ['red', 'maroon']:
            direction = "DOWN"
        
        return {
            'direction': direction,
            'velocity': velocity,
            'mom_color': mom_color
        }
    
    def _get_force(self, ladder):
        """Extrae fuerza real del ladder (15m)"""
        adx_15m = ladder.get('15m', {}).get('adx', {})
        return {
            'adx': adx_15m.get('adx_current', 0),
            'trend': adx_15m.get('adx_trend', 'estable')
        }
    
    def _decide(self, grid, pointer, force, df):
        """Decisión basada en scoring espacial - SIN BINARIOS DUROS (Time Ladder)"""
        score = 0
        reasons = []
        decision = "WAIT"
        
        current_price = df['close'].iloc[-1]
        
        # 1. LATERALIZACIÓN (Compresión Espacial)
        # Si el ancho es menor al umbral, no hay espacio físico para operar con R:R decente.
        if grid['width_atr'] < GRID_COMPRESSION_TH:
            score -= 80  # Penalización masiva (Soft Block)
            reasons.append(f"⛔ CAJA LATERAL: Compresión Extrema ({grid['width_atr']:.1f} ATRs)")
        
        # 2. ALINEACIÓN PUNTERO (Dirección inmediata)
        if pointer['direction'] == 'DOWN':
            # Validar si estamos en la parte baja rompiendo
            if current_price <= grid['floor'] * 1.002: # Rompiendo piso
                score += 20
                reasons.append("🔨 Grid: Rompiendo Piso")
            else:
                 # Simplemente bajista
                 score += 10
                 
        elif pointer['direction'] == 'UP':
            # Validar si estamos en la parte alta rompiendo
            if current_price >= grid['ceiling'] * 0.998: # Rompiendo techo
                score += 20
                reasons.append("🔨 Grid: Rompiendo Techo")
            else:
                 # Simplemente alcista
                 score += 10
        
        # 3. CONTRADICCIÓN ESPACIAL (Rebotes falsos)
        # Si el precio sube pero el puntero es BAJISTA
        if pointer['velocity'] > 0 and pointer['direction'] == 'DOWN':
             score -= 30
             reasons.append("⚠️ Grid: Rebote contra tendencia (Puntero DOWN)")
             
        # Si el precio baja pero el puntero es ALCISTA
        elif pointer['velocity'] < 0 and pointer['direction'] == 'UP':
             score -= 30
             reasons.append("⚠️ Grid: Caída contra tendencia (Puntero UP)")
        # 4. FUERZA ADX (Confirmación)
        if force['adx'] > 30:
            score += 10
            reasons.append(f"💪 Grid: ADX Soporta ({force['adx']:.1f})")
        elif force['adx'] < 20:
             # Mercado débil reduce confianza espacial
             score -= 10
        
        # 5. RESULTADO FINAL
        # Mapear score a decisión
        if score <= -50:
            decision = "BLOCK"
        elif score >= 30:
            decision = "GO"
        else:
            decision = "WAIT"
        
        return {
            'decision': decision,
            'score': score,
            'reasons': reasons,
            'grid_data': grid,
            'confidence': max(0, min(100, 50 + score)) # Base 50 +/- score
        }
