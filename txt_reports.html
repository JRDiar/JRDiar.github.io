// Support/Resistance Function
f_support_resistance() =>
    high_pivot = ta.pivothigh(high, sr_left_bars, sr_right_bars)
    low_pivot = ta.pivotlow(low, sr_left_bars, sr_right_bars)
    
    var float resistance = na
    var float support = na
    
    if not na(high_pivot)
        resistance := high_pivot
    if not na(low_pivot)
        support := low_pivot
    
    short_vol = ta.ema(volume, 5)
    long_vol = ta.ema(volume, 10)
    vol_osc = 100 * (short_vol - long_vol) / long_vol
    
    res_break_vol = ta.crossover(close, resistance) and vol_osc > sr_volume_threshold
    sup_break_vol = ta.crossunder(close, support) and vol_osc > sr_volume_threshold
    bull_wick = ta.crossover(close, resistance) and (open - low) > (close - open)
    bear_wick = ta.crossunder(close, support) and (open - close) < (high - open)
    res_break_wick = bull_wick and not res_break_vol
    sup_break_wick = bear_wick and not sup_break_vol
    
    [resistance, support, res_break_vol, sup_break_vol, res_break_wick, sup_break_wick]

// BB/KC Strategy Function
f_bb_kc_strategy() =>
    // Bollinger Bands
    basis = ta.sma(close, bbkc_length)
    dev = bbkc_mult * ta.stdev(close, bbkc_length)
    upper_bb = basis + dev
    lower_bb = basis - dev
    
    // Keltner Channels
    ma = ta.ema(close, bbkc_length)
    range_val = bbkc_use_true_range ? ta.tr : high - low
    rangema = ta.ema(range_val, bbkc_length)
    upper_kc = ma + rangema * bbkc_mult
    lower_kc = ma - rangema * bbkc_mult
    
    // Señales
    bearish = low < lower_kc
    bullish = high > upper_kc
    
    // Squeeze (cuando BB está dentro de KC)
    squeeze_on = upper_bb < upper_kc and lower_bb > lower_kc
    
    // SAR
    sar_value = ta.sar(bbkc_sar_start, bbkc_sar_increment, bbkc_sar_maximum)
    
    [basis, upper_bb, lower_bb, ma, upper_kc, lower_kc, bearish, bullish, squeeze_on, sar_value]

// EMA Envelope Function
f_ema_envelope() =>
    ema_close = ta.ema(close, emae_length)
    ema_high = ta.ema(high, emae_length)
    ema_low = ta.ema(low, emae_length)
    
    // Condiciones de tendencia
    bull_normal = high > ema_high and low > ema_low
    bear_normal = high < ema_high and low < ema_low
    sidewise = not bull_normal and not bear_normal
    
    // Fuerza de tendencia
    bull_strong = bull_normal and low > ema_high
    bear_strong = bear_normal and high < ema_low
    
    // Color de fondo
    bg_color = bull_strong ? color.new(color.lime, 90) :
               bull_normal ? color.new(color.green, 90) :
               bear_strong ? color.new(color.red, 90) :
               bear_normal ? color.new(color.orange, 90) :
               sidewise ? color.new(color.blue, 95) : na
    
    [ema_close, ema_high, ema_low, bg_color, bull_normal, bear_normal, bull_strong, bear_strong]

// Supertrend Function
f_supertrend() =>
    src = hl2
    atr2 = ta.sma(ta.tr, st_atr_period)
    atr_val = st_change_atr ? ta.atr(st_atr_period) : atr2
    
    up = src - (st_multiplier * atr_val)
    up1 = nz(up[1], up)
    up := close[1] > up1 ? math.max(up, up1) : up
    
    dn = src + (st_multiplier * atr_val)
    dn1 = nz(dn[1], dn)
    dn := close[1] < dn1 ? math.min(dn, dn1) : dn
    
    var int trend = 1
    trend := nz(trend[1], trend)
    trend := trend == -1 and close > dn1 ? 1 : trend == 1 and close < up1 ? -1 : trend
    
    buy_signal = trend == 1 and trend[1] == -1
    sell_signal = trend == -1 and trend[1] == 1
    
    [up, dn, trend, buy_signal, sell_signal]

// Delta Volume Profile Function
f_delta_vp() =>
    var max_min = array.new<float>()
    var vol_plus = array.new<float>()
    var vol_minus = array.new<float>()
    var boxes = array.new<box>()
    var labels = array.new<label>()
    
    step = 0.
    bins = 0
    
    atr_val = ta.atr(200)
    
    if barstate.islast
        // Limpiar arrays
        max_min.clear()
        
        // Limpiar boxes anteriores
        if boxes.size() > 0
            for b in boxes
                box.delete(b)
            boxes.clear()
        
        // Limpiar labels anteriores
        if labels.size() > 0
            for l in labels
                label.delete(l)
            labels.clear()
        
        // Recolectar máximos y mínimos
        for i = 0 to dvp_lookback - 1
            h = high[i]
            l = low[i]
            max_min.push(h)
            max_min.push(l)
        
        // Calcular bins y step
        bins := int((max_min.max() - max_min.min()) / atr_val)
        bins := math.max(bins, 20) // Mínimo 20 bins
        bins := math.min(bins, 150) // Máximo 150 bins
        step := (max_min.max() - max_min.min()) / bins
        
        // Inicializar arrays de volumen
        vol_plus.clear()
        vol_minus.clear()
        for i = 0 to bins - 1
            vol_plus.push(0.0)
            vol_minus.push(0.0)
        
        // Calcular volumen por nivel de precio
        for i = 0 to dvp_lookback - 1
            h = high[i]
            l = low[i]
            c = close[i]
            o = open[i]
            v = volume[i]
            
            for k = 0 to bins - 1
                loww = max_min.min() + step * k
                mid = loww + step / 2
                
                if h > mid and l < mid
                    if c > o
                        vol_plus.set(k, vol_plus.get(k) + v)
                    else
                        vol_minus.set(k, vol_minus.get(k) + v)
        
        // Dibujar perfil
        for i = 0 to bins - 1
            loww = max_min.min() + step * i
            highh = loww + step
            
            vp = vol_plus.get(i)
            vm = vol_minus.get(i)
            
            value_plus = vol_plus.sum() > 0 ? int(vp / vol_plus.sum() * 200) : 0
            value_minus = vol_minus.sum() > 0 ? int(vm / vol_minus.sum() * 200) : 0
            
            max_plus = vp == vol_plus.max()
            max_minus = vm == vol_minus.max()
            
            col_plus = color.from_gradient(vp, vol_plus.min(), vol_plus.max(), color.new(dvp_color_plus, 90), dvp_color_plus)
            col_minus = color.from_gradient(vm, vol_minus.min(), vol_minus.max(), color.new(dvp_color_minus, 90), dvp_color_minus)
            
            sum_bars = value_plus + value_minus
            col_sum = color.from_gradient(sum_bars, 0, 60, color.rgb(189, 189, 189, 100), #787b8699)
            
            // Box de fondo (perfil completo)
            b1 = box.new(bar_index - dvp_lookback, highh, bar_index, loww, max_plus ? dvp_poc_plus : max_minus ? dvp_poc_minus : color.new(chart.bg_color, 50), bgcolor=col_sum)
            
            // Box de volumen positivo (hacia la izquierda)
            b2 = box.new(bar_index + 50, highh, bar_index + 50 - value_plus, loww,chart.bg_color, bgcolor=max_plus ? dvp_poc_plus : col_plus, text=vp != 0 ? "+" + str.tostring(vp, format.volume) : "", text_color=chart.fg_color, text_size=size.small,text_halign=text.align_right)
            
            // Box de volumen negativo (hacia la derecha)
            b3 = box.new(bar_index + 50, highh, bar_index + 50 + value_minus, loww, chart.bg_color,bgcolor=max_minus ? dvp_poc_minus : col_minus, text=vm != 0 ? "-" + str.tostring(vm, format.volume) : "", text_color=chart.fg_color, text_size=size.small,text_halign=text.align_left)
            
            boxes.push(b1)
            boxes.push(b2)
            boxes.push(b3)
        
        // Calcular delta y total
        total_vol = vol_plus.sum() + vol_minus.sum()
        delta_calc = vol_minus.sum() > 0 ? ((vol_plus.sum() - vol_minus.sum()) / vol_minus.sum()) * 100 : 0
        
        // Label de Delta
        l1 = label.new(bar_index + 50, max_min.max(), 
                      "Delta: " + str.tostring(delta_calc, format.percent), 
                      color=delta_calc > 0 ? dvp_color_plus : dvp_color_minus, 
                      textcolor=chart.fg_color, 
                      style=label.style_label_down)
        
        // Label de Total
        l2 = label.new(bar_index + 50, max_min.min(), 
                      "Total: " + str.tostring(total_vol, format.volume), 
                      color=color.gray, 
                      textcolor=chart.fg_color, 
                      style=label.style_label_up)
        
        labels.push(l1)
        labels.push(l2)
