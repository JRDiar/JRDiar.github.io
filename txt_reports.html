def analyze_4h(symbol):
    """
    4H: Define contexto dominante y clima de mercado
    
    NUNCA RECHAZA - Solo informa dirección y fuerza
    
    Args:
        symbol (str): Par trading (ej: 'BTC/USDT')
    
    Returns:
        dict: Análisis completo con direction, strength, indicadores
        None: Solo si no hay datos (caso extremo)
    """
    # Fetch datos
    ohlcv = fetch_ohlcv(symbol, '4h', 100)
    if not ohlcv or len(ohlcv) < 50:
        return None  # Solo este caso rechaza (sin datos)
    
    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    current_price = float(df['close'].iloc[-1])
    
    # ===== ADX con memoria =====
    adx_data = calculate_adx_with_memory(df, period=14, lookback=10)
    
    # ADX ANTICIPATORIO: Proyectar evolución
    adx_now = adx_data['adx_current']
    adx_trend = adx_data['adx_trend']
    
    # Calcular velocidad ADX (puntos por vela)
    if len(adx_data['adx_history']) >= 5:
        adx_5_velas_atras = adx_data['adx_history'][-5]
        adx_velocity = (adx_now - adx_5_velas_atras) / 5
    else:
        adx_velocity = 0
    
    # Proyección 3 velas adelante (12 horas en 4H)
    adx_proyectado = adx_now + (adx_velocity * 3)
    
    # ===== Determinar dirección base por dominancia =====
    if adx_data['dominance'] == 'DI+':
        direction = 'BULLISH'
    elif adx_data['dominance'] == 'DI-':
        direction = 'BEARISH'
    else:
        direction = 'LATERAL'
        print(f"   ⚠️ 4H {symbol}: Dominancia neutral → LATERAL (DI+:{adx_data['di_plus_current']:.1f} DI-:{adx_data['di_minus_current']:.1f})")
    
    # ===== Evaluar fuerza ADX =====
    strength = 'WEAK'  # Inicializar en WEAK
    
    # CASO 1: ADX ya >25 (perfecto)
    if adx_now >= 25:
        if adx_now > 35:
            strength = 'STRONG'
        elif adx_now > 28:
            strength = 'MODERATE'
        else:
            strength = 'WEAK'
        print(f"   ✅ {symbol}: ADX {adx_now:.1f} (tendencia confirmada)")
    
    # CASO 2: ADX <25 pero subiendo rápido (ANTICIPAR)
    elif adx_now >= 17 and adx_velocity > 1.5:
        strength = 'MODERATE'  # Tendencia formándose
        print(f"   ⚡ ANTICIPANDO {symbol}: ADX {adx_now:.1f} subiendo {adx_velocity:+.1f}/vela → Proyección: {adx_proyectado:.1f}")
    
    # CASO 3: ADX bajo sin velocidad
    else:
        direction = 'LATERAL'  # Forzar LATERAL si ADX muy bajo
        strength = 'WEAK'
        print(f"   ⚠️ {symbol}: ADX bajo {adx_now:.1f} (vel:{adx_velocity:+.1f}/vela) → LATERAL WEAK")
    
    # Verificar caída pronunciada
    if adx_trend == 'decreciendo' and adx_velocity < -0.8:
        strength = 'WEAK'  # Degradar fuerza si cayendo rápido
        print(f"   ⚠️ {symbol}: ADX cayendo rápido ({adx_velocity:-.1f}/vela) → Fuerza degradada a WEAK")
    
    # ===== Verificar spread (ajusta fuerza, no rechaza) =====
    if direction != 'LATERAL' and abs(adx_data['spread']) < 5.0:
        if strength == 'STRONG':
            strength = 'MODERATE'
        elif strength == 'MODERATE':
            strength = 'WEAK'
        print(f"   ⚠️ {symbol}: Spread bajo ({abs(adx_data['spread']):.1f} < 5.0) → Fuerza ajustada a {strength}")
    
    # ===== Momentum =====
    momentum_data = calculate_momentum_with_memory(df, lookback=10)
    expected_color = 'green' if direction == 'BULLISH' else 'red'
    
    # Si momentum no coincide, ajustar dirección a LATERAL
    if direction != 'LATERAL' and momentum_data['color_current'] != expected_color:
        print(f"   ⚠️ {symbol}: Momentum {momentum_data['color_current']} ≠ {expected_color} → Ajustando a LATERAL")
        direction = 'LATERAL'
        strength = 'WEAK'
    
    # Si momentum débil, degradar fuerza
    if momentum_data['consecutive_count'] < 3:
        if strength == 'STRONG':
            strength = 'MODERATE'
        elif strength == 'MODERATE':
            strength = 'WEAK'
        print(f"   ⚠️ {symbol}: Momentum débil ({momentum_data['consecutive_count']} velas) → Fuerza ajustada a {strength}")
    
    # ===== RSI (solo para información, NO filtro) =====
    rsi_data = calculate_rsi_with_memory(df, period=14, lookback=10)
    
    # ===== EMAs =====
    emas_data = calculate_emas_with_memory(df, periods=[9, 21, 50, 200], lookback=10)
    
    # ===== Estructura HH/LL (INFORMATIVO) =====
    if ENABLE_4H_STRUCTURE:
        structure = detect_price_structure(df, lookback=20)
    else:
        structure = {'structure': 'UNKNOWN', 'strength': 'UNKNOWN'}
    # ===== Zig Zag (INFORMATIVO) =====
    if ENABLE_4H_ZIGZAG:
        zigzag = calculate_zigzag(df, threshold_pct=3.0)
    else:
        zigzag = {'direction': 'UNKNOWN'}
    
    # Output compacto: una línea
    print(f"📊 4H | {symbol} | {direction} {strength} | ADX:{adx_data['adx_current']:.1f} DI+:{adx_data['di_plus_current']:.1f} DI-:{adx_data['di_minus_current']:.1f} | Mom:{momentum_data['consecutive_count']}v | Estructura:{structure['structure']}")
    
    # ==== SCORING 0-40 ====
    score_4h = 0
    if strength == 'STRONG':
        score_4h += 20
    elif strength == 'MODERATE':
        score_4h += 12
    else:
        score_4h += 5
    if adx_trend == 'creciendo':
        score_4h += 10
    if momentum_data['consecutive_count'] >= 5:
        score_4h += 5
    if abs(adx_data['spread']) > 7:
        score_4h += 5
    
    return {
        'direction': direction,
        'strength': strength,
        'score_4h': score_4h,  # ← AGREGADO
        'adx': adx_data,
        'momentum': momentum_data,
        'rsi': rsi_data,
        'emas': emas_data,
        'structure': structure,
        'zigzag': zigzag,
        'current_price': current_price,
        'context': f"4H {direction} {strength} - ADX {adx_data['adx_current']:.1f}",
        'timestamp': datetime.now().isoformat()
    }

def analyze_1h(symbol):
    """1H: Confirma tendencia 4H"""
    ohlcv = fetch_ohlcv(symbol, '1h', 100)
    if not ohlcv or len(ohlcv) < 50:
        # Retornar análisis vacío con score 0
        return {
            'direction': 'UNKNOWN',
            'strength': 'WEAK',
            'score_1h': 0,
            'adx': {},
            'momentum': {},
            'rsi': {},
            'emas': {},
            'structure': {},
            'zigzag': {},
            'context': '1H - Datos insuficientes',
            'timestamp': datetime.now().isoformat()
        }
    
    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    
    # ===== ADX =====
    adx_data = calculate_adx_with_memory(df, period=14, lookback=10)
    adx_trend = adx_data['adx_trend']
    adx_current = adx_data['adx_current']
    spread = abs(adx_data['spread'])
    
    if adx_data['dominance'] == 'DI+':
        direction = 'BULLISH'
    elif adx_data['dominance'] == 'DI-':
        direction = 'BEARISH'
    else:
        direction = 'LATERAL'
    
    # ===== Momentum =====
    momentum_data = calculate_momentum_with_memory(df, lookback=10)
    expected_color = 'green' if direction == 'BULLISH' else 'red'
    momentum_color = momentum_data['color_current']
    
    # ===== RSI =====
    rsi_data = calculate_rsi_with_memory(df, period=14, lookback=10)
    
    # ===== EMAs =====
    emas_data = calculate_emas_with_memory(df, periods=[9, 21, 50, 200], lookback=10)
    
    # ===== Estructura (INFORMATIVO - NO rechazar) =====
    structure = detect_price_structure(df, lookback=20)
    
    # ===== Zig Zag (INFORMATIVO - NO rechazar) =====
    zigzag = calculate_zigzag(df, threshold_pct=2.5)
    
    # ===== Strength =====
    if adx_current > 32:
        strength = 'STRONG'
    elif adx_current > 25:
        strength = 'MODERATE'
    else:
        strength = 'WEAK'
    
    # ==== SCORING 0-30 ====
    score_1h = 0
    if adx_trend == 'creciendo':
        score_1h += 10
    if adx_current >= 22:
        score_1h += 5
    if momentum_color == expected_color:
        score_1h += 10
    if spread > 4.0:
        score_1h += 5
    
    print(f"   📈 1H | {symbol} | {direction} {strength} | ADX:{adx_current:.1f} DI+:{adx_data['di_plus_current']:.1f} DI-:{adx_data['di_minus_current']:.1f} | Mom:{momentum_data['consecutive_count']}v | Score1H:{score_1h}")
    
    return {
        'direction': direction,
        'strength': strength,
        'score_1h': score_1h,  # ← AGREGADO
        'adx': adx_data,
        'momentum': momentum_data,
        'rsi': rsi_data,
        'emas': emas_data,
        'structure': structure,
        'zigzag': zigzag,
        'context': f"1H {direction} {strength} - ADX {adx_current:.1f}",
        'timestamp': datetime.now().isoformat()
    }

        score_15m += 15

def analyze_5m(symbol):
    """5M: Data para ejecución - NO rechaza"""
    ohlcv = fetch_ohlcv(symbol, '5m', 100)
    if not ohlcv or len(ohlcv) < 50:
        return {
            'direction': 'UNKNOWN',
            'strength': 'WEAK',
            'context': '5M - Datos insuficientes',
            'timestamp': datetime.now().isoformat()
        }
    
    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    
    # ADX
    adx_data = calculate_adx_with_memory(df, period=14, lookback=10)
    
    # Dirección (puede ser cualquiera)
    if adx_data['dominance'] == 'DI+':
        direction = 'BULLISH'
    elif adx_data['dominance'] == 'DI-':
        direction = 'BEARISH'
    else:
        direction = 'LATERAL'
    
    # Momentum
    momentum_data = calculate_momentum_with_memory(df, lookback=10)
    
    # RSI
    rsi_data = calculate_rsi_with_memory(df, period=14, lookback=10)
    
    # EMAs
    emas_data = calculate_emas_with_memory(df, periods=[9, 21, 50, 200], lookback=10)
    
    # Estructura
    structure = detect_price_structure(df, lookback=10)
    
    # Zig Zag
    zigzag = calculate_zigzag(df, threshold_pct=1.5)
    
    # Support/Resistance
    sr_data = detect_support_resistance_with_memory(df, lookback=30)
    
    # Velocity & Acceleration
    velocity_data = calculate_velocity(df, lookback=5)
    acceleration_data = calculate_acceleration(df, lookback=5)
    
    # Volume Profile
    volume_profile = calculate_volume_profile_with_memory(df, num_levels=20)
    
    # Precio actual
    current_price = float(df['close'].iloc[-1])
    
    # ===== Strength =====

    elif adx_data['adx_current'] > 18:
        strength = 'MODERATE'
    else:
        strength = 'WEAK'
    
    print(f"         ⚡ 5M | {symbol} | {direction} | Precio:${current_price:.6f} | Vel:{velocity_data['velocity_current']:.6f}")
    
    return {

    }

# FUNCIÓN 5: analyze_coin_complete() - ORQUESTADOR PRINCIPAL
# ============================================================================

def analyze_coin_complete(symbol):
    """
    Análisis COMPLETO multi-timeframe con contexto acumulativo
    
    Flow ACUMULATIVO:
    1. 4H debe aprobar (contexto dominante) → Si falla, STOP
    2. 1H debe confirmar o no contradecir → Si falla, STOP
    3. 15M debe alinearse (timing) → Si falla, STOP
    4. 5M info adicional (SIEMPRE ejecuta)
    
    Args:
        symbol (str): Par trading (ej: 'BTC/USDT')
    
    Returns:
        dict: {
            'viable': bool,
            'score_data': dict con score 0-100,
            'direction': str,
            'analysis_4h': dict,
            'analysis_1h': dict,
            'analysis_15m': dict,
            'analysis_5m': dict,
            'triggers': dict (entry, stop, target),
            'context_summary': str,
            'timestamp': str
        }
    """
    # ===== PASO 0: 1D - Contexto Macro (OPCIONAL) =====
    analysis_1d = None
    if ENABLE_1D_ANALYSIS:
        analysis_1d = analyze_1d(symbol)

    # ===== PASO 1: 4H - Contexto Dominante =====
    analysis_4h = analyze_4h(symbol)

    if not analysis_4h:
        return {
            'viable': False,
            'reason': '4H no cumple criterios estrictos',
            'timestamp': datetime.now().isoformat()
        }

    direction_dominant = analysis_4h['direction']

    # Verificar coherencia 1D-4H
    conflict_1d_4h = False
    if analysis_1d:
        # Si 1D bajista fuerte + 4H alcista = posible rebote (precaución)
        if (analysis_1d['phase'] == 'IMPULSO_BAJISTA' and 
            direction_dominant == 'BULLISH'):
            conflict_1d_4h = True
            print(f"⚠️ {symbol}: 1D bajista fuerte, 4H alcista (rebote técnico?)")
        # Si 1D alcista fuerte + 4H bajista = corrección saludable (OK)
        elif (analysis_1d['phase'] == 'CORRECCIÓN_ALCISTA' and
              direction_dominant == 'BEARISH'):
            print(f"✅ {symbol}: Corrección en tendencia alcista 1D")

    # ===== PASO 2: 1H - Confirmación =====
    analysis_1h = analyze_1h(symbol)
    
    # ===== PASO 3: 15M - Timing =====
    analysis_15m = analyze_15m(symbol)
    
    # ===== PASO 4: 5M - Ejecución (SIEMPRE) =====
    analysis_5m = analyze_5m(symbol)
    
    # ===== PASO 5: Calcular Score 0-100 =====
    score_total = (
        analysis_4h.get('score_4h', 20) +
        analysis_1h.get('score_1h', 0) +
        analysis_15m.get('score_15m', 0)
    )

    # Bonus por evolución temporal
    evolution_bonus = calculate_evolution_score(symbol)
    score_total += evolution_bonus

    # Penalty si 1D contradice fuertemente 4H
    if conflict_1d_4h:
        score_total -= 10
        print(f"   -10: Conflicto 1D-4H")

    # Guardar snapshot actual
    save_analysis_snapshot(symbol, analysis_4h, analysis_1h, analysis_15m)

    # Pre-viable si score >45
    pre_viable = score_total >= 45

    if not pre_viable:
        return {
            'viable': False,
            'reason': f'Score {score_total}/100 < 45',
            'timestamp': datetime.now().isoformat()
        }

    # ===== ANÁLISIS PROFUNDO =====
    print(f"\n🔬 {symbol} pre-viable (score {score_total}) → análisis profundo...")

    deep_ctx = deep_analysis_complete(symbol, analysis_4h, analysis_1h, analysis_15m)

    # Decisión final basada en probabilidad
    viable = deep_ctx['safe_to_trade'] and deep_ctx['probability'] >= 70

    # Si no viable por contexto profundo, explicar
    if not viable:
        reason = f"Probabilidad {deep_ctx['probability']}% < 70"
        if deep_ctx['warnings']:
            reason += f" | Warnings: {', '.join(deep_ctx['warnings'][:2])}"
        return {
            'viable': False,
            'reason': reason,
            'score_total': score_total,
            'deep_context': deep_ctx,
            'timestamp': datetime.now().isoformat()
        }

    # ===== PASO 6: Calcular Triggers =====
    triggers = calculate_triggers(analysis_15m, analysis_5m, direction_dominant)

    # ===== Resumen Final =====
    context_summary = (
        f"{direction_dominant} - "
        f"4H:{analysis_4h['strength']} "
        f"1H:{analysis_1h['strength']} "
        f"15M:{analysis_15m['strength']} | "
        f"Score:{score_total}/100"
    )

    print(f"✅ {symbol} | {direction_dominant} | Score:{score_total}/100 | 4H={analysis_4h['strength']} 1H={analysis_1h['strength']} 15M={analysis_15m['strength']} | Entry:${triggers['entry_price']:.6f} R:R={triggers['rr_ratio']:.2f}:1")

    return {
        'viable': viable,
        'score_data': {
            'score': score_total,
            'clasificacion': 'ÓPTIMA' if score_total >= 80 else 'BUENA' if score_total >= 65 else 'REGULAR' if score_total >= 50 else 'DÉBIL',
            'score_4h': analysis_4h.get('score_4h', 0),
            'score_1h': analysis_1h.get('score_1h', 0),
            'score_15m': analysis_15m.get('score_15m', 0),
            'evolution_bonus': evolution_bonus
        },
        'direction': direction_dominant,
        'analysis_1d': analysis_1d,
        'analysis_4h': analysis_4h,
        'analysis_1h': analysis_1h,
        'analysis_15m': analysis_15m,
        'analysis_5m': analysis_5m,
        'sr_zones': analysis_5m.get('sr', {}),
        'triggers': triggers,
        'context_summary': context_summary,
        'timestamp': datetime.now().isoformat(),
        'deep_context': deep_ctx,
        'probability': deep_ctx['probability'],
    }

# FUNCIÓN 6: calculate_score() - SCORING INTELIGENTE 0-100
# ============================================================================

def calculate_score(analysis_4h, analysis_1h, analysis_15m, analysis_5m):
    """
    Score 0-100 basado en calidad del setup multi-timeframe
    
    BASE: 50 puntos
    
    BONOS:
    +15: Confluencia perfecta 4H-1H-15M (misma dirección + todos STRONG)
    +10: ADX creciendo en TODOS los timeframes
    +10: Estructura HH/LL perfecta en 4H
    +5: Momentum >5 velas consecutivas en 4H
    +5: EMAs perfectamente separadas en 15M
    +5: Sin obstáculos SR cercanos (<2% distancia)
    
    PENALTIES:
    -10: ADX decreciendo en cualquier TF
    -10: Momentum débil (<3 velas en 4H)
    -5: Estructura mixta en 1H
    -5: Obstáculo SR muy cercano (<1%)
    -5: RSI en zona extrema (>70 BULL o <30 BEAR)
    
    Args:
        analysis_4h, analysis_1h, analysis_15m, analysis_5m (dict): Análisis por TF
    
    Returns:
        dict: {
            'score': int (0-100),
            'details': dict con desglose,
            'clasificacion': 'ÓPTIMA'/'BUENA'/'REGULAR'/'DÉBIL'
        }
    """
    score = 50  # Base
    details = {'base': 50, 'bonos': [], 'penalties': []}
    
    # ===== BONOS =====
    
    # +15: Confluencia perfecta (todos STRONG)
    if (analysis_4h['strength'] == 'STRONG' and 
        analysis_1h['strength'] == 'STRONG' and 
        analysis_15m['strength'] == 'STRONG'):
        score += 15
        details['bonos'].append('+15 Confluencia perfecta STRONG')
    elif (analysis_4h['strength'] in ['STRONG', 'MODERATE'] and
          analysis_1h['strength'] in ['STRONG', 'MODERATE']):
        score += 8
        details['bonos'].append('+8 Confluencia moderada')
    
    # +10: ADX creciendo en TODOS
    adx_trends = [
        analysis_4h['adx']['adx_trend'],
        analysis_1h['adx']['adx_trend'],
        analysis_15m['adx']['adx_trend']
    ]
    if all(trend == 'creciendo' for trend in adx_trends):
        score += 10
        details['bonos'].append('+10 ADX creciendo en todos TF')
    
    # +10: Estructura HH/LL perfecta 4H
    if analysis_4h['structure']['structure'] == analysis_4h['direction']:
        hh_count = analysis_4h['structure'].get('hh_count', 0)
        ll_count = analysis_4h['structure'].get('ll_count', 0)
        if (analysis_4h['direction'] == 'BULLISH' and hh_count >= 3) or \
           (analysis_4h['direction'] == 'BEARISH' and ll_count >= 3):
            score += 10
            details['bonos'].append('+10 Estructura perfecta 4H')
    
    # +5: Momentum >5 velas consecutivas 4H
    if analysis_4h['momentum']['consecutive_count'] >= 5:
        score += 5
        details['bonos'].append(f"+5 Momentum fuerte 4H ({analysis_4h['momentum']['consecutive_count']} velas)")
    
    # +5: EMAs separadas óptimamente 15M
    emas_15m = analysis_15m.get('emas', {})
    if emas_15m.get('alignment') == 'aligned':
        score += 5
        details['bonos'].append('+5 EMAs alineadas 15M')
    
    # +5: Sin obstáculos SR cercanos
    sr_15m = analysis_15m.get('sr', {})
    current_price = analysis_5m.get('current_price', 0)
    if current_price > 0 and sr_15m:
        resistances = sr_15m.get('resistances', [])
        supports = sr_15m.get('supports', [])
        
        if analysis_4h['direction'] == 'BULLISH' and resistances:
            nearest_resistance = min([r for r in resistances if r > current_price], default=None)
            if nearest_resistance:
                distance_pct = ((nearest_resistance - current_price) / current_price) * 100
                if distance_pct > 2.0:
                    score += 5
                    details['bonos'].append(f'+5 Sin resistencia cercana ({distance_pct:.1f}%)')
        
        elif analysis_4h['direction'] == 'BEARISH' and supports:
            nearest_support = max([s for s in supports if s < current_price], default=None)
            if nearest_support:
                distance_pct = ((current_price - nearest_support) / current_price) * 100
                if distance_pct > 2.0:
                    score += 5
                    details['bonos'].append(f'+5 Sin soporte cercana ({distance_pct:.1f}%)')
    
    # ===== PENALTIES =====
    
    # -10: ADX decreciendo en cualquier TF
    if 'decreciendo' in adx_trends:
        score -= 10
        details['penalties'].append('-10 ADX decreciendo detectado')
    
    # -10: Momentum débil 4H
    if analysis_4h['momentum']['consecutive_count'] < 3:
        score -= 10
        details['penalties'].append(f"-10 Momentum débil 4H ({analysis_4h['momentum']['consecutive_count']} velas)")
    
    # -5: Estructura mixta 1H
    if analysis_1h['structure']['structure'] not in [analysis_1h['direction'], 'MIXED']:
        score -= 5
        details['penalties'].append('-5 Estructura mixta 1H')
    
    # -5: Obstáculo SR muy cercano (<1%)
    if current_price > 0 and sr_15m:
        if analysis_4h['direction'] == 'BULLISH' and resistances:
            nearest_resistance = min([r for r in resistances if r > current_price], default=None)
            if nearest_resistance:
                distance_pct = ((nearest_resistance - current_price) / current_price) * 100
                if distance_pct < 1.0:
                    score -= 5
                    details['penalties'].append(f'-5 Resistencia muy cercana ({distance_pct:.1f}%)')
        
        elif analysis_4h['direction'] == 'BEARISH' and supports:
            nearest_support = max([s for s in supports if s < current_price], default=None)
            if nearest_support:
                distance_pct = ((current_price - nearest_support) / current_price) * 100
                if distance_pct < 1.0:
                    score -= 5
                    details['penalties'].append(f'-5 Soporte muy cercana ({distance_pct:.1f}%)')
    
    # -5: RSI en zona extrema
    rsi_4h = analysis_4h.get('rsi', {}).get('rsi_current', 50)
    if (analysis_4h['direction'] == 'BULLISH' and rsi_4h > 70) or \
       (analysis_4h['direction'] == 'BEARISH' and rsi_4h < 30):
        score -= 5
        details['penalties'].append(f'-5 RSI extremo ({rsi_4h:.1f})')
    
    # ===== Limitar score 0-100 =====
    score = max(0, min(100, score))
    
    # ===== Clasificación =====
    if score >= 80:
        classification = 'ÓPTIMA'
    elif score >= 65:
        classification = 'BUENA'
    elif score >= 50:
        classification = 'REGULAR'
    else:
        classification = 'DÉBIL'
    
    return {
        'score': score,
        'details': details,
        'clasificacion': classification
    }

# FUNCIÓN 7: calculate_triggers() - ENTRY/STOP/TARGET
# ============================================================================

def calculate_triggers(analysis_15m, analysis_5m, direction):
    """
    Calcula precios exactos para entrada/stop/target
    
    ENTRY: Precio actual 5M
    STOP: -2.1% del entry (configurable)
    TARGET: Proyección matemática limitada por SR
    
    Proyección:
    1. Velocity + Acceleration del 15M
    2. Proyectar movimiento esperado en próximos 10-30 minutos
    3. Limitar por próxima resistencia (BULL) o soporte (BEAR)
    4. Si proyección >= SR cercano → usar SR * 0.98 (2% antes)
    
    Args:
        analysis_15m (dict): Análisis 15M con velocity/acceleration
        analysis_5m (dict): Análisis 5M con precio actual
        direction (str): 'BULLISH' o 'BEARISH'
    
    Returns:
        dict: {
            'entry_price': float,
            'stop_loss': float,
            'target_price': float,
            'rr_ratio': float,
            'tiempo_proyectado_min': int,
            'metodo_target': str
        }
    """
    # ===== Entry Price =====
    entry_price = analysis_5m.get('current_price', 0)
    
    if entry_price == 0:
        print("[TRIGGERS] ⚠️ No se pudo obtener precio actual")
        return {
            'entry_price': 0,
            'stop_loss': 0,
            'target_price': 0,
            'rr_ratio': 0,
            'tiempo_proyectado_min': 0,
            'metodo_target': 'ERROR - Sin precio'
        }
    
    # ===== Stop Loss =====
    stop_loss_pct = STOP_LOSS_PERCENT  # Desde brain_config
    
    if direction == 'BULLISH':
        stop_loss = entry_price * (1 - stop_loss_pct / 100)
    else:  # BEARISH
        stop_loss = entry_price * (1 + stop_loss_pct / 100)
    
    # ===== Target Price - Proyección combinada =====
    
    # Obtener velocity y acceleration del 15M
    velocity_15m = analysis_15m.get('velocity', {})
    acceleration_15m = analysis_15m.get('acceleration', {})
    
    velocity_value = velocity_15m.get('velocity_current', 0)  # % por minuto
    acceleration_value = acceleration_15m.get('acceleration_current', 0)
    
    # Tiempo proyectado (entre 10-30 minutos dependiendo de fuerza)
    if analysis_15m['strength'] == 'STRONG':
        tiempo_proyectado_min = 15
    elif analysis_15m['strength'] == 'MODERATE':
        tiempo_proyectado_min = 20
    else:
        tiempo_proyectado_min = 25
    
    # Proyección matemática
    # precio_futuro = precio_actual + (velocity * tiempo) + (0.5 * acceleration * tiempo^2)
    velocity_contribution = (velocity_value / 100) * entry_price * tiempo_proyectado_min
    acceleration_contribution = 0.5 * (acceleration_value / 100) * entry_price * (tiempo_proyectado_min ** 2)
    
    proyeccion_matematica = entry_price + velocity_contribution + acceleration_contribution
    
    # ===== Limitar por SR =====
    sr_15m = analysis_15m.get('sr', {})
    metodo_target = 'proyeccion_matematica'
    target_price = proyeccion_matematica
    
    if direction == 'BULLISH':
        # Buscar próxima resistencia
        resistances = sr_15m.get('resistances', [])
        resistances_above = [r for r in resistances if r > entry_price]
        
        if resistances_above:
            nearest_resistance = min(resistances_above)
            
            # Si proyección alcanza o supera resistencia → usar resistencia * 0.98
            if proyeccion_matematica >= nearest_resistance:
                target_price = nearest_resistance * 0.98  # 2% antes de resistencia
                metodo_target = f'resistencia_{nearest_resistance:.2f}_ajustada'
            else:
                target_price = proyeccion_matematica
                metodo_target = 'proyeccion_libre'
        else:
            # Sin resistencias detectadas → usar proyección pura
            target_price = proyeccion_matematica
            metodo_target = 'proyeccion_sin_SR'
    
    else:  # BEARISH
        # Buscar próximo soporte
        supports = sr_15m.get('supports', [])
        supports_below = [s for s in supports if s < entry_price]
        
        if supports_below:
            nearest_support = max(supports_below)
            
            # Si proyección alcanza o baja del soporte → usar soporte * 1.02
            if proyeccion_matematica <= nearest_support:
                target_price = nearest_support * 1.02  # 2% antes de soporte
                metodo_target = f'soporte_{nearest_support:.2f}_ajustado'
            else:
                target_price = proyeccion_matematica
                metodo_target = 'proyeccion_libre'
        else:
            target_price = proyeccion_matematica
            metodo_target = 'proyeccion_sin_SR'
    
    # ===== Risk:Reward Ratio =====
    if direction == 'BULLISH':
        risk = entry_price - stop_loss
        reward = target_price - entry_price
    else:  # BEARISH
        risk = stop_loss - entry_price
        reward = entry_price - target_price
    
    rr_ratio = reward / risk if risk > 0 else 0
    
    # ===== Validación mínima R:R =====
    if rr_ratio < 1.5:
        print(f"[TRIGGERS] ⚠️ R:R bajo ({rr_ratio:.2f}) - Ajustando target...")
        # Ajustar target para lograr R:R = 2.0 mínimo
        if direction == 'BULLISH':
            target_price = entry_price + (risk * 2.0)
        else:
            target_price = entry_price - (risk * 2.0)
        
        rr_ratio = 2.0
        metodo_target += '_ajustado_RR'
    
    return {
        'entry_price': round(entry_price, 4),
        'stop_loss': round(stop_loss, 4),
        'target_price': round(target_price, 4),
        'rr_ratio': round(rr_ratio, 2),
        'tiempo_proyectado_min': tiempo_proyectado_min,
        'metodo_target': metodo_target,
        'proyeccion_matematica': round(proyeccion_matematica, 4),
        'velocity_15m': round(velocity_value, 4),
        'acceleration_15m': round(acceleration_value, 6)
    }

# SISTEMA DE MEMORIA HISTÓRICA

coin_memory = {}  # {symbol: [snapshots históricos]}

def save_analysis_snapshot(symbol, analysis_4h, analysis_1h, analysis_15m):
    """Guarda snapshot completo para comparación futura"""
    if symbol not in coin_memory:
        coin_memory[symbol] = []
    
    snapshot = {
        'timestamp': datetime.now().isoformat(),
        'adx_4h': analysis_4h['adx']['adx_current'],
        'di_plus_4h': analysis_4h['adx']['di_plus_current'],
        'momentum_4h': analysis_4h['momentum']['consecutive_count'],
        'price': analysis_15m.get('current_price', 0)
    }
    
    coin_memory[symbol].append(snapshot)
    
    # Mantener últimos 20 snapshots (evitar memoria infinita)
    if len(coin_memory[symbol]) > 20:
        coin_memory[symbol] = coin_memory[symbol][-20:]

def get_historical_context(symbol):
    """Retorna contexto histórico si existe"""
    return coin_memory.get(symbol, [])

def calculate_evolution_score(symbol):
    """Score basado en PROGRESO temporal (+0 a +20 puntos)"""
    history = get_historical_context(symbol)
    
    if len(history) < 3:
        return 0  # Sin historia suficiente
    
    score = 0
    
    # ADX creciendo últimas 3 mediciones: +10
    adx_values = [h['adx_4h'] for h in history[-3:]]
    if adx_values[-1] > adx_values[-2] > adx_values[-3]:
        score += 10
    
    # Momentum madurando: +5
    mom_values = [h['momentum_4h'] for h in history[-3:]]
    if mom_values[-1] >= mom_values[-2] >= mom_values[-3]:
        score += 5
    
    # Precio subiendo consistente: +5
    price_values = [h['price'] for h in history[-3:]]
    if price_values[-1] > price_values[-2] > price_values[-3]:
        score += 5
    
    return score
