    def detect_market_structure(self, df):

    def detect_double_patterns(self, df):

        # Buscar dos máximos/mínimos similares en las últimas 50 velas
        recent_highs = df['high'][-50:].nlargest(5)
        recent_lows = df['low'][-50:].nsmallest(5)

        if len(recent_highs) >= 2:
            # Si la diferencia es menor al 0.5%

                return {

    def detect_triangles(self, df):
        """Triángulos - Los más confiables"""
        # Líneas de tendencia con regresión lineal en últimas 30 velas
        highs = df['high'][-30:]
        lows = df['low'][-30:]
        
        # Indices x para regresión
        x = np.arange(len(highs))
        
        # Pendientes (slope)
        high_slope = np.polyfit(x, highs, 1)[0]
        low_slope = np.polyfit(x, lows, 1)[0]

        # Triángulo ascendente: Highs planos (slope ~0), Lows subiendo (slope > 0)
        if abs(high_slope) < 0.001 and low_slope > 0.001:
            return {
                'type': 'ASCENDING_TRIANGLE',
                'breakout_level': highs.max(),
                'confidence': 85,
                'action': 'BUY_ON_BREAKOUT'
            }
        
        # Triángulo descendente: Lows planos (slope ~0), Highs bajando (slope < 0)
        elif abs(low_slope) < 0.001 and high_slope < -0.001:
            return {
                'type': 'DESCENDING_TRIANGLE',
                'breakdown_level': lows.min(),
                'confidence': 85,
                'action': 'SELL_ON_BREAKDOWN'
            }
            
        # Triángulo simétrico: Highs bajan, Lows suben
        elif high_slope < -0.001 and low_slope > 0.001:
            return {
                'type': 'SYMMETRIC_TRIANGLE',
                'confidence': 70,
                'action': 'WAIT_BREAKOUT'
            }
            
        return {'type': 'none'}

    def detect_flags(self, df):

        # Consolidación: Rango estrecho
        consolidation_range = (last_20['high'].max() - last_20['low'].min()) / last_20['close'].mean()

        # Flag alcista: Impulso fuerte positivo + consolidación estrecha

            return {

                'target': last_20['close'].iloc[-1] * (1 + impulse), # Target proyectado

                'type': 'BEAR_FLAG',
                'confidence': 80,
                'target': last_20['close'].iloc[-1] * (1 + impulse),
                'action': 'SELL_CONTINUATION'
            }

    def calculate_pattern_confluence(self, all_patterns):
        """Confluencia entre TFs"""
        score = 0
        bullish_score = 0
        bearish_score = 0
        
        weights = {'15m': 0.2, '1h': 0.4, '4h': 0.4}
        
        for tf, patterns in all_patterns.items():
            w = weights.get(tf, 0.2)
            for p in patterns:
                conf = p.get('confidence', 0) * w
                action = p.get('action', '')
                
                if 'BUY' in action or 'LONG' in action or 'UPTREND' in p.get('type', ''):
                    bullish_score += conf
                elif 'SELL' in action or 'SHORT' in action or 'DOWNTREND' in p.get('type', ''):
                    bearish_score += conf

        # Retornar dirección dominante
        if bullish_score > bearish_score * 1.5:
            return {'direction': 'BULLISH', 'strength': bullish_score}
        elif bearish_score > bullish_score * 1.5:
            return {'direction': 'BEARISH', 'strength': bearish_score}
        else:
            return {'direction': 'NEUTRAL', 'strength': 0}

    def get_strongest_pattern(self, all_patterns):
        """Retorna el patrón individual más fuerte encontrado"""
        best = None
        max_conf = 0
        best_tf = None
        
        for tf, patterns in all_patterns.items():
            for p in patterns:
                if p.get('confidence', 0) > max_conf:
                    max_conf = p.get('confidence')
                    best = p
                    best_tf = tf
        
        if not best:
            return None
            
        best['tf'] = best_tf
        return best

    def pattern_to_action(self, results):

            return 'PREPARE_SHORT'

def _extract_score(score_value, default=0):
    """
    Recursively extract integer score from potentially nested dict.
    
    Args:
        score_value: The value to extract (can be int, float, dict, or None)
        default: Default value if extraction fails

def find_nearest_resistance(current_price, resistances):
    """
    Encuentra la resistencia más cercana por encima del precio actual.
    Args:
        current_price (float): Precio actual.
        resistances (list): Lista de resistencias (floats).
    Returns:
        tuple: (precio_resistencia, distancia_pct) o None
    """
    if not resistances:
        return None
    above = [r for r in resistances if r > current_price]
    if not above:
        return None
    nearest = min(above, key=lambda r: r - current_price)
    distance_pct = ((nearest - current_price) / current_price) * 100
    return (nearest, distance_pct)

def find_nearest_support(current_price, supports):
    """
    Encuentra el soporte más cercano por debajo del precio actual.
    Args:
        current_price (float): Precio actual.
        supports (list): Lista de soportes (floats).
    Returns:
        tuple: (precio_soporte, distancia_pct) o None
    """
    if not supports:
        return None
    below = [s for s in supports if s < current_price]
    if not below:
        return None
    nearest = max(below, key=lambda s: current_price - s)
    distance_pct = ((current_price - nearest) / current_price) * 100
    return (nearest, distance_pct)

            # Toque = si vela cruzó nivel ±tolerance
            if level == 0:
                continue
            if abs((low - level) / level * 100) <= tolerance_pct or \
               abs((high - level) / level * 100) <= tolerance_pct:
                touches += 1
        return touches

    # Analizar soporte más cercano
    nearest_support_info = None
    if supports:
        nearest_support = supports[0]  # Ya viene ordenado por cercanía
        touches = count_touches(nearest_support)
        distance_pct = ((current_price - nearest_support) / current_price) * 100

        if touches >= 10:
            strength = 'MUY_FUERTE'
        elif touches >= 6:
            strength = 'FUERTE'
        elif touches >= 3:
            strength = 'MODERADO'
        else:
            strength = 'DÉBIL'

        nearest_support_info = {
            'price': round(nearest_support, 6),
            'touches': touches,
            'strength': strength,
            'distance_pct': round(distance_pct, 2)
        }

    # Analizar resistencia más cercana
    nearest_resistance_info = None
    if resistances:
        nearest_resistance = resistances[0]  # Ya viene ordenado por cercanía
        touches = count_touches(nearest_resistance)
        distance_pct = ((nearest_resistance - current_price) / current_price) * 100

        if touches >= 10:
            strength = 'MUY_FUERTE'
        elif touches >= 6:
            strength = 'FUERTE'
        elif touches >= 3:
            strength = 'MODERADO'
        else:
            strength = 'DÉBIL'

        nearest_resistance_info = {
            'price': round(nearest_resistance, 6),
            'touches': touches,
            'strength': strength,
            'distance_pct': round(distance_pct, 2)
        }

    # ¿Zona peligrosa?
    danger_zone = False
    if nearest_support_info and nearest_support_info['distance_pct'] < 2.0 and nearest_support_info['touches'] >= 8:
        danger_zone = True  # Muy cerca de soporte fuerte = no SHORT
    if nearest_resistance_info and nearest_resistance_info['distance_pct'] < 2.0 and nearest_resistance_info['touches'] >= 8:
        danger_zone = True  # Muy cerca de resistencia fuerte = no LONG

    return {
        'nearest_support': nearest_support_info,
        'nearest_resistance': nearest_resistance_info,
        'danger_zone': danger_zone
    }

def detect_structure_deterioration(symbol, timeframe='1h'):
    """
    Detecta si estructura precio se está deteriorando

    PATRÓN: HH → HL → LL = debilidad progresiva
    Ejemplo: Alto 100 → rebote 95 → nuevo alto 98 (más bajo) → siguiente rebote será <95

    Returns:
        dict: {
            'deteriorating': bool,
            'pattern': 'HH_HL_LL' o 'STRONG' o 'UNCLEAR',
            'confidence': float (0-100)
        }
    """
    ohlcv = fetch_ohlcv(symbol, timeframe, 100)
    if not ohlcv or len(ohlcv) < 50:
        return None

    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])

    # Detectar pivots significativos
    structure = detect_price_structure(df, lookback=50, pivot_window=3)

    pivot_highs = structure['pivot_highs']
    pivot_lows = structure['pivot_lows']

    if len(pivot_highs) < 3 or len(pivot_lows) < 3:
        return {
            'deteriorating': False,
            'pattern': 'UNCLEAR',
            'confidence': 0
        }

    # Últimos 3 máximos
    last_3_highs = pivot_highs[-3:]
    # Últimos 3 mínimos
    last_3_lows = pivot_lows[-3:]

    # Patrón deterioro alcista: HH → HL (máximo más bajo) + LL
    if last_3_highs[2] < last_3_highs[1] < last_3_highs[0]:  # Máximos bajando
        if last_3_lows[2] < last_3_lows[1]:  # Mínimos también bajando
            return {
                'deteriorating': True,
                'pattern': 'HH_HL_LL',
                'confidence': 85
            }

    # Patrón fuerte alcista: HH + HL consecutivos
    if last_3_highs[2] > last_3_highs[1] > last_3_highs[0]:
        if last_3_lows[2] > last_3_lows[1] > last_3_lows[0]:
            return {
                'deteriorating': False,
                'pattern': 'STRONG_BULLISH',
                'confidence': 90
            }

    # Patrón fuerte bajista: LH + LL consecutivos
    if last_3_highs[2] < last_3_highs[1] < last_3_highs[0]:
        if last_3_lows[2] < last_3_lows[1] < last_3_lows[0]:
            return {
                'deteriorating': False,
                'pattern': 'STRONG_BEARISH',
                'confidence': 90
            }

    return {
        'deteriorating': False,
        'pattern': 'MIXED',
        'confidence': 40
    }

def detect_divergence(symbol, timeframe='15m'):
    """
    Detecta divergencia ADX vs precio

    DIVERGENCIA BAJISTA: Precio sube + ADX baja = debilidad (peligro)
    DIVERGENCIA ALCISTA: Precio baja + ADX sube = fuerza oculta

    Returns:
        dict: {
            'divergence_detected': bool,
            'type': 'BEARISH'/'BULLISH'/'NONE',
            'strength': 'FUERTE'/'MODERADA'/'DÉBIL'
        }
    """
    ohlcv = fetch_ohlcv(symbol, timeframe, 50)
    if not ohlcv or len(ohlcv) < 30:
        return None

    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])

    adx_data = calculate_adx_with_memory(df, period=14, lookback=20)

    # Últimas 10 velas: tendencia precio vs tendencia ADX
    prices = df['close'].tail(10).tolist()
    adx_values = adx_data['adx_history'][-10:]

    # Precio subiendo?
    price_rising = prices[-1] > prices[0]
    price_change_pct = ((prices[-1] - prices[0]) / prices[0]) * 100

    # ADX subiendo?
    adx_rising = adx_values[-1] > adx_values[0]
    adx_change = adx_values[-1] - adx_values[0]

    # Divergencia bajista: precio sube + ADX baja
    if price_rising and not adx_rising and price_change_pct > 2.0 and adx_change < -3:
        return {
            'divergence_detected': True,
            'type': 'BEARISH',
            'strength': 'FUERTE' if abs(adx_change) > 5 else 'MODERADA'
        }

    # Divergencia alcista: precio baja + ADX sube
    if not price_rising and adx_rising and price_change_pct < -2.0 and adx_change > 3:
        return {
            'divergence_detected': True,
            'type': 'BULLISH',
            'strength': 'FUERTE' if adx_change > 5 else 'MODERADA'
        }

    return {
        'divergence_detected': False,
        'type': 'NONE',
        'strength': 'N/A'
    }
