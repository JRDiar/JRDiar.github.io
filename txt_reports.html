# v5.4.3: EARLY IGNITION DETECTION
# ===================================================================
def detect_early_ignition(symbol, direction, timeframe='5m'):
    """
    Detecta explosión EN FORMACIÓN (antes de confirmación completa).
    
    Joel: "El bot entra tarde cuando ya explotó. Necesita anticipar."
    
    Señales tempranas:
    1. Volume spike >200% promedio (capital entrando)
    2. Vela actual >0.15% sin cerrar (movimiento iniciando)
    3. Rompe máximo/mínimo anterior (estructura rompiendo)
    4. ADX velocity >2 pts/vela (fuerza acelerando)
    
    Returns:
        {
            'is_igniting': bool,
            'confidence': str ('HIGH', 'MEDIUM', 'LOW'),
            'reason': str,
            'signals': list
        }
    """
    from exchange_connection import fetch_current_price
    
    storage = get_frame_storage(symbol, timeframe)
    
    if not storage.has_enough_frames(25):
        return {'is_igniting': False, 'confidence': 'NONE', 'reason': 'Datos insuficientes'}
    
    try:
        df = storage.df
    except:
        return {'is_igniting': False, 'confidence': 'NONE', 'reason': 'Sin DataFrame'}
    
    if len(df) < 25:
        return {'is_igniting': False, 'confidence': 'NONE', 'reason': 'Datos insuficientes'}
    
    signals = []
    confidence_score = 0
    
    # ===== SEÑAL 1: VOLUME SPIKE =====
    try:
        volume_current = float(df['volume'].iloc[-1])
        volume_avg = df['volume'].iloc[-20:-1].mean()
        volume_ratio = volume_current / volume_avg if volume_avg > 0 else 0
        
        if volume_ratio > 2.5:
            signals.append(f'Volume spike x{volume_ratio:.1f}')
            confidence_score += 40
        elif volume_ratio > 1.8:
            signals.append(f'Volume elevado x{volume_ratio:.1f}')
            confidence_score += 25
    except:
        pass
    
    # ===== SEÑAL 2: VELA EN FORMACIÓN =====
    try:
        current_price = fetch_current_price(symbol)
        current_open = float(df['open'].iloc[-1])
        candle_change_pct = ((current_price - current_open) / current_open) * 100
        
        if direction == 'BULLISH':
            if candle_change_pct > 0.20:
                signals.append(f'Vela verde +{candle_change_pct:.2f}%')
                confidence_score += 30
            elif candle_change_pct > 0.10:
                signals.append(f'Vela subiendo +{candle_change_pct:.2f}%')
                confidence_score += 15
        else:  # BEARISH
            if candle_change_pct < -0.20:
                signals.append(f'Vela roja {candle_change_pct:.2f}%')
                confidence_score += 30
            elif candle_change_pct < -0.10:
                signals.append(f'Vela bajando {candle_change_pct:.2f}%')
                confidence_score += 15
    except:
        pass
    
    # ===== SEÑAL 3: RUPTURA ESTRUCTURA =====
    try:
        prev_high = float(df['high'].iloc[-2])
        prev_low = float(df['low'].iloc[-2])
        current_price = fetch_current_price(symbol)
        
        if direction == 'BULLISH' and current_price > prev_high:
            signals.append('Rompió máximo anterior')
            confidence_score += 20
        elif direction == 'BEARISH' and current_price < prev_low:
            signals.append('Rompió mínimo anterior')
            confidence_score += 20
    except:
        pass
    
    # ===== SEÑAL 4: ADX ACELERANDO =====
    try:
        adx_data = storage.get_trajectory('adx.adx_current', 5)
        if adx_data and len(adx_data) >= 3:
            adx_velocity = adx_data[-1] - adx_data[-2]
            if adx_velocity > 2:
                signals.append(f'ADX acelerando +{adx_velocity:.1f}')
                confidence_score += 10
    except:
        pass
    
    # ===== EVALUACIÓN FINAL =====
    if confidence_score >= 60:
        confidence = 'HIGH'
        is_igniting = True
    elif confidence_score >= 40:
        confidence = 'MEDIUM'
        is_igniting = True
    else:
        confidence = 'LOW'
        is_igniting = False
    
    return {
        'is_igniting': is_igniting,
        'confidence': confidence,
        'confidence_score': confidence_score,
        'reason': ', '.join(signals) if signals else 'Sin señales',
        'signals': signals
    }
