def analyze_4h(symbol):
    """
    4H: Define contexto dominante y clima de mercado
    
    NUNCA RECHAZA - Solo informa dirección y fuerza
    
    Args:
        symbol (str): Par trading (ej: 'BTC/USDT')
    
    Returns:
        dict: Análisis completo con direction, strength, indicadores
        None: Solo si no hay datos (caso extremo)
    """
    # Fetch datos
    ohlcv = fetch_ohlcv(symbol, '4h', 100)
    if not ohlcv or len(ohlcv) < 50:
        return None

    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    current_price = float(df['close'].iloc[-1])
    
    # ===== ADX con memoria =====
    adx_data = calculate_adx_with_memory(df, period=14, lookback=10)
    
    # ADX ANTICIPATORIO: Proyectar evolución
    adx_now = adx_data['adx_current']
    adx_trend = adx_data['adx_trend']
    
    # Calcular velocidad ADX (puntos por vela)
    if len(adx_data['adx_history']) >= 5:
        adx_velocity = (adx_data['adx_history'][-1] - adx_data['adx_history'][0]) / 5
    else:
        adx_velocity = 0.0
    
    # Proyección 3 velas adelante (12 horas en 4H)
    adx_proyectado = adx_now + (adx_velocity * 3)
    
    # ===== Determinar dirección base por dominancia =====
    if adx_data['dominance'] == 'DI+':
        direction = 'BULLISH'
    elif adx_data['dominance'] == 'DI-':
        direction = 'BEARISH'
    else:
        direction = 'LATERAL'

    # ===== Evaluar fuerza ADX =====
    strength = 'WEAK'  # Inicializar en WEAK
    
    # CASO 1: ADX ya >25 (perfecto)
    if adx_now >= 25:
        strength = 'STRONG'
    elif adx_now >= 17 and adx_velocity > 1.5:
        strength = 'MODERATE'
    else:
        strength = 'WEAK'

    if adx_trend == 'decreciendo' and adx_velocity < -0.8:
        strength = 'WEAK'

    if direction != 'LATERAL' and abs(adx_data['spread']) < 5.0:
        strength = 'WEAK'

    momentum_data = calculate_momentum_with_memory(df, lookback=10)
    # FIX: Aceptar 'lime' como bullish y 'maroon' como bearish
    expected_colors = ['green', 'lime'] if direction == 'BULLISH' else ['red', 'maroon']

    if direction != 'LATERAL' and momentum_data['color_current'] not in expected_colors:
        direction = 'LATERAL'

    if momentum_data['consecutive_count'] < 3:
        strength = 'WEAK'

    rsi_data = calculate_rsi_with_memory(df, period=14, lookback=10)
    emas_data = calculate_emas_with_memory(df, periods=[9, 21, 50, 200], lookback=10)

    if ENABLE_4H_STRUCTURE:
        structure = detect_price_structure(df, lookback=20)
    else:
        structure = {'structure': 'N/A'}

    if ENABLE_4H_ZIGZAG:
        zigzag = calculate_zigzag(df, threshold_pct=3.0)
    else:
        zigzag = {'zigzag': 'N/A'}

    print(f"📊 4H | {symbol} | {direction} {strength} | ADX:{adx_data['adx_current']:.1f} DI+:{adx_data['di_plus_current']:.1f} DI-:{adx_data['di_minus_current']:.1f} | Mom:{momentum_data['consecutive_count']}v | Estructura:{structure['structure']}")

    score_4h = 0
    if strength == 'STRONG':
        score_4h += 20
    elif strength == 'MODERATE':
        score_4h += 10
    else:
        score_4h += 0
    if adx_trend == 'creciendo':
        score_4h += 5
    if momentum_data['consecutive_count'] >= 5:
        score_4h += 5
    if abs(adx_data['spread']) > 7:
        score_4h += 5

    return {
        'direction': direction,
        'strength': strength,
        'score_4h': score_4h,
        'adx': adx_data,
        'momentum': momentum_data,
        'rsi': rsi_data,
        'emas': emas_data,
        'structure': structure,
        'zigzag': zigzag,
        'current_price': current_price,
        'context': f"4H {direction} {strength} - ADX {adx_data['adx_current']:.1f}",
        'timestamp': datetime.now().isoformat()
    }

def analyze_1h(symbol):
    """1H: Confirma tendencia 4H"""
    ohlcv = fetch_ohlcv(symbol, '1h', 100)
    if not ohlcv or len(ohlcv) < 50:
        return None

    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    adx_data = calculate_adx_with_memory(df, period=14, lookback=10)
    adx_trend = adx_data['adx_trend']
    adx_current = adx_data['adx_current']
    spread = abs(adx_data['spread'])

    if adx_data['dominance'] == 'DI+':
        direction = 'BULLISH'
    elif adx_data['dominance'] == 'DI-':
        direction = 'BEARISH'
    else:
        direction = 'LATERAL'

    momentum_data = calculate_momentum_with_memory(df, lookback=10)
    # FIX: Aceptar 'lime' como bullish y 'maroon' como bearish
    expected_colors = ['green', 'lime'] if direction == 'BULLISH' else ['red', 'maroon']
    momentum_color = momentum_data['color_current']

    rsi_data = calculate_rsi_with_memory(df, period=14, lookback=10)
    emas_data = calculate_emas_with_memory(df, periods=[9, 21, 50, 200], lookback=10)
    structure = detect_price_structure(df, lookback=20)
    zigzag = calculate_zigzag(df, threshold_pct=2.5)

    if adx_current > 32:
        strength = 'STRONG'
    elif adx_current > 25:
        strength = 'MODERATE'
    else:
        strength = 'WEAK'

    score_1h = 0
    if adx_trend == 'creciendo':
        score_1h += 5
    if adx_current >= 22:
        score_1h += 5
    if momentum_color in expected_colors:
        score_1h += 5
    if spread > 4.0:
        score_1h += 5

    adx_val = adx_data['adx_current']
    momentum = momentum_data['color_current']
    rsi = rsi_data['rsi_current']
    emas_aligned = emas_data.get('emas_aligned', False)

    if adx_val >= 25:
        score_1h += 5
    elif adx_val >= 20:
        score_1h += 2

    if momentum in ['green', 'lime', 'red', 'maroon']:
        score_1h += 2

    if 40 <= rsi <= 70:
        score_1h += 2

    if emas_aligned:
        score_1h += 2

    return {
        'direction': direction,
        'strength': strength,
        'score_1h': score_1h,
        'adx': adx_data,
        'momentum': momentum_data,
        'rsi': rsi_data,
        'emas': emas_data,
        'structure': structure,
        'zigzag': zigzag,
        'context': f"1H {direction} {strength} - ADX {adx_current:.1f}",
        'timestamp': datetime.now().isoformat()
    }

    if adx_data['dominance'] == 'DI+':
        direction = 'BULLISH'
    elif adx_data['dominance'] == 'DI-':
        direction = 'BEARISH'
    else:
        direction = 'LATERAL'

    momentum_data = calculate_momentum_with_memory(df, lookback=10)
    rsi_data = calculate_rsi_with_memory(df, period=14, lookback=10)
    emas_data = calculate_emas_with_memory(df, periods=[9, 21], lookback=10)
    structure = detect_price_structure(df, lookback=10)
    zigzag = calculate_zigzag(df, threshold_pct=1.5)
    sr_data = detect_support_resistance_with_memory(df, lookback=30)
    velocity_data = calculate_velocity(df, lookback=5)
    acceleration_data = calculate_acceleration(df, lookback=5)
    volume_profile = calculate_volume_profile_with_memory(df, num_levels=20)
    current_price = float(df['close'].iloc[-1])

    if adx_data['adx_current'] > 25:
        strength = 'STRONG'
    elif adx_data['adx_current'] > 18:
        strength = 'MODERATE'
    else:
        strength = 'WEAK'

    score_5m = 0
    if adx_data['adx_current'] > 25:
        score_5m += 10
    elif adx_data['adx_current'] > 18:
        score_5m += 5
    if velocity_data.get('velocity_current', 0) > 0:
        score_5m += 5
    if rsi_data['rsi_current'] and 30 <= rsi_data['rsi_current'] <= 70:
        score_5m += 5

    print(f"         ⚡ 5M | {symbol} | {direction} | Precio:${current_price:.6f} | Vel:{velocity_data['velocity_current']:.6f} | Score5M:{score_5m}")

    # Contexto de Volume Profile
    vp = calculate_volume_profile_with_memory(df, lookback=50)
    profile_high = vp.get('profile_high', current_price * 1.1)
    breaking_profile = current_price > profile_high
    
    # Distancia a EMA 20 (Bollinger Middle)
    bb = calculate_bollinger_bands(df, period=20)
    ema_20 = bb['middle'][-1]
    dist_ema_pct = ((current_price - ema_20) / ema_20) * 100

    return {
        'direction': direction,
        'strength': strength,
        'score_5m': score_5m,
        'current_price': current_price,
        'adx': adx_data,
        'momentum': momentum_data,

        'acceleration': acceleration_data,
        'volume_profile': vp,
        'breaking_profile': breaking_profile,
        'dist_ema_pct': dist_ema_pct,
        'context': f"5M {direction} - Ejecución inmediata",
        'timestamp': datetime.now().isoformat()
    }
