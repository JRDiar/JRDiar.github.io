def validate_5m_entry_conditions(symbol, direction):
    """
    Validación final de timing antes de entrar.
    Verifica:
    1. Que la vela 5M sea consistente con la dirección.
    2. Que el momentum sea joven (no > 7 velas).
    3. Que no estemos en un pullback profundo inmediato.
    """
    from exchange_connection import fetch_ohlcv
    from brain_calculator import calculate_momentum_with_memory
    import pandas as pd
    
    # Default score
    result = {'score': 100, 'missing': []}
    
    ohlcv = fetch_ohlcv(symbol, '5m', 20)
    if not ohlcv:
        return {'score': 0, 'missing': ['No Data']}
        
    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    mom_data = calculate_momentum_with_memory(df)
    
    last_candle = df.iloc[-1]
    is_green = last_candle['close'] > last_candle['open']
    
    # 1. Color Match
    if direction == 'BULLISH' and not is_green:
        result['score'] -= 30
        result['missing'].append("Red Candle 5M")
    elif direction == 'BEARISH' and is_green:
        result['score'] -= 30
        result['missing'].append("Green Candle 5M")
        
    # 2. Momentum Age
    consecutive = mom_data['consecutive_count']
    if consecutive > 7:
        result['score'] -= 15
        result['missing'].append(f"Momentum Stale ({consecutive}v)")
        
    # 3. Strength Check
    if mom_data['strength'] == 'WEAK':
        result['score'] -= 10
        result['missing'].append("Weak Momentum")
        
    return result
