def load_cache():
    """Cargar datos históricos desde CSVs"""
    cache = {}
    files = os.listdir(DATA_DIR)
    
    if not files:
        print(f"{RED}❌ Directorio vacío{RESET}")
        return None
    
    print(f"   📂 Cargando {len(files)} archivos...")
    
    # FIX 2026-01-02: Cargar CSVs en lugar de JSONs
    for filename in files:
        if not filename.endswith('.csv'):
            continue
        
        try:
            filepath = os.path.join(DATA_DIR, filename)
            df = pd.read_csv(filepath)
            
            # Extraer symbol del filename (ej: BTC_USDT_USDT.csv → BTC/USDT:USDT)
            symbol_raw = filename.replace('.csv', '')
            # Primera _ es /, segunda _ es :
            parts = symbol_raw.split('_')
            if len(parts) >= 3:
                symbol = f"{parts[0]}/{parts[1]}:{parts[2]}"
            else:
                continue  # Skip archivos con formato incorrecto
            
            # Convertir DataFrame a lista de listas [timestamp, O, H, L, C, V]
            required_cols = ['timestamp', 'open', 'high', 'low', 'close', 'volume']
            if not all(col in df.columns for col in required_cols):
                print(f"   ⚠️ {filename}: columnas faltantes")
                continue
            
            # Convertir a lista de listas para compatibilidad con FrameStorage
            ohlcv_list = df[required_cols].values.tolist()
            
            cache[symbol] = {
                '5m': ohlcv_list
            }
        except Exception as e:
            print(f"   ⚠️ Error en {filename}: {e}")
    
    print(f"   ✅ {len(cache)} símbolos cargados")
    return cache

def simulate_trade_outcome(symbol, direction, entry_frame, current_frame, cache):
    """
    Simula el resultado de un trade mirando frames futuros.
    Retorna None si aún no hay outcome, o PnL% si hay SL/TP/TimeExit
    """
    if symbol not in cache or '5m' not in cache[symbol]:
        return -STOP_LOSS_PCT  # Asumir loss si no hay data
    
    ohlcv = cache[symbol]['5m']
    
    if entry_frame >= len(ohlcv) or current_frame >= len(ohlcv):
        return None
    
    entry_price = ohlcv[entry_frame][4]  # Close al entrar
    
    # Revisar frames desde entrada hasta ahora
    for frame_idx in range(entry_frame + 1, current_frame + 1):
        if frame_idx >= len(ohlcv):
            break
            
        candle = ohlcv[frame_idx]
        high = candle[2]
        low = candle[3]
        
        if direction == 'BULLISH':
            # Check SL
            pnl_low = (low - entry_price) / entry_price * 100
            if pnl_low <= -STOP_LOSS_PCT:
                return -STOP_LOSS_PCT
            
            # Check TP
            pnl_high = (high - entry_price) / entry_price * 100
            if pnl_high >= TAKE_PROFIT_PCT:
                return TAKE_PROFIT_PCT
        else:
            # BEARISH
            pnl_high = (entry_price - high) / entry_price * 100
            if pnl_high <= -STOP_LOSS_PCT:
                return -STOP_LOSS_PCT
            
            pnl_low = (entry_price - low) / entry_price * 100
            if pnl_low >= TAKE_PROFIT_PCT:
                return TAKE_PROFIT_PCT
    
    # Check time exit
    frames_held = current_frame - entry_frame
    if frames_held >= MAX_HOLD_FRAMES:
        current_close = ohlcv[min(current_frame, len(ohlcv)-1)][4]
        if direction == 'BULLISH':
            return (current_close - entry_price) / entry_price * 100
        else:
            return (entry_price - current_close) / entry_price * 100
    
    return None  # Trade aún activo

def record_trade_result(symbol, direction, pnl_pct):
    """Registra resultado de trade de forma thread-safe"""
    global trade_count, results
    
    with results_lock:
        trade_count += 1
        current = trade_count
        
        is_win = pnl_pct > 0
        
        if is_win:
            results['wins'] += 1
        else:
            results['losses'] += 1
        
        results['total_pnl'] += pnl_pct
        pnl_usd = (pnl_pct / 100) * results['capital']
        results['capital'] += pnl_usd
        
        if pnl_pct > results['best_trade']:
            results['best_trade'] = pnl_pct
        if pnl_pct < results['worst_trade']:
            results['worst_trade'] = pnl_pct
        
        results['trades'].append({
            'symbol': symbol,
            'direction': direction,
            'pnl': pnl_pct
        })
        
        # Progress
        color = GREEN if is_win else RED
        icon = "✅" if is_win else "❌"
        total = results['wins'] + results['losses']
        wr = (results['wins'] / total * 100) if total > 0 else 0
        
        print(f"{icon} [{current:3d}/{TARGET_TRADES}] {color}{symbol:20s}{RESET} | "
              f"{direction:7s} | PnL: {color}{pnl_pct:+6.2f}%{RESET} | "
              f"WR: {wr:5.1f}% | Cap: ${results['capital']:.2f}")
        
        if current % SHOW_PROGRESS_EVERY == 0:
            print(f"\n{CYAN}--- STATS @ {total} trades ---{RESET}")
            print(f"    WR: {wr:.1f}% | PnL: {results['total_pnl']:+.2f}%\n")
    
    return current
