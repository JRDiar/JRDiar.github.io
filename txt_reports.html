# ===================================================================
# 5M VETO SYSTEM - Time Ladder v5.3.0
# ===================================================================
# El 5M puede vetar entrada si detecta condiciones críticas.
# Sigue Time Ladder: penalties acumulativas, no hard blocks individuales.
# Veto solo si penalty total >= 100
# ===================================================================
def check_5m_veto(symbol, direction, ladder_data):
    """
    5M puede vetar entrada si detecta condiciones críticas.
    
    Time Ladder Puro:
    - No hay hard blocks individuales
    - Penalties se acumulan
    - Veto solo si total >= 100
    
    Args:
        symbol: Par de trading
        direction: 'BULLISH' o 'BEARISH' del trade propuesto
        ladder_data: Dict con análisis de todos los timeframes
        
    Returns:
        {
            'should_veto': bool,
            'reason': str,
            'penalty': int,
            'details': list
        }
    """
    from brain_config import (
        ENABLE_5M_VETO,
        VETO_5M_ADX_MIN,
        VETO_5M_MOMENTUM_PENALTY,
        VETO_5M_LATERAL_PENALTY,
        VETO_5M_CONFLICT_PENALTY,
        VETO_5M_THRESHOLD
    )
    
    # Kill switch
    if not ENABLE_5M_VETO:
        return {'should_veto': False, 'reason': 'Veto disabled', 'penalty': 0, 'details': []}
    
    # Obtener data 5M
    data_5m = ladder_data.get('5m', {})
    
    # Edge case: Sin datos de 5M
    if not data_5m:
        print(f"   ⚠️ 5M VETO: Sin datos 5M para {symbol}")
        return {'should_veto': False, 'reason': 'No 5M data', 'penalty': 0, 'details': []}
    
    # Extraer campos con validación
    adx_5m = data_5m.get('adx', {})
    adx_current = adx_5m.get('adx_current', None)
    adx_trend = adx_5m.get('adx_trend', 'insuficiente')
    adx_velocity = adx_5m.get('adx_velocity', 0)
    
    mom_5m = data_5m.get('momentum', {})
    mom_color = mom_5m.get('color_current', 'gray')
    
    direction_5m = data_5m.get('direction', 'UNKNOWN')
    
    # Edge case: ADX current es None (sin datos válidos)
    if adx_current is None:
        print(f"   ⚠️ 5M VETO: ADX 5M sin valor para {symbol}")
        return {'should_veto': False, 'reason': 'No ADX data', 'penalty': 0, 'details': []}
    
    # Acumuladores Time Ladder
    penalties = []
    vetos = []
    
    # ===============================================
    # CONDICIÓN 1: ADX débil + decreciendo (COMBO)
    # No es veto individual, es penalty
    # ===============================================
    if adx_current < VETO_5M_ADX_MIN and adx_trend == 'decreciendo':
        penalty = 50
        penalties.append(penalty)
        vetos.append(f'ADX 5M {adx_current:.1f} < {VETO_5M_ADX_MIN} + decreciendo')
        print(f"   ⚠️ 5M: ADX {adx_current:.1f} débil + decreciendo → penalty -{penalty}")
    
    # ===============================================
    # CONDICIÓN 2: ADX colapsando rápido
    # Velocity < -2 pts/frame = colapso crítico
    # ===============================================
    if adx_velocity < -2:
        penalty = 40
        penalties.append(penalty)
        vetos.append(f'ADX 5M colapsando ({adx_velocity:.1f} pts/frame)')
        print(f"   🚨 5M: ADX colapsando rápido ({adx_velocity:.1f} pts/frame) → penalty -{penalty}")
    
    # ===============================================
    # CONDICIÓN 3: Direction LATERAL en 5M
    # Penalty gradual, no veto absoluto
    # ===============================================
    if direction_5m == 'LATERAL':
        penalty = VETO_5M_LATERAL_PENALTY  # 40
        penalties.append(penalty)
        vetos.append('5M Direction LATERAL')
        print(f"   ⚠️ 5M: Direction LATERAL → penalty -{penalty}")
    
    # ===============================================
    # CONDICIÓN 4: Direction UNKNOWN (sin claridad)
    # ===============================================
    if direction_5m == 'UNKNOWN':
        penalty = 30
        penalties.append(penalty)
        vetos.append('5M Direction UNKNOWN')
        print(f"   ⚠️ 5M: Direction UNKNOWN → penalty -{penalty}")
    
    # ===============================================
    # CONDICIÓN 5: Momentum 5M contradice trade (CRÍTICO)
    # ===============================================
    if direction == 'BULLISH' and mom_color in ['red', 'maroon']:
        penalty = VETO_5M_MOMENTUM_PENALTY  # 60
        penalties.append(penalty)
        vetos.append(f'Momentum 5M {mom_color} contradice LONG')
        print(f"   🚨 5M: Momentum {mom_color} contradice LONG → penalty -{penalty}")
    elif direction == 'BEARISH' and mom_color in ['green', 'lime']:
        penalty = VETO_5M_MOMENTUM_PENALTY  # 60
        penalties.append(penalty)
        vetos.append(f'Momentum 5M {mom_color} contradice SHORT')
        print(f"   🚨 5M: Momentum {mom_color} contradice SHORT → penalty -{penalty}")
    
    # ===============================================
    # CONDICIÓN 6: Conflicto de dirección 5M vs trade
    # ===============================================
    if direction_5m == 'BEARISH' and direction == 'BULLISH':
        penalty = VETO_5M_CONFLICT_PENALTY  # 50
        penalties.append(penalty)
        vetos.append('5M BEARISH pero trade es BULLISH')
        print(f"   🚨 5M: Direction {direction_5m} contradice trade {direction} → penalty -{penalty}")
    elif direction_5m == 'BULLISH' and direction == 'BEARISH':
        penalty = VETO_5M_CONFLICT_PENALTY  # 50
        penalties.append(penalty)
        vetos.append('5M BULLISH pero trade es BEARISH')
        print(f"   🚨 5M: Direction {direction_5m} contradice trade {direction} → penalty -{penalty}")
    
    # ===============================================
    # DECISIÓN FINAL: Veto si penalty total >= threshold
    # ===============================================
    total_penalty = sum(penalties)
    should_veto = total_penalty >= VETO_5M_THRESHOLD  # 100
    
    if should_veto:
        print(f"\n   🚫 5M VETO ACTIVADO: Penalty total {total_penalty} >= {VETO_5M_THRESHOLD}")
        print(f"      Razones: {' | '.join(vetos)}")
    elif total_penalty > 0:
        print(f"   ⚠️ 5M Penalty: {total_penalty} (< {VETO_5M_THRESHOLD}, no veto)")
    
    return {
        'should_veto': should_veto,
        'reason': ' | '.join(vetos) if vetos else '',
        'penalty': total_penalty,
        'details': vetos
    }
