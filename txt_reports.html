class SheetsManager:
    """Gestiona conexión y escritura a Google Sheets"""
    
    def __init__(self):
        self.client = None
        self.spreadsheet = None
        self.connected = False
        
        if ENABLE_SHEETS and GOOGLE_CREDENTIALS_FILE and GOOGLE_SPREADSHEET_ID:
            self._connect()
    
    def _connect(self):
        """Establece conexión con Google Sheets"""
        try:
            scope = [
                'https://spreadsheets.google.com/feeds',
                'https://www.googleapis.com/auth/drive'
            ]
            
            creds = ServiceAccountCredentials.from_json_keyfile_name(
                GOOGLE_CREDENTIALS_FILE, scope
            )
            
            self.client = gspread.authorize(creds)
            self.spreadsheet = self.client.open_by_key(GOOGLE_SPREADSHEET_ID)
            self.connected = True
            
            print("✅ Google Sheets conectado")
            
        except Exception as e:
            print(f"❌ Error conectando Google Sheets: {e}")
            self.connected = False
    
    def _get_sheet(self, sheet_name):
        """Obtiene o crea una hoja"""
        if not self.connected:
            return None
        
        try:
            return self.spreadsheet.worksheet(sheet_name)
        except gspread.WorksheetNotFound:
            # Crear hoja si no existe
            return self.spreadsheet.add_worksheet(title=sheet_name, rows=1000, cols=20)
    
    def write_row(self, sheet_name, row_data):
        """
        Escribe una fila en una hoja
        
        Args:
            sheet_name: Nombre de la hoja
            row_data: Lista de valores para la fila
        
        Returns:
            bool: True si éxito, False si error
        """
        if not self.connected:
            return False
        
        try:
            sheet = self._get_sheet(sheet_name)
            if not sheet:
                return False
            
            sheet.append_row(row_data)
            return True
            
        except Exception as e:
            print(f"❌ Error escribiendo fila en {sheet_name}: {e}")
            return False
    
    def write_batch(self, sheet_name, rows_data):
        """
        Escribe múltiples filas en una hoja (batch)
        
        Args:
            sheet_name: Nombre de la hoja
            rows_data: Lista de listas [[row1], [row2], ...]
        
        Returns:
            bool: True si éxito, False si error
        """
        if not self.connected or not BATCH_WRITE:
            return False
        
        try:
            sheet = self._get_sheet(sheet_name)
            if not sheet:
                return False
            
            # Escribir TODAS las filas en un solo request API
            sheet.append_rows(rows_data, value_input_option='USER_ENTERED')
            return True
            
        except Exception as e:
            print(f"❌ Error escribiendo batch en {sheet_name}: {e}")
            return False
    
    def clear_sheet(self, sheet_name):
        """Limpia todo el contenido de una hoja"""
        if not self.connected:
            return False
        
        try:
            sheet = self._get_sheet(sheet_name)
            if not sheet:
                return False
            
            sheet.clear()
            return True
            
        except Exception as e:
            print(f"❌ Error limpiando {sheet_name}: {e}")
            return False
    
    # ========================================================================
    # FUNCIONES ESPECÍFICAS POR HOJA
    # ========================================================================
    
    def write_pre_filtro(self, symbol, weekly_candles, status):
        """
        Escribe en BRAIN_PRE_FILTRO
        
        Args:
            symbol: Símbolo (ej: BTC/USDT:USDT)
            weekly_candles: Número de velas semanales
            status: APROBADO o RECHAZADO
        """
        timestamp = datetime.now(ARGENTINA_TZ).strftime("%Y-%m-%d %H:%M:%S")
        
        row = [
            timestamp,
            symbol,
            weekly_candles,
            status
        ]
        
        return self.write_row(SHEET_BRAIN_PRE_FILTRO, row)
    
    def write_watchlist(self, symbol, score, direction, emas_distance, structure, edad, tiempo_proyectado_min, rr_ratio):
        """
        Escribe en BRAIN_WATCHLIST
        
        Args:
            symbol: Símbolo
            score: Score 0-100
            direction: BULLISH o BEARISH
            emas_distance: Distancia entre EMAs (%)
            structure: SANA, REGULAR, MALA
            edad: JOVEN, MADURA
            tiempo_proyectado_min: Minutos proyectados a target
            rr_ratio: Risk/Reward ratio
        """
        timestamp = datetime.now(ARGENTINA_TZ).strftime("%Y-%m-%d %H:%M:%S")
        
        row = [
            timestamp,
            symbol,
            score,
            direction,
            emas_distance,
            structure,
            edad,
            tiempo_proyectado_min,
            rr_ratio
        ]
        
        return self.write_row(SHEET_BRAIN_WATCHLIST, row)
    
    def write_trigger(self, symbol, trigger_type, adx_value, momentum, volume_ratio):
        """
        Escribe en BRAIN_TRIGGERS
        
        Args:
            symbol: Símbolo
            trigger_type: ADX, MOMENTUM, VOLUMEN
            adx_value: Valor ADX actual
            momentum: Velas consecutivas momentum
            volume_ratio: Ratio volumen vs promedio
        """
        timestamp = datetime.now(ARGENTINA_TZ).strftime("%Y-%m-%d %H:%M:%S")
        
        row = [
            timestamp,
            symbol,
            trigger_type,
            adx_value,
            momentum,
            volume_ratio
        ]
        
        return self.write_row(SHEET_BRAIN_TRIGGERS, row)
    
    def write_proyeccion(self, symbol, current_price, velocity, acceleration, 
                        time_to_target, rr_ratio, target_price, stop_loss):
        """
        Escribe en BRAIN_PROYECCIONES
        
        Args:
            symbol: Símbolo
            current_price: Precio actual
            velocity: Velocidad (precio/min)
            acceleration: Aceleración
            time_to_target: Minutos estimados
            rr_ratio: Risk/Reward ratio
            target_price: Precio objetivo
            stop_loss: Stop loss
        """
        timestamp = datetime.now(ARGENTINA_TZ).strftime("%Y-%m-%d %H:%M:%S")
        
        row = [
            timestamp,
            symbol,
            current_price,
            velocity,
            acceleration,
            time_to_target,
            rr_ratio,
            target_price,
            stop_loss
        ]
        
        return self.write_row(SHEET_BRAIN_PROYECCIONES, row)
    
    def write_trade_entry(self, symbol, direction, entry_price, stop_loss, 
                         take_profit, score, rr_ratio):
        """
        Escribe entrada de trade en BRAIN_TRADES
        
        Args:
            symbol: Símbolo
            direction: BULLISH o BEARISH
            entry_price: Precio entrada
            stop_loss: Stop loss
            take_profit: Take profit
            score: Score Brain
            rr_ratio: Risk/Reward ratio
        """
        timestamp = datetime.now(ARGENTINA_TZ).strftime("%Y-%m-%d %H:%M:%S")
        
        row = [
            timestamp,
            symbol,
            "ENTRADA",
            direction,
            entry_price,
            stop_loss,
            take_profit,
            score,
            rr_ratio,
            "",  # PnL (se llena en salida)
            "",  # Duración (se llena en salida)
            ""   # Razón salida (se llena en salida)
        ]
        
        return self.write_row(SHEET_BRAIN_TRADES, row)
    
    def write_trade_exit(self, symbol, exit_price, pnl_pct, pnl_usd, 
                        duration_min, exit_reason):
        """
        Escribe salida de trade en BRAIN_TRADES
        
        Args:
            symbol: Símbolo
            exit_price: Precio salida
            pnl_pct: PnL en %
            pnl_usd: PnL en USD
            duration_min: Duración en minutos
            exit_reason: Razón de salida
        """
        timestamp = datetime.now(ARGENTINA_TZ).strftime("%Y-%m-%d %H:%M:%S")
        
        row = [
            timestamp,
            symbol,
            "SALIDA",
            "",  # Direction (vacío en salida)
            exit_price,
            "",  # Stop loss (vacío en salida)
            "",  # Take profit (vacío en salida)
            "",  # Score (vacío en salida)
            "",  # RR (vacío en salida)
            f"{pnl_pct:+.2f}% | ${pnl_usd:+.2f}",
            f"{duration_min:.1f} min",
            exit_reason
        ]
        
        return self.write_row(SHEET_BRAIN_TRADES, row)
    
    def write_monitoreo(self, symbol, current_price, pnl_pct, time_in_trade, 
                       stop_status, nearest_resistance):
        """
        Escribe en BRAIN_MONITOREO (estado actual trades activos)
        
        Args:
            symbol: Símbolo
            current_price: Precio actual
            pnl_pct: PnL actual en %
            time_in_trade: Minutos en trade
            stop_status: ORIGINAL, BREAKEVEN, TRAILING
            nearest_resistance: Resistencia más cercana
        """
        timestamp = datetime.now(ARGENTINA_TZ).strftime("%Y-%m-%d %H:%M:%S")
        
        row = [
            timestamp,
            symbol,
            current_price,
            pnl_pct,
            time_in_trade,
            stop_status,
            nearest_resistance
        ]
        
        return self.write_row(SHEET_BRAIN_MONITOREO, row)
    
    def write_salida_inteligente(self, symbol, signal_type, description):
        """
        Escribe señales de salida inteligente en BRAIN_SALIDAS
        
        Args:
            symbol: Símbolo
            signal_type: DESACELERACION, VOLUMEN_BAJO, ZONA_CRITICA, etc.

        """
        timestamp = datetime.now(ARGENTINA_TZ).strftime("%Y-%m-%d %H:%M:%S")
        
        row = [
            timestamp,

        ]

    def write_stats(self, total_trades, wins, losses, win_rate, avg_pnl, 
                   total_pnl, best_trade, worst_trade):
        """
        Escribe estadísticas en BRAIN_STATS
        
        Args:
            total_trades: Total de trades
            wins: Trades ganadores
            losses: Trades perdedores
            win_rate: Tasa de acierto (%)
            avg_pnl: PnL promedio (%)
            total_pnl: PnL total (USD)
            best_trade: Mejor trade (%)
            worst_trade: Peor trade (%)
        """

        row = [
            timestamp,
            total_trades,
            wins,
            losses,
            f"{win_rate:.2f}%",
            f"{avg_pnl:+.2f}%",
            f"${total_pnl:+.2f}",
            f"{best_trade:+.2f}%",
            f"{worst_trade:+.2f}%"
        ]
        
        return self.write_row(SHEET_BRAIN_STATS, row)

def fetch_ohlcv(symbol, timeframe='5m', limit=100):
    """
    Obtiene datos OHLCV para símbolo y timeframe
    
    Args:
        symbol: Símbolo (ej: 'BTC/USDT:USDT')
        timeframe: Timeframe (1d, 4h, 1h, 15m, 5m)
        limit: Número de velas
    
    Returns:
        list: OHLCV [[timestamp, open, high, low, close, volume], ...]
        None: Si error
    """
    try:
        ohlcv = exchange.fetch_ohlcv(symbol, timeframe, limit=limit)
        
        if not ohlcv or len(ohlcv) < 10:
            return None
            
        return ohlcv
        
    except Exception as e:
        print(f"❌ Error fetch_ohlcv {symbol} {timeframe}: {e}")
        return None


def fetch_markets():
    """
    Obtiene lista de mercados disponibles
    
    Returns:
        dict: Markets {symbol: market_info}
        None: Si error
    """
    try:
        markets = exchange.load_markets()
        return markets
    except Exception as e:
        print(f"❌ Error fetch_markets: {e}")
        return None


def get_futures_symbols():
    """
    Obtiene lista de símbolos futuros USDT
    
    Returns:
        list: ['BTC/USDT:USDT', 'ETH/USDT:USDT', ...]
        None: Si error
    """
    try:
        markets = fetch_markets()
        if not markets:
            return None
        
        # Filtrar solo futuros USDT
        symbols = [
            symbol for symbol, market in markets.items()
            if market.get('type') == 'swap' and symbol.endswith(':USDT')
        ]
        
        return sorted(symbols)
        
    except Exception as e:
        print(f"❌ Error get_futures_symbols: {e}")
        return None


def get_top_symbols_by_volume(limit=100):
    """
    Obtiene símbolos ordenados por volumen 24h (top movers)
    
    Args:
        limit: Cantidad de símbolos a retornar
    
    Returns:
        list: Símbolos ordenados por volumen descendente
    """
    try:
        tickers = exchange.fetch_tickers()
        
        # Filtrar y ordenar por volumen
        volume_list = []
        for symbol, ticker in tickers.items():
            if ':USDT' in symbol and ticker.get('quoteVolume'):
                volume_list.append((symbol, ticker['quoteVolume']))
        
        # Ordenar por volumen descendente
        volume_list.sort(key=lambda x: x[1], reverse=True)
        
        # Retornar top N
        return [symbol for symbol, _ in volume_list[:limit]]
        
    except Exception as e:
        print(f"❌ Error obteniendo top symbols: {e}")
        return []

def send_telegram(message):
    """
    Envía mensaje a Telegram
    
    Args:
        message: Texto del mensaje
    
    Returns:
        bool: True si éxito, False si error
    """
    if not ENABLE_TELEGRAM or not TELEGRAM_BOT_TOKEN or not TELEGRAM_CHAT_ID:
        return False
    
    try:
        url = f"https://api.telegram.org/bot{TELEGRAM_BOT_TOKEN}/sendMessage"
        payload = {
            'chat_id': TELEGRAM_CHAT_ID,
            'text': message,
            'parse_mode': 'HTML'
        }
        
        response = requests.post(url, data=payload, timeout=10)
        return response.status_code == 200
        
    except Exception as e:
        print(f"❌ Error Telegram: {e}")
        return False
