import numpy as np
import pandas as pd
from exchange_connection import fetch_ohlcv

class MarketPatternEngine:
    """Motor de detección de patrones reales basado en matemática"""
    
    def __init__(self):
        self.patterns = {
            'reversal': ['double_top', 'double_bottom', 'hch', 'wedge'],
            'continuation': ['flag', 'triangle', 'channel', 'pennant'],
            'structure': ['hh_hl', 'lh_ll', 'bos', 'choch']
        }

    def detect_all_patterns(self, symbol):
        """Detecta TODOS los patrones en múltiples TFs"""
        results = {
            '15m': self.scan_patterns(symbol, '15m'),
            '1h': self.scan_patterns(symbol, '1h'),
            '4h': self.scan_patterns(symbol, '4h')
        }
        
        # --- NUEVO: Detectar Momentum Drive (Fuerza Pura) ---
        mom_drive = self.detect_momentum_drive(symbol)
        if mom_drive['valid']:
            results['1h'].append(mom_drive)
        # ----------------------------------------------------

        # Confluencia de patrones
        confidence = self.calculate_pattern_confluence(results)
        
        return {
            'patterns': results,
            'best_pattern': self.get_strongest_pattern(results),
            'confidence': confidence,
            'action': self.pattern_to_action(results)
        }

    def detect_momentum_drive(self, symbol):
        """
        Detecta patrón de FUERZA PURA (Momentum Drive).
        Lógica Estricta:
        - ADX > 30 (Fuerza brutal)
        - ADX subiendo (Aceleración)
        - Spread DI > 10 (Claridad)
        - Momentum >= 5 velas (Sostenibilidad)
        """
        try:
            # Importación local para evitar dependencia circular
            from brain_calculator import calculate_adx_with_memory, calculate_momentum_with_memory
            
            # 1. Obtener datos 1H (50 velas para cálculo preciso)
            ohlcv_1h = fetch_ohlcv(symbol, '1h', 50)
            if not ohlcv_1h or len(ohlcv_1h) < 20:
                return {'valid': False}
            
            df = pd.DataFrame(ohlcv_1h, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
            
            # 2. Calcular Indicadores Reales
            adx_data = calculate_adx_with_memory(df, period=14, lookback=10)
            mom_data = calculate_momentum_with_memory(df, lookback=10)
            
            # 3. Extraer Valores
            adx_current = adx_data.get('adx_current', 0)
            adx_trend = adx_data.get('adx_trend', 'estable')
            di_plus = adx_data.get('di_plus_current', 0)
            di_minus = adx_data.get('di_minus_current', 0)
            consecutive = mom_data.get('consecutive_count', 0)
            di_spread = abs(di_plus - di_minus)
            
            # 4. CONDICIONES DE FUERZA BRUTAL
            adx_brutal = adx_current > 30
            adx_acelerando = adx_trend == 'creciendo'
            momentum_sostenido = consecutive >= 5
            spread_claro = di_spread > 10
            
            if adx_brutal and adx_acelerando and momentum_sostenido and spread_claro:
                direction = 'BULLISH' if di_plus > di_minus else 'BEARISH'
                action = 'BUY_CONTINUATION' if direction == 'BULLISH' else 'SELL_CONTINUATION'
                
                # Confianza dinámica (60 base + exceso sobre 30)
                confidence = min(95, 60 + int(adx_current - 30))
                
                return {
                    'valid': True,
                    'type': 'MOMENTUM_DRIVE',
                    'confidence': confidence,
                    'action': action,
                    'tf': '1h',
                    'details': f"ADX {adx_current:.1f} (Rising) | Spread {di_spread:.1f}"
                }
            
            return {'valid': False}
            
        except Exception as e:
            print(f"⚠️ Error detecting momentum drive for {symbol}: {e}")
            return {'valid': False}

    def scan_patterns(self, symbol, tf):
        """Escanea patrones específicos por TF"""
        # Usamos 200 velas para tener suficiente historia para patrones grandes
        ohlcv = fetch_ohlcv(symbol, tf, 200)
        if not ohlcv:
            return []
            
        df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
        patterns_found = []

        # 1. ESTRUCTURA DE MERCADO
        structure = self.detect_market_structure(df)
        if structure['valid']:
            patterns_found.append(structure)

        # 2. DOBLE TECHO/SUELO
        double = self.detect_double_patterns(df)
        if double['found']:
            patterns_found.append(double)

        # 3. TRIÁNGULOS
        triangle = self.detect_triangles(df)
        if triangle['type'] != 'none':
            patterns_found.append(triangle)

        # 4. FLAGS/PENNANTS
        flag = self.detect_flags(df)
        if flag['valid']:
            patterns_found.append(flag)

        return patterns_found

    def detect_market_structure(self, df):
        """HH/HL o LH/LL - La base de todo"""
        highs = []
        lows = []
        
        # Encontrar pivotes con lógica simple (high mayor a vecinos)
        # ZigZag simplificado
        for i in range(2, len(df)-2):
            if df['high'].iloc[i] > df['high'].iloc[i-1] and df['high'].iloc[i] > df['high'].iloc[i+1]:
                highs.append((i, df['high'].iloc[i]))
            if df['low'].iloc[i] < df['low'].iloc[i-1] and df['low'].iloc[i] < df['low'].iloc[i+1]:
                lows.append((i, df['low'].iloc[i]))
        
        if len(highs) < 2 or len(lows) < 2:
            return {'valid': False}

        # Detectar estructura reciente
        last_highs = [h[1] for h in highs[-2:]]
        last_lows = [l[1] for l in lows[-2:]]

        # UPTREND: HH + HL
        if last_highs[-1] > last_highs[-2] and last_lows[-1] > last_lows[-2]:
            return {
                'valid': True,
                'type': 'UPTREND',
                'strength': 90,
                'action': 'BUY_CONTINUATION'
            }
        # DOWNTREND: LH + LL
        elif last_highs[-1] < last_highs[-2] and last_lows[-1] < last_lows[-2]:
            return {
                'valid': True,
                'type': 'DOWNTREND',
                'strength': 90,
                'action': 'SELL_CONTINUATION'
            }
        # CAMBIO DE ESTRUCTURA (BOS) - Bullish
        elif last_lows[-1] > last_lows[-2] and last_highs[-1] > last_highs[-2]:
             # Esto es uptrend, BOS sería romper el último high válido previo en downtrend
             # Simplificación: Higher High después de Lower Highs
             return {
                'valid': True,
                'type': 'BOS_BULLISH',
                'strength': 85,
                'action': 'BUY_REVERSAL'
            }
            
        return {'valid': False}

    def detect_double_patterns(self, df):
        """Doble techo/suelo - Reversión clásica"""
        # Buscar dos máximos/mínimos similares en las últimas 50 velas
        recent_highs = df['high'][-50:].nlargest(5)
        recent_lows = df['low'][-50:].nsmallest(5)

        # Doble techo
        if len(recent_highs) >= 2:
            # Si la diferencia es menor al 0.5%
            if abs(recent_highs.iloc[0] - recent_highs.iloc[1]) / recent_highs.iloc[0] < 0.005:
                return {
                    'found': True,
                    'type': 'DOUBLE_TOP',
                    'price_level': recent_highs.iloc[0],
                    'confidence': 75,
                    'action': 'PREPARE_SHORT'
                }

        # Doble suelo
        if len(recent_lows) >= 2:
            if abs(recent_lows.iloc[0] - recent_lows.iloc[1]) / recent_lows.iloc[0] < 0.005:
                return {
                    'found': True,
                    'type': 'DOUBLE_BOTTOM',
                    'price_level': recent_lows.iloc[0],
                    'confidence': 75,
                    'action': 'PREPARE_LONG'
                }
        
        return {'found': False}

    def detect_triangles(self, df):
        """Triángulos - Los más confiables"""
        # Líneas de tendencia con regresión lineal en últimas 30 velas
        highs = df['high'][-30:]
        lows = df['low'][-30:]
        
        # Indices x para regresión
        x = np.arange(len(highs))
        
        # Pendientes (slope)
        high_slope = np.polyfit(x, highs, 1)[0]
        low_slope = np.polyfit(x, lows, 1)[0]

        # Triángulo ascendente: Highs planos (slope ~0), Lows subiendo (slope > 0)
        if abs(high_slope) < 0.001 and low_slope > 0.001:
            return {
                'type': 'ASCENDING_TRIANGLE',
                'breakout_level': highs.max(),
                'confidence': 85,
                'action': 'BUY_ON_BREAKOUT'
            }
        
        # Triángulo descendente: Lows planos (slope ~0), Highs bajando (slope < 0)
        elif abs(low_slope) < 0.001 and high_slope < -0.001:
            return {
                'type': 'DESCENDING_TRIANGLE',
                'breakdown_level': lows.min(),
                'confidence': 85,
                'action': 'SELL_ON_BREAKDOWN'
            }
            
        # Triángulo simétrico: Highs bajan, Lows suben
        elif high_slope < -0.001 and low_slope > 0.001:
            return {
                'type': 'SYMMETRIC_TRIANGLE',
                'confidence': 70,
                'action': 'WAIT_BREAKOUT'
            }
            
        return {'type': 'none'}

    def detect_flags(self, df):
        """Banderas - Continuación de tendencia"""
        # Detectar impulso fuerte seguido de consolidación
        # Últimas 20 velas vs las 20 anteriores
        last_20 = df[-20:]
        prev_20 = df[-40:-20]
        
        if len(prev_20) < 20:
            return {'valid': False}

        # Cambio de precio en los períodos
        impulse = (prev_20['close'].iloc[-1] - prev_20['close'].iloc[0]) / prev_20['close'].iloc[0]
        
        # Consolidación: Rango estrecho
        consolidation_range = (last_20['high'].max() - last_20['low'].min()) / last_20['close'].mean()

        # Flag alcista: Impulso fuerte positivo + consolidación estrecha
        if impulse > 0.05 and consolidation_range < 0.03: # 5% subida, 3% rango
            return {
                'valid': True,
                'type': 'BULL_FLAG',
                'confidence': 80,
                'target': last_20['close'].iloc[-1] * (1 + impulse), # Target proyectado
                'action': 'BUY_CONTINUATION'
            }
            
        # Flag bajista
        elif impulse < -0.05 and consolidation_range < 0.03:
            return {
                'valid': True,
                'type': 'BEAR_FLAG',
                'confidence': 80,
                'target': last_20['close'].iloc[-1] * (1 + impulse),
                'action': 'SELL_CONTINUATION'
            }
            
        return {'valid': False}

    def calculate_pattern_confluence(self, all_patterns):
        """Confluencia entre TFs"""
        score = 0
        bullish_score = 0
        bearish_score = 0
        
        weights = {'15m': 0.2, '1h': 0.4, '4h': 0.4}
        
        for tf, patterns in all_patterns.items():
            w = weights.get(tf, 0.2)
            for p in patterns:
                conf = p.get('confidence', 0) * w
                action = p.get('action', '')
                
                if 'BUY' in action or 'LONG' in action or 'UPTREND' in p.get('type', ''):
                    bullish_score += conf
                elif 'SELL' in action or 'SHORT' in action or 'DOWNTREND' in p.get('type', ''):
                    bearish_score += conf

        # Retornar dirección dominante
        if bullish_score > bearish_score * 1.5:
            return {'direction': 'BULLISH', 'strength': bullish_score}
        elif bearish_score > bullish_score * 1.5:
            return {'direction': 'BEARISH', 'strength': bearish_score}
        else:
            return {'direction': 'NEUTRAL', 'strength': 0}

    def get_strongest_pattern(self, all_patterns):
        """Retorna el patrón individual más fuerte encontrado"""
        best = None
        max_conf = 0
        
        for tf, patterns in all_patterns.items():
            for p in patterns:
                if p.get('confidence', 0) > max_conf:
                    max_conf = p.get('confidence')
                    best = p
                    best['tf'] = tf # Agregar TF donde se encontró
        
        if not best:
            return None
        
        return best

    def pattern_to_action(self, results):
        """Determina acción final basada en patrones"""
        best = self.get_strongest_pattern(results)
        if not best:
            return 'WAIT'
            
        action = best.get('action', 'WAIT')
        if 'BUY' in action or 'LONG' in action:
            return 'PREPARE_LONG'
        elif 'SELL' in action or 'SHORT' in action:
            return 'PREPARE_SHORT'
        return 'WAIT'
