def analyze_4h(symbol):
    """4H: Define contexto dominante"""
    # Fetch datos
    ohlcv = fetch_ohlcv(symbol, '4h', 100)
    if not ohlcv or len(ohlcv) < 50:
        return None  # Solo este caso rechaza (sin datos)
    
    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    current_price = float(df['close'].iloc[-1])
    
    # ===== ADX con memoria =====
    adx_data = calculate_adx_with_memory(df, period=14, lookback=10)
    
    # ADX ANTICIPATORIO: Proyectar evolución
    adx_now = adx_data['adx_current']
    adx_trend = adx_data['adx_trend']
    
    # Calcular velocidad ADX (puntos por vela)
    if len(adx_data['adx_history']) >= 5:
        adx_5_velas_atras = adx_data['adx_history'][-5]
        adx_velocity = (adx_now - adx_5_velas_atras) / 5
    else:
        adx_velocity = 0
    
    # Proyección 3 velas adelante (12 horas en 4H)
    adx_proyectado = adx_now + (adx_velocity * 3)
    
    # ===== Determinar dirección base por dominancia =====
    if adx_data['dominance'] == 'DI+':
        direction = 'BULLISH'
    elif adx_data['dominance'] == 'DI-':
        direction = 'BEARISH'
    else:
        direction = 'LATERAL'
        print(f"   ⚠️ 4H {symbol}: Dominancia neutral → LATERAL (DI+:{adx_data['di_plus_current']:.1f} DI-:{adx_data['di_minus_current']:.1f})")
    
    # ===== Evaluar fuerza ADX =====
    strength = 'WEAK'  # Inicializar en WEAK
    
    # CASO 1: ADX ya >25 (perfecto)
    if adx_now >= 25:
        if adx_now > 35:
            strength = 'STRONG'
        elif adx_now > 28:
            strength = 'MODERATE'
        else:
            strength = 'WEAK'
        print(f"   ✅ {symbol}: ADX {adx_now:.1f} (tendencia confirmada)")
    
    # CASO 2: ADX <25 pero subiendo rápido (ANTICIPAR)
    elif adx_now >= 17 and adx_velocity > 1.5:
        strength = 'MODERATE'  # Tendencia formándose
        print(f"   ⚡ ANTICIPANDO {symbol}: ADX {adx_now:.1f} subiendo {adx_velocity:+.1f}/vela → Proyección: {adx_proyectado:.1f}")
    
    # CASO 3: ADX bajo sin velocidad
    else:
        direction = 'LATERAL'  # Forzar LATERAL si ADX muy bajo
        strength = 'WEAK'
        print(f"   ⚠️ {symbol}: ADX bajo {adx_now:.1f} (vel:{adx_velocity:+.1f}/vela) → LATERAL WEAK")
    
    # Verificar caída pronunciada
    if adx_trend == 'decreciendo' and adx_velocity < -0.8:
        strength = 'WEAK'  # Degradar fuerza si cayendo rápido
        print(f"   ⚠️ {symbol}: ADX cayendo rápido ({adx_velocity:-.1f}/vela) → Fuerza degradada a WEAK")
    
    # ===== Verificar spread (ajusta fuerza, no rechaza) =====
    if direction != 'LATERAL' and abs(adx_data['spread']) < 5.0:
        if strength == 'STRONG':
            strength = 'MODERATE'
        elif strength == 'MODERATE':
            strength = 'WEAK'
        print(f"   ⚠️ {symbol}: Spread bajo ({abs(adx_data['spread']):.1f} < 5.0) → Fuerza ajustada a {strength}")
    
    # ===== Momentum =====
    momentum_data = calculate_momentum_with_memory(df, lookback=10)
    expected_color = 'green' if direction == 'BULLISH' else 'red'
    
    # Si momentum no coincide, ajustar dirección a LATERAL
    if direction != 'LATERAL' and momentum_data['color_current'] != expected_color:
        print(f"   ⚠️ {symbol}: Momentum {momentum_data['color_current']} ≠ {expected_color} → Ajustando a LATERAL")
        direction = 'LATERAL'
        strength = 'WEAK'
    
    # Si momentum débil, degradar fuerza
    if momentum_data['consecutive_count'] < 3:
        if strength == 'STRONG':
            strength = 'MODERATE'
        elif strength == 'MODERATE':
            strength = 'WEAK'
        print(f"   ⚠️ {symbol}: Momentum débil ({momentum_data['consecutive_count']} velas) → Fuerza ajustada a {strength}")
    
    # ===== RSI (solo para información, NO filtro) =====
    rsi_data = calculate_rsi_with_memory(df, period=14, lookback=10)
    
    # ===== EMAs =====
    emas_data = calculate_emas_with_memory(df, periods=[9, 21, 50, 200], lookback=10)
    
    # ===== Estructura HH/LL (INFORMATIVO) =====
    structure = detect_price_structure(df, lookback=20)
    
    # ===== Zig Zag (INFORMATIVO) =====
    zigzag = calculate_zigzag(df, threshold_pct=3.0)
    
    # ===== SCORING 4H (0-40 puntos) =====
    score_4h = 20  # Base
    
    # REGLA 1: Fuerza ADX (+15)
    if strength == 'STRONG':
        score_4h += 15
    elif strength == 'MODERATE':
        score_4h += 8
    
    # REGLA 2: Momentum fuerte (+5)
    if momentum_data['consecutive_count'] >= 5:
        score_4h += 5
    
    # Output compacto: una línea
    print(f"📊 4H | {symbol} | {direction} {strength} | Score: {score_4h}/40 | ADX:{adx_data['adx_current']:.1f} DI+:{adx_data['di_plus_current']:.1f} DI-:{adx_data['di_minus_current']:.1f} | Mom:{momentum_data['consecutive_count']}v | Estructura:{structure['structure']}")
    
    return {
        'direction': direction,
        'strength': strength,
        'score_4h': score_4h,
        'adx': adx_data,
        'momentum': momentum_data,
        'rsi': rsi_data,
        'emas': emas_data,
        'structure': structure,
        'zigzag': zigzag,
        'current_price': current_price,
        'context': f"4H {direction} {strength} - ADX {adx_data['adx_current']:.1f}",
        'timestamp': datetime.now().isoformat()
    }

def analyze_coin_complete(symbol):
    """
    Análisis COMPLETO multi-timeframe con contexto acumulativo
    
    Flow ACUMULATIVO:
    1. 4H debe aprobar (contexto dominante) → Si falla, STOP
    2. 1H debe confirmar o no contradecir → Si falla, STOP
    3. 15M debe alinearse (timing) → Si falla, STOP
    4. 5M info adicional (SIEMPRE ejecuta)
    
    Args:
        symbol (str): Par trading (ej: 'BTC/USDT')
    
    Returns:
        dict: {
            'viable': bool,
            'score_data': dict con score 0-100,
            'direction': str,
            'analysis_4h': dict,
            'analysis_1h': dict,
            'analysis_15m': dict,
            'analysis_5m': dict,
            'triggers': dict (entry, stop, target),
            'context_summary': str,
            'timestamp': str
        }
    """
    # ===== PASO 1: 4H - Contexto Dominante =====
    analysis_4h = analyze_4h(symbol)
    
    if not analysis_4h:
        return {
            'viable': False,
            'reason': '4H no cumple criterios estrictos',
            'timestamp': datetime.now().isoformat()
        }
    
    direction_dominant = analysis_4h['direction']
    
    # Si 4H es LATERAL, no viable para trading
    if direction_dominant == 'LATERAL':
        return {'viable': False, 'reason': '4H lateral - sin dirección clara'}
    
    # ===== PASO 2: 1H - Confirmación ADAPTATIVA =====
    analysis_1h = analyze_1h(symbol, analysis_4h)
    
    if not analysis_1h:
        return {
            'viable': False,
            'reason': '1H datos insuficientes',
            'analysis_4h': analysis_4h,
            'timestamp': datetime.now().isoformat()
        }
    
    # ===== PASO 3: 15M - Timing ADAPTATIVO =====
    analysis_15m = analyze_15m(symbol, analysis_4h, analysis_1h)
    
    if not analysis_15m:
        return {
            'viable': False,
            'reason': '15M datos insuficientes',
            'analysis_4h': analysis_4h,
            'analysis_1h': analysis_1h,
            'timestamp': datetime.now().isoformat()
        }
    
    # ===== PASO 4: 5M - Ejecución (SIEMPRE) =====
    analysis_5m = analyze_5m(symbol)
    
    # ===== PASO 5: Scoring TOTAL =====
    # Score = 4H (0-40) + 1H (0-30) + 15M (0-30) = 0-100
    score_total = (
        analysis_4h.get('score_4h', 0) +
        analysis_1h.get('score_1h', 0) +
        analysis_15m.get('score_15m', 0)
    )
    
    # Clasificación (RANGOS MÁS PERMISIVOS)
    if score_total >= 70:
        classification = 'ÓPTIMA'
    elif score_total >= 55:
        classification = 'BUENA'
    elif score_total >= 40:
        classification = 'REGULAR'
    else:
        classification = 'DÉBIL'
    
    # VIABLE si score >45 (MÁS PERMISIVO)
    viable = score_total > 45
    
    if not viable:
        return {
            'viable': False,
            'reason': f'Score {score_total}/100 < 45',
            'analysis_4h': analysis_4h,
            'analysis_1h': analysis_1h,
            'analysis_15m': analysis_15m,
            'timestamp': datetime.now().isoformat()
        }
    
    # ===== PASO 6: Calcular Triggers (Entry/Stop/Target) =====
    direction = analysis_4h['direction']
    triggers = calculate_triggers(analysis_15m, analysis_5m, direction)
    
    # ===== PASO 7: Datos SR (Support/Resistance) =====
    sr_zones = analysis_15m.get('sr', {
        'resistances': [],
        'supports': [],
        'nearest_resistance': (None, None),
        'nearest_support': (None, None)
    })
    
    # ===== PASO 8: Resumen Contextual =====
    context_summary = (
        f"{direction} setup | "
        f"4H:{analysis_4h['strength']} "

def calculate_triggers(analysis_15m, analysis_5m, direction):
    """
    Calcula precios exactos para entrada/stop/target
    
    ENTRY: Precio actual 5M
    STOP: -2.1% del entry (configurable)
    TARGET: Proyección matemática limitada por SR
    
    Proyección:
    1. Velocity + Acceleration del 15M
    2. Proyectar movimiento esperado en próximos 10-30 minutos
    3. Limitar por próxima resistencia (BULL) o soporte (BEAR)
    4. Si proyección >= SR cercano → usar SR * 0.98 (2% antes)
    
    Args:
        analysis_15m (dict): Análisis 15M con velocity/acceleration
        analysis_5m (dict): Análisis 5M con precio actual
        direction (str): 'BULLISH' o 'BEARISH'
    
    Returns:
        dict: {
            'entry_price': float,
            'stop_loss': float,
            'target_price': float,
            'rr_ratio': float,
            'tiempo_proyectado_min': int,
            'metodo_target': str
        }
    """
    # ===== Entry Price =====
    entry_price = analysis_5m.get('current_price', 0)
    
    if entry_price == 0:
        print("[TRIGGERS] ⚠️ No se pudo obtener precio actual")
        return {
            'entry_price': 0,
            'stop_loss': 0,
            'target_price': 0,
            'rr_ratio': 0,
            'tiempo_proyectado_min': 0,
            'metodo_target': 'ERROR - Sin precio'
        }
    
    # ===== Stop Loss =====
    stop_loss_pct = STOP_LOSS_PERCENT  # Desde brain_config
    
    if direction == 'BULLISH':
        stop_loss = entry_price * (1 - stop_loss_pct / 100)
    else:  # BEARISH
        stop_loss = entry_price * (1 + stop_loss_pct / 100)
    
    # ===== Target Price - Proyección combinada =====
    
    # Obtener velocity y acceleration del 15M
    velocity_15m = analysis_15m.get('velocity', {})
    acceleration_15m = analysis_15m.get('acceleration', {})
    
    velocity_value = velocity_15m.get('velocity_current', 0)  # % por minuto
    acceleration_value = acceleration_15m.get('acceleration_current', 0)
    
    # Tiempo proyectado (entre 10-30 minutos dependiendo de fuerza)
    if analysis_15m['strength'] == 'STRONG':
        tiempo_proyectado_min = 15
    elif analysis_15m['strength'] == 'MODERATE':
        tiempo_proyectado_min = 20
    else:
        tiempo_proyectado_min = 25
    
    # Proyección matemática
    # precio_futuro = precio_actual + (velocity * tiempo) + (0.5 * acceleration * tiempo^2)
    velocity_contribution = (velocity_value / 100) * entry_price * tiempo_proyectado_min
    acceleration_contribution = 0.5 * (acceleration_value / 100) * entry_price * (tiempo_proyectado_min ** 2)
    
    proyeccion_matematica = entry_price + velocity_contribution + acceleration_contribution
    
    # ===== Limitar por SR =====
    sr_15m = analysis_15m.get('sr', {})
    metodo_target = 'proyeccion_matematica'
    target_price = proyeccion_matematica
    
    if direction == 'BULLISH':
        # Buscar próxima resistencia
        resistances = sr_15m.get('resistances', [])
        resistances_above = [r for r in resistances if r > entry_price]
        
        if resistances_above:
            nearest_resistance = min(resistances_above)
            
            # Si proyección alcanza o supera resistencia → usar resistencia * 0.98
            if proyeccion_matematica >= nearest_resistance:
                target_price = nearest_resistance * 0.98  # 2% antes de resistencia
                metodo_target = f'resistencia_{nearest_resistance:.2f}_ajustada'
            else:
                target_price = proyeccion_matematica
                metodo_target = 'proyeccion_libre'
        else:
            # Sin resistencias detectadas → usar proyección pura
            target_price = proyeccion_matematica
            metodo_target = 'proyeccion_sin_SR'
    
    else:  # BEARISH
        # Buscar próximo soporte
        supports = sr_15m.get('supports', [])
        supports_below = [s for s in supports if s < entry_price]
        
        if supports_below:
            nearest_support = max(supports_below)
            
            # Si proyección alcanza o baja del soporte → usar soporte * 1.02
            if proyeccion_matematica <= nearest_support:
                target_price = nearest_support * 1.02  # 2% antes de soporte
                metodo_target = f'soporte_{nearest_support:.2f}_ajustado'
            else:
                target_price = proyeccion_matematica
                metodo_target = 'proyeccion_libre'
        else:
            target_price = proyeccion_matematica
            metodo_target = 'proyeccion_sin_SR'
    
    # ===== Risk:Reward Ratio =====
    if direction == 'BULLISH':
        risk = entry_price - stop_loss
        reward = target_price - entry_price
    else:  # BEARISH
        risk = stop_loss - entry_price
        reward = entry_price - target_price
    
    rr_ratio = reward / risk if risk > 0 else 0
    
    # ===== Validación mínima R:R =====
    if rr_ratio < 1.5:
        print(f"[TRIGGERS] ⚠️ R:R bajo ({rr_ratio:.2f}) - Ajustando target...")
        # Ajustar target para lograr R:R = 2.0 mínimo
        if direction == 'BULLISH':
            target_price = entry_price + (risk * 2.0)
        else:
            target_price = entry_price - (risk * 2.0)
        
        rr_ratio = 2.0
        metodo_target += '_ajustado_RR'
    
    return {
        'entry_price': round(entry_price, 4),
        'stop_loss': round(stop_loss, 4),
        'target_price': round(target_price, 4),
        'rr_ratio': round(rr_ratio, 2),
        'tiempo_proyectado_min': tiempo_proyectado_min,
        'metodo_target': metodo_target,
        'proyeccion_matematica': round(proyeccion_matematica, 4),
        'velocity_15m': round(velocity_value, 4),
        'acceleration_15m': round(acceleration_value, 6)
    }

def calculate_linear_regression_slope(values):
    """
    Calcula pendiente de regresión lineal simple
    
    Args:
        values: Lista de valores (últimas N velas)
        
    Returns:
        float: Pendiente (slope)
    """
    n = len(values)
    if n < 2:
        return 0.0
    
    x = np.arange(n)
    y = np.array(values)
    
    # y = mx + b → m = (n*Σxy - Σx*Σy) / (n*Σx² - (Σx)²)
    x_mean = np.mean(x)
    y_mean = np.mean(y)
    
    numerator = np.sum((x - x_mean) * (y - y_mean))
    denominator = np.sum((x - x_mean) ** 2)
    
    if denominator == 0:
        return 0.0
    
    slope = numerator / denominator
    return slope


def classify_trend(slope, threshold=0.3):
    """
    Clasifica tendencia según pendiente
    
    Args:
        slope: Pendiente calculada

    Returns:
        str: 'creciendo', 'decreciendo', 'estable'
    """
    if slope > threshold:
        return 'creciendo'
    elif slope < -threshold:
        return 'decreciendo'
    else:
        return 'estable'


def calculate_velocity(df, lookback=10):
    """
    Calcula velocidad de precio con memoria temporal
    
    Args:
        df: DataFrame con columna 'close'
        lookback: Velas a analizar
        
    Returns:
        dict: {
            'velocity_current': float (cambio precio por vela),
            'velocity_history': [list últimos lookback],
            'trend': 'acelerando'/'desacelerando'/'constante'
        }
    """
    if len(df) < lookback + 1:
        return {
            'velocity_current': 0.0,
            'velocity_history': [],
            'trend': 'constante'
        }
    
    prices = df['close'].values
    velocities = []
    
    # Calcular velocidad para cada par de velas
    for i in range(len(prices) - lookback, len(prices)):
        if i > 0:
            vel = prices[i] - prices[i-1]
            velocities.append(vel)
    
    velocity_current = velocities[-1] if velocities else 0.0
    
    # Calcular tendencia (velocidad creciendo/decreciendo)
    if len(velocities) >= 3:
        # Pendiente de las velocidades
        x = list(range(len(velocities)))
        y = velocities
        n = len(x)
        
        sum_x = sum(x)
        sum_y = sum(y)
        sum_xy = sum(x[i] * y[i] for i in range(n))
        sum_x2 = sum(x[i] ** 2 for i in range(n))
        
        slope = (n * sum_xy - sum_x * sum_y) / (n * sum_x2 - sum_x ** 2) if (n * sum_x2 - sum_x ** 2) != 0 else 0
        
        if slope > 0.0001:
            trend = 'acelerando'
        elif slope < -0.0001:
            trend = 'desacelerando'
        else:
            trend = 'constante'
    else:
        trend = 'constante'
    
    return {
        'velocity_current': velocity_current,
        'velocity_history': velocities,
        'trend': trend
    }


def calculate_acceleration(df, lookback=5):
    """
    Calcula aceleración (cambio de velocidad) con memoria temporal
    
    Args:

        lookback: Velas a analizar
        
    Returns:

    }

def calculate_adx_with_memory(df, period=14, lookback=10):
    """
    ADX con memoria temporal - Wilder's Smoothing EXACTO (manual)
    
    Args:
        df: DataFrame con columnas ['high', 'low', 'close']
        period: Período para ADX (default 14)
        lookback: Cuántos valores históricos guardar
        
    Returns:
        dict: {
            'adx_current': float,
            'adx_history': [list de últimos lookback valores],
            'adx_trend': 'creciendo'/'decreciendo'/'estable',
            'di_plus_current': float,
            'di_plus_history': [list],
            'di_plus_trend': str,
            'di_minus_current': float,
            'di_minus_history': [list],
            'di_minus_trend': str,
            'dominance': 'DI+'/'DI-'/'neutral',
            'spread': float (diferencia DI+ - DI-)
        }
    """
    
    df = df.copy()
    n = len(df)
    
    # Inicializar arrays
    tr = np.zeros(n)
    dm_plus = np.zeros(n)
    dm_minus = np.zeros(n)
    
    # 1. Calcular True Range y Directional Movements
    for i in range(1, n):
        high = df['high'].iloc[i]
        low = df['low'].iloc[i]
        prev_close = df['close'].iloc[i-1]
        prev_high = df['high'].iloc[i-1]
        prev_low = df['low'].iloc[i-1]
        
        # True Range = max(high-low, |high-prev_close|, |low-prev_close|)
        tr[i] = max(
            high - low,
            abs(high - prev_close),
            abs(low - prev_close)
        )
        
        # Directional Movements
        high_diff = high - prev_high
        low_diff = prev_low - low
        
        # DM+ si high_diff > low_diff Y high_diff > 0
        if high_diff > low_diff and high_diff > 0:
            dm_plus[i] = high_diff
        else:
            dm_plus[i] = 0
        
        # DM- si low_diff > high_diff Y low_diff > 0
        if low_diff > high_diff and low_diff > 0:
            dm_minus[i] = low_diff
        else:
            dm_minus[i] = 0
    
    # 2. Wilder's Smoothing (manual, NO pandas.ewm)
    smoothed_tr = np.zeros(n)
    smoothed_dm_plus = np.zeros(n)
    smoothed_dm_minus = np.zeros(n)
    
    # Primera suavización = suma de los primeros 'period' valores
    smoothed_tr[period] = np.sum(tr[1:period+1])
    smoothed_dm_plus[period] = np.sum(dm_plus[1:period+1])
    smoothed_dm_minus[period] = np.sum(dm_minus[1:period+1])
    
    # Wilder's smoothing: valor[i] = valor[i-1] - (valor[i-1]/period) + valor_actual[i]
    for i in range(period + 1, n):
        smoothed_tr[i] = smoothed_tr[i-1] - (smoothed_tr[i-1] / period) + tr[i]
        smoothed_dm_plus[i] = smoothed_dm_plus[i-1] - (smoothed_dm_plus[i-1] / period) + dm_plus[i]
        smoothed_dm_minus[i] = smoothed_dm_minus[i-1] - (smoothed_dm_minus[i-1] / period) + dm_minus[i]
    
    # 3. Calcular DI+ y DI-
    di_plus = np.zeros(n)
    di_minus = np.zeros(n)
    
    for i in range(period, n):
        if smoothed_tr[i] != 0:
            di_plus[i] = (smoothed_dm_plus[i] / smoothed_tr[i]) * 100
            di_minus[i] = (smoothed_dm_minus[i] / smoothed_tr[i]) * 100
    
    # 4. Calcular DX
    dx = np.zeros(n)
    for i in range(period, n):
        di_sum = di_plus[i] + di_minus[i]
        if di_sum != 0:
            dx[i] = (abs(di_plus[i] - di_minus[i]) / di_sum) * 100
    
    # 5. Calcular ADX (smoothing de DX)
    adx = np.zeros(n)
    adx[period * 2 - 1] = np.mean(dx[period:period * 2])
    
    for i in range(period * 2, n):
        adx[i] = ((adx[i-1] * (period - 1)) + dx[i]) / period
    
    # 6. Extraer últimos lookback valores
    start_idx = max(0, n - lookback)
    valid_start = max(period * 2, start_idx)
    
    adx_history = adx[valid_start:n].tolist()
    di_plus_history = di_plus[valid_start:n].tolist()
    di_minus_history = di_minus[valid_start:n].tolist()
    
    # 7. Calcular tendencias (pendiente lineal últimas 5 velas)
    adx_slope = calculate_linear_regression_slope(adx_history[-5:]) if len(adx_history) >= 5 else 0
    di_plus_slope = calculate_linear_regression_slope(di_plus_history[-5:]) if len(di_plus_history) >= 5 else 0
    di_minus_slope = calculate_linear_regression_slope(di_minus_history[-5:]) if len(di_minus_history) >= 5 else 0
    
    adx_trend = classify_trend(adx_slope)
    di_plus_trend = classify_trend(di_plus_slope)
    di_minus_trend = classify_trend(di_minus_slope)
    
    # 8. Valores actuales
    adx_current = adx[-1]
    di_plus_current = di_plus[-1]
    di_minus_current = di_minus[-1]
    
    # 9. Dominancia y spread
    spread = di_plus_current - di_minus_current
    if spread > 5:
        dominance = 'DI+'
    elif spread < -5:
        dominance = 'DI-'
    else:
        dominance = 'neutral'
    
    return {
        'adx_current': round(adx_current, 2),
        'adx_history': [round(x, 2) for x in adx_history],
        'adx_trend': adx_trend,
        'di_plus_current': round(di_plus_current, 2),
        'di_plus_history': [round(x, 2) for x in di_plus_history],
        'di_plus_trend': di_plus_trend,
        'di_minus_current': round(di_minus_current, 2),
        'di_minus_history': [round(x, 2) for x in di_minus_history],
        'di_minus_trend': di_minus_trend,
        'dominance': dominance,
        'spread': round(spread, 2)
    }


def calculate_momentum_with_memory(df, ema_period=34, lookback=10):
    """
    Momentum color con historia (LazyBear style)
    
    Args:
        df: DataFrame con columna 'close'
        ema_period: Período para EMA (default 34)
        lookback: Cuántos valores históricos guardar
        
    Returns:
        dict: {
            'color_current': 'green'/'red'/'neutral',
            'color_history': [list],
            'consecutive_count': int (velas consecutivas mismo color),
            'strength': 'fuerte'/'moderada'/'débil'
        }
    """
    
    df = df.copy()
    n = len(df)
    
    # Calcular EMA manualmente
    multiplier = 2 / (ema_period + 1)
    ema = np.zeros(n)
    ema[0] = df['close'].iloc[0]
    
    for i in range(1, n):
        ema[i] = (df['close'].iloc[i] * multiplier) + (ema[i-1] * (1 - multiplier))
    
    # Determinar color por vela (close vs EMA)
    colors = []
    for i in range(n):
        if df['close'].iloc[i] > ema[i]:
            colors.append('green')
        elif df['close'].iloc[i] < ema[i]:
            colors.append('red')
        else:
            colors.append('neutral')
    
    # Extraer últimos lookback colores
    start_idx = max(0, n - lookback)
    color_history = colors[start_idx:n]
    color_current = colors[-1]
    
    # Contar velas consecutivas mismo color (desde el final)
    consecutive_count = 1
    for i in range(len(color_history) - 2, -1, -1):
        if color_history[i] == color_current:
            consecutive_count += 1
        else:
            break
    
    # Determinar strength
    if consecutive_count >= 5:
        strength = 'fuerte'
    elif consecutive_count >= 3:
        strength = 'moderada'
    else:
        strength = 'débil'
    
    return {
        'color_current': color_current,
        'color_history': color_history,
        'consecutive_count': consecutive_count,
        'strength': strength
    }


def calculate_rsi_with_memory(df, period=14, lookback=10):
    """
    RSI con tendencia
    
    Args:
        df: DataFrame con columna 'close'
        period: Período para RSI (default 14)
        lookback: Cuántos valores históricos guardar
        
    Returns:
        dict: {
            'rsi_current': float,
            'rsi_history': [list],
            'zone': 'sobrecompra'/'sobreventa'/'neutral',
            'trend': 'subiendo'/'bajando'/'estable'
        }
    """
    
    df = df.copy()
    n = len(df)
    
    # Calcular cambios de precio
    deltas = df['close'].diff()
    
    gains = deltas.copy()
    losses = deltas.copy()
    
    gains[gains < 0] = 0
    losses[losses > 0] = 0
    losses = abs(losses)
    
    # Wilder's smoothing para average gain/loss
    avg_gain = np.zeros(n)
    avg_loss = np.zeros(n)
    
    # Primera media = promedio simple de primeros 'period' valores
    avg_gain[period] = gains.iloc[1:period+1].mean()
    avg_loss[period] = losses.iloc[1:period+1].mean()
    
    # Wilder's smoothing
    for i in range(period + 1, n):
        avg_gain[i] = (avg_gain[i-1] * (period - 1) + gains.iloc[i]) / period
        avg_loss[i] = (avg_loss[i-1] * (period - 1) + losses.iloc[i]) / period
    
    # Calcular RSI
    rsi = np.zeros(n)
    for i in range(period, n):
        if avg_loss[i] == 0:
            rsi[i] = 100
        else:
            rs = avg_gain[i] / avg_loss[i]
            rsi[i] = 100 - (100 / (1 + rs))
    
    # Extraer últimos lookback valores
    start_idx = max(period, n - lookback)
    rsi_history = rsi[start_idx:n].tolist()
    rsi_current = rsi[-1]
    
    # Calcular tendencia (pendiente lineal últimas 5 velas)
    rsi_slope = calculate_linear_regression_slope(rsi_history[-5:]) if len(rsi_history) >= 5 else 0
    
    if rsi_slope > 0.5:
        trend = 'subiendo'
    elif rsi_slope < -0.5:
        trend = 'bajando'
    else:
        trend = 'estable'
    
    # Determinar zona
    if rsi_current >= 70:
        zone = 'sobrecompra'
    elif rsi_current <= 30:
        zone = 'sobreventa'
    else:
        zone = 'neutral'
    
    return {
        'rsi_current': round(rsi_current, 2),
        'rsi_history': [round(x, 2) for x in rsi_history],
        'zone': zone,
        'trend': trend
    }


def calculate_emas_with_memory(df, periods=None, fast=10, slow=50, lookback=10):
    """
    EMAs con evolución - Soporta múltiples períodos o fast/slow tradicional
    
    Args:
        df: DataFrame con columna 'close'
        periods: Lista de períodos [9, 21, 50, 200] (si se proporciona, ignora fast/slow)
        fast: Período EMA rápida (default 10, usado si periods=None)
        slow: Período EMA lenta (default 50, usado si periods=None)
        lookback: Cuántos valores históricos guardar
        
    Returns:
        dict: {
            # Si periods proporcionado:
            'emas': {9: valor, 21: valor, ...},
            'emas_aligned': bool (todas ordenadas correctamente),
            'separation_pct': float (entre más rápida y más lenta),
            
            # Si fast/slow tradicional:
            'ema_fast_current': float,
            'ema_slow_current': float,
            'separation_pct': float,
            'separation_history': [list],
            'trend': 'alejándose'/'acercándose'/'estable',
            'classification': 'CERCANAS'/'MODERADAS'/'SEPARADAS'
        }
    """
    
    df = df.copy()
    n = len(df)
    
    # MODO 1: Múltiples períodos (usado por brain_analyzer)
    if periods is not None:
        emas_values = {}
        
        for period in periods:
            multiplier = 2 / (period + 1)
            ema = np.zeros(n)
            ema[0] = df['close'].iloc[0]
            
            for i in range(1, n):
                ema[i] = (df['close'].iloc[i] * multiplier) + (ema[i-1] * (1 - multiplier))
            
            emas_values[period] = round(ema[-1], 4)
        
        # Verificar alineación (orden correcto según tendencia)
        sorted_periods = sorted(periods)
        emas_aligned = True
        
        # Para BULLISH: EMA9 > EMA21 > EMA50 > EMA200
        for i in range(len(sorted_periods) - 1):
            if emas_values[sorted_periods[i]] <= emas_values[sorted_periods[i+1]]:
                emas_aligned = False
                break
        
        # Calcular separación entre más rápida y más lenta
        fastest = min(periods)
        slowest = max(periods)
        
        if emas_values[slowest] != 0:
            separation_pct = ((emas_values[fastest] - emas_values[slowest]) / emas_values[slowest]) * 100
        else:
            separation_pct = 0.0
        
        return {
            'emas': emas_values,
            'emas_aligned': emas_aligned,
            'separation_pct': round(separation_pct, 2)
        }
    
    # MODO 2: Fast/Slow tradicional (legacy)
    # Calcular EMA rápida
    multiplier_fast = 2 / (fast + 1)
    ema_fast = np.zeros(n)
    ema_fast[0] = df['close'].iloc[0]
    
    for i in range(1, n):
        ema_fast[i] = (df['close'].iloc[i] * multiplier_fast) + (ema_fast[i-1] * (1 - multiplier_fast))
    
    # Calcular EMA lenta
    multiplier_slow = 2 / (slow + 1)
    ema_slow = np.zeros(n)
    ema_slow[0] = df['close'].iloc[0]
    
    for i in range(1, n):
        ema_slow[i] = (df['close'].iloc[i] * multiplier_slow) + (ema_slow[i-1] * (1 - multiplier_slow))
    
    # Calcular separación porcentual
    separation_pct_array = np.zeros(n)
    for i in range(n):
        if ema_slow[i] != 0:
            separation_pct_array[i] = ((ema_fast[i] - ema_slow[i]) / ema_slow[i]) * 100
    
    # Extraer últimos lookback valores
    start_idx = max(0, n - lookback)
    separation_history = separation_pct_array[start_idx:n].tolist()
    
    ema_fast_current = ema_fast[-1]
    ema_slow_current = ema_slow[-1]
    separation_pct_current = separation_pct_array[-1]
    
    # Calcular tendencia (pendiente lineal últimas 5 velas)
    sep_slope = calculate_linear_regression_slope(separation_history[-5:]) if len(separation_history) >= 5 else 0
    
    if sep_slope > 0.05:
        trend = 'alejándose'
    elif sep_slope < -0.05:
        trend = 'acercándose'
    else:
        trend = 'estable'
    
    # Clasificación
    abs_sep = abs(separation_pct_current)
    if abs_sep < 1:
        classification = 'CERCANAS'
    elif abs_sep < 3:
        classification = 'MODERADAS'
    else:
        classification = 'SEPARADAS'
    
    return {
        'ema_fast_current': round(ema_fast_current, 4),
        'ema_slow_current': round(ema_slow_current, 4),
        'separation_pct': round(separation_pct_current, 2),
        'separation_history': [round(x, 2) for x in separation_history],
        'trend': trend,
        'classification': classification
    }

def detect_price_structure(df, lookback=20, pivot_window=5):
    """
    Detecta estructura de precio HH/HL (BULLISH) o LH/LL (BEARISH)
    
    Args:
        df: DataFrame con columnas ['high', 'low', 'close']
        lookback: Cuántas velas analizar
        pivot_window: Ventana para detectar pivots (máx/mín locales)
        
    Returns:
        dict: {
            'structure': 'BULLISH'/'BEARISH'/'LATERAL',
            'pivot_highs': [list de máximos],
            'pivot_lows': [list de mínimos],
            'strength': 'fuerte'/'moderada'/'débil'
        }
    """
    
    df = df.copy()
    n = len(df)
    start_idx = max(0, n - lookback)
    df_window = df.iloc[start_idx:n].reset_index(drop=True)
    
    # Encontrar pivots (máximos/mínimos locales)
    pivot_highs = []
    pivot_lows = []
    
    for i in range(pivot_window, len(df_window) - pivot_window):
        # Pivot High: high[i] es mayor que ventana antes y después
        is_pivot_high = True
        for j in range(i - pivot_window, i + pivot_window + 1):
            if j != i and df_window['high'].iloc[j] >= df_window['high'].iloc[i]:
                is_pivot_high = False
                break
        
        if is_pivot_high:
            pivot_highs.append({
                'index': i,
                'price': df_window['high'].iloc[i]
            })
        
        # Pivot Low: low[i] es menor que ventana antes y después
        is_pivot_low = True
        for j in range(i - pivot_window, i + pivot_window + 1):
            if j != i and df_window['low'].iloc[j] <= df_window['low'].iloc[i]:
                is_pivot_low = False
                break
        
        if is_pivot_low:
            pivot_lows.append({
                'index': i,
                'price': df_window['low'].iloc[i]
            })
    
    # Analizar estructura HH/HL vs LH/LL
    hh_count = 0  # Higher Highs
    lh_count = 0  # Lower Highs
    hl_count = 0  # Higher Lows
    ll_count = 0  # Lower Lows
    
    # Comparar pivots consecutivos
    for i in range(1, len(pivot_highs)):
        if pivot_highs[i]['price'] > pivot_highs[i-1]['price']:
            hh_count += 1
        else:
            lh_count += 1
    
    for i in range(1, len(pivot_lows)):
        if pivot_lows[i]['price'] > pivot_lows[i-1]['price']:
            hl_count += 1
        else:
            ll_count += 1
    
    # Determinar estructura
    bullish_score = hh_count + hl_count
    bearish_score = lh_count + ll_count
    
    if bullish_score > bearish_score and bullish_score >= 2:
        structure = 'BULLISH'
        strength_value = bullish_score
    elif bearish_score > bullish_score and bearish_score >= 2:
        structure = 'BEARISH'
        strength_value = bearish_score
    else:
        structure = 'LATERAL'
        strength_value = 0
    
    # Clasificar fuerza
    if strength_value >= 4:
        strength = 'fuerte'
    elif strength_value >= 2:
        strength = 'moderada'
    else:
        strength = 'débil'
    
    return {
        'structure': structure,
        'pivot_highs': [p['price'] for p in pivot_highs],
        'pivot_lows': [p['price'] for p in pivot_lows],
        'strength': strength
    }


def calculate_zigzag(df, threshold_pct=3.0):
    """
    Zig Zag para identificar tendencia claramente
    
    Args:
        df: DataFrame con columnas ['high', 'low', 'close']
        threshold_pct: Porcentaje mínimo de cambio para marcar swing
        
    Returns:
        dict: {
            'direction': 'up'/'down'/'sideways',
            'last_swing_high': float,
            'last_swing_low': float,
            'swings': [list de pivots significativos]
        }
    """
    
    df = df.copy()
    n = len(df)
    
    swings = []
    current_trend = None  # 'up' o 'down'
    last_swing_price = df['close'].iloc[0]
    last_swing_idx = 0
    
    for i in range(1, n):
        high = df['high'].iloc[i]
        low = df['low'].iloc[i]
        
        # Calcular cambio porcentual desde último swing
        change_from_swing = ((high - last_swing_price) / last_swing_price) * 100
        
        if current_trend is None:
            # Inicializar tendencia
            if change_from_swing >= threshold_pct:
                current_trend = 'up'
                swings.append({'type': 'low', 'price': last_swing_price, 'index': last_swing_idx})
                last_swing_price = high
                last_swing_idx = i
            elif change_from_swing <= -threshold_pct:
                current_trend = 'down'
                swings.append({'type': 'high', 'price': last_swing_price, 'index': last_swing_idx})
                last_swing_price = low
                last_swing_idx = i
        
        elif current_trend == 'up':
            # En tendencia alcista, buscamos nuevo máximo o reversión
            if high > last_swing_price:
                last_swing_price = high
                last_swing_idx = i
            
            change_down = ((low - last_swing_price) / last_swing_price) * 100
            if change_down <= -threshold_pct:
                # Reversión bajista
                swings.append({'type': 'high', 'price': last_swing_price, 'index': last_swing_idx})
                current_trend = 'down'
                last_swing_price = low
                last_swing_idx = i
        
        elif current_trend == 'down':
            # En tendencia bajista, buscamos nuevo mínimo o reversión
            if low < last_swing_price:
                last_swing_price = low
                last_swing_idx = i
            
            change_up = ((high - last_swing_price) / last_swing_price) * 100
            if change_up >= threshold_pct:
                # Reversión alcista
                swings.append({'type': 'low', 'price': last_swing_price, 'index': last_swing_idx})
                current_trend = 'up'
                last_swing_price = high
                last_swing_idx = i
    
    # Determinar dirección general
    if len(swings) < 2:
        direction = 'sideways'
        last_swing_high = df['high'].max()
        last_swing_low = df['low'].min()
    else:
        # Comparar últimos swings
        last_highs = [s['price'] for s in swings if s['type'] == 'high']
        last_lows = [s['price'] for s in swings if s['type'] == 'low']
        
        last_swing_high = last_highs[-1] if last_highs else df['high'].iloc[-1]
        last_swing_low = last_lows[-1] if last_lows else df['low'].iloc[-1]
        
        if len(last_highs) >= 2 and len(last_lows) >= 2:
            if last_highs[-1] > last_highs[-2] and last_lows[-1] > last_lows[-2]:
                direction = 'up'
            elif last_highs[-1] < last_highs[-2] and last_lows[-1] < last_lows[-2]:
                direction = 'down'
            else:
                direction = 'sideways'
        else:
            direction = current_trend if current_trend else 'sideways'
    
    return {
        'direction': direction,
        'last_swing_high': round(last_swing_high, 4),
        'last_swing_low': round(last_swing_low, 4),
        'swings': [{'type': s['type'], 'price': round(s['price'], 4)} for s in swings]
    }

def detect_support_resistance_with_memory(df, num_clusters=5, lookback=100):
    """
    SR con evolución temporal usando clustering de precios
    
    Args:
        df: DataFrame con columnas ['high', 'low', 'close']
        num_clusters: Número de niveles SR a detectar
        lookback: Cuántas velas analizar
        
    Returns:
        dict: {
            'resistances': [list ordenado descendente],
            'supports': [list ordenado ascendente],
            'nearest_resistance': (precio, distancia_pct),
            'nearest_support': (precio, distancia_pct)
        }
    """
    
    df = df.copy()
    n = len(df)
    start_idx = max(0, n - lookback)
    df_window = df.iloc[start_idx:n]
    
    # Recolectar todos los máximos y mínimos
    all_levels = []
    all_levels.extend(df_window['high'].tolist())
    all_levels.extend(df_window['low'].tolist())
    
    # Clustering simple por proximidad
    all_levels = sorted(set(all_levels))
    
    if len(all_levels) < num_clusters:
        clusters = all_levels
    else:
        # Dividir en clusters por rango
        min_price = min(all_levels)
        max_price = max(all_levels)
        step = (max_price - min_price) / num_clusters
        
        clusters = []
        for i in range(num_clusters):
            cluster_min = min_price + (i * step)
            cluster_max = min_price + ((i + 1) * step)
            
            # Encontrar niveles en este rango
            cluster_levels = [p for p in all_levels if cluster_min <= p < cluster_max]
            
            if cluster_levels:
                # Nivel del cluster = precio más tocado (moda aproximada)
                cluster_level = np.median(cluster_levels)
                clusters.append(cluster_level)
    
    # Separar en soportes y resistencias
    current_price = df['close'].iloc[-1]
    
    resistances = sorted([c for c in clusters if c > current_price], reverse=True)
    supports = sorted([c for c in clusters if c < current_price], reverse=True)
    
    # Nearest resistance/support
    if resistances:
        nearest_resistance = resistances[-1]  # La más cercana (menor)
        resistance_dist_pct = ((nearest_resistance - current_price) / current_price) * 100
    else:
        nearest_resistance = None
        resistance_dist_pct = None
    
    if supports:

def calculate_volume_profile_with_memory(df, num_levels=10, lookback=50):
    """
    Volume Profile con POC temporal (Point of Control)
    
    Args:
        df: DataFrame con columnas ['high', 'low', 'close', 'volume']
        num_levels: Número de niveles de precio para profile
        lookback: Cuántas velas analizar
        
    Returns:
        dict: {
            'poc_current': float,
            'poc_history': [list de POCs históricos],
            'poc_trend': 'subiendo'/'bajando'/'estable',
            'high_volume_nodes': [list de precios con alto volumen],
            'low_volume_nodes': [list de precios con bajo volumen]
        }
    """
    
    df = df.copy()
    n = len(df)
    
    # Calcular POC para múltiples ventanas (para obtener historia)
    poc_history = []
    window_size = 20  # Ventana para cada cálculo de POC
    
    for end_idx in range(window_size, n + 1, 5):  # Cada 5 velas
        start_idx = max(0, end_idx - lookback)
        df_window = df.iloc[start_idx:end_idx]
        
        # Crear niveles de precio
        min_price = df_window['low'].min()
        max_price = df_window['high'].max()
        
        if max_price == min_price:
            continue
        
        price_levels = np.linspace(min_price, max_price, num_levels)
        volume_per_level = np.zeros(num_levels)
        
        # Distribuir volumen por nivel
        for i, row in df_window.iterrows():
            candle_range = row['high'] - row['low']
            if candle_range == 0:
                continue
            
            # Asignar volumen a niveles que intersectan la vela
            for j, level in enumerate(price_levels):
                if row['low'] <= level <= row['high']:
                    volume_per_level[j] += row['volume'] / candle_range
        
        # POC = nivel con más volumen
        poc_idx = np.argmax(volume_per_level)
        poc = price_levels[poc_idx]
        poc_history.append(poc)
    
    # POC actual (última ventana)
    start_idx = max(0, n - lookback)
    df_current = df.iloc[start_idx:n]
    
    min_price = df_current['low'].min()
    max_price = df_current['high'].max()
    price_levels = np.linspace(min_price, max_price, num_levels)
    volume_per_level = np.zeros(num_levels)
    
    for i, row in df_current.iterrows():
        candle_range = row['high'] - row['low']
        if candle_range == 0:
            continue
        
        for j, level in enumerate(price_levels):
            if row['low'] <= level <= row['high']:
                volume_per_level[j] += row['volume'] / candle_range
    
    poc_current = price_levels[np.argmax(volume_per_level)]
    
    # Calcular tendencia del POC
    if len(poc_history) >= 5:
        poc_slope = calculate_linear_regression_slope(poc_history[-5:])
        poc_trend = classify_trend(poc_slope, threshold=0.0001)  # Threshold menor para precios
    else:
        poc_trend = 'estable'
    
    # High/Low volume nodes
    volume_threshold_high = np.percentile(volume_per_level, 70)
    volume_threshold_low = np.percentile(volume_per_level, 30)
    
    high_volume_nodes = [price_levels[i] for i in range(num_levels) if volume_per_level[i] >= volume_threshold_high]
    low_volume_nodes = [price_levels[i] for i in range(num_levels) if volume_per_level[i] <= volume_threshold_low]
    
    return {
        'poc_current': round(poc_current, 4),
        'poc_history': [round(p, 4) for p in poc_history[-10:]],  # Últimos 10
        'poc_trend': poc_trend,
        'high_volume_nodes': [round(h, 4) for h in high_volume_nodes],
        'low_volume_nodes': [round(l, 4) for l in low_volume_nodes]
    }


def calculate_fibonacci_levels(df, lookback=100):
    """
    Fibonacci basado en swing high/low recientes
    
    Args:
        df: DataFrame con columnas ['high', 'low', 'close']
        lookback: Cuántas velas analizar para encontrar swing
        
    Returns:
        dict: {
            'levels': {0.236: precio, 0.382: precio, ...},
            'nearest_level': (level_name, precio, distancia_pct),
            'swing_high': float,
            'swing_low': float
        }
    """
    
    df = df.copy()
    n = len(df)
    start_idx = max(0, n - lookback)
    df_window = df.iloc[start_idx:n]
    
    # Encontrar swing high/low
    swing_high = df_window['high'].max()
    swing_low = df_window['low'].min()
    
    # Determinar dirección (si estamos más cerca del high o low)
    current_price = df['close'].iloc[-1]
    
    # Calcular niveles de Fibonacci
    diff = swing_high - swing_low
    
    fib_ratios = {
        '0.0': swing_low,
        '0.236': swing_low + (diff * 0.236),
        '0.382': swing_low + (diff * 0.382),
        '0.5': swing_low + (diff * 0.5),
        '0.618': swing_low + (diff * 0.618),
        '0.786': swing_low + (diff * 0.786),
        '1.0': swing_high,
        '1.272': swing_high + (diff * 0.272),
        '1.618': swing_high + (diff * 0.618)
    }
    
    # Encontrar nivel más cercano
    min_distance = float('inf')
    nearest_level_name = None
    nearest_level_price = None
    
    for level_name, level_price in fib_ratios.items():
        distance = abs(current_price - level_price)
        if distance < min_distance:
            min_distance = distance
            nearest_level_name = level_name
            nearest_level_price = level_price
    
    distance_pct = (min_distance / current_price) * 100
    
    return {
        'levels': {k: round(v, 4) for k, v in fib_ratios.items()},
        'nearest_level': (nearest_level_name, round(nearest_level_price, 4), round(distance_pct, 2)),
        'swing_high': round(swing_high, 4),
        'swing_low': round(swing_low, 4)
    }

def find_nearest_resistance(current_price, resistances):
    """
    Encuentra resistencia más cercana por encima del precio actual
    
    Args:
        current_price: Precio actual
        resistances: Lista de niveles de resistencia
        
    Returns:
        tuple: (precio, distancia_pct) o None si no hay resistencias
    """

    # Filtrar solo resistencias por encima del precio actual
    above = [r for r in resistances if r > current_price]
    if not above:
        return None
    
    # Tomar la más cercana
    nearest = min(above)
    distance_pct = ((nearest - current_price) / current_price) * 100
    
    return (round(nearest, 4), round(distance_pct, 2))


def find_nearest_support(current_price, supports):
    """
    Encuentra soporte más cercano por debajo del precio actual
    
    Args:
        current_price: Precio actual
        supports: Lista de niveles de soporte
        
    Returns:
        tuple: (precio, distancia_pct) o None si no hay soportes
    """
    if not supports:
        return None
    
    # Filtrar solo soportes por debajo del precio actual
    below = [s for s in supports if s < current_price]
    if not below:
        return None
    
    # Tomar el más cercano
    nearest = max(below)
    distance_pct = ((current_price - nearest) / current_price) * 100
    
    return (round(nearest, 4), round(distance_pct, 2))


def calculate_time_to_target(current_price, target_price, velocity, candle_duration_min=5):
    """
    Proyecta tiempo para alcanzar objetivo basado en velocidad
    
    Args:
        current_price: Precio actual
        target_price: Precio objetivo
        velocity: Velocidad de movimiento (cambio por vela o dict)
        candle_duration_min: Duración de vela en minutos
        
    Returns:
        float: Minutos estimados o None si no alcanzable
    """
    # Extraer valores si vienen como dict
    if isinstance(velocity, dict):
        velocity = velocity.get('velocity_current', 0)
    
    # Extraer valor si candle_duration_min viene como dict
    if isinstance(candle_duration_min, dict):
        candle_duration_min = 5  # Default 5 minutos
    elif not isinstance(candle_duration_min, (int, float)):
        candle_duration_min = 5
    
    if velocity == 0:
        return None
    
    distance = target_price - current_price
    candles_needed = distance / velocity
    
    # Si necesita velas negativas, no es alcanzable
    if candles_needed < 0:
        return None
    
    return candles_needed * candle_duration_min

    def remove_from_watchlist(self, symbol, reason=""):
        """Remueve moneda de watchlist"""
        if symbol in self.watchlist:
            del self.watchlist[symbol]
            print(f"❌ Removido de watchlist: {symbol} ({reason})")
    
    def get_watchlist(self):
        """Retorna watchlist ordenada por score"""
        sorted_list = sorted(
            self.watchlist.items(),
            key=lambda x: x[1]['score'],
            reverse=True
        )
        return sorted_list
    
    def update_watchlist_from_analysis(self, symbols_analyzed):
        """
        Actualiza watchlist con nuevos análisis
        
        Args:
            symbols_analyzed: List de (symbol, analysis_data) tuples
        """
        added_count = 0
        
        for symbol, analysis_data in symbols_analyzed:
            if not analysis_data or not analysis_data.get('viable'):
                continue
            
            # Si ya está en watchlist, actualizar score
            if symbol in self.watchlist:
                old_score = self.watchlist[symbol]['score']
                new_score = analysis_data['score_data']['score']
                
                # Si score bajó mucho (>15 puntos), remover
                if new_score < old_score - 15:
                    self.remove_from_watchlist(symbol, "score bajó significativamente")
                else:
                    # Actualizar score
                    self.watchlist[symbol]['score'] = new_score
            
            # Si no está, agregar
            else:
                if self.add_to_watchlist(symbol, analysis_data):
                    added_count += 1
        
        print(f"📊 Watchlist actualizada: {len(self.watchlist)} monedas ({added_count} nuevas)")
        self.save_to_file()
    
    def cleanup_stale_symbols(self, max_age_minutes=30, max_checks=20):
        """
        Remueve monedas que llevan mucho tiempo sin activar trigger
        
        Args:
            max_age_minutes: Tiempo máximo en watchlist sin trigger (30min)
            max_checks: Verificaciones máximas sin activar (20 checks)
        
        Returns:
            int: Cantidad de monedas removidas
        """
        now = datetime.now(ARGENTINA_TZ)
        to_remove = []
        
        for symbol, data in self.watchlist.items():
            # Calcular tiempo en watchlist
            added = datetime.fromisoformat(data['added_timestamp'])
            age_minutes = (now - added).total_seconds() / 60
            
            # Calcular checks sin trigger
            checks = data.get('checks_without_trigger', 0)
            
            # Remover si lleva mucho tiempo O muchos checks sin trigger
            if age_minutes > max_age_minutes or checks > max_checks:
                to_remove.append(symbol)
                print(f"🗑️ Removiendo {symbol} de watchlist:")
                print(f"   Tiempo: {age_minutes:.1f}min | Checks sin trigger: {checks}")
        
        for symbol in to_remove:
            del self.watchlist[symbol]
        
        if to_remove:
            print(f"✅ Limpieza watchlist: {len(to_remove)} monedas removidas")
            self.save_to_file()
        
        return len(to_remove)
    
    def increment_check_counter(self, symbol):
        """Incrementa contador de verificaciones sin trigger"""
        if symbol in self.watchlist:
            self.watchlist[symbol]['checks_without_trigger'] = \
                self.watchlist[symbol].get('checks_without_trigger', 0) + 1
            self.watchlist[symbol]['last_check_timestamp'] = \
                datetime.now(ARGENTINA_TZ).isoformat()
    
    def reset_check_counter(self, symbol):
        """Resetea contador cuando trigger se activa (casi)"""
        if symbol in self.watchlist:
            self.watchlist[symbol]['checks_without_trigger'] = 0
    
    def clear_watchlist(self):
        """Limpia watchlist completa"""
        self.watchlist = {}
        self.save_to_file()
        print("🗑️ Watchlist limpiada")
