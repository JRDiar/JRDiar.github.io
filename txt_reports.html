"""
================================================================================
BRAIN ADX GRANULAR - Análisis Separado de Componentes ADX
================================================================================
Filosofía: Cada línea del ADX es un módulo independiente
- ADX (negro): Fuerza pura del mercado
- DI+ (verde): Fuerza alcista
- DI- (rojo): Fuerza bajista

Uso:
    from brain_adx_granular import analyze_adx_granular
    adx_data = analyze_adx_granular(df, period=14)
"""

import pandas as pd
import numpy as np
from typing import Dict


class ADXGranular:
    """
    Análisis granular del ADX separando cada componente.
    """
    
    def __init__(self, period: int = 14):
        self.period = period
    
    def calculate_granular_adx(self, df: pd.DataFrame) -> Dict:
        """
        Calcula ADX, DI+, DI- por separado con análisis individual.
        
        Returns:
            dict: {
                'adx': {...},
                'di_plus': {...},
                'di_minus': {...},
                'state': {...}
            }
        """
        # Calcular True Range
        high = df['high']
        low = df['low']
        close = df['close']
        
        tr1 = high - low
        tr2 = abs(high - close.shift(1))
        tr3 = abs(low - close.shift(1))
        tr = pd.DataFrame({'tr1': tr1, 'tr2': tr2, 'tr3': tr3}).max(axis=1)
        
        # Calcular movimientos direccionales
        high_diff = high.diff()
        low_diff = -low.diff()
        
        plus_dm = np.where((high_diff > low_diff) & (high_diff > 0), high_diff, 0)
        minus_dm = np.where((low_diff > high_diff) & (low_diff > 0), low_diff, 0)
        
        # Suavizar con ATR
        atr = tr.rolling(self.period).mean()
        
        # Calcular DI+ y DI-
        plus_di = 100 * (pd.Series(plus_dm).rolling(self.period).mean() / atr)
        minus_di = 100 * (pd.Series(minus_dm).rolling(self.period).mean() / atr)
        
        # Calcular ADX
        dx = 100 * abs(plus_di - minus_di) / (plus_di + minus_di)
        adx = dx.rolling(self.period).mean()
        
        # Análisis individual de cada componente
        return {
            'adx': self._analyze_adx_pure(adx),
            'di_plus': self._analyze_di_plus(plus_di),
            'di_minus': self._analyze_di_minus(minus_di),
            'state': self._analyze_combined_state(adx, plus_di, minus_di)
        }
    
    def _analyze_adx_pure(self, adx: pd.Series) -> Dict:
        """
        Analiza ADX puro (línea negra) - La fuerza del mercado.
        """
        current = adx.iloc[-1]
        prev_5 = adx.iloc[-5:-1].mean()
        
        # Tendencia del ADX
        if current > prev_5:
            trend = 'RISING'
            velocity = ((current - prev_5) / prev_5) * 100 if prev_5 > 0 else 0
        else:
            trend = 'FALLING'
            velocity = ((prev_5 - current) / prev_5) * 100 if prev_5 > 0 else 0
        
        # Clasificación de fuerza
        if current < 20:
            strength = 'WEAK'  # ❌ SIN FUERZA - PELIGRO
        elif current < 25:
            strength = 'MODERATE'
        elif current < 40:
            strength = 'STRONG'
        else:
            strength = 'VERY_STRONG'
        
        return {
            'adx_current': float(current),
            'adx_prev_5': float(prev_5),
            'trend': trend,
            'velocity': float(velocity),
            'strength': strength,
            'tradeable': current >= 20  # Mínimo para operar
        }
    
    def _analyze_di_plus(self, di_plus: pd.Series) -> Dict:
        """
        Analiza DI+ (línea verde) - Fuerza alcista.
        """
        current = di_plus.iloc[-1]
        prev_5 = di_plus.iloc[-5:-1].mean()
        
        # Tendencia
        trend = 'RISING' if current > prev_5 else 'FALLING'
        
        # Fuerza alcista
        if current > 25:
            strength = 'STRONG_BULLISH'
        elif current > 20:
            strength = 'MODERATE_BULLISH'
        else:
            strength = 'WEAK_BULLISH'
        
        return {
            'di_plus_current': float(current),
            'di_plus_prev_5': float(prev_5),
            'trend': trend,
            'strength': strength
        }
    
    def _analyze_di_minus(self, di_minus: pd.Series) -> Dict:
        """
        Analiza DI- (línea roja) - Fuerza bajista.
        """
        current = di_minus.iloc[-1]
        prev_5 = di_minus.iloc[-5:-1].mean()
        
        # Tendencia
        trend = 'RISING' if current > prev_5 else 'FALLING'
        
        # Fuerza bajista
        if current > 25:
            strength = 'STRONG_BEARISH'
        elif current > 20:
            strength = 'MODERATE_BEARISH'
        else:
            strength = 'WEAK_BEARISH'
        
        return {
            'di_minus_current': float(current),
            'di_minus_prev_5': float(prev_5),
            'trend': trend,
            'strength': strength
        }
    
    def _analyze_combined_state(self, adx: pd.Series, di_plus: pd.Series, di_minus: pd.Series) -> Dict:
        """
        Analiza el estado combinado para detectar lateralización.
        
        CRÍTICO: Detecta cuando DI+ y DI- están superpuestos = LATERAL
        """
        adx_current = adx.iloc[-1]
        di_plus_current = di_plus.iloc[-1]
        di_minus_current = di_minus.iloc[-1]
        
        # Calcular spread entre DI+ y DI-
        spread = abs(di_plus_current - di_minus_current)
        
        # Dominio
        if di_plus_current > di_minus_current:
            dominant = 'BULLISH'
            dominant_strength = di_plus_current
        else:
            dominant = 'BEARISH'
            dominant_strength = di_minus_current
        
        # DETECCIÓN DE LATERALIZACIÓN (CLAVE)
        is_lateral = False
        lateral_reasons = []
        
        # Condición 1: ADX bajo (<20)
        if adx_current < 20:
            is_lateral = True
            lateral_reasons.append(f"ADX bajo ({adx_current:.1f} < 20)")
        
        # Condición 2: Spread pequeño (<5) - DI+/DI- superpuestos
        if spread < 5:
            is_lateral = True
            lateral_reasons.append(f"DI+/DI- superpuestos (spread {spread:.1f} < 5)")
        
        # Condición 3: ADX cayendo mientras precio lateral
        if adx_current < adx.iloc[-5:-1].mean():
            lateral_reasons.append("ADX cayendo (pérdida de fuerza)")
        
        # Estado del mercado
        if is_lateral:
            market_state = 'LATERAL'
            tradeable = False
        elif adx_current >= 25 and spread >= 10:
            market_state = 'TRENDING'
            tradeable = True
        else:
            market_state = 'TRANSITION'
            tradeable = False
        
        return {
            'market_state': market_state,
            'is_lateral': is_lateral,
            'lateral_reasons': lateral_reasons,
            'dominant_direction': dominant,
            'dominant_strength': float(dominant_strength),
            'spread': float(spread),
            'tradeable': tradeable
        }


# Función helper para usar en analyzers
def analyze_adx_granular(df: pd.DataFrame, period: int = 14) -> Dict:
    """
    Wrapper para usar en brain_analyzer.py
    
    Args:
        df: DataFrame con columnas ['high', 'low', 'close']
        period: Período del ADX (default 14)
    
    Returns:
        dict: Análisis granular completo del ADX
    """
    adx_granular = ADXGranular(period)
    return adx_granular.calculate_granular_adx(df)

"""
================================================================================
BRAIN MOMENTUM GRANULAR - Análisis Separado de Momentum
================================================================================
Filosofía: Separar componentes del LazyBear Momentum
- Squeeze: Estado de compresión
- Momentum Bar: Dirección e intensidad
- Color: Verde/Rojo/Gris

Uso:
    from brain_momentum_granular import analyze_momentum_granular
    momentum_data = analyze_momentum_granular(df)
"""

import pandas as pd
import numpy as np
from typing import Dict


class MomentumGranular:
    """
    Análisis granular del momentum LazyBear.
    """
    
    def __init__(self, bb_length: int = 20, kc_length: int = 20):
        self.bb_length = bb_length
        self.kc_length = kc_length
    
    def calculate_granular_momentum(self, df: pd.DataFrame) -> Dict:
        """
        Calcula momentum con análisis granular.
        
        Returns:
            dict: {
                'squeeze_state': {...},
                'momentum_bar': {...},
                'color_state': {...},
                'combined': {...}
            }
        """
        close = df['close']
        high = df['high']
        low = df['low']
        
        # Bollinger Bands
        bb_basis = close.rolling(self.bb_length).mean()
        bb_std = close.rolling(self.bb_length).std()
        bb_upper = bb_basis + (2 * bb_std)
        bb_lower = bb_basis - (2 * bb_std)
        
        # Keltner Channels
        kc_basis = close.rolling(self.kc_length).mean()
        tr = pd.DataFrame({
            'hl': high - low,
            'hc': abs(high - close.shift(1)),
            'lc': abs(low - close.shift(1))
        }).max(axis=1)
        kc_range = tr.rolling(self.kc_length).mean() * 1.5
        kc_upper = kc_basis + kc_range
        kc_lower = kc_basis - kc_range
        
        # Squeeze detection
        squeeze_on = (bb_lower > kc_lower) & (bb_upper < kc_upper)
        
        # Momentum calculation
        highest = high.rolling(self.kc_length).max()
        lowest = low.rolling(self.kc_length).min()
        avg_hl = (highest + lowest) / 2
        momentum = close - avg_hl
        
        return {
            'squeeze_state': self._analyze_squeeze(squeeze_on),
            'momentum_bar': self._analyze_momentum_bar(momentum),
            'color_state': self._analyze_color(momentum),
            'combined': self._analyze_combined(squeeze_on, momentum)
        }
    
    def _analyze_squeeze(self, squeeze_on: pd.Series) -> Dict:
        """
        Analiza el estado de squeeze (compresión).
        """
        current_squeeze = bool(squeeze_on.iloc[-1])
        
        # Contar velas consecutivas en squeeze
        consecutive_squeeze = 0
        for val in reversed(squeeze_on.tail(20).values):
            if val:
                consecutive_squeeze += 1
            else:
                break
        
        return {
            'is_squeezed': current_squeeze,
            'consecutive_candles': consecutive_squeeze,
            'state': 'COMPRESSED' if current_squeeze else 'RELEASED'
        }
    
    def _analyze_momentum_bar(self, momentum: pd.Series) -> Dict:
        """
        Analiza la barra de momentum (intensidad y dirección).
        """
        current = momentum.iloc[-1]
        prev = momentum.iloc[-2]
        avg_5 = momentum.iloc[-5:].mean()
        
        # Dirección
        direction = 'BULLISH' if current > 0 else 'BEARISH'
        
        # Intensidad
        intensity = abs(current)
        if intensity > abs(avg_5) * 1.5:
            strength = 'STRONG'
        elif intensity > abs(avg_5):
            strength = 'MODERATE'
        else:
            strength = 'WEAK'
        
        # Tendencia
        if current > prev:
            trend = 'RISING'
        else:
            trend = 'FALLING'
        
        return {
            'momentum_current': float(current),
            'direction': direction,
            'strength': strength,
            'trend': trend,
            'intensity': float(intensity)
        }
    
    def _analyze_color(self, momentum: pd.Series) -> Dict:
        """
        Analiza el color del momentum (verde/rojo/gris).
        """
        current = momentum.iloc[-1]
        prev = momentum.iloc[-2]
        
        # Determinar color
        if current > 0:
            if current > prev:
                color = 'lime'  # Verde claro
            else:
                color = 'green'  # Verde oscuro
        else:
            if current < prev:
                color = 'maroon'  # Rojo oscuro
            else:
                color = 'red'  # Rojo claro
        
        # Contar consecutivos del mismo color
        consecutive_count = 1
        for i in range(2, min(11, len(momentum))):
            curr_mom = momentum.iloc[-i]
            
            # Mismo color?
            same_color = False
            if current > 0 and curr_mom > 0:
                same_color = True
            elif current < 0 and curr_mom < 0:
                same_color = True
            
            if same_color:
                consecutive_count += 1
            else:
                break
        
        return {
            'color_current': color,
            'consecutive_count': consecutive_count,
            'is_mature': consecutive_count >= 3
        }
    
    def _analyze_combined(self, squeeze_on: pd.Series, momentum: pd.Series) -> Dict:
        """
        Análisis combinado para detectar condiciones de NO operación.
        """
        current_squeeze = bool(squeeze_on.iloc[-1])
        current_momentum = momentum.iloc[-1]
        momentum_avg = abs(momentum.iloc[-10:].mean())
        
        # DETECCIÓN DE MOMENTUM PLANO (LATERAL)
        is_flat = False
        flat_reasons = []
        
        # Condición 1: Momentum muy bajo (cerca de 0)
        if abs(current_momentum) < momentum_avg * 0.3:
            is_flat = True
            flat_reasons.append(f"Momentum débil ({abs(current_momentum):.4f})")
        
        # Condición 2: Squeeze activo por mucho tiempo
        consecutive_squeeze = 0
        for val in reversed(squeeze_on.tail(10).values):
            if val:
                consecutive_squeeze += 1
            else:
                break
        
        if consecutive_squeeze > 5:
            is_flat = True
            flat_reasons.append(f"Squeeze prolongado ({consecutive_squeeze} velas)")
        
        # Estado del momentum
        if is_flat:
            state = 'FLAT'
            tradeable = False
        elif abs(current_momentum) > momentum_avg * 1.5:
            state = 'STRONG'
            tradeable = True
        else:
            state = 'MODERATE'
            tradeable = abs(current_momentum) > momentum_avg
        
        return {
            'state': state,
            'is_flat': is_flat,
            'flat_reasons': flat_reasons,
            'tradeable': tradeable
        }

"""
================================================================================
BRAIN ZIGZAG TTI - ZigZag usando Technical Trading Indicators
================================================================================
Reemplaza el cálculo manual de pandas por TTI library

INSTALACIÓN REQUERIDA:
    pip install tti

Uso:
    from brain_zigzag_tti import analyze_zigzag_tti
    zigzag_data = analyze_zigzag_tti(df, threshold=5.0)
"""

import pandas as pd
import numpy as np
from typing import Dict, List

try:
    from tti.indicators import ZigZag
    TTI_AVAILABLE = True
except ImportError:
    TTI_AVAILABLE = False
    print("⚠️ TTI library not installed. Run: pip install tti")


class ZigZagTTI:
    """
    ZigZag usando TTI library para mayor precisión.
    """
    
    def __init__(self, threshold: float = 5.0):
        """
        Args:
            threshold: Umbral de cambio porcentual mínimo (default 5%)
        """
        self.threshold = threshold
    
    def calculate_zigzag(self, df: pd.DataFrame) -> Dict:
        """
        Calcula ZigZag usando TTI library.
        
        Returns:
            dict: {
                'pivots': [...],
                'pivot_types': [...],
                'current_trend': str,
                'structure': str
            }
        """
        if not TTI_AVAILABLE:
            return self._fallback_zigzag(df)
        
        try:
            # Calcular ZigZag con TTI
            zz = ZigZag(
                input_data=df,
                threshold=self.threshold,
                fill_values_method='zeros'
            )
            
            zigzag_values = zz.getTiData()
            
            # Encontrar pivots (donde zigzag != 0)
            pivots = []
            pivot_indices = []
            
            for i, val in enumerate(zigzag_values):
                if val != 0:
                    pivots.append(float(val))
                    pivot_indices.append(i)
            
            # Clasificar pivots (HH, HL, LH, LL)
            pivot_types = self._classify_pivots(pivots)
            
            # Analizar estructura
            structure_analysis = self._analyze_structure(pivot_types)
            
            return {
                'pivots': pivots,
                'pivot_indices': pivot_indices,
                'pivot_types': pivot_types,
                'current_trend': structure_analysis['trend'],
                'structure': structure_analysis['structure'],
                'last_pivot_type': pivot_types[-1] if pivot_types else 'UNKNOWN',
                'valid': len(pivots) >= 3
            }
            
        except Exception as e:
            print(f"⚠️ Error calculating ZigZag with TTI: {e}")
            return self._fallback_zigzag(df)
    
    def _classify_pivots(self, pivots: List[float]) -> List[str]:
        """
        Clasifica pivots en HH, HL, LH, LL.
        """
        if len(pivots) < 3:
            return []
        
        pivot_types = []
        
        for i in range(2, len(pivots)):
            prev_prev = pivots[i-2]
            prev = pivots[i-1]
            current = pivots[i]
            
            # Determinar si es High o Low
            is_high = prev > prev_prev and prev > current
            is_low = prev < prev_prev and prev < current
            
            if is_high:
                # Comparar con el high anterior
                prev_highs = [p for j, p in enumerate(pivots[:i]) if j % 2 == (i-1) % 2]
                if prev_highs:
                    last_high = prev_highs[-1]
                    if prev > last_high:
                        pivot_types.append('HH')  # Higher High
                    else:
                        pivot_types.append('LH')  # Lower High
                else:
                    pivot_types.append('H')
                    
            elif is_low:
                # Comparar con el low anterior
                prev_lows = [p for j, p in enumerate(pivots[:i]) if j % 2 == (i-1) % 2]
                if prev_lows:
                    last_low = prev_lows[-1]
                    if prev > last_low:
                        pivot_types.append('HL')  # Higher Low
                    else:
                        pivot_types.append('LL')  # Lower Low
                else:
                    pivot_types.append('L')
        
        return pivot_types
    
    def _analyze_structure(self, pivot_types: List[str]) -> Dict:
        """
        Analiza la estructura del mercado basada en pivots.
        """
        if not pivot_types or len(pivot_types) < 2:
            return {'trend': 'UNKNOWN', 'structure': 'INSUFFICIENT_DATA'}
        
        # Últimos 3 pivots
        recent_pivots = pivot_types[-3:] if len(pivot_types) >= 3 else pivot_types
        
        # Detectar tendencia
        if all(p in ['HH', 'HL'] for p in recent_pivots):
            trend = 'BULLISH'
            structure = 'UPTREND'
        elif all(p in ['LH', 'LL'] for p in recent_pivots):
            trend = 'BEARISH'
            structure = 'DOWNTREND'
        elif 'HH' in recent_pivots and 'LL' in recent_pivots:
            trend = 'NEUTRAL'
            structure = 'CHOPPY'
        else:
            trend = 'NEUTRAL'
            structure = 'LATERAL'
        
        return {
            'trend': trend,
            'structure': structure
        }
    
    def _fallback_zigzag(self, df: pd.DataFrame) -> Dict:
        """
        Fallback si TTI falla o no está instalado.
        Usa cálculo manual simple.
        """
        high = df['high'].values
        low = df['low'].values
        close = df['close'].values
        
        threshold_pct = self.threshold / 100
        
        pivots = []
        pivot_indices = []
        
        # Algoritmo simple de ZigZag
        direction = 0  # 1 = up, -1 = down
        last_pivot_price = close[0]
        last_pivot_idx = 0
        
        for i in range(1, len(close)):
            if direction == 0:
                # Inicializar dirección
                if close[i] > last_pivot_price * (1 + threshold_pct):
                    direction = 1
                    pivots.append(last_pivot_price)
                    pivot_indices.append(last_pivot_idx)
                elif close[i] < last_pivot_price * (1 - threshold_pct):
                    direction = -1
                    pivots.append(last_pivot_price)
                    pivot_indices.append(last_pivot_idx)
                    
            elif direction == 1:
                # Tendencia alcista
                if high[i] > last_pivot_price:
                    last_pivot_price = high[i]
                    last_pivot_idx = i
                elif close[i] < last_pivot_price * (1 - threshold_pct):
                    # Cambio de dirección
                    pivots.append(last_pivot_price)
                    pivot_indices.append(last_pivot_idx)
                    direction = -1
                    last_pivot_price = low[i]
                    last_pivot_idx = i
                    
            elif direction == -1:
                # Tendencia bajista
                if low[i] < last_pivot_price:
                    last_pivot_price = low[i]
                    last_pivot_idx = i
                elif close[i] > last_pivot_price * (1 + threshold_pct):
                    # Cambio de dirección
                    pivots.append(last_pivot_price)
                    pivot_indices.append(last_pivot_idx)
                    direction = 1
                    last_pivot_price = high[i]
                    last_pivot_idx = i
        
        # Agregar último pivot
        if pivots:
            pivots.append(last_pivot_price)
            pivot_indices.append(last_pivot_idx)
        
        pivot_types = self._classify_pivots(pivots)
        structure_analysis = self._analyze_structure(pivot_types)
        
        return {
            'pivots': pivots,
            'pivot_indices': pivot_indices,
            'pivot_types': pivot_types,
            'current_trend': structure_analysis['trend'],
            'structure': structure_analysis['structure'],
            'last_pivot_type': pivot_types[-1] if pivot_types else 'UNKNOWN',
            'valid': len(pivots) >= 3,
            'method': 'FALLBACK'
        }
