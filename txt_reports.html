    # PASO 1: Verificar FUERZA mínima (Gatekeeper)

        ready = False

        print(f"  READY: 🔴 NO (Sin fuerza)")
        
    elif total_score < 50:
         # FUERZA PERO SCORE MUY BAJO: Rechazar
        ready = False
        summary = f"❌ REJECTED: Low Quality Setup (Score {total_score} < 50)"
        print(f"  READY: 🔴 NO (Low score)")

        # HAY FUERZA Y CALIDAD DECENTE -> Evaluar Timing

                missing_reasons.append("Low confidence")
