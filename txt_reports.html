def _validate_explosion_entry(self, symbol, direction, force_result, ladder_data):
    """
    v5.2.1: TIME LADDER - Retorna score + can_enter
    """
    from brain_config import (
        MIN_DIST_BLOCK, MIN_DIST_PENALTY, 
        SPACE_BASE_PENALTY, VELOCITY_PENALTY_REDUCTION_MAX
    )
    
    data_15m = ladder_data.get('15m', {})
    current_price = data_15m.get('current_price', 0)
    
    if current_price == 0:
        return {'can_enter': False, 'score': 0, 'reason': 'No price'}
    
    # Obtener velocity del force_result
    velocity_pct = abs(force_result.get('velocity', 0))
    
    # Calcular factor de reducción (0 a 0.70)
    velocity_factor = min(velocity_pct / 0.30, VELOCITY_PENALTY_REDUCTION_MAX)
    
    score = 100  # Score base
    blocks = []
    
    # === OBTENER DATOS DE ZONAS ===
    vp_data = data_15m.get('volume_profile', {})
    vah = vp_data.get('value_area_high', 0)
    val = vp_data.get('value_area_low', 0)
    
    # === EVALUAR ESPACIO - BULLISH ===
    if direction == 'BULLISH' and vah > 0 and current_price < vah:
        dist_pct = ((vah - current_price) / current_price) * 100
        
        if dist_pct < MIN_DIST_BLOCK:  # < 0.20%
            blocks.append(f'EN resistencia VAH ({dist_pct:.2f}%)')
        elif dist_pct < MIN_DIST_PENALTY:  # < 0.50%
            # Penalty gradual reducida por velocity
            actual_penalty = SPACE_BASE_PENALTY * (1 - velocity_factor)
            score -= actual_penalty
            print(f"   📏 Dist VAH: {dist_pct:.2f}% | Vel: {velocity_pct:.2f}%")
            print(f"   📉 Penalty: -{actual_penalty:.0f} (base {SPACE_BASE_PENALTY} x {1-velocity_factor:.0%})")
    
    # === EVALUAR ESPACIO - BEARISH ===
    elif direction == 'BEARISH' and val > 0 and current_price > val:
        dist_pct = ((current_price - val) / current_price) * 100
        
        if dist_pct < MIN_DIST_BLOCK:
            blocks.append(f'EN soporte VAL ({dist_pct:.2f}%)')
        elif dist_pct < MIN_DIST_PENALTY:
            actual_penalty = SPACE_BASE_PENALTY * (1 - velocity_factor)
            score -= actual_penalty
    
    # === EVALUAR DIAGONAL (si existe) ===
    grid_data = data_15m.get('grid', {})
    diagonal = grid_data.get('diagonal', {})
    
    if diagonal.get('valid'):
        if direction == 'BULLISH':
            res_diag = diagonal.get('resistencia_diagonal', 0)
            if res_diag > current_price:
                dist_diag = ((res_diag - current_price) / current_price) * 100
                if dist_diag < MIN_DIST_BLOCK:
                    blocks.append(f'EN diagonal ({dist_diag:.2f}%)')
                elif dist_diag < MIN_DIST_PENALTY:
                    penalty = SPACE_BASE_PENALTY * 0.7 * (1 - velocity_factor)
                    score -= penalty
    
    # === DECISIÓN FINAL ===
    if blocks:
        return {
            'can_enter': False, 
            'score': 0, 
            'reason': f"BLOCK: {blocks[0]}",
            'validation_type': 'HARD_BLOCK'
        }
    
    score = max(0, min(100, score))  # Clamp 0-100
    
    return {
        'can_enter': score >= 60,  # Threshold interno
        'score': score,
        'reason': f'Validator Score: {score:.0f}',
        'velocity_factor': velocity_factor,
        'validation_type': 'TIME_LADDER'
    }

@staticmethod
def is_perfect_setup(symbol: str, score: float, decision: dict) -> bool:
    """
    v5.2.1: Time Ladder con scores combinados
    """
    from brain_config import (
        ENABLE_MOTOR_VELOCITY, ENABLE_MOTOR_CHANNEL,
        SCORE_WEIGHT_ANALYZER, SCORE_WEIGHT_VALIDATOR,
        SCORE_THRESHOLD_ENTER
    )
    from brain_frames import check_price_velocity
    from brain_entry_validator import entry_validator
    
    full_analysis = decision.get('full_analysis', {})
    ladder_data = full_analysis.get('ladder_data', {})
    data_15m = ladder_data.get('15m', {})
    direction = decision.get('direction', 'LATERAL')
    
    if direction == 'LATERAL':
        return False
    
    # === OBTENER VELOCITY ===
    velocity_check = check_price_velocity(symbol, direction, '5m')
    
    # Fallback a 15m si no hay datos
    if 'insuficientes' in velocity_check.get('reason', '').lower():
        velocity_check = check_price_velocity(symbol, direction, '15m')
    
    velocity_value = velocity_check.get('velocity', 0)
    velocity_mode = velocity_check.get('mode', 'NORMAL')
    
    print(f"   🏎️ Velocity: {velocity_value:.3f}% | Mode: {velocity_mode}")
    
    # === CREAR FORCE_RESULT CON VELOCITY ===
    force_result = full_analysis.get('force_result', {}).copy()
    force_result['velocity'] = velocity_value
    force_result['velocity_mode'] = velocity_mode
    
    # === VALIDAR CON ENTRY_VALIDATOR (retorna score) ===
    validation = entry_validator._validate_explosion_entry(
        symbol, direction, force_result, ladder_data
    )
    
    validator_score = validation.get('score', 0)
    analyzer_score = score  # Score del analyzer
    
    # === COMBINAR SCORES ===
    combined_score = (analyzer_score * SCORE_WEIGHT_ANALYZER) + \
                     (validator_score * SCORE_WEIGHT_VALIDATOR)
    
    print(f"   📊 Scores: Analyzer={analyzer_score:.0f} | Validator={validator_score:.0f}")
    print(f"   🎯 Combined: {combined_score:.0f} (threshold: {SCORE_THRESHOLD_ENTER})")
    
    # === VERIFICACIONES ADICIONALES ===
    
    # Momentum color
    mom_color = data_15m.get('momentum_color', 'gray')
    if direction == 'BULLISH' and mom_color in ['red', 'maroon']:
        print(f"   🚫 Momentum {mom_color} contradice BULLISH")
        return False
    elif direction == 'BEARISH' and mom_color in ['lime', 'green']:
        print(f"   🚫 Momentum {mom_color} contradice BEARISH")
        return False
    
    # Grid position (usar diagonal si existe)
    grid_data = data_15m.get('grid', {})
    diagonal = grid_data.get('diagonal', {})
    
    if diagonal.get('valid'):
        grid_position = diagonal.get('price_position_pct', 50)
    else:
        grid_position = data_15m.get('position_in_range', 50)
    
    if direction == 'BULLISH' and grid_position > 70:
        print(f"   🚫 Grid {grid_position:.0f}% muy alto")
        return False
    elif direction == 'BEARISH' and grid_position < 30:
        print(f"   🚫 Grid {grid_position:.0f}% muy bajo")
        return False
    
    # === DECISIÓN FINAL POR SCORE COMBINADO ===
    if combined_score >= SCORE_THRESHOLD_ENTER:
        print(f"   ✅ APROBADO: Score {combined_score:.0f} >= {SCORE_THRESHOLD_ENTER}")
        return True
    else:
        print(f"   ❌ RECHAZADO: Score {combined_score:.0f} < {SCORE_THRESHOLD_ENTER}")
        return False
