def check_btc_alignment(score=0, strict=None):
    """Verifica si BTC está estable. Ahora CONDICIONAL por score.
    
    Args:
        score: Score de la moneda analizada
        strict: Si True, bloquea siempre. Si False/None, solo bloquea scores bajos
    
    Returns:
        bool: True si OK para operar, False si debe bloquearse
    """
    if strict is None:
        strict = BTC_ALIGNMENT_STRICT  # Usar config
    
    try:
        btc_ohlcv = fetch_ohlcv('BTC/USDT:USDT', '15m', 10)
        if not btc_ohlcv or len(btc_ohlcv) < 5:
            return True  # Sin datos, no bloqueamos
        
        df_btc = pd.DataFrame(btc_ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
        last_5_closes = df_btc['close'].tail(5).values
        btc_change_pct = ((last_5_closes[-1] - last_5_closes[0]) / last_5_closes[0]) * 100
        
        # Threshold más suave: -3% en vez de -2%
        if btc_change_pct < -3.0:
            # Si score es alto (>70), permitir operar igual (desacoplamiento)
            if not strict and score >= 70:
                print(f"🔓 BTC cayendo {btc_change_pct:.2f}% PERO score {score} alto. Permitiendo.")
                return True
            else:
                print(f"⚠️ BTC cayendo {btc_change_pct:.2f}% en 15M. Protección activada.")
                return False
        return True
    except Exception as e:
        print(f"⚠️ Error verificando BTC: {e}")
        return True  # Error = no bloquear

def analyze_4h(symbol):
    """
    4H: Define contexto dominante y clima de mercado
    
    NUNCA RECHAZA - Solo informa dirección y fuerza
    
    Args:
        symbol (str): Par trading (ej: 'BTC/USDT')
    
    Returns:
        dict: Análisis completo con direction, strength, indicadores
        None: Solo si no hay datos (caso extremo)
    """
    # Fetch datos
    ohlcv = fetch_ohlcv(symbol, '4h', 100)
    if not ohlcv or len(ohlcv) < 50:
        return None

    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    current_price = float(df['close'].iloc[-1])
    
    # ===== ADX con memoria =====
    adx_data = calculate_adx_with_memory(df, period=14, lookback=10)
    
    # ADX ANTICIPATORIO: Proyectar evolución
    adx_now = adx_data['adx_current']
    adx_trend = adx_data['adx_trend']
    
    # Calcular velocidad ADX (puntos por vela)
    if len(adx_data['adx_history']) >= 5:
        adx_velocity = (adx_data['adx_history'][-1] - adx_data['adx_history'][0]) / 5
    else:
        adx_velocity = 0.0
    
    # Proyección 3 velas adelante (12 horas en 4H)
    adx_proyectado = adx_now + (adx_velocity * 3)
    
    # ===== Determinar dirección base por dominancia =====
    di_direction = 'LATERAL'
    if adx_data['dominance'] == 'DI+':
        di_direction = 'BULLISH'
    elif adx_data['dominance'] == 'DI-':
        di_direction = 'BEARISH'
    
    # ===== Calcular momentum e indicadores ANTES de validación =====
    momentum_data = calculate_momentum_with_memory(df, lookback=10)
    expected_colors_bullish = ['green', 'lime']
    expected_colors_bearish = ['red', 'maroon']
    
    # Calcular zigzag UNA SOLA VEZ (se usará más adelante también)
    if ENABLE_4H_ZIGZAG:
        zigzag = calculate_zigzag(df, threshold_pct=3.0)
        last_pivot = zigzag.get('pivot_types', [''])[-1] if zigzag.get('pivot_types') else ''
    else:
        zigzag = None
        last_pivot = ''
    
    # ===== VALIDACIÓN CRUZADA (Bug #28 Fix + ULTIMA Fix) =====
    score_4h = 0  # FIX: Inicializar antes de usar en gradiente
    
    # === GRADIENTE BULLISH ===
    if di_direction == 'BULLISH':
        expected_colors_bullish = ['green', 'lime', 'gray']
        
        if momentum_data['color_current'] not in expected_colors_bullish:
            if adx_now > 30 and abs(adx_data['spread']) > 10:
                direction = 'BULLISH' # Override por fuerza bruta
                score_4h -= 10
            else:
                direction = 'LATERAL'
        elif last_pivot in ['H', 'LH']:
            if adx_now > 35 and abs(adx_data['spread']) > 12:
                direction = 'BULLISH' # Override por fuerza bruta extrema
                score_4h -= 15
            else:
                direction = 'LATERAL'
        else:
            direction = 'BULLISH'

    # === GRADIENTE BEARISH (SIMÉTRICO) ===
    elif di_direction == 'BEARISH':
        expected_colors_bearish = ['red', 'maroon', 'gray']
        
        if momentum_data['color_current'] not in expected_colors_bearish:
            if adx_now > 30 and abs(adx_data['spread']) > 10:
                direction = 'BEARISH'
                score_4h -= 10
            else:
                direction = 'LATERAL'
        elif last_pivot in ['L', 'HL']:
            if adx_now > 35 and abs(adx_data['spread']) > 12:
                direction = 'BEARISH'
                score_4h -= 15
            else:
                direction = 'LATERAL'
        else:
            direction = 'BEARISH'
    else:
        direction = 'LATERAL'
    
    # ===== Evaluar fuerza ADX =====
    strength = 'WEAK'  # Inicializar en WEAK
    
    # CASO 1: ADX ya >25 (perfecto)
    if adx_now >= 25:
        strength = 'STRONG'
    elif adx_now >= 17 and adx_velocity > 1.5:
        strength = 'MODERATE'
    else:
        strength = 'WEAK'

    if adx_trend == 'decreciendo' and adx_velocity < -0.8:
        strength = 'WEAK'

    if direction != 'LATERAL' and abs(adx_data['spread']) < 5.0:
        strength = 'WEAK'

    if momentum_data['consecutive_count'] < 3:
        strength = 'WEAK'

    rsi_data = calculate_rsi_with_memory(df, period=14, lookback=10)
    emas_data = calculate_emas_with_memory(df, periods=[9, 21, 50, 200], lookback=10)

    if ENABLE_4H_STRUCTURE:
        structure = detect_price_structure(df, lookback=20)
    else:
        structure = {'structure': 'N/A'}
    
    # zigzag ya calculado arriba - ELIMINAR DUPLICADO
    if not ENABLE_4H_ZIGZAG:
        zigzag = {'zigzag': 'N/A'}

    # ROADMAP: Detectar Soportes y Resistencias Mayores
    sr_data = detect_support_resistance_with_memory(df, lookback=50)
    
    # VOLUME PROFILE (Exponer para Market Force)
    vp_data = calculate_volume_profile_with_memory(df, lookback=200)

    print(f"📊 4H | {symbol} | {direction} {strength} | ADX:{adx_data['adx_current']:.1f} DI+:{adx_data['di_plus_current']:.1f} DI-:{adx_data['di_minus_current']:.1f} | Mom:{momentum_data['consecutive_count']}v | Estructura:{structure['structure']}")

    score_4h = 0
    if strength == 'STRONG':
        score_4h += 20
    elif strength == 'MODERATE':
        score_4h += 10
    else:
        score_4h += 0
    if adx_trend == 'creciendo':
        score_4h += 5
    if momentum_data['consecutive_count'] >= 5:
        score_4h += 5
    if abs(adx_data['spread']) > 7:
        score_4h += 5

    return {
        'direction': direction,
        'strength': strength,
        'score_4h': score_4h,
        'adx': adx_data,
        'momentum': momentum_data,
        'rsi': rsi_data,
        'emas': emas_data,
        'structure': structure,
        'zigzag': zigzag,
        'sr': sr_data, # ROADMAP DATA
        'volume_profile': vp_data, # EXPOSED FOR MARKET FORCE
        'current_price': current_price,
        'context': f"4H {direction} {strength} - ADX {adx_data['adx_current']:.1f}",
        'timestamp': datetime.now().isoformat()
    }

def analyze_1h(symbol):
    """1H: Confirma tendencia 4H"""
    ohlcv = fetch_ohlcv(symbol, '1h', 100)
    if not ohlcv or len(ohlcv) < 50:
        return None

    df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
    adx_data = calculate_adx_with_memory(df, period=14, lookback=10)
    adx_trend = adx_data['adx_trend']
    adx_current = adx_data['adx_current']
    spread = abs(adx_data['spread'])

    # ===== Determinar dirección con validación cruzada (Bug #28 Fix) =====
    di_direction = 'LATERAL'
    if adx_data['dominance'] == 'DI+':
        di_direction = 'BULLISH'
    elif adx_data['dominance'] == 'DI-':
        di_direction = 'BEARISH'
    
    # Calcular momentum PRIMERO
    momentum_data = calculate_momentum_with_memory(df, lookback=10)
    zigzag = calculate_zigzag(df, threshold_pct=2.5)
    last_pivot = zigzag.get('pivot_types', [''])[-1] if zigzag.get('pivot_types') else ''
    
    # Validación cruzada CON DEGRADACIÓN CONSISTENTE
    if di_direction == 'BULLISH':
        mom_bullish = momentum_data['color_current'] in ['green', 'lime']
        struct_bullish = last_pivot not in ['H', 'LH']
        
        if mom_bullish and struct_bullish:
            direction = 'BULLISH'
        else:
            direction = 'LATERAL'  # Degrada, no invierte
    elif di_direction == 'BEARISH':
        mom_bearish = momentum_data['color_current'] in ['red', 'maroon']
        struct_bearish = last_pivot not in ['L', 'HL']
        
        if mom_bearish and struct_bearish:
            direction = 'BEARISH'
        else:
            direction = 'LATERAL'  # Degrada, no invierte
    else:
        direction = 'LATERAL'
    
    # FIX: Aceptar 'lime' como bullish y 'maroon' como bearish
    expected_colors = ['green', 'lime'] if direction == 'BULLISH' else ['red', 'maroon']
    momentum_color = momentum_data['color_current']

    rsi_data = calculate_rsi_with_memory(df, period=14, lookback=10)
    emas_data = calculate_emas_with_memory(df, periods=[9, 21, 50, 200], lookback=10)
    structure = detect_price_structure(df, lookback=20)
    
    # ROADMAP: Detectar Soportes y Resistencias Intermedios
    sr_data = detect_support_resistance_with_memory(df, lookback=50)
    
    # VOLUME PROFILE (Exponer para Market Force)
    vp_data = calculate_volume_profile_with_memory(df, lookback=100)

    if adx_current > 32:
        strength = 'STRONG'
    elif adx_current > 25:
        strength = 'MODERATE'
    else:
        strength = 'WEAK'

    score_1h = 0
    if adx_trend == 'creciendo':
        score_1h += 5
    if adx_current >= 22:
        score_1h += 5
    if momentum_color in expected_colors:
        score_1h += 5
    if spread > 4.0:
        score_1h += 5

    adx_val = adx_data['adx_current']
    momentum = momentum_data['color_current']
    rsi = rsi_data['rsi_current']
    emas_aligned = emas_data.get('emas_aligned', False)

    if adx_val >= 25:
        score_1h += 5
    elif adx_val >= 20:
        score_1h += 2

    if momentum in ['green', 'lime', 'red', 'maroon']:
        score_1h += 2

    if 40 <= rsi <= 70:
        score_1h += 2

    if emas_aligned:
        score_1h += 2

    return {
        'direction': direction,
        'strength': strength,
        'score_1h': score_1h,
        'adx': adx_data,
        'momentum': momentum_data,
        'rsi': rsi_data,
        'emas': emas_data,
        'structure': structure,
        'zigzag': zigzag,
        'sr': sr_data, # ROADMAP DATA
        'volume_profile': vp_data, # EXPOSED FOR MARKET FORCE
        'context': f"1H {direction} {strength} - ADX {adx_current:.1f}",
        'timestamp': datetime.now().isoformat()
    }

    pump_penalty = 0

        # Pump extremo: penalización -50
        pump_penalty = 50
        print(f"⚠️ PUMP EXTREMO: Vol {last_vol/vol_avg:.1f}x, Vela {last_candle_pct:.1f}% → Penalty -50")
    elif last_vol > vol_avg * 15 and last_candle_pct > 20:
        # Pump moderado: penalización -30
        pump_penalty = 30
        print(f"⚠️ Pump moderado: Vol {last_vol/vol_avg:.1f}x, Vela {last_candle_pct:.1f}% → Penalty -30")

    di_minus_granular = analyze_di_minus(df, lookback=14)

    # Análisis granular de momentum según color
    momentum_green = analyze_momentum_green(momentum_data)
    momentum_red = analyze_momentum_red(momentum_data)

    # SISTEMA DE PELÍCULA: Guardar frame + Analizar movimiento
    # ===================================================================
    # Concepto Joel: "Como un dibujante que apila dibujos y los mueve
    # rápido para generar animación"

    frame_storage = get_frame_storage(symbol, '15m')

        lateral_penalty += 15

    if zigzag_granular.get('current_trend') == 'LATERAL':
        lateral_penalty += 15
        lateral_reasons.append(f"Estructura: {zigzag_granular.get('structure', 'LATERAL')}")
    
    # >>> NUEVO: Logging si lateralización detectada
    if lateral_penalty > 0:
        print(f"⚠️  LATERALIZACIÓN | {symbol} | Penalty: -{lateral_penalty} | Razones: {', '.join(lateral_reasons)}")
    
    # Determinar dirección con validación de momentum (Bug #28 Fix + ULTIMA Fix)
    di_direction = 'LATERAL'
    if adx_data['dominance'] == 'DI+':
        di_direction = 'BULLISH'
    elif adx_data['dominance'] == 'DI-':
        di_direction = 'BEARISH'
    
    # Validar con momentum
    if di_direction == 'BULLISH':
        if momentum_data['color_current'] in ['green', 'lime']:
            direction = 'BULLISH'
        else:
            direction = 'LATERAL'
    elif di_direction == 'BEARISH':
        if momentum_data['color_current'] in ['red', 'maroon']:
            direction = 'BEARISH'
        else:
            direction = 'LATERAL'

                direction = 'LATERAL'

            direction = 'BULLISH'

        if mom_color not in ['red', 'maroon']:
            direction = 'LATERAL'
        elif last_pivot in ['L', 'HL']:
            if adx_data['adx_current'] > 40 and abs(adx_data['spread']) > 15 and momentum_data['consecutive_count'] >= 2:
                direction = 'BEARISH'
            else:
                direction = 'LATERAL'

    dist_ema_pct = ((current_price - ema_20) / ema_20) * 100

def analyze_temporal_context(symbol):
    """
    Analiza el contexto temporal reciente (últimos 60 min) usando la DB.
    Busca estabilidad y consistencia.
    """
    try:
        history = db_manager.get_recent_context(symbol, limit_minutes=60)
        if not history or len(history) < 10:
            return {'stability': 'DESCONOCIDO', 'bonus': 0}
        
        # 1. Verificar cambios de dirección (inestabilidad)
        directions = [row['direction'] for row in history]
        flips = 0
        for i in range(1, len(directions)):
            if directions[i] != directions[i-1]:
                flips += 1
        
        # 2. Verificar consistencia de ADX
        adx_values = [row['adx_15m'] for row in history]
        adx_rising = 0
        for i in range(1, len(adx_values)):
            if adx_values[i] > adx_values[i-1]:
                adx_rising += 1
        
        # Evaluación
        bonus = 0
        stability = 'ESTABLE'
        
        if flips > 3:
            stability = 'INESTABLE'
            bonus -= 10  # Penalización por inestabilidad
        elif flips == 0:
            stability = 'MUY_ESTABLE'
            bonus += 10  # Bonus por tendencia sostenida
            
        if adx_rising > len(adx_values) * 0.7:
            bonus += 5  # Bonus si ADX sube la mayoría del tiempo
            
        return {'stability': stability, 'bonus': bonus, 'flips': flips}
        
    except Exception as e:
        print(f"⚠️ Error analizando contexto temporal: {e}")
        return {'stability': 'ERROR', 'bonus': 0}

def analyze_trend_evolution(symbol):
    """
    ANÁLISIS DE EVOLUCIÓN (TIME LADDER / ROADMAP)
    Analiza velas de 1M para detectar tendencias explosivas en tiempo real.
    Permite 'bypassear' filtros estrictos si la moneda está explotando ahora mismo.
    """
    try:
        # Buscar 30 velas de 1M (media hora de evolución)
        ohlcv = fetch_ohlcv(symbol, '1m', 30)
        if not ohlcv or len(ohlcv) < 15:
            return {'is_explosive': False, 'score': 0, 'reason': 'Sin datos 1M'}

        df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
        
        # 1. Análisis de ADX en 1M (Velocidad de tendencia)
        adx_data = calculate_adx_with_memory(df, period=14, lookback=5)
        adx_current = adx_data['adx_current']
        adx_history = adx_data['adx_history']
        
        # 2. Análisis de Precio (Escalera de precios)
        closes = df['close'].values
        price_velocity = calculate_velocity(df, lookback=5)
        
        # DETECCIÓN DE EXPLOSIÓN
        is_explosive = False
        score = 0
        reason = []

        # Criterio A: ADX Disparándose (Roadmap de fuerza)
        # Si ADX subió más de 5 puntos en las últimas 5 velas 1M
        if len(adx_history) >= 5:
            adx_change = adx_history[-1] - adx_history[0]
            if adx_change > 3.0 and adx_current > 20:
                score += 40
                reason.append(f"ADX 1M disparándose (+{adx_change:.1f})")
                if adx_current > 25:
                    is_explosive = True

        # Criterio B: Escalera de Precios (ZigZag alcista agresivo)
        # Verificar si las últimas 3 velas son verdes y con cuerpos crecientes o fuertes
        last_3_closes = closes[-3:]
        last_3_opens = df['open'].values[-3:]
        is_green_ladder = all(c > o for c, o in zip(last_3_closes, last_3_opens))
        
        if is_green_ladder and price_velocity['velocity_current'] > 0:
            score += 30
            reason.append("Escalera de precios 1M alcista")
            if price_velocity['trend'] == 'acelerando':
                is_explosive = True
                score += 20

        # Criterio C: Volumen Climático
        volumes = df['volume'].values
        vol_avg = volumes[:-1].mean()
        if volumes[-1] > vol_avg * 2:
            score += 20
            reason.append("Volumen 1M climático")

        # Decisión final de explosividad
        if score >= 70:
            is_explosive = True
        
        return {
            'is_explosive': is_explosive,
            'score': score,
            'adx_1m': adx_current,
            'reason': ", ".join(reason),
            'timestamp': datetime.now().isoformat()
        }

    except Exception as e:
        print(f"⚠️ Error en analyze_trend_evolution: {e}")
        return {'is_explosive': False, 'score': 0, 'reason': 'Error'}

def _extract_score(score_value, default=0):
    """
    Recursively extract integer score from potentially nested dict.
    
    Args:
        score_value: The value to extract (can be int, float, dict, or None)
        default: Default value if extraction fails
    
    Returns:
        int: Extracted integer score
    """
    if score_value is None:
        return default
    if isinstance(score_value, (int, float)):
        return int(score_value)
    if isinstance(score_value, dict):
        # Recursively extract from nested dict
        return _extract_score(score_value.get('score', default), default)
    # Fallback for any other type
    return default

            pass  # Fallback a fixed percentage
        except:
            pass
    
    if atr > 0:
        # Multiplicador dinámico: 2.0x ATR es estándar
        atr_multiplier = 2.0
        stop_loss_dist = atr * atr_multiplier
        
        if direction == 'BULLISH':
            stop_loss = entry_price - stop_loss_dist
        else:
            stop_loss = entry_price + stop_loss_dist
            
        # Sanity Check: Ensure SL is not too close (< 0.5%) or too far (> 5%)
        # If too close, force min 0.5%
        min_dist = entry_price * 0.015  # 1.5% mínimo
        if stop_loss_dist < min_dist:
            if direction == 'BULLISH': stop_loss = entry_price - min_dist
            else: stop_loss = entry_price + min_dist
            
    else:
        # Fallback: Fixed 2%
        stop_loss = entry_price * 0.98 if direction == 'BULLISH' else entry_price * 1.02

    # Target: 1.5x Risk (Simple initial target)
    risk = abs(entry_price - stop_loss)
    target_dist = risk * 1.5
    
    if direction == 'BULLISH':
        target_price = entry_price + target_dist
    else:
        target_price = entry_price - target_dist

    rr_ratio = target_dist / risk if risk > 0 else 0
    tiempo_proyectado_min = 15

    return {
        'entry_price': entry_price,
        'stop_loss': stop_loss,
        'target_price': target_price,
        'rr_ratio': rr_ratio,
        'tiempo_proyectado_min': tiempo_proyectado_min,
        'metodo_target': 'atr_dynamic' if atr > 0 else 'fixed_pct'
    }

"""
brain_frames.py - Sistema de Frames Temporales (PELÍCULA)
=========================================================
Guarda historia de indicadores como "frames" de una película.

Concepto de Joel:
"Es como un dibujante que apila sus dibujos con secuencia y los 
mueve rápido para generar el dibujo animado."

Cada frame contiene:
- ADX, DI+, DI-, Spread
- Momentum color, age
- Precio, volumen
- Grid posición
- ZigZag estado
- Timestamp

Los frames se apilan secuencialmente para ver MOVIMIENTO.

Autor: Joel Rodríguez Diarte (concepto)
Implementación: Claude (Anthropic)
Fecha: 2025-12-21
"""

from datetime import datetime

import json

class FrameStorage:
    """
    Almacena frames temporales de indicadores.
    
    Permite ver la PELÍCULA del mercado, no solo fotos estáticas.
    
    Ejemplo:
        Frame -5: ADX = 45
        Frame -4: ADX = 48  (+3)
        Frame -3: ADX = 52  (+4)
        Frame -2: ADX = 56  (+4)
        Frame -1: ADX = 58  (+2)
        
        → ADX ACELERANDO → EXPLOSIÓN EN CURSO
    """
    
    def __init__(self, max_frames=20):
        """
        Args:
            max_frames: Máximo de frames a guardar (20 = últimas 20 velas)
        """
        self.max_frames = max_frames
        self.frames = deque(maxlen=max_frames)
    
    def add_frame(self, symbol, timeframe, indicators):
        """
        Agrega un frame nuevo (un "dibujo" del momento actual)
        
        Args:
            symbol: Par (ej: RAVE/USDT:USDT)
            timeframe: TF (5m, 15m, 1h, 4h)
            indicators: Dict con todos los indicadores actuales
        """
        frame = {
            'timestamp': datetime.now().isoformat(),
            'symbol': symbol,
            'timeframe': timeframe,
            
            # ADX (líneas del indicador)
            'adx': {
                'value': indicators.get('adx', {}).get('adx_current', 0),
                'di_plus': indicators.get('adx', {}).get('di_plus_current', 0),
                'di_minus': indicators.get('adx', {}).get('di_minus_current', 0),
                'spread': abs(
                    indicators.get('adx', {}).get('di_plus_current', 0) -
                    indicators.get('adx', {}).get('di_minus_current', 0)
                ),
                'trend': indicators.get('adx', {}).get('adx_trend', 'unknown')
            },
            
            # Momentum (barras)
            'momentum': {
                'color': indicators.get('momentum', {}).get('color_current', 'gray'),
                'age': indicators.get('momentum', {}).get('consecutive_count', 0),
                'intensity': indicators.get('momentum', {}).get('intensity', 0)
            },
            
            # >>> NUEVO: OHLC Completo para ver "dibujo" del precio
            'ohlc': {
                'open': indicators.get('open', 0),
                'high': indicators.get('high', 0),
                'low': indicators.get('low', 0),
                'close': indicators.get('close', indicators.get('price', 0))
            },
            
            # Precio (legacy - mantener compatibilidad)
            'price': indicators.get('price', indicators.get('close', 0)),
            
            # Volumen
            'volume': indicators.get('volume', 0),
            'volume_ratio': indicators.get('volume_ratio', 100),
            
            # Grid
            'grid': {
                'position_pct': indicators.get('position_in_range', 50),
                'rompiendo': indicators.get('position_in_range', 50) > 85
            },
            
            # Score
            'score': indicators.get('score_15m', 0)
        }
        
        self.frames.append(frame)
    
    def get_frames(self, n=None):
        """
        Obtiene últimos N frames
        
        Args:
            n: Número de frames (None = todos)
            
        Returns:
            list: Lista de frames [más viejo ... más nuevo]
        """
        if n is None:
            return list(self.frames)
        else:
            return list(self.frames)[-n:]
    
    def get_trajectory(self, indicator_path, n=5):
        """
        Obtiene trayectoria de un indicador (la "animación")
        
        Args:
            indicator_path: Ruta al indicador (ej: 'adx.value')
            n: Frames a analizar
            
        Returns:
            list: Valores históricos [más viejo ... más nuevo]
            
        Example:
            trajectory = storage.get_trajectory('adx.value', 5)
            # [45, 48, 52, 56, 58] → subiendo
        """
        frames = self.get_frames(n)
        values = []
        
        for frame in frames:
            # Navegar por el path (ej: 'adx.value')
            parts = indicator_path.split('.')
            value = frame
            
            try:
                for part in parts:
                    if isinstance(value, dict):
                        value = value.get(part, 0)
                    else:
                        value = 0
                        break
                values.append(value)
            except:
                values.append(0)
        
        return values
    
    def get_frame_count(self):
        """Retorna cantidad de frames almacenados"""
        return len(self.frames)
    
    def clear(self):
        """Limpia todos los frames"""
        self.frames.clear()
    
    def get_latest_frame(self):
        """Obtiene el frame más reciente"""
        if self.frames:
            return self.frames[-1]
        return None
    
    def has_enough_frames(self, min_frames=3):
        """Verifica si hay suficientes frames para análisis"""
        return len(self.frames) >= min_frames

def load_frames_from_ohlcv(symbol: str, timeframe: str, ohlcv_data: list, num_frames: int = 10):
    """
    CARGA INSTANTÁNEA DE FRAMES DESDE OHLCV HISTÓRICO.
    
    CONCEPTO DE JOEL:
    "El Time Ladder debe ser rápido, no en slow motion"
    
    Esta función carga los últimos N frames inmediatamente desde los datos
    OHLCV históricos del exchange, en lugar de esperar múltiples ciclos.
    
    Args:
        symbol: Par (ej: RAVE/USDT:USDT)
        timeframe: TF (5m, 15m, 1h, 4h)
        ohlcv_data: Lista de OHLCV [[timestamp, open, high, low, close, volume], ...]
        num_frames: Cantidad de frames a cargar (default 10)
        
    Returns:
        int: Cantidad de frames cargados
    """
    if not ohlcv_data or len(ohlcv_data) < 5:
        return 0
    
    storage = get_frame_storage(symbol, timeframe)
    
    # Si ya tiene suficientes frames, no recargar
    if storage.has_enough_frames(num_frames):
        return storage.get_frame_count()
    
    # Limpiar frames antiguos si los hay
    storage.clear()
    
    # Cargar últimos N candles como frames
    candles_to_load = ohlcv_data[-num_frames:]
    
    for candle in candles_to_load:
        if len(candle) >= 6:
            timestamp, open_p, high_p, low_p, close_p, volume = candle[:6]
            
            # Crear indicadores simulados para el frame
            indicators = {
                'open': open_p,
                'high': high_p,
                'low': low_p,
                'close': close_p,
                'price': close_p,
                'volume': volume,
                'volume_ratio': 100,  # Se calculará después si es necesario
                'position_in_range': 50,  # Se calculará después
                
                # ADX y momentum se llenarán con datos reales cuando se analice
                'adx': {'adx_current': 0, 'di_plus_current': 0, 'di_minus_current': 0},
                'momentum': {'color_current': 'gray', 'consecutive_count': 0}
            }
            
            storage.add_frame(symbol, timeframe, indicators)
    
    loaded = storage.get_frame_count()
    if loaded >= 5:
        print(f"   ⚡ Time Ladder INSTANTÁNEO: {symbol} - {loaded} frames cargados")
    
    return loaded

def detect_price_pattern(symbol, direction='BULLISH', timeframe='15m'):
    """
    Detecta patrón de precio usando Time Ladder.
    
    CONCEPTO DE JOEL:
    "El bot no esperaba el piso y luego el impulso, entraba sin ver ese piso."
    
    Esta función ve la PELÍCULA del precio frame a frame para detectar:
    - LONG: Corrección → Piso → Rebote confirmado → ENTRAR
    - SHORT: Rally → Techo → Caída confirmada → ENTRAR
    
    Args:
        symbol: Par a analizar
        direction: 'BULLISH' o 'BEARISH'
        timeframe: TF para análisis
        
    Returns:
        dict: {
            'ready': bool - Si el patrón está listo para entrar
            'pattern': str - Nombre del patrón detectado
            'phase': str - Fase actual (CORRECCION, PISO, IMPULSO, etc)
            'reason': str - Razón descriptiva
            'wait_for': str - Qué esperar si no está listo
        }
    """
    storage = get_frame_storage(symbol, timeframe)
    
    # Necesitamos mínimo 5 frames para ver la película
    if not storage.has_enough_frames(5):
        return {
            'ready': False,
            'pattern': 'INSUFFICIENT_DATA',
            'phase': 'UNKNOWN',
            'reason': f'Solo {storage.get_frame_count()} frames, necesita 5',
            'wait_for': 'Más datos de precio'
        }
    
    # Obtener trayectorias
    closes = storage.get_trajectory('ohlc.close', 5)
    lows = storage.get_trajectory('ohlc.low', 5)
    highs = storage.get_trajectory('ohlc.high', 5)
    
    # Fallback si OHLC no está disponible
    if not closes or closes[-1] == 0:
        closes = storage.get_trajectory('price', 5)
        lows = closes.copy()
        highs = closes.copy()
    
    if not closes or len(closes) < 5:
        return {
            'ready': False,
            'pattern': 'NO_PRICE_DATA',
            'phase': 'UNKNOWN',
            'reason': 'Sin datos de precio',
            'wait_for': 'Datos de precio',
            'structure': None
        }
    
    # ===================================================================
    # >>> NUEVO: DETECCIÓN DE HH/HL/LL/LH (ESTRUCTURA DE TENDENCIA)
    # ===================================================================
    # Esto es como ver el gráfico completo de velas pero con el precio
    
    # Identificar swing highs y swing lows (pivots locales)
    swing_highs = []  # [(idx, price)]
    swing_lows = []   # [(idx, price)]
    
    for i in range(1, len(highs) - 1):
        # Swing High: precio mayor que vecinos
        if highs[i] > highs[i-1] and highs[i] >= highs[i+1]:
            swing_highs.append((i, highs[i]))
        # Swing Low: precio menor que vecinos  
        if lows[i] < lows[i-1] and lows[i] <= lows[i+1]:
            swing_lows.append((i, lows[i]))
    
    # Analizar estructura HH/HL/LL/LH
    structure = {
        'swing_highs': swing_highs,
        'swing_lows': swing_lows,
        'trend': 'LATERAL',
        'hh_count': 0,
        'hl_count': 0,
        'll_count': 0,
        'lh_count': 0,
        'last_pivot': 'NONE'
    }
    
    # Contar HH (Higher High) y LH (Lower High)
    for i in range(1, len(swing_highs)):
        if swing_highs[i][1] > swing_highs[i-1][1]:
            structure['hh_count'] += 1
        else:
            structure['lh_count'] += 1
    
    # Contar HL (Higher Low) y LL (Lower Low)
    for i in range(1, len(swing_lows)):
        if swing_lows[i][1] > swing_lows[i-1][1]:
            structure['hl_count'] += 1
        else:
            structure['ll_count'] += 1
    
    # Determinar tendencia por estructura
    alcista_score = structure['hh_count'] + structure['hl_count']
    bajista_score = structure['ll_count'] + structure['lh_count']
    
    if alcista_score > bajista_score + 1:
        structure['trend'] = 'UPTREND'
        structure['last_pivot'] = 'HL' if swing_lows and swing_lows[-1][0] > (swing_highs[-1][0] if swing_highs else 0) else 'HH'
    elif bajista_score > alcista_score + 1:
        structure['trend'] = 'DOWNTREND'
        structure['last_pivot'] = 'LL' if swing_lows and swing_lows[-1][0] > (swing_highs[-1][0] if swing_highs else 0) else 'LH'
    else:
        structure['trend'] = 'LATERAL'
    
    print(f"         📊 Estructura: {structure['trend']} | HH:{structure['hh_count']} HL:{structure['hl_count']} LL:{structure['ll_count']} LH:{structure['lh_count']}")
    
    # === ANÁLISIS PARA LONG (BULLISH) ===
    if direction == 'BULLISH':
        # Calcular movimientos
        lowest_idx = lows.index(min(lows)) if lows else -1
        current_close = closes[-1]
        previous_close = closes[-2]
        lowest_low = min(lows) if lows else 0
        highest_high = max(highs) if highs else 0
        
        # >>> NUEVO: BLOQUEAR LONG EN ZONA DE RESISTENCIA
        # El bot no debe entrar LONG justo en la resistencia (probable rechazo)
        if highest_high > 0:
            dist_to_resistance = abs(highest_high - current_close) / highest_high * 100
            if dist_to_resistance < 1.0:  # Menos de 1% del máximo
                return {
                    'ready': False,
                    'pattern': 'AT_RESISTANCE',
                    'phase': 'RESISTENCIA',
                    'reason': f'LONG en zona de RESISTENCIA (a {dist_to_resistance:.2f}% del máximo)',
                    'wait_for': 'Esperar rechazo o ruptura de resistencia',
                    'structure': structure
                }
        
        # Detectar fases
        # FASE 1: Corrección (precio bajando)
        if closes[-3] > closes[-2] > closes[-1]:
            return {
                'ready': False,
                'pattern': 'CORRECTION_IN_PROGRESS',
                'phase': 'CORRECCION',
                'reason': 'Precio bajando - esperar piso',
                'wait_for': 'Rebote desde piso',
                'structure': structure
            }
        
        # FASE 2: Piso detectado (mínimo tocado, precio rebotando)
        if lowest_idx >= 2 and current_close > lowest_low * 1.005:
            # Verificar que hay rebote confirmado
            if current_close > previous_close:
                return {
                    'ready': True,
                    'pattern': 'FLOOR_BOUNCE',
                    'phase': 'IMPULSO',
                    'reason': f'Rebote desde piso ${lowest_low:.4f} confirmado',
                    'wait_for': None,
                    'structure': structure
                }
            else:
                return {
                    'ready': False,
                    'pattern': 'FLOOR_TOUCHED',
                    'phase': 'PISO',
                    'reason': 'Piso tocado, esperando confirmación de rebote',
                    'wait_for': 'Vela verde desde piso',
                    'structure': structure
                }
        
        # FASE 3: Impulso en curso (subiendo)
        if closes[-2] < closes[-1] and closes[-3] < closes[-2]:
            # >>> NUEVO: Verificar que el movimiento es FUERTE (no solo subiendo)
            price_change_pct = ((closes[-1] - closes[-3]) / closes[-3]) * 100
            
            if price_change_pct >= 0.5:  # Subida fuerte (>=0.5% en 3 velas)
                return {
                    'ready': True,
                    'pattern': 'IMPULSE_STRONG',
                    'phase': 'EXPLOSIÓN',
                    'reason': f'Impulso FUERTE confirmado (+{price_change_pct:.2f}% en 3 velas)',

                    'ready': True,

                    'phase': 'IMPULSO',

                    'wait_for': None,
                    'structure': structure,
                    'price_momentum': price_change_pct
                }
        
        # FASE 4: Lateral (sin dirección clara)
        return {
            'ready': False,
            'pattern': 'LATERAL',
            'phase': 'CONSOLIDACION',
            'reason': 'Sin patrón claro - lateralizando',
            'wait_for': 'Dirección definida',

        lowest_low = min(lows) if lows else 0

        # El bot no debe entrar SHORT justo en el soporte (probable rebote)

            dist_to_support = abs(current_close - lowest_low) / lowest_low * 100
            if dist_to_support < 1.0:  # Menos de 1% del mínimo
                return {
                    'ready': False,
                    'pattern': 'AT_SUPPORT',
                    'phase': 'SOPORTE',
                    'reason': f'SHORT en zona de SOPORTE (a {dist_to_support:.2f}% del mínimo)',
                    'wait_for': 'Esperar rebote o ruptura del soporte',
                    'structure': structure
                }
        
        # FASE 1: Rally (precio subiendo - contra-tendencia)
        if closes[-3] < closes[-2] < closes[-1]:
            return {
                'ready': False,
                'pattern': 'RALLY_IN_PROGRESS',
                'phase': 'RALLY',
                'reason': 'Precio subiendo - esperar techo',
                'wait_for': 'Rechazo desde techo',
                'structure': structure
            }

                    'ready': True,

                    'reason': f'Rechazo desde techo ${highest_high:.4f} confirmado',

                return {

                    'pattern': 'CEILING_TOUCHED',
                    'phase': 'TECHO',
                    'reason': 'Techo tocado, esperando confirmación de caída',
                    'wait_for': 'Vela roja desde techo',
                    'structure': structure
                }
        
        # FASE 3: Caída en curso
        if closes[-2] > closes[-1] and closes[-3] > closes[-2]:
            # >>> NUEVO: Verificar que el movimiento es FUERTE (no solo cayendo)
            price_change_pct = ((closes[-3] - closes[-1]) / closes[-3]) * 100
            
            if price_change_pct >= 0.5:  # Caída fuerte (>=0.5% en 3 velas)
                return {
                    'ready': True,
                    'pattern': 'DROP_STRONG',
                    'phase': 'EXPLOSIÓN',

                    'wait_for': None,
                    'structure': structure,
                    'price_momentum': -price_change_pct
                }
            else:
                return {
                    'ready': True,
                    'pattern': 'DROP_CONFIRMED',
                    'phase': 'CAIDA',
                    'reason': f'Caída confirmada (-{price_change_pct:.2f}%)',
