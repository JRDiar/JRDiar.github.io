"""BRAIN GRID v6.4.8 - Navegación Espacial (CORREGIDO)"""
import pandas as pd
from brain_config import GRID_LOOKBACK, GRID_COMPRESSION_MULTIPLIERS
from brain_calculator import calculate_atr
class MarketGridNavigator:
    def navigate(self, symbol, df_5m, ladder):
        if df_5m is None or len(df_5m) < 30:
            return {'decision': 'WAIT', 'score': -100, 'reasons': ['No Data'], 'grid_data': {}}
        
        grid = self._build_grid(df_5m)
        vol_class = self._classify_volatility(df_5m) # FIX: Sin pasar atr
        grid['vol_class'] = vol_class
        # FIX BUG #1: REEMPLAZAR, no multiplicar
        grid['threshold'] = GRID_COMPRESSION_MULTIPLIERS.get(vol_class, 2.5)
        
        pointer = self._calc_pointer(df_5m, ladder)
        force = self._get_force(ladder)
        
        return self._decide(grid, pointer, force, df_5m)
    
    def _build_grid(self, df):
        ceiling = df['high'].rolling(GRID_LOOKBACK).max().iloc[-1]
        floor = df['low'].rolling(GRID_LOOKBACK).min().iloc[-1]
        atr_data = calculate_atr(df, period=14)
        atr = atr_data.get('atr_current', 0)
        width_atr = (ceiling - floor) / atr if atr > 0 else 0
        return {'ceiling': ceiling, 'floor': floor, 'width_atr': width_atr, 'atr_current': atr}
    
    def _classify_volatility(self, df):
        # FIX MEJORA #1: Usar 50 velas
        if len(df) < 50:
            return 'medium_vol'
        
        atr_data = calculate_atr(df, period=14)
        atr_avg = atr_data['atr_current']
        price_avg = df['close'].tail(50).mean()
        atr_pct = (atr_avg / price_avg) * 100 if price_avg > 0 else 0
        
        if atr_pct > 2.0: return 'high_vol'
        if atr_pct < 0.8: return 'low_vol'
        return 'medium_vol'
    
    def _calc_pointer(self, df, ladder):
        closes = df['close'].values
        velocity = closes[-1] - closes[-3] if len(closes) > 3 else 0
        
        ladder_15m = ladder.get('15m') if ladder else None
        mom_data = ladder_15m.get('momentum', {}) if ladder_15m else {}
        mom_color = mom_data.get('color_current', 'neutral')
        
        direction = "NEUTRAL"
        if mom_color in ['green', 'lime']: direction = "UP"
        elif mom_color in ['red', 'maroon']: direction = "DOWN"
        return {'direction': direction, 'velocity': velocity}
    
    def _get_force(self, ladder):
        ladder_15m = ladder.get('15m') if ladder else None
        if not ladder_15m:
            return {'adx': 0, 'mom_color': 'neutral'}
        
        adx_data = ladder_15m.get('adx', {})
        mom_data = ladder_15m.get('momentum', {})
        
        return {
            'adx': adx_data.get('adx_current', 0),
            'mom_color': mom_data.get('color_current', 'neutral')
        }
    
    def _decide(self, grid, pointer, force, df):
        score = 0
        reasons = []
        current_price = df['close'].iloc[-1]
        
        # 1. COMPRESSION
        if grid['width_atr'] < grid['threshold']:
            score -= 100
            reasons.append(f"⛔ LATERAL ({grid['width_atr']:.1f} ATRs)")
        
        # 2. FORCE
        adx_val = force['adx']
        if adx_val < 22:
            score -= 50
            reasons.append(f"⚠️ ADX débil ({adx_val:.1f})")
        elif adx_val > 30:
            score += 15
            reasons.append(f"💪 ADX {adx_val:.1f}")
        
        # 3. POSITION (FIX BUG #4: division por zero)
        denominator = grid['ceiling'] - grid['floor']
        if denominator < 0.0001:
            position = 0.5  # Neutral si flat
        else:
            position = (current_price - grid['floor']) / denominator
        
        # BREAKOUT UP
        if position > 0.85 and pointer['direction'] == 'UP':
            if force['mom_color'] in ['green', 'lime'] and adx_val > 25:
                score += 50
                reasons.append("🚀 Breakout + Mom + ADX")
            else:
                score += 10
        
        # BREAKDOWN DOWN
        elif position < 0.15 and pointer['direction'] == 'DOWN':
            if force['mom_color'] in ['red', 'maroon'] and adx_val > 25:
                score += 50
                reasons.append("🚀 Breakdown + Mom + ADX")
            else:
                score += 10
        
        # NEUTRAL ZONE
        elif 0.3 <= position <= 0.7:
            score += 25
            reasons.append("✅ Zona neutral")
        
        # CONTRA-TENDENCIA
        elif position > 0.7 and pointer['direction'] == 'DOWN':
            score -= 50
            reasons.append("⚠️ SHORT cerca techo")
        elif position < 0.3 and pointer['direction'] == 'UP':
            score -= 50
            reasons.append("⚠️ LONG cerca piso")
        
        decision = "GO" if score >= 35 else "WAIT"
        return {'decision': decision, 'score': score, 'reasons': reasons, 'grid_data': grid}
