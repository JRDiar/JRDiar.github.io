class BotBrainRecorder:
    """
    Records BOT's decision-making process at each historical frame.
    
    TIME LADDER COMPLIANT:
    - Uses windowed context (lookback=5) for velocity calculations
    - Analyzes TRAJECTORIES not snapshots
    - Real velocities for ADX, exhaustion, momentum
    """
    
    LOOKBACK = 5  # Frames of context for velocity calculation
    
    def record_decision_at_frame(self, symbol, frame_idx, all_frames, direction):
        """
        Simulates bot's decision with FULL CONTEXT.
        
        Args:
            symbol: str - Trading pair
            frame_idx: int - Current frame position in all_frames
            all_frames: list - ALL frames (for velocity calculation)
            direction: str - BULLISH/BEARISH
        
        Returns:
            dict with confidence_factors, decision, velocities
        """
        try:
            # Get context window for velocity calculation
            start_idx = max(0, frame_idx - self.LOOKBACK)
            context_window = all_frames[start_idx:frame_idx + 1]
            current_frame = all_frames[frame_idx]
            
            # Calculate REAL velocities from context
            velocities = self._calculate_velocities(context_window)
            
            # Build trajectory with REAL data
            trajectory = self._build_trajectory(
                current_frame['indicators'],
                velocities,
                direction
            )
            
            # Build ladder analysis
            ladder_analysis = self._build_ladder_analysis(
                symbol,
                current_frame['indicators'],
                velocities,
                direction
            )
            
            # Call UnifiedVision with real data
            from brain_vision_unified import UnifiedVision
            vision = UnifiedVision()
            result = vision.evaluate_confidence(trajectory, direction, ladder_analysis)
            
            # Extract breakdown
            breakdown = self._extract_confidence_breakdown(result, velocities)
            
            return breakdown
            
        except Exception as e:
            return {
                'confidence_factors': [],
                'total_confidence': 0,
                'decision': 'ERROR',
                'decision_reason': f'Simulation failed: {str(e)[:50]}',
                'velocities': {},
                'error': str(e)
            }
    
    def _calculate_velocities(self, context_window):
        """
        Calculates REAL velocities from context window.
        
        TIME LADDER: This is the core of trajectory analysis.
        """
        if len(context_window) < 2:
            return {
                'adx': 0, 'exhaustion': 0, 'room': 0,
                'momentum': 0, 'spread': 0
            }
        
        n = len(context_window)
        
        # Extract series (with None handling)
        def safe_get(frames, key):
            values = [f['indicators'].get(key) for f in frames]
            return [v if v is not None else 0 for v in values]
        
        adx_series = safe_get(context_window, 'adx')
        exhaustion_series = safe_get(context_window, 'exhaustion_pct')
        room_series = safe_get(context_window, 'room_pct')
        momentum_series = safe_get(context_window, 'momentum_age')
        spread_series = safe_get(context_window, 'spread')
        
        # Calculate velocities (change per frame)
        def velocity(series):
            if len(series) < 2:
                return 0
            return (series[-1] - series[0]) / (len(series) - 1)
        
        # Calculate accelerations (velocity change)
        def acceleration(series):
            if len(series) < 4:
                return 0
            mid = len(series) // 2
            vel1 = velocity(series[:mid])
            vel2 = velocity(series[mid:])
            return vel2 - vel1
        
        return {
            'adx': round(velocity(adx_series), 2),
            'adx_accel': round(acceleration(adx_series), 3),
            'exhaustion': round(velocity(exhaustion_series), 2),
            'exhaustion_accel': round(acceleration(exhaustion_series), 3),
            'room': round(velocity(room_series), 3),
            'momentum': round(velocity(momentum_series), 2),
            'spread': round(velocity(spread_series), 2)
        }
    
    def _build_trajectory(self, indicators, velocities, direction):
        """
        Builds trajectory with REAL velocities (not zeros).
        
        TIME LADDER COMPLIANT: All deltas are real calculated values.
        """
        return {
            'direction': direction,
            'adx_end': indicators.get('adx', 0),
            'adx_delta': velocities['adx'],  # ✅ REAL velocity
            'candle_ratio': 1.0,  # TODO: calculate from price data
            'momentum_end': indicators.get('momentum_age', 0),
            'momentum_delta': velocities['momentum'],  # ✅ REAL velocity
            'exhaustion_pct': indicators.get('exhaustion_pct', 50),
            'room_pct': indicators.get('room_pct', 0),
            'exhaustion_velocity': velocities['exhaustion'],  # ✅ REAL velocity
            'pivot_types': [],
            'position_pct': indicators.get('grid_position', 50)
        }
    
    def _build_ladder_analysis(self, symbol, indicators, velocities, direction):
        """
        Constructs ladder_analysis with velocity-aware phase detection.
        """
        adx = indicators.get('adx', 0)
        di_plus = indicators.get('di_plus', 0)
        di_minus = indicators.get('di_minus', 0)
        exhaustion_pct = indicators.get('exhaustion_pct', 50)
        room_pct = indicators.get('room_pct', 0)
        momentum_age = indicators.get('momentum_age', 0)
        
        # TIME LADDER: Phase based on velocity, not static thresholds
        phase = self._get_phase_time_ladder(
            exhaustion_pct, 
            velocities['exhaustion'],
            velocities['exhaustion_accel']
        )
        
        # ADX trend from velocity
        if velocities['adx'] > 0.5:
            adx_trend = 'creciendo'
        elif velocities['adx'] < -0.5:
            adx_trend = 'decreciendo'
        else:
            adx_trend = 'estable'
        
        ladder_15m = {
            'direction': direction,
            'adx': {
                'adx_current': adx,
                'di_plus': di_plus,
                'di_minus': di_minus,
                'spread': abs(di_plus - di_minus),
                'trend': adx_trend,
                'velocity': velocities['adx']  # NEW: velocity included
            },
            'exhaustion_pct': exhaustion_pct,
            'room_pct': room_pct,
            'exhaustion_phase': phase,
            'exhaustion_velocity': velocities['exhaustion'],  # NEW
            'momentum': {
                'consecutive_count': momentum_age,
                'current_color': 'green' if momentum_age > 0 else 'red',
                'velocity': velocities['momentum']  # NEW
            }
        }
        
        return {
            '15m': ladder_15m,
            '5m': {},
            '1h': {},
            '4h': {}
        }
    
    def _get_phase_time_ladder(self, exhaustion_pct, velocity, acceleration):
        """
        TIME LADDER COMPLIANT phase detection.
        
        Uses TRAJECTORY (velocity + acceleration), not snapshot.
        """
        # Acceleration-based detection (most important)
        if acceleration > 0.5:
            return 'ACCELERATING_EXHAUSTION'  # Getting worse fast
        elif acceleration < -0.5:
            return 'RECOVERING_FAST'  # Getting better fast
        
        # Velocity-based detection
        if velocity > 5:
            return 'BURNING_BRUTAL'  # Very fast exhaustion
        elif velocity > 2:
            return 'BURNING_FAST'
        elif velocity > 0:
            return 'DEVELOPING'
        elif velocity < -3:
            return 'RECOVERING'
        elif velocity < 0:
            return 'COOLING'
        else:
            return 'STABLE'
    
    def _extract_confidence_breakdown(self, result, velocities):
        """
        Extracts confidence factors with velocity context.
        """
        action = result.get('action', 'WAIT')
        confidence = result.get('confidence', 0)
        details = result.get('details', {})
        
        factors = []
        
        # Extract from details if available
        if 'confidence_factors' in details:
            for factor_tuple in details['confidence_factors']:
                if len(factor_tuple) >= 2:
                    factors.append({
                        'name': factor_tuple[0],
                        'value': round(factor_tuple[1], 1) if factor_tuple[1] else 0,
                        'reason': str(factor_tuple[2]) if len(factor_tuple) > 2 else ''
                    })
        
        if not factors:
            factors.append({
                'name': 'Total',
                'value': confidence if confidence else 0,
                'reason': 'Details not available'
            })
        
        decision = 'ENTER' if action == 'OPERAR' else 'WAIT'
        threshold = 40
        
        if confidence and confidence >= threshold:
            decision_reason = f"Confidence {confidence:.0f}% >= {threshold}% threshold"
        else:
            decision_reason = f"Confidence {confidence:.0f}% < {threshold}% threshold" if confidence else "No confidence calculated"
        
        return {
            'confidence_factors': factors,
            'total_confidence': round(confidence, 1) if confidence else 0,
            'decision': decision,
            'decision_reason': decision_reason,
            'velocities': velocities  # Include velocities in output
        }

def _detect_bugs_time_ladder(self):
        """
        TIME LADDER COMPLIANT bug detection.
        
        Analyzes ACCELERATIONS and TRAJECTORIES, not threshold crossings.
        """
        bugs = []
        
        brain_frames = [f.get('bot_brain', {}) for f in self.frames if 'bot_brain' in f]
        
        if not brain_frames:
            return bugs
        
        # BUG 1: Factor never used (same as before)
        factor_usage = {}
        for brain in brain_frames:
            for factor in brain.get('confidence_factors', []):
                name = factor.get('name', 'Unknown')
                value = factor.get('value', 0)
                
                if name not in factor_usage:
                    factor_usage[name] = []
                factor_usage[name].append(value if value else 0)
        
        for factor_name, values in factor_usage.items():
            if values and all(v == 0 for v in values):
                bugs.append({
                    'bug': f'{factor_name.upper()}_NEVER_USED',
                    'evidence': f'{factor_name} = 0 pts in ALL {len(values)} frames',
                    'impact': f'Bot blind to {factor_name}',
                    'fix': f'Implement {factor_name} in evaluate_confidence()',
                    'severity': 'HIGH'
                })
        
        # BUG 2: TIME LADDER - Entered while decelerating
        confidences = [b.get('total_confidence', 0) for b in brain_frames]
        
        if len(confidences) >= 4:
            # Calculate confidence velocities
            conf_velocities = [
                confidences[i] - confidences[i-1] 
                for i in range(1, len(confidences))
            ]
            
            # Check if decelerating at entry
            mid = len(conf_velocities) // 2
            vel_first_half = sum(conf_velocities[:mid]) / mid if mid > 0 else 0
            vel_second_half = sum(conf_velocities[mid:]) / (len(conf_velocities) - mid) if (len(conf_velocities) - mid) > 0 else 0
            
            # Decelerating if second half velocity < first half
            if vel_second_half < vel_first_half - 2:
                bugs.append({
                    'bug': 'ENTERED_WHILE_DECELERATING',
                    'evidence': f'Confidence velocity: {vel_first_half:.1f} → {vel_second_half:.1f}',
                    'impact': 'Entered when momentum was dying',
                    'fix': 'Add acceleration check before entry',
                    'severity': 'HIGH'
                })
        
        # BUG 3: Exhaustion accelerating at entry
        entry_brain = None
        for f in self.frames:
            if 'bot_brain' in f:
                rel_min = int((f['timestamp'] - self.trade_data['entry_time']).total_seconds() / 60)
                if abs(rel_min) <= 5:  # Within 5 min of entry
                    entry_brain = f['bot_brain']
                    break
        
        if entry_brain:
            velocities = entry_brain.get('velocities', {})
            exh_accel = velocities.get('exhaustion_accel', 0)
            
            if exh_accel > 0.3:
                bugs.append({
                    'bug': 'EXHAUSTION_ACCELERATING_AT_ENTRY',
                    'evidence': f'Exhaustion acceleration: {exh_accel:.2f}',
                    'impact': 'Trend burning out faster and faster',
                    'fix': 'Block entry when exhaustion is accelerating',
                    'severity': 'CRITICAL'
                })
        
        # BUG 4: Contradictory factor (kept from before, fixed)
        if entry_brain:
            for factor in entry_brain.get('confidence_factors', []):
                reason_lower = factor.get('reason', '').lower()
                value = factor.get('value', 0)
                
                negative_signals = ['closing', 'cayendo', 'débil', 'contradice', 'bajando', 'muriendo']
                if any(sig in reason_lower for sig in negative_signals) and value and value > 0:
                    bugs.append({
                        'bug': f'{factor["name"].upper()}_BONUS_ON_NEGATIVE',
                        'evidence': f"+{value} pts despite '{factor['reason']}'",
                        'impact': 'Rewards bad signals',
                        'fix': f'Fix {factor["name"]} calculation',
                        'severity': 'HIGH'
                    })
        
        return bugs
    
    def _analyze_confidence_trajectory(self):
        """
        TIME LADDER: Analyzes confidence VELOCITY and ACCELERATION.
        """
        brain_frames = [f.get('bot_brain', {}) for f in self.frames if 'bot_brain' in f]
        
        if not brain_frames:
            return {}
        
        times = []
        for f in self.frames:
            if 'bot_brain' in f:
                rel = int((f['timestamp'] - self.trade_data['entry_time']).total_seconds() / 60)
                times.append(rel)
        
        confidences = [b.get('total_confidence', 0) for b in brain_frames]
        
        # Calculate velocity trajectory
        velocities = []
        for i in range(1, len(confidences)):
            velocities.append(round(confidences[i] - confidences[i-1], 1))
        
        # Calculate acceleration
        accelerations = []
        for i in range(1, len(velocities)):
            accelerations.append(round(velocities[i] - velocities[i-1], 2))
        
        # Interpretation based on trajectories
        avg_velocity = sum(velocities) / len(velocities) if velocities else 0
        avg_acceleration = sum(accelerations) / len(accelerations) if accelerations else 0
        
        if avg_acceleration > 0.5:
            interpretation = f"Confidence ACCELERATING (+{avg_acceleration:.1f}/frame²) - strengthening"
        elif avg_acceleration < -0.5:
            interpretation = f"Confidence DECELERATING ({avg_acceleration:.1f}/frame²) - weakening"
        elif avg_velocity > 2:
            interpretation = f"Confidence GROWING steadily (+{avg_velocity:.1f}/frame)"
        elif avg_velocity < -2:
            interpretation = f"Confidence FALLING ({avg_velocity:.1f}/frame)"
        else:
            interpretation = f"Confidence STABLE (vel: {avg_velocity:.1f}, accel: {avg_acceleration:.1f})"
        
        return {
            'frames': times,
            'values': [round(c, 1) for c in confidences],
            'velocities': velocities,
            'accelerations': accelerations,
            'avg_velocity': round(avg_velocity, 2),
            'avg_acceleration': round(avg_acceleration, 2),
            'interpretation': interpretation
        }
    
    def _analyze_factor_contributions(self):
        """
        TIME LADDER: Analyzes factor TRENDS, not just averages.
        """
        brain_frames = [f.get('bot_brain', {}) for f in self.frames if 'bot_brain' in f]
        
        if not brain_frames:
            return {}
        
        factor_series = {}
        
        for brain in brain_frames:
            for factor in brain.get('confidence_factors', []):
                name = factor.get('name', 'Unknown')
                value = factor.get('value', 0) or 0
                
                if name not in factor_series:
                    factor_series[name] = []
                factor_series[name].append(value)
        
        contributions = {}
        for name, values in factor_series.items():
            if not values:
                continue
                
            # Safe calculations
            min_val = min(values)
            max_val = max(values)
            avg_val = sum(values) / len(values)
            
            # Calculate trend (velocity)
            if len(values) >= 2:
                trend = (values[-1] - values[0]) / (len(values) - 1)
            else:
                trend = 0
            
            # TIME LADDER: Classify by TREND not absolute value
            impact = self._classify_impact_time_ladder(values, trend)
            
            contributions[name] = {
                'range': [round(min_val, 1), round(max_val, 1)],
                'avg': round(avg_val, 1),
                'trend': round(trend, 2),
                'impact': impact
            }
        
        return contributions
    
    def _classify_impact_time_ladder(self, values, trend):
        """
        TIME LADDER: Classify impact by TREND, not absolute thresholds.
        """
        if not values:
            return 'NONE'
        
        max_val = max(values)
        avg_val = sum(values) / len(values)
        
        # All zeros = bug
        if max_val == 0:
            return 'NONE - BUG: never used'
        
        # Trend-based classification
        if abs(trend) > 2:
            if trend > 0:
                return 'GROWING - increasing contribution'
            else:
                return 'SHRINKING - decreasing contribution'
        
        # Variance-based
        variance = sum((v - avg_val) ** 2 for v in values) / len(values)
        if variance > 25:
            return 'VOLATILE - high variance'
        
        # Relative contribution
        if avg_val > 10:
            return 'HIGH - major contributor'
        elif avg_val > 5:
            return 'MEDIUM - moderate contributor'
        else:
            return 'LOW - minor contributor'
