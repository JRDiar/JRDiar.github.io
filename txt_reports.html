def calculate_momentum_with_memory(df, lookback=10):
    """
    LazyBear Momentum (versión 3.5) con historia y fuerza.
    Devuelve color, historia, fuerza y valores de bandas.
    Requiere columnas: 'high', 'low', 'close', 'volume'
    """
    df = df.copy()
    n = len(df)
    if n < max(BB_LENGTH, KC_LENGTH) + 2:
        return {
            'color_current': 'neutral',
            'color_history': [],
            'consecutive_count': 0,
            'strength': 'débil',
            'value': 0.0,
            'bb_upper': 0.0,
            'bb_lower': 0.0,
            'kc_upper': 0.0,
            'kc_lower': 0.0
        }

    # 1. Cálculo de Media y Desviación (Bollinger)
    close = df['close'].values
    bb_mid = np.zeros(n)
    bb_std = np.zeros(n)
    for i in range(BB_LENGTH - 1, n):
        window = close[i - BB_LENGTH + 1:i + 1]
        bb_mid[i] = np.mean(window)
        bb_std[i] = np.std(window)

    bb_upper = bb_mid + BB_MULT * bb_std
    bb_lower = bb_mid - BB_MULT * bb_std

    # 2. Keltner Channel (con True Range o rango simple)
    high = df['high'].values
    low = df['low'].values
    tr = np.zeros(n)
    for i in range(1, n):
        tr[i] = max(
            high[i] - low[i],
            abs(high[i] - close[i - 1]),
            abs(low[i] - close[i - 1])
        ) if USE_TRUE_RANGE else (high[i] - low[i])

    atr = np.zeros(n)
    atr[KC_LENGTH] = np.mean(tr[1:KC_LENGTH + 1])
    for i in range(KC_LENGTH + 1, n):
        atr[i] = (atr[i - 1] * (KC_LENGTH - 1) + tr[i]) / KC_LENGTH

    kc_mid = np.zeros(n)
    for i in range(KC_LENGTH - 1, n):
        kc_mid[i] = np.mean(close[i - KC_LENGTH + 1:i + 1])

    kc_upper = kc_mid + KC_MULT * atr
    kc_lower = kc_mid - KC_MULT * atr

    # 3. Momentum Value (LazyBear)
    value = (close - (bb_upper + bb_lower + kc_upper + kc_lower) / 4)

    # 4. Color y fuerza
    colors = []
    for i in range(n):
        if value[i] > 0:
            colors.append('green')
        elif value[i] < 0:
            colors.append('red')
        else:
            colors.append('neutral')

    # Historia y fuerza
    start_idx = max(0, n - lookback)
    color_history = colors[start_idx:n]
    color_current = colors[-1]
    consecutive_count = 1
    for i in range(len(color_history) - 2, -1, -1):
        if color_history[i] == color_current:
            consecutive_count += 1
        else:
            break

    if consecutive_count >= 5:
        strength = 'fuerte'
    elif consecutive_count >= 3:
        strength = 'moderada'
    else:
        strength = 'débil'

    # Verificar squeeze (DI juntos = no operar)
    squeeze_on = df['SQUEEZE_ON'].iloc[-1] if 'SQUEEZE_ON' in df.columns else False

    if squeeze_on:
        return {
            'color_current': 'neutral',
            'color_history': [],
            'consecutive_count': 0,
            'strength': 'débil',
            'squeeze_active': True
        }

    return {
        'color_current': color_current,
        'color_history': color_history,
        'consecutive_count': consecutive_count,
        'strength': strength,
        'value': float(value[-1]),
        'bb_upper': float(bb_upper[-1]),
        'bb_lower': float(bb_lower[-1]),
        'kc_upper': float(kc_upper[-1]),
        'kc_lower': float(kc_lower[-1])
    }
