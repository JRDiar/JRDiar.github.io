# NUEVO: Verificar velocidad de precio
def check_price_velocity(symbol, direction, timeframe='5m'):
    """
    Verifica si el precio se mueve en la dirección correcta.
    LONG: precio debe estar aumentando minuto a minuto
    SHORT: precio debe estar disminuyendo minuto a minuto
    """
    storage = get_frame_storage(symbol, timeframe)
    closes = storage.get_trajectory('ohlc.close', 5)
    
    if not closes or len(closes) < 3:
        return {'valid': False, 'reason': 'Datos insuficientes'}
    
    # Calcular velocidad (cambio por frame)
    velocity = (closes[-1] - closes[-3]) / closes[-3] * 100
    
    if direction == 'BULLISH':
        if velocity > 0.1:  # Subiendo > 0.1% en 3 frames
            return {'valid': True, 'velocity': velocity, 'reason': f'Precio subiendo +{velocity:.2f}%'}
        else:
            return {'valid': False, 'velocity': velocity, 'reason': f'Precio NO sube ({velocity:.2f}%)'}
    else:  # BEARISH
        if velocity < -0.1:  # Bajando > 0.1%
            return {'valid': True, 'velocity': velocity, 'reason': f'Precio bajando {velocity:.2f}%'}
        else:
            return {'valid': False, 'velocity': velocity, 'reason': f'Precio NO baja ({velocity:.2f}%)'}

def unified_time_ladder_vision(symbol, direction, zigzag_data, grid_data, timeframe='15m'):
    """
    VISIÓN UNIFICADA: ZigZag + Grid + OHLC
    Joel: "Las tres cosas unirlas para sacar una conclusión"
    """
    # 1. Obtener patrón de precio (OHLC)
    price_pattern = detect_price_pattern(symbol, direction, timeframe)
    
    # 2. Verificar ZigZag
    zigzag_trend = zigzag_data.get('current_trend', 'UNKNOWN')
    zigzag_color = 'GREEN' if zigzag_trend == 'BULLISH' else 'RED' if zigzag_trend == 'BEARISH' else 'GRAY'
    
    # 3. Verificar Grid/Malla
    grid_position = grid_data.get('position_pct', 50)
    near_resistance = grid_position > 80
    near_support = grid_position < 20
    
    # 4. Verificar velocidad de precio
    velocity_check = check_price_velocity(symbol, direction, '5m')
    
    # === DECISIÓN UNIFICADA ===
    issues = []
    
    # ZigZag debe alinearse con dirección
    if direction == 'BULLISH' and zigzag_color == 'RED':
        issues.append('ZigZag ROJO (corrección activa)')
    elif direction == 'BEARISH' and zigzag_color == 'GREEN':
        issues.append('ZigZag VERDE (rally activo)')
    
    # Precio debe moverse en dirección correcta
    if not velocity_check['valid']:
        issues.append(velocity_check['reason'])
    
    # LONG no debe estar en resistencia
    if direction == 'BULLISH' and near_resistance:
        issues.append('Cerca de resistencia (grid > 80%)')
    
    # SHORT no debe estar en soporte
    if direction == 'BEARISH' and near_support:
        issues.append('Cerca de soporte (grid < 20%)')
    
    return {
        'approved': len(issues) == 0,
        'issues': issues,
        'price_pattern': price_pattern,
        'zigzag_color': zigzag_color,
        'grid_position': grid_position,
        'velocity': velocity_check
    }
