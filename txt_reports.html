"""
VAR SYSTEM - FRAME RECORDER
============================
Graba frames históricos de indicadores para análisis post-trade.
NO modifica lógica de decisión - solo almacena datos.
"""
from collections import deque
from datetime import datetime
import json
import os
class FrameRecorder:
    """
    Almacena frames históricos por símbolo para calcular velocities/accelerations.
    
    Cada frame contiene snapshot completo de todos los indicadores.
    """
    
    def __init__(self, max_frames=10):
        """
        Args:
            max_frames: Cantidad máxima de frames a almacenar por símbolo
        """
        self.max_frames = max_frames
        # {symbol: deque(frames)}
        self.frames_by_symbol = {}
        
        # Directorio para guardar JSONs
        self.output_dir = 'var_recordings'
        os.makedirs(self.output_dir, exist_ok=True)
    
    def record_frame(self, symbol, frame_data):
        """
        Agrega un nuevo frame al historial del símbolo.
        
        Args:
            symbol: Símbolo (ej: "BTC/USDT:USDT")
            frame_data: Dict con todos los indicadores del frame actual
        
        frame_data debe contener:
        {
            'timestamp': timestamp,
            'adx': float,
            'di_plus': float,
            'di_minus': float,
            'exhaustion_pct': float,
            'room_pct': float,
            'momentum_age': int,
            'candle_size': float,
            'spread': float,
            'grid_position': float,
            # ... otros indicadores
        }
        """
        if symbol not in self.frames_by_symbol:
            self.frames_by_symbol[symbol] = deque(maxlen=self.max_frames)
        
        # Agregar timestamp si no existe
        if 'timestamp' not in frame_data:
            frame_data['timestamp'] = datetime.now().timestamp()
        
        self.frames_by_symbol[symbol].append(frame_data)
    
    def get_frames(self, symbol):
        """Retorna todos los frames almacenados para un símbolo"""
        return list(self.frames_by_symbol.get(symbol, []))
    
    def calculate_velocities(self, symbol):
        """
        Calcula velocities de todos los indicadores.
        
        velocity = (valor_final - valor_inicial) / num_frames
        
        Returns:
            Dict con velocities de cada indicador
        """
        frames = self.get_frames(symbol)
        
        if len(frames) < 2:
            return {}
        
        first_frame = frames[0]
        last_frame = frames[-1]
        num_frames = len(frames)
        
        velocities = {}
        
        # Lista de indicadores numéricos a calcular velocity
        numeric_indicators = [
            'adx', 'di_plus', 'di_minus',
            'exhaustion_pct', 'room_pct',
            'momentum_age', 'candle_size',
            'spread', 'grid_position'
        ]
        
        for indicator in numeric_indicators:
            if indicator in first_frame and indicator in last_frame:
                initial = first_frame[indicator]
                final = last_frame[indicator]
                
                if initial is not None and final is not None:
                    velocities[f'{indicator}_velocity'] = (final - initial) / num_frames
        
        return velocities
    
    def calculate_accelerations(self, symbol):
        """
        Calcula accelerations (segunda derivada).
        
        acceleration = velocity_reciente - velocity_anterior
        
        Returns:
            Dict con accelerations de cada indicador
        """
        frames = self.get_frames(symbol)
        
        if len(frames) < 3:
            return {}
        
        # Dividir en dos mitades
        mid = len(frames) // 2
        first_half = frames[:mid+1]
        second_half = frames[mid:]
        
        accelerations = {}
        
        numeric_indicators = [
            'adx', 'di_plus', 'di_minus',
            'exhaustion_pct', 'room_pct',
            'momentum_age', 'candle_size',
            'spread', 'grid_position'
        ]
        
        for indicator in numeric_indicators:
            # Velocity primera mitad
            if indicator in first_half[0] and indicator in first_half[-1]:
                v1 = (first_half[-1][indicator] - first_half[0][indicator]) / len(first_half)
            else:
                continue
            
            # Velocity segunda mitad
            if indicator in second_half[0] and indicator in second_half[-1]:
                v2 = (second_half[-1][indicator] - second_half[0][indicator]) / len(second_half)
            else:
                continue
            
            # Acceleration
            accelerations[f'{indicator}_accel'] = v2 - v1
        
        return accelerations
    
    def generate_trade_json(self, symbol, trade_data):
        """
        Genera JSON completo de un trade con frames históricos.
        
        Args:
            symbol: Símbolo
            trade_data: Dict con datos del trade
            {
                'entry_time': datetime,
                'exit_time': datetime,
                'entry_price': float,
                'exit_price': float,
                'pnl_pct': float,
                'outcome': 'WIN'/'LOSS',
                'direction': 'BULLISH'/'BEARISH'
            }
        
        Returns:
            Nombre del archivo JSON guardado
        """
        frames = self.get_frames(symbol)
        
        if not frames:
            return None
        
        # Calcular velocities y accelerations
        velocities = self.calculate_velocities(symbol)
        accelerations = self.calculate_accelerations(symbol)
        
        # Estructura completa
        var_data = {
            'meta': {
                'symbol': symbol,
                'entry_time': trade_data.get('entry_time').isoformat() if trade_data.get('entry_time') else None,
                'exit_time': trade_data.get('exit_time').isoformat() if trade_data.get('exit_time') else None,
                'entry_price': trade_data.get('entry_price'),
                'exit_price': trade_data.get('exit_price'),
                'pnl_pct': trade_data.get('pnl_pct'),
                'outcome': trade_data.get('outcome'),
                'direction': trade_data.get('direction'),
                'generated_at': datetime.now().isoformat()
            },
            'frames': frames,
            'velocities': velocities,
            'accelerations': accelerations,
            'analysis': {
                'frame_count': len(frames),
                'time_span_seconds': frames[-1]['timestamp'] - frames[0]['timestamp'] if len(frames) > 0 else 0
            }
        }
        
        # Nombre de archivo
        timestamp_str = datetime.now().strftime('%Y%m%d_%H%M%S')
        symbol_clean = symbol.replace('/', '').replace(':', '')
        outcome = trade_data.get('outcome', 'UNKNOWN')
        
        filename = f"var_{timestamp_str}_{symbol_clean}_{outcome}.json"
        filepath = os.path.join(self.output_dir, filename)
        
        # Guardar
        with open(filepath, 'w') as f:
            json.dump(var_data, f, indent=2)
        
        return filename
    
    def clear_symbol(self, symbol):
        """Limpia frames de un símbolo (llamar después de cerrar trade)"""
        if symbol in self.frames_by_symbol:
            del self.frames_by_symbol[symbol]
