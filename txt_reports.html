class SheetsManager:
    """Gestiona conexión y escritura a Google Sheets"""
    
    def __init__(self):
        self.client = None
        self.spreadsheet = None
        self.connected = False
        
        if ENABLE_SHEETS and GOOGLE_CREDENTIALS_FILE and GOOGLE_SPREADSHEET_ID:
            self._connect()
    
    def _connect(self):
        """Establece conexión con Google Sheets"""
        try:
            scope = [
                'https://spreadsheets.google.com/feeds',
                'https://www.googleapis.com/auth/drive'
            ]
            
            creds = ServiceAccountCredentials.from_json_keyfile_name(
                GOOGLE_CREDENTIALS_FILE, scope
            )
            
            self.client = gspread.authorize(creds)
            self.spreadsheet = self.client.open_by_key(GOOGLE_SPREADSHEET_ID)
            self.connected = True
            
            print("✅ Google Sheets conectado")
            
        except Exception as e:
            print(f"❌ Error conectando Google Sheets: {e}")
            self.connected = False
    
    def _get_sheet(self, sheet_name):
        """Obtiene o crea una hoja"""
        if not self.connected:
            return None
        
        try:
            return self.spreadsheet.worksheet(sheet_name)
        except gspread.WorksheetNotFound:
            # Crear hoja si no existe
            return self.spreadsheet.add_worksheet(title=sheet_name, rows=1000, cols=20)
    
    def write_row(self, sheet_name, row_data):
        """
        Escribe una fila en una hoja
        
        Args:
            sheet_name: Nombre de la hoja
            row_data: Lista de valores para la fila
        
        Returns:
            bool: True si éxito, False si error
        """
        if not self.connected:
            return False
        
        try:
            sheet = self._get_sheet(sheet_name)
            if not sheet:
                return False
            
            sheet.append_row(row_data)
            return True
            
        except Exception as e:
            print(f"❌ Error escribiendo fila en {sheet_name}: {e}")
            return False
    
    def write_batch(self, sheet_name, rows_data):
        """
        Escribe múltiples filas en una hoja (batch)
        
        Args:
            sheet_name: Nombre de la hoja
            rows_data: Lista de listas [[row1], [row2], ...]
        
        Returns:
            bool: True si éxito, False si error
        """
        if not self.connected or not BATCH_WRITE:
            return False
        
        try:
            sheet = self._get_sheet(sheet_name)
            if not sheet:
                return False
            
            # Escribir TODAS las filas en un solo request API
            sheet.append_rows(rows_data, value_input_option='USER_ENTERED')
            return True
            
        except Exception as e:
            print(f"❌ Error escribiendo batch en {sheet_name}: {e}")
            return False
    
    def clear_sheet(self, sheet_name):
        """Limpia todo el contenido de una hoja"""
        if not self.connected:
            return False
        
        try:
            sheet = self._get_sheet(sheet_name)
            if not sheet:
                return False
            
            sheet.clear()
            return True
            
        except Exception as e:
            print(f"❌ Error limpiando {sheet_name}: {e}")
            return False

        Args:
            symbol: Símbolo (ej: BTC/USDT:USDT)
            weekly_candles: Número de velas semanales
            status: APROBADO o RECHAZADO

        row = [
            timestamp,
            symbol,
            weekly_candles,
            status
        ]
        
        return self.write_row(SHEET_BRAIN_PRE_FILTRO, row)

    def write_trade_entry(self, symbol, direction, entry_price, stop_loss, 
                         take_profit, score, rr_ratio):
        """
        Escribe entrada de trade en BRAIN_TRADES
        
        Args:
            symbol: Símbolo
            direction: BULLISH o BEARISH
            entry_price: Precio entrada
            stop_loss: Stop loss
            take_profit: Take profit
            score: Score Brain
            rr_ratio: Risk/Reward ratio
        """
        timestamp = datetime.now(ARGENTINA_TZ).strftime("%Y-%m-%d %H:%M:%S")
        
        row = [
            timestamp,
            symbol,
            "ENTRADA",
            direction,
            entry_price,
            stop_loss,
            take_profit,
            score,
            rr_ratio,
            "",  # PnL (se llena en salida)
            "",  # Duración (se llena en salida)
            ""   # Razón salida (se llena en salida)
        ]
        
        return self.write_row(SHEET_BRAIN_TRADES, row)
    
    def write_trade_exit(self, symbol, exit_price, pnl_pct, pnl_usd, 
                        duration_min, exit_reason):
        """
        Escribe salida de trade en BRAIN_TRADES
        
        Args:
            symbol: Símbolo
            exit_price: Precio salida
            pnl_pct: PnL en %
            pnl_usd: PnL en USD
            duration_min: Duración en minutos
            exit_reason: Razón de salida
        """
        timestamp = datetime.now(ARGENTINA_TZ).strftime("%Y-%m-%d %H:%M:%S")
        
        row = [
            timestamp,
            symbol,
            "SALIDA",
            "",  # Direction (vacío en salida)
            exit_price,
            "",  # Stop loss (vacío en salida)
            "",  # Take profit (vacío en salida)
            "",  # Score (vacío en salida)
            "",  # RR (vacío en salida)
            f"{pnl_pct:+.2f}% | ${pnl_usd:+.2f}",
            f"{duration_min:.1f} min",
            exit_reason
        ]
        
        return self.write_row(SHEET_BRAIN_TRADES, row)

    def write_stats(self, total_trades, wins, losses, win_rate, avg_pnl, 
                   total_pnl, best_trade, worst_trade):
        """
        Escribe estadísticas en BRAIN_STATS
        
        Args:
            total_trades: Total de trades
            wins: Trades ganadores
            losses: Trades perdedores
            win_rate: Tasa de acierto (%)
            avg_pnl: PnL promedio (%)
            total_pnl: PnL total (USD)

        """
        timestamp = datetime.now(ARGENTINA_TZ).strftime("%Y-%m-%d %H:%M:%S")
        
        row = [

            wins,
            losses,
            f"{win_rate:.2f}%",
            f"{avg_pnl:+.2f}%",
            f"${total_pnl:+.2f}",

    def write_rejection(self, symbol, score, reason, adx_4h, adx_1h, adx_15m, 
                    mom_4h, mom_1h, mom_15m, vp_position, sr_danger, 
                    structure, divergence, precio_actual, warnings,
                    probability, score_4h, score_1h, score_15m, evolution_bonus):
        """Registra monedas rechazadas con contexto completo"""
        timestamp = datetime.now(ARGENTINA_TZ).strftime("%Y-%m-%d %H:%M:%S")
        
        # Usar warnings directamente
        warnings_text = warnings if isinstance(warnings, str) else ""
        
        row = [
            timestamp, symbol, score, reason,
            adx_4h, adx_1h, adx_15m,
            mom_4h, mom_1h, mom_15m,
            vp_position if isinstance(vp_position, str) else vp_position.get('position', 'N/A'),
            'SI' if sr_danger else 'NO',
            structure if isinstance(structure, str) else structure.get('pattern', 'N/A'),
            divergence if isinstance(divergence, str) else divergence.get('type', 'NONE'),
            precio_actual,
            warnings_text,
            probability,
            score_4h,
            score_1h,
            score_15m,
            evolution_bonus
        ]
        
        # Crear hoja con headers si no existe
        sheet_name = 'BRAIN_RECHAZOS'
        sheet = self._get_sheet(sheet_name)
        if sheet and sheet.row_count == 0:
            headers = [
                'Timestamp','Symbol','Score','Reason','ADX_4H','ADX_1H','ADX_15M',
                'Mom_4H','Mom_1H','Mom_15M','VP_Position','SR_Danger','Structure',
                'Divergence','Precio','Warnings',
                'Probability','Score_4H','Score_1H','Score_15M','Evolution_Bonus'
            ]
            sheet.append_row(headers)
        
        self.write_row(sheet_name, row)

        # SQLite (ilimitado)
        db_manager.insert_rejection(
            symbol, score, reason, adx_4h, adx_1h, adx_15m,
            mom_4h, mom_1h, mom_15m, vp_position, sr_danger,
            structure, divergence, precio_actual, warnings_text,
            probability, score_4h, score_1h, score_15m, evolution_bonus
        )
