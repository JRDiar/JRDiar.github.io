"""
brain_adx_zones.py - ADX Cortometraje con Rangos
================================================
Analiza ADX como movimiento entre ZONAS, no valores estáticos.

Zonas:
1. LATERAL (0-20): Sin tendencia
2. INICIANDO (20-25): Naciendo
3. TENDENCIA (25-35): Confirmada
4. FUERTE (35-50): Fuerte
5. BRUTAL (50-70): Explosión
6. EXTREMO (70+): Sobreextensión
"""

class ADXZoneAnalyzer:
    """
    Analiza ADX como cortometraje con zonas
    """
    
    # Definición de zonas
    ZONES = {
        'LATERAL': (0, 20),
        'INICIANDO': (20, 25),
        'TENDENCIA': (25, 35),
        'FUERTE': (35, 50),
        'BRUTAL': (50, 70),
        'EXTREMO': (70, 100)
    }
    
    def get_zone(self, adx_value):
        """
        Determina zona del ADX
        
        Args:
            adx_value: Valor ADX
            
        Returns:
            str: Nombre de la zona
        """
        for zone_name, (min_val, max_val) in self.ZONES.items():
            if min_val <= adx_value < max_val:
                return zone_name
        
        return 'EXTREMO'  # >70
    
    def analyze_zone_trajectory(self, adx_frames):
        """
        Analiza trayectoria del ADX entre zonas
        
        Args:
            adx_frames: Lista de valores ADX [más viejo ... más nuevo]
                       Ej: [28, 30, 33, 35, 38, 41, 45]
        
        Returns:
            dict: {
                'zones': list,           # Zonas de cada frame
                'current_zone': str,     # Zona actual
                'zone_changes': list,    # Cruces de zona
                'direction': str,        # ASCENDING, DESCENDING, STABLE
                'strength': str,         # Fuerza del movimiento
                'story': str,            # Historia del cortometraje
                'score_bonus': int       # Bonus por movimiento
            }
        """
        
        if len(adx_frames) < 3:
            return self._default_result()
        
        # === MAPEAR ZONAS ===
        zones = [self.get_zone(adx) for adx in adx_frames]
        current_zone = zones[-1]
        
        # === DETECTAR CRUCES DE ZONA ===
        zone_changes = []
        for i in range(1, len(zones)):
            if zones[i] != zones[i-1]:
                zone_changes.append({
                    'from': zones[i-1],
                    'to': zones[i],
                    'frame': i,
                    'adx': adx_frames[i]
                })
        
        # === ANALIZAR DIRECCIÓN ===
        delta = adx_frames[-1] - adx_frames[0]
        
        if delta > 5:
            direction = 'ASCENDING'
        elif delta < -5:
            direction = 'DESCENDING'
        else:
            direction = 'STABLE'
        
        # === CALCULAR FUERZA DEL MOVIMIENTO ===
        velocity = delta / len(adx_frames)
        
        if abs(velocity) > 3:
            strength = 'EXPLOSIVE'
        elif abs(velocity) > 1.5:
            strength = 'STRONG'
        elif abs(velocity) > 0.5:
            strength = 'MODERATE'
        else:
            strength = 'WEAK'
        
        # === CREAR HISTORIA ===
        story = self._create_story(
            adx_frames, 
            zones, 
            zone_changes, 
            direction, 
            strength
        )
        
        # === CALCULAR SCORE BONUS ===
        score_bonus = self._calculate_zone_bonus(
            zones,
            zone_changes,
            direction,
            strength,
            current_zone
        )
        
        return {
            'zones': zones,
            'current_zone': current_zone,
            'zone_changes': zone_changes,
            'direction': direction,
            'strength': strength,
            'story': story,
            'score_bonus': score_bonus,
            'adx_frames': adx_frames,
            'valid': True
        }
    
    def _create_story(self, adx_frames, zones, zone_changes, direction, strength):
        """
        Crea historia del cortometraje ADX
        """
        
        # Frame inicial y final
        start_adx = adx_frames[0]
        end_adx = adx_frames[-1]
        start_zone = zones[0]
        end_zone = zones[-1]
        
        # Historia base
        if len(zone_changes) == 0:
            # Sin cruces de zona
            story = f"ADX {start_adx:.1f}→{end_adx:.1f} ({direction.lower()}) en zona {end_zone}"
        
        elif len(zone_changes) == 1:
            # Un cruce
            change = zone_changes[0]
            story = f"ADX cruzó de {change['from']} a {change['to']} ({start_adx:.1f}→{end_adx:.1f})"
        
        else:
            # Múltiples cruces
            zones_crossed = ' → '.join([c['to'] for c in zone_changes])
            story = f"ADX cruzó {len(zone_changes)} zonas: {start_zone} → {zones_crossed} ({start_adx:.1f}→{end_adx:.1f})"
        
        # Agregar fuerza
        story += f" | Fuerza: {strength}"
        
        return story
    
    def _calculate_zone_bonus(self, zones, zone_changes, direction, strength, current_zone):
        """
        Calcula bonus basado en movimiento entre zonas
        """
        
        bonus = 0
        
        # === BONUS POR ZONA ACTUAL ===
        zone_bonus = {
            'LATERAL': -40,      # Muy malo
            'INICIANDO': 0,      # Neutral
            'TENDENCIA': 10,     # Bueno
            'FUERTE': 20,        # Muy bueno
            'BRUTAL': 30,        # Excelente
            'EXTREMO': -10       # Precaución
        }
        bonus += zone_bonus.get(current_zone, 0)
        
        # === BONUS POR CRUCES ASCENDENTES ===
        if direction == 'ASCENDING':
            for change in zone_changes:
                # LATERAL → INICIANDO
                if change['from'] == 'LATERAL' and change['to'] == 'INICIANDO':
                    bonus += 10
                
                # INICIANDO → TENDENCIA (confirmación)
                elif change['from'] == 'INICIANDO' and change['to'] == 'TENDENCIA':
                    bonus += 20
                
                # TENDENCIA → FUERTE (fortalecimiento)
                elif change['from'] == 'TENDENCIA' and change['to'] == 'FUERTE':
                    bonus += 30
                
                # FUERTE → BRUTAL (explosión)
                elif change['from'] == 'FUERTE' and change['to'] == 'BRUTAL':
                    bonus += 40
        
        # === PENALTY POR CRUCES DESCENDENTES ===
        elif direction == 'DESCENDING':
            for change in zone_changes:
                # BRUTAL → FUERTE
                if change['from'] == 'BRUTAL' and change['to'] == 'FUERTE':
                    bonus -= 20
                
                # FUERTE → TENDENCIA
                elif change['from'] == 'FUERTE' and change['to'] == 'TENDENCIA':
                    bonus -= 30
                
                # TENDENCIA → INICIANDO
                elif change['from'] == 'TENDENCIA' and change['to'] == 'INICIANDO':
                    bonus -= 40
        
        # === BONUS POR FUERZA ===
        if strength == 'EXPLOSIVE' and direction == 'ASCENDING':
            bonus += 25
        elif strength == 'STRONG' and direction == 'ASCENDING':
            bonus += 15
        
        return bonus
    
    def _default_result(self):
        return {
            'zones': [],
            'current_zone': 'UNKNOWN',
            'zone_changes': [],
            'direction': 'UNKNOWN',
            'strength': 'UNKNOWN',
            'story': 'Datos insuficientes',
            'score_bonus': 0,
            'valid': False
        }


# === INTEGRACIÓN CON brain_trajectory.py ===

```
## 📋 **ARQUITECTURA NUEVA: ALIGNMENT DETECTOR**
```
================================================================================
SISTEMA DEFINITIVO: DETECTOR DE ALINEACIÓN GLOBAL
================================================================================

🎯 FILOSOFÍA:

NO importa si ADX es 30 o 50
NO importa si momentum es 3v o 7v
NO importa si volumen es 80% o 150%

LO QUE IMPORTA:
¿TODO apunta en la MISMA dirección?
¿La historia es COHERENTE?
¿Todos los indicadores "cantan la misma canción"?

================================================================================
ARCHIVO: brain_alignment.py
================================================================================

"""
brain_alignment.py - Detector de Alineación Global
==================================================
Detecta cuando TODOS los indicadores están alineados en la misma dirección.

NO busca perfección de valores individuales.
Busca COHERENCIA direccional del conjunto.
"""

class AlignmentDetector:
    """
    Detecta alineación global de todos los indicadores
    """
    
    def analyze_complete_alignment(self, symbol, direction):
        """
        Analiza alineación COMPLETA de todos los componentes
        
        Returns:
            dict: {
                'alignment_score': float,  # 0-100%
                'aligned_signals': list,   # Qué está alineado
                'conflicting_signals': list, # Qué NO está alineado
                'story': str,              # Historia coherente
                'decision': bool           # ENTRAR o NO
            }
        """
        
        # === OBTENER TODOS LOS DATOS ===
        
        from brain_frames import get_frame_storage
        from brain_trajectory import TrajectoryAnalyzer
        from brain_calculator import (
            calculate_adx_with_memory,
            calculate_momentum_with_memory,
            calculate_zigzag
        )
        from exchange_connection import fetch_ohlcv
        
        df = fetch_ohlcv(symbol, '5m', 100)
        
        adx_data = calculate_adx_with_memory(df)
        momentum_data = calculate_momentum_with_memory(df)
        zigzag_data = calculate_zigzag(df)
        
        storage = get_frame_storage(symbol, '15m')
        analyzer = TrajectoryAnalyzer()
        
        # === ANALIZAR CADA COMPONENTE ===
        
        signals = {}
        
        # 1. ADX DIRECCIÓN
        adx_values = storage.get_trajectory('adx.value', 5)
        if len(adx_values) >= 3:
            adx_delta = adx_values[-1] - adx_values[0]
            if direction == 'BULLISH':
                signals['adx'] = 'ALIGNED' if adx_delta > 0 else 'CONFLICT'
            else:
                signals['adx'] = 'ALIGNED' if adx_delta > 0 else 'CONFLICT'
        
        # 2. PRECIO DIRECCIÓN
        price_values = storage.get_trajectory('price', 5)
        if len(price_values) >= 3:
            price_delta = price_values[-1] - price_values[0]
            if direction == 'BULLISH':
                signals['price'] = 'ALIGNED' if price_delta > 0 else 'CONFLICT'
            else:
                signals['price'] = 'ALIGNED' if price_delta < 0 else 'CONFLICT'
        
        # 3. ZIGZAG ESTRUCTURA
        pivot_types = zigzag_data.get('pivot_types', [])
        if pivot_types and len(pivot_types) >= 2:
            recent = pivot_types[-2:]
            
            if direction == 'BULLISH':
                # LONG necesita HH o HL
                signals['zigzag'] = 'ALIGNED' if any(p in ['HH', 'HL'] for p in recent) else 'CONFLICT'
            else:
                # SHORT necesita LL o LH
                signals['zigzag'] = 'ALIGNED' if any(p in ['LL', 'LH'] for p in recent) else 'CONFLICT'
        
        # 4. MOMENTUM COLOR
        mom_color = momentum_data.get('color_current', 'gray')
        
        if direction == 'BULLISH':
            signals['momentum'] = 'ALIGNED' if mom_color in ['lime', 'green'] else 'CONFLICT'
        else:
            signals['momentum'] = 'ALIGNED' if mom_color in ['red', 'maroon'] else 'CONFLICT'
        
        # 5. DI DOMINANCIA
        di_plus = adx_data.get('di_plus_current', 0)
        di_minus = adx_data.get('di_minus_current', 0)
        
        if direction == 'BULLISH':
            signals['di'] = 'ALIGNED' if di_plus > di_minus else 'CONFLICT'
        else:
            signals['di'] = 'ALIGNED' if di_minus > di_plus else 'CONFLICT'
        
        # 6. VOLUMEN TENDENCIA
        volume_values = storage.get_trajectory('volume', 5)
        if len(volume_values) >= 3:
            volume_delta = volume_values[-1] - volume_values[0]

"""
brain_adx_zones.py - ADX Cortometraje con Zonas
================================================
Analiza ADX como movimiento entre ZONAS, no valores estáticos.
"""
class ADXZoneAnalyzer:
    """
    Analiza ADX como cortometraje con zonas
    """
    
    # Definición de zonas
    ZONES = {
        'LATERAL': (0, 20),
        'INICIANDO': (20, 25),
        'TENDENCIA': (25, 35),
        'FUERTE': (35, 50),
        'BRUTAL': (50, 70),
        'EXTREMO': (70, 100)
    }
    
    # Bonus por zona actual
    ZONE_BONUS = {
        'LATERAL': -40,
        'INICIANDO': 0,
        'TENDENCIA': 10,
        'FUERTE': 20,
        'BRUTAL': 30,
        'EXTREMO': -10
    }
    
    def get_zone(self, adx_value):
        """Determina zona del ADX"""
        for zone_name, (min_val, max_val) in self.ZONES.items():
            if min_val <= adx_value < max_val:
                return zone_name
        return 'EXTREMO'
    
    def analyze_zone_trajectory(self, adx_frames):
        """
        Analiza trayectoria del ADX entre zonas
        
        Args:
            adx_frames: Lista [más viejo ... más nuevo]
                       Ej: [28, 30, 33, 35, 38, 41, 45]
        
        Returns:
            dict con zonas, cruces, dirección, historia, bonus
        """
        if len(adx_frames) < 3:
            return self._default_result()
        
        # Mapear zonas
        zones = [self.get_zone(adx) for adx in adx_frames]
        current_zone = zones[-1]
        
        # Detectar cruces de zona
        zone_changes = []
        for i in range(1, len(zones)):
            if zones[i] != zones[i-1]:
                zone_changes.append({
                    'from': zones[i-1],
                    'to': zones[i],
                    'adx': adx_frames[i]
                })
        
        # Analizar dirección
        delta = adx_frames[-1] - adx_frames[0]
        if delta > 5:
            direction = 'ASCENDING'
        elif delta < -5:
            direction = 'DESCENDING'
        else:
            direction = 'STABLE'
        
        # Calcular fuerza
        velocity = delta / len(adx_frames)
        if abs(velocity) > 3:
            strength = 'EXPLOSIVE'
        elif abs(velocity) > 1.5:
            strength = 'STRONG'
        elif abs(velocity) > 0.5:
            strength = 'MODERATE'
        else:
            strength = 'WEAK'
        
        # Calcular bonus
        bonus = self._calculate_zone_bonus(
            current_zone, zone_changes, direction, strength
        )
        
        # Crear historia
        story = self._create_story(
            adx_frames, zones, zone_changes, direction, strength
        )
        
        return {
            'zones': zones,
            'current_zone': current_zone,
            'zone_changes': zone_changes,
            'direction': direction,
            'strength': strength,
            'story': story,
            'bonus': bonus,
            'is_aligned': direction == 'ASCENDING' and current_zone not in ['LATERAL'],
            'valid': True
        }
    
    def _calculate_zone_bonus(self, current_zone, zone_changes, direction, strength):
        """Calcula bonus por zonas y cruces"""
        bonus = self.ZONE_BONUS.get(current_zone, 0)
        
        # Bonus por cruces ascendentes
        if direction == 'ASCENDING':
            for change in zone_changes:
                if change['from'] == 'INICIANDO' and change['to'] == 'TENDENCIA':
                    bonus += 20  # Confirmación
                elif change['from'] == 'TENDENCIA' and change['to'] == 'FUERTE':
                    bonus += 30  # Fortalecimiento
                elif change['from'] == 'FUERTE' and change['to'] == 'BRUTAL':
                    bonus += 40  # Explosión
            
            # Bonus por fuerza
            if strength == 'EXPLOSIVE':
                bonus += 25
            elif strength == 'STRONG':
                bonus += 15
        
        # Penalty por cruces descendentes
        elif direction == 'DESCENDING':
            for change in zone_changes:
                if change['from'] == 'BRUTAL' and change['to'] == 'FUERTE':
                    bonus -= 20
                elif change['from'] == 'FUERTE' and change['to'] == 'TENDENCIA':
                    bonus -= 30
        
        return bonus
    
    def _create_story(self, adx_frames, zones, zone_changes, direction, strength):
        """Crea historia del cortometraje"""
        start_adx = adx_frames[0]
        end_adx = adx_frames[-1]
        
        if len(zone_changes) == 0:
            return f"ADX {start_adx:.0f}→{end_adx:.0f} ({direction}) en zona {zones[-1]}"
        elif len(zone_changes) == 1:
            c = zone_changes[0]
            return f"ADX cruzó {c['from']}→{c['to']} ({start_adx:.0f}→{end_adx:.0f})"
        else:
            return f"ADX cruzó {len(zone_changes)} zonas ({start_adx:.0f}→{end_adx:.0f}) | {strength}"
    
    def _default_result(self):
        return {
            'zones': [],
            'current_zone': 'UNKNOWN',
            'zone_changes': [],
            'direction': 'UNKNOWN',
            'strength': 'UNKNOWN',
            'story': 'Datos insuficientes',
            'bonus': 0,
            'is_aligned': False,
            'valid': False
        }

        structure = zigzag_data.get('structure', '')

            'aligned_count': aligned_count,
            'conflict_count': conflict_count,

            'aligned_list': aligned_list,
            'conflict_list': conflict_list,
            'neutral_list': neutral_list,

        }
