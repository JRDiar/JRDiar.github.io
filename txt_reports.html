def determine_strategy(patterns, ladder, scenarios=None):
    """
    Define Take Profit y Stop Loss.
    Usa Proyección Dinámica (Escenarios) si están disponibles.
    """
    pattern_type = patterns.get('best_pattern', {}).get('type', 'STANDARD')
    
    # Valores Base
    stop_mult = 2.0
    target_mult = 2.0
    position = 0.75
    
    # Ajuste por Patrón
    if 'DOUBLE' in pattern_type or 'BOS' in pattern_type:
        target_mult = 2.5
        position = 0.5
    elif 'FLAG' in pattern_type:
        target_mult = 3.0
        position = 1.0

    # Ajuste por Escenario (Proyección)
    if scenarios:
        primary = scenarios.get('primary', {})
        prob = primary.get('probability', 0)
        rr = primary.get('rr_ratio', 0)
        
        if prob > 60 and rr > 2.0:
            target_mult = rr # Usar target real del gráfico
            position = min(position + 0.25, 1.0)
        elif prob < 50:
            position *= 0.8 # Reducir mano

    return {
        'type': 'SCENARIO' if scenarios else 'PATTERN',
        'stop_multiplier': stop_mult,
        'target_multiplier': target_mult,
        'position_size': position
    }

def check_market_force(ladder):
    """
    Verifica FUERZA DE MERCADO (Blindada v5.0).
    Reglas: ADX > 25, Spread > 5, Momentum Alineado.
    """

    if not a1h: return {'detected': False, 'reason': "No data 1H"}

    adx_data = a1h.get('adx', {})
    mom_data = a1h.get('momentum', {})
    
    adx_val = adx_data.get('adx_current', 0)
    di_plus = adx_data.get('di_plus_current', 0)
    di_minus = adx_data.get('di_minus_current', 0)
    mom_color = mom_data.get('color_current', 'gray')
    
    force_direction = 'NEUTRAL'
    if di_plus > di_minus: force_direction = 'BULLISH'; spread = di_plus - di_minus
    else: force_direction = 'BEARISH'; spread = di_minus - di_plus
        
    if adx_val < 25: return {'detected': False, 'reason': f"ADX débil ({adx_val:.1f})"}
    if spread < 5.0: return {'detected': False, 'reason': f"Spread sucio ({spread:.1f})"}
    
    aligned = False
    if force_direction == 'BULLISH' and mom_color in ['green', 'lime']: aligned = True
    elif force_direction == 'BEARISH' and mom_color in ['red', 'maroon']: aligned = True
    
    if not aligned: return {'detected': False, 'reason': f"Divergencia Mom/ADX"}

    return {
        'detected': True,
        'direction': force_direction,
        'reason': f"💪 FUERZA: ADX {adx_val:.1f} {force_direction}"
    }

def check_adx_alignment(ladder):
    """Verifica si ADX apoya la tendencia en TFs clave"""
    a1h = ladder.get('1h') or {}
    a15m = ladder.get('15m') or {}
    
    def get_val(data):
        if isinstance(data, dict):
            return data.get('adx_current', 0)
        return float(data) if data else 0

    adx_1h = get_val(a1h.get('adx'))
    adx_15m = get_val(a15m.get('adx'))

    # Alineación: Al menos uno debe mostrar fuerza real
    return adx_1h > 20 or adx_15m > 20

def check_momentum_propagation(ladder):
    """Verifica si el momentum se propaga de 4H a 15M"""
    dirs = []
    for tf in ['4h', '1h', '15m']:
        analysis = ladder.get(tf)
        if analysis and isinstance(analysis, dict):
            dirs.append(analysis.get('direction', 'NEUTRAL'))
            
    # Si tenemos 3 direcciones y son iguales (y no LATERAL)
    if len(dirs) >= 3 and len(set(dirs)) == 1 and dirs[0] in ['BULLISH', 'BEARISH']:
        return True
    return False
