def check_market_force(ladder):
    """
    Verifica FUERZA DE MERCADO.
    Reglas: ADX > 25, Spread > 5, Momentum Alineado en 15M.
    """
    # Usar 15M para setup preciso (según imagen de Joel)
    a15m = ladder.get('15m') or {}
    if not a15m: 
        return {'detected': False, 'reason': "No data 15M"}
    adx_data = a15m.get('adx', {})
    mom_data = a15m.get('momentum', {})
    
    adx_val = adx_data.get('adx_current', 0)
    di_plus = adx_data.get('di_plus_current', 0)
    di_minus = adx_data.get('di_minus_current', 0)
    mom_color = mom_data.get('color_current', 'gray')
    
    # Determinar dirección por DI spread
    if di_plus > di_minus:
        force_direction = 'BULLISH'
        spread = di_plus - di_minus
    else:
        force_direction = 'BEARISH'
        spread = di_minus - di_plus
        
    # Validaciones
    if adx_val < 25: 
        return {'detected': False, 'reason': f"ADX débil ({adx_val:.1f})"}
    if spread < 5.0: 
        return {'detected': False, 'reason': f"Spread sucio ({spread:.1f})"}
    
    # Momentum DEBE alinear
    aligned = False
    if force_direction == 'BULLISH' and mom_color in ['green', 'lime']: 
        aligned = True
    elif force_direction == 'BEARISH' and mom_color in ['red', 'maroon']: 
        aligned = True
    
    if not aligned: 
        return {'detected': False, 'reason': f"Momentum ({mom_color}) NO alinea con {force_direction}"}
    return {
        'detected': True,
        'direction': force_direction,
        'adx': adx_val,
        'spread': spread,
        'momentum': mom_color,
        'reason': f"💪 FUERZA 15M: ADX {adx_val:.1f} | {force_direction} | Mom {mom_color}"
    }
