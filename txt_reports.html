import time
def fetch_current_price(symbol, max_retries=3):
    """
    Obtiene precio actual (ticker) con retry y backoff
    """
    for attempt in range(max_retries):
        try:
            ticker = exchange.fetch_ticker(symbol)
            return float(ticker['last'])
        except Exception as e:
            if '510' in str(e) or 'too frequent' in str(e).lower():
                # Rate limit - esperar y reintentar
                wait_time = (attempt + 1) * 2  # 2, 4, 6 segundos
                print(f"⏳ Rate limit, esperando {wait_time}s...")
                time.sleep(wait_time)
            else:
                print(f"❌ Error fetch_current_price {symbol}: {e}")
                return None
    return None
