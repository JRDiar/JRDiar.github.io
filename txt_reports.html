def detect_live_pattern(self, df, trade_data):
    """Detecta patrones en vivo durante el trade (3M granular con confirmación)"""
    try:
        # 0. PERIODO DE GRACIA (5 min)
        # Evitar pánico en los primeros minutos de volatilidad
        now = datetime.now(ARGENTINA_TZ)
        entry_time = trade_data['entry_time']
        duration_min = (now - entry_time).total_seconds() / 60
        
        if duration_min < 5:
            return None # Demasiado temprano para patrones de reversión
        
        # 0.5 CONFIRMACIÓN ANTI-RUIDO (3M)
        # Exigir 2+ velas rojas consecutivas antes de considerar reversión
        direction = trade_data.get('direction', 'BULLISH')
        
        if len(df) >= 3:
            last_3_closes = df['close'].values[-3:]
            last_3_opens = df['open'].values[-3:]
            
            if direction == 'BULLISH':
                # Contar velas rojas (bearish)
                red_count = sum(1 for c, o in zip(last_3_closes, last_3_opens) if c < o)
                
                # Si detectamos 2+ velas rojas consecutivas, es señal de reversión confirmada
                if red_count >= 2:
                    print(f"  ⚠️ 3M REVERSIÓN: {red_count}/3 velas rojas detectadas")
                    # Permitir continuar a Double Top detection para confirmar
            else:  # BEARISH
                # Contar velas verdes (bullish)
                green_count = sum(1 for c, o in zip(last_3_closes, last_3_opens) if c > o)
                
                if green_count >= 2:
                    print(f"  ⚠️ 3M REVERSIÓN: {green_count}/3 velas verdes detectadas")
            
        # 1. Flag Pattern (Continuación)
        # Lógica simplificada: Consolidación después de impulso
        closes = df['close'].values
Cambio: Antes de buscar patrones, primero cuenta cuántas velas van en contra de la dirección del trade. Si hay 2+ de 3, marca que hay posible reversión confirmada (no solo ruido).
