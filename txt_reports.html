1. brain_grid.py - El Nuevo Mapa Dinámico
Este es el corazón del cambio. Calcula el canal diagonal (piso y techo) usando matemáticas de regresión lineal (como lo hacen los indicadores de TradingView).

"""
brain_grid.py - Generador de Grid Dinámico (Regresión Lineal)
=============================================================
Calcula soportes y resistencias DIAGONALES en tiempo real.
Define el "campo de juego" dinámico del bot.
"""

import numpy as np
import pandas as pd
from scipy.stats import linregress

class GridManager:
    def __init__(self):
        self.cached_grid = {}

    def calculate_grid(self, df):
        """
        Calcula el canal dinámico usando Regresión Lineal.
        Devuelve: {
            'soporte_diagonal': float,
            'resistencia_diagonal': float,
            'linea_media': float,
            'pendiente': float (para saber si la tendencia general es alcista/bajista)
        }
        """
        if df is None or len(df) < 50:
            return {'valid': False}

        # Configuración del Grid (Ajustable)
        LOOKBACK_PERIOD = 50  # Cuántas velas atrás mirar para trazar la línea
        DESVIACION_STD = 2.0  # Ancho del canal (2 desviaciones estándar es lo común)

        # Usamos los datos recientes
        recent_df = df.tail(LOOKBACK_PERIOD).copy()
        
        # Preparamos ejes X (tiempo/índice) e Y (precio)
        x = np.arange(len(recent_df))
        y = recent_df['close'].values

        # Calcular Regresión Lineal (La línea central)
        slope, intercept, r_value, p_value, std_err = linregress(x, y)

        # Calcular la línea media teórica para el último punto (el precio actual)
        last_x = x[-1]
        linea_media_actual = slope * last_x + intercept

        # Calcular Desviación Estándar (La anchura del canal)
        # Calculamos la distancia de cada precio a la línea media
        regression_line = slope * x + intercept
        residuals = y - regression_line
        std_deviation = np.std(residuals)

        # Calcular Techo y Piso Dinámicos
        techo_diagonal = linea_media_actual + (std_deviation * DESVIACION_STD)
        piso_diagonal = linea_media_actual - (std_deviation * DESVIACION_STD)

        # Proyectar al futuro (opcional, para la siguiente vela)
        # future_x = last_x + 1
        # future_mean = slope * future_x + intercept

        result = {
            'valid': True,
            'soporte_diagonal': piso_diagonal,
            'resistencia_diagonal': techo_diagonal,
            'linea_media': linea_media_actual,
            'pendiente': slope,
            'ancho_canal_pct': ((techo_diagonal - piso_diagonal) / piso_diagonal) * 100
        }
        
        # Guardar en cache
        self.cached_grid = result
        return result

# Instancia global
grid_manager = GridManager()

import pandas as pd

def check_price_velocity(symbol, direction, timeframe='1m'): # Ojo: Default a 1m para precisión
    """
    Mide la aceleración del precio minuto a minuto.
    Detecta si la vela actual está creciendo agresivamente.
    """
    storage = get_frame_storage(symbol, timeframe)
    
    # Necesitamos al menos la vela actual y la anterior cerrada
    if len(storage.df) < 2:
        return {'valid': False, 'velocity': 0, 'reason': 'Faltan datos'}

    # 1. Obtener precio AHORA (tiempo real) y precio hace 1 minuto (cierre anterior)
    current_price = storage.get_current_price() # Precio actual tick a tick
    prev_close = storage.df['close'].iloc[-1]   # Cierre de la vela anterior (hace 1 min)
    
    # 2. Calcular Velocidad Instantánea (% de cambio en este minuto)
    velocity_pct = ((current_price - prev_close) / prev_close) * 100
    
    # 3. Calcular Aceleración (comparado con la vela anterior a esa)
    prev_prev_close = storage.df['close'].iloc[-2]
    prev_velocity = ((prev_close - prev_prev_close) / prev_prev_close) * 100
    
    acceleration = velocity_pct - prev_velocity

    print(f"   🏎️ VELOCIDAD {symbol} ({timeframe}): {velocity_pct:.3f}% (Accel: {acceleration:.3f})")

    # UMBRALES DE SCALPING (Ajustables)
    # Ejemplo: Si sube más de 0.3% en 1 minuto, es un movimiento BRUTAL en scalping
    MIN_VELOCITY_TRIGGER = 0.25 

    if direction == 'BULLISH':
        if velocity_pct > MIN_VELOCITY_TRIGGER:
            return {
                'valid': True, 
                'mode': 'VELOCITY_SCALP', 
                'velocity': velocity_pct,
                'reason': f'🚀 EXPLOSIÓN detectada: +{velocity_pct:.2f}% en 1 min'
            }
        elif velocity_pct < -0.05:
            return {'valid': False, 'reason': 'Precio frenado o cayendo'}
            
    elif direction == 'BEARISH':
        if velocity_pct < -MIN_VELOCITY_TRIGGER:
             return {
                'valid': True, 
                'mode': 'VELOCITY_SCALP', 
                'velocity': velocity_pct,
                'reason': f'🔻 DESPLOME detectado: {velocity_pct:.2f}% en 1 min'
            }
        elif velocity_pct > 0.05:
            return {'valid': False, 'reason': 'Precio frenado o subiendo'}

    # Si no hay velocidad explosiva, devuelve "NEUTRAL" (para que use la estrategia de canal)
    return {'valid': False, 'reason': 'Velocidad normal', 'mode': 'NORMAL'}
