def orchestrate_intelligence(symbol):

    Orquesta TODO el bot con inteligencia real de patrones.
    Combina:
    1. Detección de Patrones (Triángulos, Flags, Estructura)
    2. Análisis Time Ladder (Indicadores matemáticos)
    3. Confluencia Real
    """
    print(f"\n🧠 ORCHESTRATING INTELLIGENCE: {symbol}")
    
    # 1. DETECTAR PATRONES (La nueva "visión" del bot)
    print("   Scanning patterns...")
    patterns_result = pattern_engine.detect_all_patterns(symbol)
    
    if patterns_result['best_pattern']:
        p = patterns_result['best_pattern']
        print(f"   ✨ PATTERN FOUND: {p['type']} in {p['tf']} (Conf: {p['confidence']})")

    print("   Analyzing Time Ladder...")

        '1h': analyze_1h(symbol),

    print(f"   1. ¿QUÉ VEO? (Percepción)")

    print(f"      - Patrón Detectado: {pat_str}")

    else:

    print("-" * 60)

def make_pattern_based_decision(patterns, ladder, signals):
    """Decisión basada en confluencia REAL"""
    score = 0
    reasons = []
    
    # 0. DEFINIR MULTIPLICADOR DE MODO
    penalty_multiplier = 1.0  # Default (CONSERVATIVE)
    if TRADING_MODE == 'AGGRESSIVE':
        penalty_multiplier = 0.6
    elif TRADING_MODE == 'BALANCED':
        penalty_multiplier = 0.8
    
    # CRÍTICO: Bloquear LATERAL inmediatamente
    pattern_dir = patterns['confidence'].get('direction', 'NEUTRAL')
    if pattern_dir in ['LATERAL', 'NEUTRAL']:
        return {
            'action': 'WAIT',
            'confidence': 0,
            'missing': 100,
            'reasons': ['❌ Dirección LATERAL/NEUTRAL bloqueada']
        }
    
    # Validar que al menos 3 TFs tengan dirección clara
    valid_dirs = [
        ladder.get('4h', {}).get('direction'),
        ladder.get('1h', {}).get('direction'),
        ladder.get('15m', {}).get('direction'),
        ladder.get('5m', {}).get('direction')
    ]
    
    non_lateral = [d for d in valid_dirs if d in ['BULLISH', 'BEARISH']]
    
    # Bloqueo explícito si hay presencia significativa de LATERAL
    lateral_count = valid_dirs.count('LATERAL')
    if lateral_count >= 2:
        return {
            'action': 'WAIT',
            'confidence': 0,
            'reasons': [f'❌ Mercado LATERAL detectado en {lateral_count} TFs']
        }

    if len(non_lateral) < 3:
        return {
            'action': 'WAIT',
            'confidence': 0,
            'reasons': ['❌ Insuficientes TFs con dirección clara']
        }
    
    # 1. Patrón principal detectado (Base sólida)
    if patterns['best_pattern']:
        pattern = patterns['best_pattern']
        score += pattern['confidence'] * 0.4
        reasons.append(f"Pattern: {pattern['type']} ({pattern['tf']})")

    # FIX: Calcular confluence de patrones
    confluence_bonus = 0
    if patterns['best_pattern']:
        # Si múltiples TFs tienen mismo patrón
        pattern_count = sum(1 for tf in patterns['patterns'].values() if len(tf) > 0)
        if pattern_count >= 2:
            confluence_bonus = 10
            reasons.append(f"Pattern Confluence (+{confluence_bonus})")
    score += confluence_bonus
    
    # 2. Estructura de mercado (Alineación)
    pattern_dir = patterns['confidence']['direction']  # BULLISH/BEARISH
    
    structure_aligned = 0
    for tf, analysis in ladder.items():
        if analysis and analysis.get('direction') == pattern_dir:
            structure_aligned += 1
            
    if structure_aligned >= 3:
        score += 30
        reasons.append(f"Trend aligned {structure_aligned}/4")
    elif structure_aligned >= 2:
        score += 15
        reasons.append(f"Trend aligned {structure_aligned}/4")

    # 3. Indicadores matemáticos (Confirmación)
    if signals['adx_aligned']:
        score += 15
        reasons.append("ADX aligned")
    
    if signals['momentum_propagating']:
        score += 15
        reasons.append("Momentum propagating")
        
    if signals['volume_confirming']:
        score += 10
        reasons.append("Volume confirming")

    # 4. SMART CONTEXT LOGIC (La "Inteligencia" real)
    a5m = ladder.get('5m', {})
    if a5m:
        current_price = a5m.get('current_price', 0)
        
        # A. DYNAMIC OVEREXTENSION (ATR-based)
        dist_ema = abs(a5m.get('dist_ema_pct', 0))
        atr_pct = a5m.get('atr_pct', 1.0)
        
        extension_ratio = dist_ema / atr_pct if atr_pct > 0 else 0
        
        if extension_ratio > 4.0:  # >4 ATRs lejos - Más tolerante
            score -= int(15 * penalty_multiplier)  # Menos penalización
            reasons.append(f"⚠️ Overextended ({extension_ratio:.1f} ATRs)")
        elif extension_ratio > 2.0:
            score -= int(10 * penalty_multiplier)
            reasons.append(f"⚠️ Extended ({extension_ratio:.1f} ATRs)")
        elif extension_ratio < 0.5:
            score += 5
            reasons.append("In Value Zone")

        # B. ZONE AWARENESS (Soportes y Resistencias)
        sr_data = a5m.get('sr', {})
        if pattern_dir == 'BULLISH':
            resistances = sr_data.get('resistances', [])
            if resistances:
                nearest_res = resistances[0]
                dist_to_res = (nearest_res - current_price) / current_price * 100
                if 0 < dist_to_res < 1.0:
                    score -= int(35 * penalty_multiplier)
                    reasons.append(f"⚠️ Resistance ahead (+{dist_to_res:.2f}%)")
                elif 1.0 <= dist_to_res < 2.0:
                    score -= int(15 * penalty_multiplier)
                    reasons.append(f"⚠️ Near resistance (+{dist_to_res:.2f}%)")
        elif pattern_dir == 'BEARISH':
            supports = sr_data.get('supports', [])
            if supports:
                nearest_sup = supports[0]
                dist_to_sup = (current_price - nearest_sup) / current_price * 100
                if 0 < dist_to_sup < 1.0:
                    score -= int(35 * penalty_multiplier)
                    reasons.append(f"⚠️ Support below (-{dist_to_sup:.2f}%)")
                elif 1.0 <= dist_to_sup < 2.0:
                    score -= int(15 * penalty_multiplier)
                    reasons.append(f"⚠️ Near support (-{dist_to_sup:.2f}%)")

        # C. RSI TIMING (Evitar extremos sin fuerza)
        rsi_val = a5m.get('rsi', {}).get('rsi_current', 50)
        mom_strength = a5m.get('momentum', {}).get('strength', 'débil')
        
        if pattern_dir == 'BULLISH':
            if rsi_val > 75 and mom_strength != 'fuerte':
                score -= int(15 * penalty_multiplier)
                reasons.append(f"⚠️ RSI Overbought ({rsi_val:.0f})")
        elif pattern_dir == 'BEARISH':
            if rsi_val < 25 and mom_strength != 'fuerte':
                score -= int(15 * penalty_multiplier)
                reasons.append(f"⚠️ RSI Oversold ({rsi_val:.0f})")

    # D. TIMING ÓPTIMO (Flags en brain_config.py)
    # ----------------------------------------------------------------
    
    # 1. PULLBACK CHECK (Evitar comprar en extensión máxima)
    # Requisito: >3.0 ATRs Y lejos del POC (>2%)
    if ENABLE_PULLBACK_CHECK:
        poc = a5m.get('volume_profile', {}).get('poc_current', 0)
        dist_to_poc = abs(current_price - poc) / current_price * 100 if poc > 0 else 0

    # 3. HH PENALTY (Evitar comprar el pico exacto - Buying Top)
    if ENABLE_HH_PENALTY and pattern_dir == 'BULLISH':
        recent_highs = a5m.get('zigzag', {}).get('pivot_highs', [])
        if recent_highs:
            hh = max(recent_highs)
            if current_price < hh:
                dist_to_hh = ((hh - current_price) / current_price) * 100
                
                if dist_to_hh < 0.2:  # < 0.2% del máximo (Peligro Extremo)
                    score -= int(40 * penalty_multiplier)
                    reasons.append(f"⛔ Buying Top (Dist to HH: {dist_to_hh:.2f}%)")
                elif dist_to_hh < 0.5: # < 0.5% del máximo (Peligro)
                    score -= int(20 * penalty_multiplier)
                    reasons.append(f"⚠️ Near Top (Dist to HH: {dist_to_hh:.2f}%)")

    # 4. CLEAN SPACE BONUS (Premio por camino libre)
    if ENABLE_CLEAN_SPACE_BONUS and pattern_dir == 'BULLISH':
        resistances = sr_data.get('resistances', [])
        recent_highs = a5m.get('zigzag', {}).get('pivot_highs', [])
        
        # Verificar Resistencias
        res_clear = not resistances or ((resistances[0] - current_price) / current_price * 100 > 3.0)
        
        # Verificar HH (ZigZag)
        hh_clear = True
        if recent_highs:
            hh = max(recent_highs)
            if current_price < hh: # Si estamos debajo del HH
                if ((hh - current_price) / current_price * 100) <= 3.0:
                    hh_clear = False
        
        if res_clear and hh_clear:
            score += 10
            reasons.append("✨ Clean Space (>3% room)")

    # (Bloque de penalizaciones acumulativas eliminado por causar piso artificial de 70 puntos)

    # Clamp score to 0-100
    score = max(0, min(100, score))

    # DECISIÓN FINAL
    if score >= 65:
        confidence_level = 'HIGH' if score >= 70 else 'MEDIUM'
        return {
            'action': 'ENTER',
            'confidence': score,
            'confidence_level': confidence_level,
            'pattern': patterns['best_pattern']['type'] if patterns['best_pattern'] else "STRUCTURE_PLAY",
            'reasons': reasons,
            'strategy': determine_strategy(patterns, ladder)
        }
    else:
        return {
            'action': 'WAIT',
            'confidence': score,
            'missing': 65 - score,
            'reasons': reasons
        }

def determine_strategy(patterns, ladder):
    """Define estrategia según patrón"""
    pattern_type = "STANDARD"
    if patterns['best_pattern']:
        pattern_type = patterns['best_pattern']['type']
    
    # Reversión
    if 'DOUBLE' in pattern_type or 'BOS' in pattern_type:
        return {
            'type': 'REVERSAL',
            'stop_multiplier': 1.5,
            'target_multiplier': 2.0,
            'position_size': 0.5
        }
    # Continuación
    elif 'FLAG' in pattern_type or 'TRIANGLE' in pattern_type or 'TREND' in pattern_type or 'MOMENTUM' in pattern_type:
        return {
            'type': 'CONTINUATION',
            'stop_multiplier': 2.0,
            'target_multiplier': 3.0,
            'position_size': 1.0
        }
    
    # Default
    return {
        'type': 'STANDARD',
        'stop_multiplier': 2.0,
        'target_multiplier': 2.0,
        'position_size': 0.75
    }

def check_adx_alignment(ladder):
    """Verifica si ADX apoya la tendencia en TFs clave"""
    a1h = ladder.get('1h')
    a15m = ladder.get('15m')
    
    if not a1h or not a15m:
        return False
    
    adx_1h_val = a1h.get('adx', 0)
    if isinstance(adx_1h_val, dict):
        adx_1h_val = adx_1h_val.get('adx_current', adx_1h_val.get('current', 0))
        
    adx_15m_val = a15m.get('adx', 0)
    if isinstance(adx_15m_val, dict):
        adx_15m_val = adx_15m_val.get('adx_current', adx_15m_val.get('current', 0))

    strong_1h = float(adx_1h_val) > 20
    strong_15m = float(adx_15m_val) > 20
    
    return strong_1h or strong_15m

def check_momentum_propagation(ladder):
    """Verifica si el momentum se propaga de 4H a 5M"""
    dirs = []
    for tf in ['4h', '1h', '15m']:
        if ladder.get(tf):
            dirs.append(ladder[tf].get('direction'))
            
    if len(dirs) == 3 and len(set(dirs)) == 1 and dirs[0] in ['BULLISH', 'BEARISH']:
        return True
    return False

def check_volume_confirmation(ladder):
    """Verifica volumen en 15M/5M con lógica real"""
    a5m = ladder.get('5m')
    if not a5m:
        return False
    
    vp_data = a5m.get('volume_profile', {})
    vol_ratio = vp_data.get('volume_ratio', 0)
    
    return vol_ratio > 1.2
