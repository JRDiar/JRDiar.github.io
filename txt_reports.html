def analyze_trend_evolution(symbol):
    """
    ANÁLISIS DE EVOLUCIÓN (TIME LADDER / ROADMAP)
    Analiza velas de 1M para detectar tendencias explosivas en tiempo real.
    Permite 'bypassear' filtros estrictos si la moneda está explotando ahora mismo.
    """
    try:
        # Buscar 30 velas de 1M (media hora de evolución)
        ohlcv = fetch_ohlcv(symbol, '1m', 30)
        if not ohlcv or len(ohlcv) < 15:
            return {'is_explosive': False, 'score': 0, 'reason': 'Sin datos 1M'}

        df = pd.DataFrame(ohlcv, columns=['timestamp', 'open', 'high', 'low', 'close', 'volume'])
        
        # 1. Análisis de ADX en 1M (Velocidad de tendencia)
        adx_data = calculate_adx_with_memory(df, period=14, lookback=5)
        adx_current = adx_data['adx_current']
        adx_history = adx_data['adx_history']
        
        # 2. Análisis de Precio (Escalera de precios)
        closes = df['close'].values
        price_velocity = calculate_velocity(df, lookback=5)
        
        # DETECCIÓN DE EXPLOSIÓN
        is_explosive = False
        score = 0
        reason = []

        # Criterio A: ADX Disparándose (Roadmap de fuerza)
        # Si ADX subió más de 5 puntos en las últimas 5 velas 1M
        if len(adx_history) >= 5:
            adx_change = adx_history[-1] - adx_history[0]
            if adx_change > 3.0 and adx_current > 20:
                score += 40
                reason.append(f"ADX 1M disparándose (+{adx_change:.1f})")
                if adx_current > 25:
                    is_explosive = True

        # Criterio B: Escalera de Precios (ZigZag alcista agresivo)
        # Verificar si las últimas 3 velas son verdes y con cuerpos crecientes o fuertes
        last_3_closes = closes[-3:]
        last_3_opens = df['open'].values[-3:]
        is_green_ladder = all(c > o for c, o in zip(last_3_closes, last_3_opens))
        
        if is_green_ladder and price_velocity['velocity_current'] > 0:
            score += 30
            reason.append("Escalera de precios 1M alcista")
            if price_velocity['trend'] == 'acelerando':
                is_explosive = True
                score += 20

        # Criterio C: Volumen Climático
        volumes = df['volume'].values
        vol_avg = volumes[:-1].mean()
        if volumes[-1] > vol_avg * 2:
            score += 20
            reason.append("Volumen 1M climático")

        # Decisión final de explosividad
        if score >= 50:
            is_explosive = True
        
        return {
            'is_explosive': is_explosive,
            'score': score,
            'adx_1m': adx_current,
            'reason': ", ".join(reason),
            'timestamp': datetime.now().isoformat()
        }

    except Exception as e:
        print(f"⚠️ Error en analyze_trend_evolution: {e}")
        return {'is_explosive': False, 'score': 0, 'reason': 'Error'}

def _extract_score(score_value, default=0):
    """
    Recursively extract integer score from potentially nested dict.
    
    Args:
        score_value: The value to extract (can be int, float, dict, or None)
        default: Default value if extraction fails
    
    Returns:
        int: Extracted integer score
    """
    if score_value is None:
        return default
    if isinstance(score_value, (int, float)):
        return int(score_value)
    if isinstance(score_value, dict):\
        # Recursively extract from nested dict
        return _extract_score(score_value.get('score', default), default)
    # Fallback for any other type
    return default

def calculate_triggers(analysis_15m, analysis_5m, direction):
    entry_price = analysis_5m['current_price']
    
    # --- DYNAMIC RISK MANAGEMENT (Phase 4.1) ---
    # Use ATR for Stop Loss if available
    atr = analysis_15m.get('atr', 0)
    
    if atr > 0:
        # Multiplicador dinámico: 2.0x ATR es estándar
        atr_multiplier = 2.0
        stop_loss_dist = atr * atr_multiplier
        
        if direction == 'BULLISH':
            stop_loss = entry_price - stop_loss_dist
        else:
            stop_loss = entry_price + stop_loss_dist
            
        # Sanity Check: Ensure SL is not too close (< 0.5%) or too far (> 5%)
        # If too close, force min 0.5%
        min_dist = entry_price * 0.005
        if stop_loss_dist < min_dist:
            if direction == 'BULLISH': stop_loss = entry_price - min_dist
            else: stop_loss = entry_price + min_dist
            
    else:
        # Fallback: Fixed 2%
        stop_loss = entry_price * 0.98 if direction == 'BULLISH' else entry_price * 1.02

    # Target: 1.5x Risk (Simple initial target)
    risk = abs(entry_price - stop_loss)
    target_dist = risk * 1.5
    
    if direction == 'BULLISH':
        target_price = entry_price + target_dist
    else:
        target_price = entry_price - target_dist

    rr_ratio = target_dist / risk if risk > 0 else 0
    tiempo_proyectado_min = 15

    return {
        'entry_price': entry_price,
        'stop_loss': stop_loss,
        'target_price': target_price,
        'rr_ratio': rr_ratio,
        'tiempo_proyectado_min': tiempo_proyectado_min,
        'metodo_target': 'atr_dynamic' if atr > 0 else 'fixed_pct'
    }
