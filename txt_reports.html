def verify_15m_confluence(self, symbol):
    """
    FASE 3: Verificar confluencia 15M antes de entrar en 5M
    Evita entradas sin respaldo de temporalidad superior
    """
    try:
        # Obtener datos de 15M
        ohlcv_15m = exchange.fetch_ohlcv(symbol, '15m', limit=50)
        if not ohlcv_15m or len(ohlcv_15m) < 50:
            return False, "DATOS_INSUFICIENTES_15M"
        
        df_15m = pd.DataFrame(ohlcv_15m, columns=['timestamp','open','high','low','close','volume'])
        
        # Calcular indicadores en 15M
        calculate_lazybear_momentum(df_15m)
        calculate_manual_adx(df_15m)
        
        # Obtener valores actuales de 15M
        momentum_15m = df_15m['LAZYBEAR_MOMENTUM'].iloc[-1]
        momentum_color_15m = df_15m['MOMENTUM_COLOR'].iloc[-1]
        adx_15m = df_15m['ADX_MANUAL'].iloc[-1]
        di_plus_15m = df_15m['DI_PLUS_MANUAL'].iloc[-1]
        di_minus_15m = df_15m['DI_MINUS_MANUAL'].iloc[-1]
        
        # CRITERIOS DE CONFLUENCIA 15M
        confluence_score = 0
        reasons = []
        
        # 1. Momentum debe ser positivo y fuerte
        if momentum_color_15m in ['lime', 'green'] and momentum_15m > 0:
            confluence_score += 40
            reasons.append(f"15M_MOM_BULLISH_{momentum_color_15m}")
        elif momentum_color_15m in ['red', 'maroon'] and momentum_15m < 0:
            confluence_score += 40
            reasons.append(f"15M_MOM_BEARISH_{momentum_color_15m}")
        else:
            reasons.append(f"15M_MOMENTUM_DÉBIL_{momentum_color_15m}")
            
        # 2. ADX debe mostrar tendencia
        if adx_15m >= 20:
            confluence_score += 30
            reasons.append(f"15M_ADX_{adx_15m:.1f}")
        else:
            reasons.append(f"15M_ADX_BAJO_{adx_15m:.1f}")
            
        # 3. Dirección clara
        if abs(di_plus_15m - di_minus_15m) > 5:
            confluence_score += 30
            reasons.append("15M_DIRECCIÓN_CLARA")
        
        # Necesita 70+ puntos para confirmar
        if confluence_score >= 70:
            print(f"   15M ✅ CONFLUENCIA CONFIRMADA: Score {confluence_score} | {' | '.join(reasons)}")
            return True, reasons
        else:
            print(f"   15M ❌ SIN CONFLUENCIA: Score {confluence_score} | {' | '.join(reasons)}")
            return False, reasons
            
    except Exception as e:
        print(f"   15M ⚠️ Error verificando confluencia: {e}")
        return False, "ERROR_15M"

        try:
            print(f"   5M 📊 Analizando {symbol} - FASE 3...")

                continue
                
            # SEGUNDO: Si hay confluencia, analizar 5M
            signal_5m = self.fetch_and_analyze_ultra(symbol, '5m')

                # Verificar cascada completa
                signal_4h = next(s for sym, s in self.signals_4h if sym == symbol)
                signal_1h = next(s for sym, s in self.signals_1h if sym == symbol)
                signal_30m = next(s for sym, s in self.signals_30m if sym == symbol)

                if all(s['direction'] == signal_5m['direction'] for s in [signal_4h, signal_1h, signal_30m, signal_15m]):
                    # Agregar info de confluencia

•	Eficiencia de contexto: Un artefacto vs 10+ artefactos
•	Trazabilidad clara: Cada corrección marcada con comentarios explicativos

def plot_trade_evolution(self, trade_id):
    # Filtrar datos de monitoreo para este trade
    df_trade = self.df_monitoring[self.df_monitoring['Trade_ID'] == trade_id]
    
    # Convertir timestamps
    df_trade['Minutes'] = pd.to_datetime(df_trade['Current_Timestamp'])
    df_trade['Minutes'] = (df_trade['Minutes'] - df_trade['Minutes'].min()).dt.total_seconds() / 60
    
    # Crear figura con subplots
    fig, axes = plt.subplots(3, 1, figsize=(12, 10))
    
    # Subplot 1: PnL evolution
    axes[0].plot(df_trade['Minutes'], df_trade['Current_PnL_Percent'])
    axes[0].axhline(y=df_trade['Current_PnL_Percent'].max(), color='g', linestyle='--', label='Max PnL')
    axes[0].set_ylabel('PnL %')
    axes[0].legend()
    axes[0].grid(True)
    
    # Subplot 2: ADX evolution
    axes[1].plot(df_trade['Minutes'], df_trade['ADX_Current'])
    axes[1].set_ylabel('ADX')
