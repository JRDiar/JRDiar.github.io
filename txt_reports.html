"""
================================================================================
BRAIN VISION UNIFIED v1.0 - VISIÓN HOLÍSTICA DEL MERCADO
================================================================================
Joel: "Hay que darle inteligencia, que vea la película y decida como un 
trader humano. El bot está preparado para eso."
Este módulo:
1. RECOPILA información de TODOS los módulos existentes
2. VE el cuadro completo (no snapshots aislados)
3. DECIDE basado en CONTEXTO, no en scoring aritmético
NO usa:
- Hard blocks (if > X: BLOQUEAR)
- Penalties aritméticas (score -= 50)
- Thresholds rígidos
SÍ usa:
- Análisis contextual
- Decisión como trader humano
- Time Ladder (frames en movimiento)
================================================================================
"""
from brain_movie_engine import MovieEngine
from brain_grid import grid
from brain_trend_exhaustion import analyze_trend_exhaustion
from brain_frames import get_frame_storage
from brain_zigzag_tti import get_zigzag_data
from brain_force_filter import force_filter
from exchange_connection import fetch_ohlcv
import pandas as pd
class UnifiedVision:
    """
    Cerebro unificado que ve TODO el contexto del mercado.
    
    Decide como un trader humano, no como una calculadora.
    """
    
    def __init__(self):
        self.movie_engine = MovieEngine()
    
    # =========================================================================
    # FASE 1: RECOPILAR TODA LA INTELIGENCIA
    # =========================================================================
    
    def gather_intelligence(self, symbol, ladder_analysis):
        """
        Recopila información de TODOS los módulos existentes.
        
        Returns:
            dict: Toda la información del mercado en un solo lugar
        """
        
        # 1. MOVIEENGINE - ¿La película cuenta una historia bullish/bearish?
        direction = ladder_analysis.get('15m', {}).get('direction', 'LATERAL')
        movie_result = self.movie_engine.analyze_complete_movie(symbol, direction)
        
        # 2. GRID - ¿Dónde estoy? ¿Hacia dónde voy?
        ohlcv = fetch_ohlcv(symbol, '5m', 50)
        if ohlcv:
            df = pd.DataFrame(ohlcv, columns=['timestamp','open','high','low','close','volume'])
            grid_result = grid.analyze(symbol, df, ladder_analysis)
        else:
            grid_result = {'status': 'ERROR', 'score': 0}
        
        # 3. EXHAUSTION - ¿Cuánto recorrí vs cuánto queda?
        exhaustion_result = analyze_trend_exhaustion(symbol, direction)
        
        # 4. FRAMES - ¿ADX/Momentum acelerando o desacelerando?
        frames_15m = get_frame_storage(symbol, '15m').get_frames(5)
        frames_5m = get_frame_storage(symbol, '5m').get_frames(5)
        
        # 5. ZIGZAG - ¿Estructura sana o deteriorándose?
        try:
            zigzag_data = get_zigzag_data(symbol, '15m')
        except:
            zigzag_data = {'pivot_types': [], 'pivots': []}
        
        # 6. FORCE - ¿Hay fuerza desarrollándose?
        force_result = force_filter.has_force(symbol, ladder_analysis)
        
        return {
            'movie': movie_result,
            'grid': grid_result,
            'exhaustion': exhaustion_result,
            'frames': {'15m': frames_15m, '5m': frames_5m},
            'zigzag': zigzag_data,
            'force': force_result,
            'direction': direction
        }
    
    # =========================================================================
    # FASE 2: VER LA PELÍCULA COMPLETA
    # =========================================================================
    
    def see_the_movie(self, intelligence):
        """
        Ve la PELÍCULA completa y entiende el contexto.
        
        NO usa if/else binarios.
        ENTIENDE el contexto como un trader humano.
        
        Returns:
            dict: Interpretación holística del mercado
        """
        vision = {}
        
        # ---------- A. CALIDAD DE LA EXPLOSIÓN ----------
        # ¿La fuerza está NACIENDO, CRECIENDO, o MURIENDO?
        
        movie = intelligence.get('movie', {})
        scenes = movie.get('scenes', {})
        scene_5m = scenes.get('5m', {})
        
        adx_phase = scene_5m.get('adx_phase', 'UNKNOWN')
        adx_velocity = scene_5m.get('adx_trajectory', {}).get('velocity', 0)
        
        if adx_phase in ['EXPLOSION_NASCENT', 'EXPLOSION_BUILDING']:
            vision['explosion'] = 'NACIENDO'  # 🟢 Excelente
        elif adx_phase in ['EXPLOSION_SUSTAINED', 'EXPLOSION_STRONG']:
            vision['explosion'] = 'ACTIVA'    # 🟡 Buena
        elif adx_phase in ['EXPLOSION_CLIMAX', 'EXPLOSION_FADING']:
            vision['explosion'] = 'MURIENDO'  # 🔴 Peligrosa
        elif adx_phase in ['FORCE_DYING', 'FORCE_WEAK', 'LATERAL']:
            vision['explosion'] = 'MUERTA'    # ⚫ No hay
        else:
            vision['explosion'] = 'DESCONOCIDA'
        
        # ADX acelerando vs desacelerando
        if adx_velocity > 0.5:
            vision['adx_trend'] = 'ACELERANDO'
        elif adx_velocity < -0.5:
            vision['adx_trend'] = 'DESACELERANDO'
        else:
            vision['adx_trend'] = 'ESTABLE'
        
        # ---------- B. RUNWAY (ESPACIO LIBRE) ----------
        # ¿Cuánto espacio hay para que el precio se mueva?
        
        exhaustion = intelligence.get('exhaustion', {})
        room_pct = exhaustion.get('room_pct', 0)
        moved_pct = exhaustion.get('moved_pct', 0)
        exhaustion_pct = exhaustion.get('exhaustion_pct', 50)
        
        if room_pct > 4:
            vision['runway'] = 'AMPLIO'       # 🟢 Mucho espacio
        elif room_pct > 2:
            vision['runway'] = 'MODERADO'     # 🟡 Espacio aceptable
        elif room_pct > 1:
            vision['runway'] = 'ESTRECHO'     # 🟠 Poco espacio
        else:
            vision['runway'] = 'BLOQUEADO'    # 🔴 Sin espacio
        
        # Fase de exhaustion
        if exhaustion_pct < 30:
            vision['exhaustion_phase'] = 'FRESH'
        elif exhaustion_pct < 50:
            vision['exhaustion_phase'] = 'DEVELOPING'
        elif exhaustion_pct < 70:
            vision['exhaustion_phase'] = 'MATURE'
        else:
            vision['exhaustion_phase'] = 'EXHAUSTED'
        
        # ---------- C. ESTRUCTURA ZIGZAG ----------
        # ¿La estructura es SANA (HH→HL) o ROTA (HH→LH→LL)?
        
        zigzag = intelligence.get('zigzag', {})
        pivot_types = zigzag.get('pivot_types', [])
        
        if len(pivot_types) >= 2:
            last_two = pivot_types[-2:]
            
            direction = intelligence.get('direction', 'LATERAL')
            
            if direction == 'BULLISH':
                # Bullish sano: HH, HL
                if last_two[-1] in ['HH', 'HL'] and last_two[-2] in ['HH', 'HL']:
                    vision['structure'] = 'SANA'
                elif 'LH' in last_two or 'LL' in last_two:
                    vision['structure'] = 'DETERIORANDO'
                else:
                    vision['structure'] = 'AMBIGUA'
            
            elif direction == 'BEARISH':
                # Bearish sano: LL, LH
                if last_two[-1] in ['LL', 'LH'] and last_two[-2] in ['LL', 'LH']:
                    vision['structure'] = 'SANA'
                elif 'HH' in last_two or 'HL' in last_two:
                    vision['structure'] = 'DETERIORANDO'
                else:
                    vision['structure'] = 'AMBIGUA'
            else:
                vision['structure'] = 'LATERAL'
        else:
            vision['structure'] = 'INSUFICIENTE'
        
        # ---------- D. POSICIÓN EN GRID ----------
        # ¿Estoy comprando TECHO o cerca de SOPORTE?
        
        grid_data = intelligence.get('grid', {})
        grid_status = grid_data.get('status', 'UNKNOWN')
        contexto = grid_data.get('contexto', 'UNKNOWN')
        
        # Posición en diagonal (si está disponible)
        diagonal = grid_data.get('diagonal_data', {})
        position_pct = diagonal.get('price_position_pct', 50)
        
        if position_pct < 25:
            vision['position'] = 'CERCA_SOPORTE'   # 🟢 Ideal para LONG
        elif position_pct < 50:
            vision['position'] = 'ZONA_BAJA'       # 🟡 Buena
        elif position_pct < 75:
            vision['position'] = 'ZONA_ALTA'       # 🟠 Riesgosa
        else:
            vision['position'] = 'CERCA_TECHO'     # 🔴 Peligro
        
        vision['grid_status'] = grid_status
        vision['grid_contexto'] = contexto
        
        # ---------- E. NIVEL DE FUERZA ----------
        
        force = intelligence.get('force', {})
        vision['has_force'] = force.get('has_force', False)
        vision['force_level'] = force.get('force_level', 'NONE')
        
        # ---------- F. FRAMES - ACELERACIÓN ----------
        # ¿Los indicadores están mejorando o empeorando?
        
        frames = intelligence.get('frames', {})
        frames_15m = frames.get('15m', [])
        
        if len(frames_15m) >= 3:
            # Extraer ADX de últimos 3 frames
            try:
                adx_values = [f.get('adx', {}).get('value', 0) for f in frames_15m[-3:]]
                if adx_values[2] > adx_values[1] > adx_values[0]:
                    vision['indicators'] = 'MEJORANDO'
                elif adx_values[2] < adx_values[1] < adx_values[0]:
                    vision['indicators'] = 'EMPEORANDO'
                else:
                    vision['indicators'] = 'MIXTO'
            except:
                vision['indicators'] = 'DESCONOCIDO'
        else:
            vision['indicators'] = 'INSUFICIENTE'
        
        return vision
    
    # =========================================================================
    # FASE 3: DECIDIR COMO TRADER HUMANO
    # =========================================================================
    
    def make_human_decision(self, vision, intelligence):
        """
        Decide como Joel decidía.
        
        NO usa scoring aritmético.
        USA contexto completo para decidir.
        
        Returns:
            dict: Decisión con razón humana
        """
        
        # Extraer valores clave
        explosion = vision.get('explosion', 'DESCONOCIDA')
        runway = vision.get('runway', 'DESCONOCIDO')
        structure = vision.get('structure', 'AMBIGUA')
        position = vision.get('position', 'DESCONOCIDA')
        exhaustion_phase = vision.get('exhaustion_phase', 'UNKNOWN')
        adx_trend = vision.get('adx_trend', 'ESTABLE')
        has_force = vision.get('has_force', False)
        indicators = vision.get('indicators', 'DESCONOCIDO')
        
        # =====================================================================
        # ESCENARIOS DE DECISIÓN (como Joel pensaba)
        # =====================================================================
        
        # ----- ESCENARIO 1: SETUP IDEAL -----
        # Explosión naciendo + espacio libre + estructura sana + indicadores mejorando
        if (explosion in ['NACIENDO', 'ACTIVA'] and
            runway in ['AMPLIO', 'MODERADO'] and
            structure == 'SANA' and
            position in ['CERCA_SOPORTE', 'ZONA_BAJA'] and
            indicators in ['MEJORANDO', 'MIXTO']):
            
            return {
                'decision': 'ENTER',
                'confidence': 90,
                'reason': f"🎯 SETUP IDEAL: Explosión {explosion}, runway {runway}, estructura {structure}",
                'archetype': 'IDEAL_SETUP'
            }
        
        # ----- ESCENARIO 2: EXPLOSIÓN FUERTE -----
        # Fuerza brutal compensado deficiencias menores
        if (explosion == 'NACIENDO' and
            has_force and
            adx_trend == 'ACELERANDO' and
            runway != 'BLOQUEADO'):
            
            return {
                'decision': 'ENTER',
                'confidence': 85,
                'reason': f"⚡ EXPLOSIÓN NACIENDO: ADX acelerando, fuerza detectada",
                'archetype': 'EXPLOSION_ENTRY'
            }
        
        # ----- ESCENARIO 3: CONTINUACIÓN DE TENDENCIA -----
        # Tendencia activa con espacio y estructura
        if (explosion == 'ACTIVA' and
            runway in ['AMPLIO', 'MODERADO'] and
            structure == 'SANA' and
            exhaustion_phase in ['FRESH', 'DEVELOPING']):
            
            return {
                'decision': 'ENTER',
                'confidence': 75,
                'reason': f"📈 CONTINUACIÓN: Tendencia activa, fase {exhaustion_phase}",
                'archetype': 'TREND_CONTINUATION'
            }
        
        # ----- ESCENARIO 4: FUERZA MURIENDO -----
        # Explosión terminando = esperar
        if explosion in ['MURIENDO', 'MUERTA']:
            return {
                'decision': 'WAIT',
                'confidence': 80,
                'reason': f"⏸️ ESPERAR: Explosión {explosion}, ADX {adx_trend}",
                'archetype': 'DYING_FORCE'
            }
        
        # ----- ESCENARIO 5: SIN ESPACIO (COMPRAR TECHO) -----
        if runway == 'BLOQUEADO' or position == 'CERCA_TECHO':
            return {
                'decision': 'WAIT',
                'confidence': 85,
                'reason': f"🚫 SIN ESPACIO: Runway {runway}, posición {position}",
                'archetype': 'NO_RUNWAY'
            }
        
        # ----- ESCENARIO 6: ESTRUCTURA ROTA -----
        if structure == 'DETERIORANDO':
            return {
                'decision': 'WAIT',
                'confidence': 75,
                'reason': f"⚠️ ESTRUCTURA ROTA: ZigZag deteriorándose",
                'archetype': 'BROKEN_STRUCTURE'
            }
        
        # ----- ESCENARIO 7: EXHAUSTED -----
        if exhaustion_phase == 'EXHAUSTED' and runway in ['ESTRECHO', 'BLOQUEADO']:
            return {
                'decision': 'WAIT',
                'confidence': 90,
                'reason': f"🛑 EXHAUSTED: Tendencia agotada ({exhaustion_phase}), sin espacio ({runway})",
                'archetype': 'EXHAUSTED_TREND'
            }
        
        # ----- ESCENARIO 8: AMBIGUO -----
        # No está claro - mejor esperar
        return {
            'decision': 'WAIT',
            'confidence': 50,
            'reason': f"❓ AMBIGUO: explosion={explosion}, runway={runway}, structure={structure}",
            'archetype': 'AMBIGUOUS'
        }
    
    # =========================================================================
    # FUNCIÓN PRINCIPAL
    # =========================================================================
    
    def decide_with_unified_vision(self, symbol, ladder_analysis):
        """
        Punto de entrada principal.
        
        Toma decisión viendo la PELÍCULA COMPLETA del mercado.
        
        Args:
            symbol: Par a analizar
            ladder_analysis: Análisis multi-TF de brain_analyzer
            
        Returns:
            dict: Decisión con toda la información de visión
        """
        
        # 1. Recopilar TODA la inteligencia
        intelligence = self.gather_intelligence(symbol, ladder_analysis)
        
        # 2. Ver la PELÍCULA COMPLETA
        vision = self.see_the_movie(intelligence)
        
        # 3. Decidir como trader HUMANO
        decision = self.make_human_decision(vision, intelligence)
        
        # 4. Agregar visión al resultado
        decision['vision'] = vision
        decision['intelligence'] = {
            'exhaustion_pct': intelligence['exhaustion'].get('exhaustion_pct', 50),
            'room_pct': intelligence['exhaustion'].get('room_pct', 0),
            'force_level': intelligence['force'].get('force_level', 'NONE'),
            'grid_status': intelligence['grid'].get('status', 'UNKNOWN')
        }
        
        return decision

def decide_unified(symbol, ladder_analysis):
    """Wrapper para uso fácil."""
    unified = UnifiedVision()
    return unified.decide_with_unified_vision(symbol, ladder_analysis)
