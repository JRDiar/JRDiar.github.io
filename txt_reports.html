    sessions = {
        (0,6): ('ASIA_LATE', 0.6),

        (20,24): ('US_CLOSE', 0.8)
    }
    
    for hours, (name, mult) in sessions.items():
        if hours[0] <= hour_utc < hours[1]:
            return mult
    return 1.0

pythondef calculate_weighted_confluence(analyses):
    weights = {'4h': 0.35, '1h': 0.30, '15m': 0.25, '5m': 0.10}
    
    score = 0
    for tf, analysis in analyses.items():
        if analysis and analysis.get('strength') == 'STRONG':
            score += weights.get(tf, 0) * 100
            
            # Bonus por alineación perfecta
            if analysis.get('adx') > 30:
                score *= 1.1
                
    return min(100, score)
