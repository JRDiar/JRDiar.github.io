class MarketEntity:
    """Mercado como entidad física."""
    
    def __init__(self, symbol):
        self.position = get_price(symbol)  # Posición
        self.velocity = 0                  # Velocidad
        self.acceleration = 0              # Aceleración
        self.force = 0                     # Fuerza
        self.friction = 0                  # Exhaustion
    
    def update(self, dt=1):
        """Física del mercado."""
        self.acceleration = self.force
        self.velocity += self.acceleration * dt
        self.velocity *= (1 - self.friction)
        self.position += self.velocity * dt
    
    def detect_collision(self, obstacle):
        """Resistencias = paredes."""
        if self.position >= obstacle:
            self.velocity *= -0.5  # Rebote
            return 'BOUNCE'
        return 'FREE'
