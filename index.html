"""
MÉTRICAS CEREBRO - BOT TRADING ANALYSIS FASE 3
Funciones para analizar efectividad del filtrado inteligente del cerebro.
"""

import pandas as pd
import numpy as np

# ============================================================================
# ANÁLISIS DEL CEREBRO (BRAIN FILTERING)
# ============================================================================

def analyze_brain_rejections(df_brain):
    """
    Analiza rechazos del cerebro agrupados por razón.
    
    Returns:
        dict con counts y porcentajes por Rejection_Reason
    """
    print("\n🧠 Analizando rechazos del cerebro...")
    
    if df_brain.empty:
        print("  ⚠️  No hay datos de cerebro")
        return None
    
    # Filtrar solo rechazos
    rejections = df_brain[df_brain['Brain_Decision'] == 'REJECT']
    
    if rejections.empty:
        print("  ✅ No hay rechazos (todas las señales fueron aprobadas)")
        return {
            'total_rejections': 0,
            'reasons': {},
            'summary': 'Sin rechazos'
        }
    
    total_rejections = len(rejections)
    
    # Agrupar por razón
    if 'Rejection_Reason' in rejections.columns:
        reason_counts = rejections['Rejection_Reason'].value_counts()
        reasons_dict = {}
        
        for reason, count in reason_counts.items():
            reasons_dict[reason] = {
                'count': int(count),
                'percentage': float(count / total_rejections * 100)
            }
        
        print(f"  📊 Total rechazos: {total_rejections}")
        for reason, data in reasons_dict.items():
            print(f"    - {reason}: {data['count']} ({data['percentage']:.1f}%)")
    else:
        reasons_dict = {}

def analyze_brain_classifications(df_brain, df_trades):
    """
    Analiza Entry_Classification cruzando con resultados de trades.
    
    Args:
        df_brain: DataFrame con análisis del cerebro
        df_trades: DataFrame con trades completados
    
    Returns:
        dict con métricas por clasificación (TEMPRANA/MEDIA/TARDÍA)
    """
    print("\n🎯 Analizando clasificaciones de entrada...")
    
    if df_brain.empty or df_trades.empty:
        print("  ⚠️  Datos insuficientes para análisis")
        return None
    
    # Verificar columnas necesarias
    if 'Entry_Classification' not in df_brain.columns:
        print("  ⚠️  Columna 'Entry_Classification' no encontrada")
        return None
    
    # Convertir PnL a numérico
    df_trades['PnL_Percent'] = pd.to_numeric(df_trades['PnL_Percent'], errors='coerce')
    
    classifications_data = {}
    
    # Analizar cada clasificación
    for classification in ['TEMPRANA', 'MEDIA', 'TARDÍA']:
        # Filtrar señales con esta clasificación
        brain_class = df_brain[df_brain['Entry_Classification'] == classification]
        
        if brain_class.empty:
            continue
        
        # Cruzar con trades por Symbol
        if 'Symbol' in brain_class.columns and 'Symbol' in df_trades.columns:
            symbols = brain_class['Symbol'].unique()
            trades_class = df_trades[df_trades['Symbol'].isin(symbols)]
            
            if not trades_class.empty:
                winners = len(trades_class[trades_class['Win_Loss'] == 'W'])
                losers = len(trades_class[trades_class['Win_Loss'] == 'L'])
                total = len(trades_class)
                avg_pnl = trades_class['PnL_Percent'].mean()
                
                classifications_data[classification] = {
                    'total_signals': len(brain_class),
                    'total_trades': total,
                    'winners': winners,
                    'losers': losers,
                    'win_rate': float(winners / total * 100) if total > 0 else 0,
                    'avg_pnl': float(avg_pnl) if pd.notna(avg_pnl) else 0
                }
                
                print(f"  📊 {classification}:")
                print(f"    Señales: {len(brain_class)} | Trades: {total}")
                print(f"    W/L: {winners}/{losers} ({classifications_data[classification]['win_rate']:.1f}%)")
                print(f"    PnL avg: {classifications_data[classification]['avg_pnl']:.2f}%")
    
    if not classifications_data:
        print("  ⚠️  No se pudo cruzar información")
        return None
    
    # Calcular deltas
    summary = {
        'classifications': classifications_data,
        'delta_temprana_tardia': 0,
        'best_classification': None
    }
    
    if 'TEMPRANA' in classifications_data and 'TARDÍA' in classifications_data:
        delta = classifications_data['TEMPRANA']['avg_pnl'] - classifications_data['TARDÍA']['avg_pnl']
        summary['delta_temprana_tardia'] = float(delta)
        print(f"\n  💡 Delta TEMPRANA vs TARDÍA: {delta:+.2f}%")
    
    # Mejor clasificación por PnL
    if classifications_data:
        best = max(classifications_data.items(), key=lambda x: x[1]['avg_pnl'])
        summary['best_classification'] = best[0]
        print(f"  🏆 Mejor clasificación: {best[0]} ({best[1]['avg_pnl']:.2f}%)")
    
    return summary


def validate_brain_effectiveness(df_brain, df_trades):
    """
    Valida efectividad del cerebro analizando rechazos vs resultados reales.
    
    Pregunta: ¿Los rechazos evitaron malos trades?
    
    Args:
        df_brain: DataFrame con análisis del cerebro
        df_trades: DataFrame con trades completados
    
    Returns:
        dict con métricas de efectividad
    """
    print("\n✅ Validando efectividad del cerebro...")
    
    if df_brain.empty or df_trades.empty:
        print("  ⚠️  Datos insuficientes para validación")
        return None
    
    # Filtrar rechazos
    rejections = df_brain[df_brain['Brain_Decision'] == 'REJECT']
    
    if rejections.empty:
        print("  ✅ No hay rechazos para validar")
        return {
            'total_rejections': 0,
            'validated': 0,
            'effectiveness': 100.0,
            'summary': 'Sin rechazos (todas señales aprobadas)'
        }
    
    # Convertir PnL a numérico
    df_trades['PnL_Percent'] = pd.to_numeric(df_trades['PnL_Percent'], errors='coerce')
    
    validated_rejections = 0
    good_rejections = 0  # Rechazos que evitaron pérdidas
    bad_rejections = 0   # Rechazos que evitaron ganancias
    
    # Para cada rechazo, buscar si ese symbol operó en otro momento
    for _, rejection in rejections.iterrows():
        symbol = rejection.get('Symbol', None)
        timestamp = rejection.get('Timestamp', None)
        
        if not symbol or not timestamp:
            continue
        
        # Buscar trades del mismo symbol en otro momento
        symbol_trades = df_trades[df_trades['Symbol'] == symbol]
        
        if not symbol_trades.empty:
            validated_rejections += 1
            
            # Analizar resultados de esos trades
            avg_pnl = symbol_trades['PnL_Percent'].mean()
            
            if pd.notna(avg_pnl):
                if avg_pnl < 0:
                    good_rejections += 1  # Evitó pérdida
                else:
                    bad_rejections += 1   # Evitó ganancia
    
    total_rejections = len(rejections)
    effectiveness = (good_rejections / validated_rejections * 100) if validated_rejections > 0 else 0
    
    print(f"  📊 Total rechazos: {total_rejections}")
    print(f"  🔍 Validados (symbol operó después): {validated_rejections}")
    print(f"  ✅ Buenos rechazos (evitaron pérdidas): {good_rejections}")
    print(f"  ❌ Malos rechazos (evitaron ganancias): {bad_rejections}")
    print(f"  📈 Efectividad: {effectiveness:.1f}%")
    
    return {
        'total_rejections': total_rejections,
        'validated': validated_rejections,
        'good_rejections': good_rejections,
        'bad_rejections': bad_rejections,
        'effectiveness': float(effectiveness),
        'summary': f"{effectiveness:.1f}% efectividad en rechazos validados"
    }
